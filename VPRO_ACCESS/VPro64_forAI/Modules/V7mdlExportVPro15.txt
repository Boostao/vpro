Public Sub ExportVPro15()

'Dec. 13, 2019
Dim RemoteDB As Database, MyDB As Database
Dim ProjectName As String, MyFile As String, Msg As String, Title As String
Set MyDB = CurrentDb

'get a name for the new project
    Msg = "Enter a name for your exported project.  The name must start with a letter, contain no punctuation characters, preferably no spaces, and be less than 40 characters in length"
    Title = "Export your project"
    ProjectName = InputBox(Msg, Title)
    If ProjectName = "" Then GoTo MyExit
    If CheckForExistingProject(MyDB, ProjectName & "_Env") Then 'project exists in the local db
        MsgBox "A project of that name is currently attached to VPro.  Please detach or try a different name.", vbInformation, "VPro"
        GoTo MyExit
    End If

'get/create the name of the new or existing db
    MyFile = GetDbName
    If MyFile = "" Then GoTo MyExit
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(MyFile)

'look for a project of that name in the db
    If CheckForExistingProject(RemoteDB, ProjectName & "_Env") Then 'project exists in the remote db
        MsgBox "A project of that name already exists in that db.  Please try a different name or db.", vbInformation, "VPro"
        GoTo MyExit
    End If

'clean up any failed attempt
    DeleteXXXRK

'attach the VPro 15 tables
    AttachVPro15Tables RemoteDB

'append the data from the current project
    AppendVPro15Data clsVProReg.CurrProject

'rename the project
    RenameProject ProjectName, RemoteDB

'unlink the tables
    RemoveLinks ProjectName

'cleanup
    DeleteXXXRK
    
'done
    MsgBox "Done", vbOKOnly, "VPro 19"

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Set RemoteDB = Nothing
    Set MyDB = Nothing

End Sub

Private Function GetDbName() As String

'create new db
Dim MyFile As String, NewFile As Boolean
    
    MyFile = GetFileName("VPro", True, "VPro 15 (*.accdb)", "*.accdb")
    GetDbName = MyFile
    If Dir(MyFile) = "" Then
        DBEngine.Workspaces(0).CreateDatabase Trim(MyFile), DB_LANG_GENERAL
    End If

Private Function CheckForExistingProject(db As Database, ProjectName As String) As Boolean

'Dec. 13, 2019
Dim MyName As String

On Error GoTo errCheckForExistingProject

    MyName = db.TableDefs(ProjectName).Name
    CheckForExistingProject = True

MyExit:
    On Error Resume Next
    
    Exit Function
errCheckForExistingProject:
    If Err.Number = 3265 Then
        CheckForExistingProject = False
    End If
    Resume MyExit
End Function

Private Function Delete15Relationship(RemoteDB As String, Project As String)

'June 30, 2015
On Error GoTo errDeleteRelationship
Dim MyDB As Database, MyRel As Relation, MyFld As Field, RelationName As String
Dim VegTable As String, HumusTable As String, MineralTable As String, MetadataTable As String
Dim AuditTable As String, OtherTable As String, i As Integer
Dim MyRelName As String, X As Integer
Set MyDB = CurrentDb()
Dim EnvTable As String, db As Database ', MyWS As Workspace
EnvTable = Project
    
Set db = DBEngine.Workspaces(0).OpenDatabase(RemoteDB)

'set variable values
    RelationName = EnvTable
    EnvTable = RelationName & "_Env"
    VegTable = RelationName & "_Veg"
    HumusTable = RelationName & "_Humus"
    MineralTable = RelationName & "_Mineral"
    OtherTable = RelationName & "_Other"
    AuditTable = RelationName & "_Audit"
    MetadataTable = RelationName & "_Metadata"

'check for existing relationship
    If Find15Relation(RelationName) = True Then GoTo MyExit

'delete old relationships
    If db.Relations.count > 0 Then
        On Error Resume Next
        For i = 1 To 9
            db.Relations.Delete (Project & i)
        Next i
    End If

MyExit:
    On Error Resume Next
    Set db = Nothing
    Set MyDB = Nothing
    Exit Function
errDeleteRelationship:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function Create15Relationship(RemoteDB As String, Project As String)

'June 30, 2015
On Error GoTo errCreateRelationship
Dim MyDB As Database, MyRel As Relation, MyFld As Field, RelationName As String
Dim VegTable As String, HumusTable As String, MineralTable As String, MetadataTable As String
Dim AuditTable As String, OtherTable As String, i As Integer
Dim TreePlotTable As String, TreesTable As String, CwdTable As String, CwdPileTable As String
Dim EnvTable As String, db As Database
Set MyDB = CurrentDb()

    EnvTable = Project
    Set db = DBEngine.Workspaces(0).OpenDatabase(RemoteDB)

'set variable values
    RelationName = EnvTable
    EnvTable = RelationName & "_Env"
    VegTable = RelationName & "_Veg"
    HumusTable = RelationName & "_Humus"
    MineralTable = RelationName & "_Mineral"
    OtherTable = RelationName & "_Other"
    AuditTable = RelationName & "_Audit"
    MetadataTable = RelationName & "_Metadata"

'is there already a relationship with that name?
    If Find15Relation(RelationName) = True Then GoTo MyExit

'delete old relationships
    If db.Relations.count > 0 Then
        On Error Resume Next
        For i = 1 To db.Relations.count
            MsgBox db.Relations(RelationName & i)
            db.Relations.Delete (RelationName & i)
        Next i
    End If

'set veg relationship
    Set MyRel = db.CreateRelation(RelationName & "1", EnvTable, VegTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set humus relations
    Set MyRel = db.CreateRelation(RelationName & "2", EnvTable, HumusTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set mineral relations
    Set MyRel = db.CreateRelation(RelationName & "3", EnvTable, MineralTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set other relations
    Set MyRel = db.CreateRelation(RelationName & "4", EnvTable, OtherTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set audit relations
    Set MyRel = db.CreateRelation(RelationName & "5", EnvTable, AuditTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set metadata relations
    Set MyRel = db.CreateRelation(RelationName & "6", EnvTable, MetadataTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

On Error GoTo 0

MyExit:
    On Error Resume Next
    Set db = Nothing
    Set MyDB = Nothing
    Exit Function
errCreateRelationship:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function Find15Relation(RelationName As String) As Boolean

'June 30, 2015
Dim MyDB As Database, MyRel As Relation
Set MyDB = CurrentDb()
Find15Relation = False

    For Each MyRel In MyDB.Relations
        If MyRel.Table = RelationName Then
            Find15Relation = True
        End If
    Next

MyExit:
    Set MyDB = Nothing

End Function
