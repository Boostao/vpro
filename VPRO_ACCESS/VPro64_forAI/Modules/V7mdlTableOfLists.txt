Public Function AssignedSiteUnitList(Val As Integer) As String

Dim SqlText As String

Select Case Val
    Case 1
        SqlText = "SELECT UsysEnv.UserSiteUnit From UsysEnv GROUP BY UsysEnv.UserSiteUnit HAVING (((UsysEnv.UserSiteUnit) Is Not Null));"
    Case 2
        'SqlText = "SELECT USysMasterSiteUnitList.Name, USysMasterSiteUnitList.UnitLongName From USysMasterSiteUnitList WHERE (((USysMasterSiteUnitList.Level) = 11)) ORDER BY USysMasterSiteUnitList.Name;"
        SqlText = "SELECT USysMasterSiteUnitList.SiteSeries, USysMasterSiteUnitList.SiteSeriesLongName From USysMasterSiteUnitList WHERE (((USysMasterSiteUnitList.Level) = 11)) ORDER BY USysMasterSiteUnitList.SiteSeries;"
    Case 3
        If clsVProReg.CurrPlotlist <> "None" Then
            SqlText = "SELECT SiteUnit From " & clsVProReg.CurrPlotlist & "_SU GROUP BY SiteUnit HAVING (((SiteUnit) Is Not Null));"
        Else
            MsgBox "Select an SU table first.", vbInformation, "VPro"
            SqlText = "DANGER"
        End If
    Case Else
        SqlText = "SELECT USysMasterSiteUnitList.Name, USysMasterSiteUnitList.UnitLongName From USysMasterSiteUnitList WHERE (((USysMasterSiteUnitList.Level) = 11)) ORDER BY USysMasterSiteUnitList.Name;"
End Select
    
    AssignedSiteUnitList = SqlText

End Function

Public Function ProjectVersionTableOfLists()

'Nov. 11, 2008
Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String
ProjectVersionTableOfLists = ""

Set MyDB = CurrentDb
SqlText = "SELECT Table, EditWhen, AfterEdit"
SqlText = SqlText & " FROM [" & clsVProReg.CurrProject & "_Audit]"
SqlText = SqlText & " WHERE (((Table)='USysTableOfLists'))"
SqlText = SqlText & " ORDER BY EditWhen DESC;"
Set MyRS = MyDB.OpenRecordset(SqlText)

If MyRS.RecordCount = 0 Then
    ProjectVersionTableOfLists = "Unknown"
    GoTo MyExit
End If

If IsNull(MyRS!AfterEdit) Or MyRS!AfterEdit = "" Then
    ProjectVersionTableOfLists = "Unknown"
Else
    ProjectVersionTableOfLists = MyRS!AfterEdit
End If

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
End Function

Public Function ChangeTableOfLists()

'Nov. 11, 2008
    
    Dim MyDB As Database, RemoteDB As Database
    Dim RemoteDbName As String, Success As Boolean, Question As Variant
    Dim MyList As Variant, i As Integer
    Dim MyProject As String
    
    Question = MsgBox("There are dangers involved in not using the 'Standard' version of USysTableOfLists.  If you understand these dangers and wish to proceed, click Yes.", vbYesNo, "VPro")
    If Question = vbNo Then GoTo MyExit
    Success = False
    Set MyDB = CurrentDb()
    RemoteDbName = GetRemoteDb()
    If RemoteDbName = "" Then GoTo MyExit
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(RemoteDbName)

    For i = 0 To RemoteDB.TableDefs.count - 1
        If RemoteDB.TableDefs(i).Name = "USysTableOfLists" Then
            DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, "USysTableOfLists", "USysTableOfListsRK"
            DoCmd.DeleteObject acTable, "USysTableOfLists"
            DoCmd.Rename "USysTableOfLists", acTable, "USysTableOfListsRK"
            Success = True
            Exit For
        End If
    Next i

If Not Success Then
    MsgBox "VPro could not find a USysTableOfLists table in " & RemoteDB.Name & ".  Your USysTableOfLists table has not been changed.", vbInformation, "VPro"
Else
    MsgBox "The USysTableOfLists table from '" & RemoteDB.Name & "' has been attached.  It has the version name '" & TableOfListsVersion & "'.", vbInformation, "VPro"
    LogProjectOut
    LogProjectIn
End If

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Set RemoteDB = Nothing
    Exit Function

End Function
