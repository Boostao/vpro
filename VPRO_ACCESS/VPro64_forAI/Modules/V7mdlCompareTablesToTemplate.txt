Option Compare Database
'The code in this module compares the project in question to the project templates
'Any differences are reported using Excel

Type FieldInfo
    Name As String
    type As String
    Size As String
End Type

Type FieldErrors
    Table As String
    Name As String
    type As String
End Type

Dim ErrorData() As FieldErrors
Dim GoldData As FieldInfo, MyData As FieldInfo
Private Sub TestFunction()

    CheckThisProject "Sample"

End Sub

Public Function CheckThisProject(Project As String)

'Sept 15, 2014

ReDim ErrorData(0)

'Compares the project in question using the CompareTables procedure
    CompareTables "USysAudit", Project & "_Audit"
    CompareTables "USysEnvTable", Project & "_Env"
    CompareTables "USysSoilHumusTable", Project & "_Humus"
    CompareTables "USysSoilMineralTable", Project & "_Mineral"
    CompareTables "USysOtherTable", Project & "_Other"
    CompareTables "USysVegTable", Project & "_Veg"
    CompareTables "USysMetadataTable", Project & "_Metadata"
    CompareTables "USysAdminTable", Project & "_Admin"

'report
    If UBound(ErrorData) > 0 Then
        ReportData
    Else
        MsgBox "No difference was found between your project and the current templates", vbInformation, clsVProReg.ProgramName
    End If

End Function

Private Sub CompareTables(Table1 As String, Table2 As String)

'Sept 15, 2014
Dim MyDB As Database, TableDef1 As TableDef, TableDef2 As TableDef
Dim i As Integer, fldName As String
Set MyDB = CurrentDb
Set TableDef1 = MyDB.TableDefs(Table1)
Set TableDef2 = MyDB.TableDefs(Table2)

'Compares Table1 to Table2 field-by-field
    For i = 0 To TableDef1.Fields.count - 1
        fldName = TableDef1.Fields(i).Name
        If FindField(fldName, TableDef2) Then
            SetGoldData fldName, TableDef1
            CompareFields Table2
        Else
            ReDim Preserve ErrorData(UBound(ErrorData) + 1)
            ErrorData(UBound(ErrorData)).Table = Table2
            ErrorData(UBound(ErrorData)).Name = TableDef1.Fields(i).Name
            ErrorData(UBound(ErrorData)).type = "Missing"
        End If
    Next i

MyExit:
    On Error Resume Next
    Set TableDef1 = Nothing
    Set TableDef2 = Nothing
    Set MyDB = Nothing
    Exit Sub
    
End Sub

Private Function FindField(FieldName1 As String, TD2 As TableDef) As Boolean
'Jan. 7, 2008
'Sets the size and type values in the array MyData.
'Returns false if there was a problem such as a missing field
On Error GoTo errFindField
Dim TestName As Variant
TestName = TD2.Fields(FieldName1).Name

FindField = True
MyData.Name = FieldName1
MyData.Size = TD2.Fields(FieldName1).Size
MyData.type = TD2.Fields(FieldName1).type

Exit Function
errFindField:
If Err.Number = 3265 Then
    FindField = False
End If

End Function

Private Sub ReportData()
'Jan. 7, 2008
'Transfers the array "ErrorData" to Excel
    Dim xl As Excel.Application
    
    CreateExcel xl
    xl.Visible = True
    xl.Workbooks.Add
    xl.Sheets.Add
    
    xl.Cells(1, 1) = "Table"
    xl.Cells(1, 2) = "Field"
    xl.Cells(1, 3) = "Error Type"
    
    For i = 1 To UBound(ErrorData)
        xl.Cells(i + 1, 1) = ErrorData(i).Table
        xl.Cells(i + 1, 2) = ErrorData(i).Name
        xl.Cells(i + 1, 3) = ErrorData(i).type
    Next i
    
    xl.Rows(1).Font.Bold = True
    xl.Columns("A:C").EntireColumn.AutoFit

End Sub

Private Sub SetGoldData(FieldName1 As String, TD2 As TableDef)
'Jan. 7, 2008
'Copies field size and type information from the template to the array "GoldData"
On Error GoTo errSetGoldData

GoldData.Name = FieldName1
GoldData.Size = TD2.Fields(FieldName1).Size
GoldData.type = TD2.Fields(FieldName1).type

Exit Sub
errSetGoldData:
    MsgBox Err.Description
    Exit Sub
End Sub

Private Function CompareFields(Table2 As String) As Boolean
'Jan. 7, 2008
'Compares the data in the "GoldData" array to the data in "MyData" array
If MyData.Size <> GoldData.Size Then
    ReDim Preserve ErrorData(UBound(ErrorData) + 1)
    ErrorData(UBound(ErrorData)).Table = Table2
    ErrorData(UBound(ErrorData)).Name = MyData.Name
    ErrorData(UBound(ErrorData)).type = "Size"
End If

If MyData.type <> GoldData.type Then
    ReDim Preserve ErrorData(UBound(ErrorData) + 1)
    ErrorData(UBound(ErrorData)).Table = Table2
    ErrorData(UBound(ErrorData)).Name = MyData.Name
    ErrorData(UBound(ErrorData)).type = "Type"
End If

MyExit:

End Function
