Option Compare Database

Function ExportUserSiteUnits()

''May 25, 2009
'Exports users' site unit list
On Error Resume Next
DoCmd.DeleteObject acTable, "TempUnitList"
On Error GoTo errExportUserSiteUnits
Dim MyDB As Database, RemoteDB As Database, i As Integer
Dim MyFile As String, MyTitle As String, strTestTable As String
Dim ListInDB As Boolean, MyDir As String
ListInDB = False

    Set MyDB = CurrentDb()
    MyFile = ""
    MyTitle = ProgramName() & " - pick a database to store your list in"
    MyFile = GetFileName(MyTitle, False)
    If MyFile = "" Then Exit Function
    
AfterCreation:
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(MyFile)
        
        For i = 0 To RemoteDB.TableDefs.count - 1
            strTestTable = RemoteDB.TableDefs(i).Name
            If strTestTable = "UserSiteUnitList" Then
                ListInDB = True
            End If
        Next i
        
    If ListInDB Then
        MsgBox MyFile & " already has a user site unit list in it.  Please pick a different database or create a new database."
        GoTo MyExit
    End If

    MyDir = clsVProReg.Location & "VUser.accda"
    DoCmd.TransferDatabase acImport, "Microsoft Access", MyDir, acTable, "UserSiteUnitList", "TempUnitList"
    DoCmd.TransferDatabase acExport, "Microsoft Access", RemoteDB.Name, acTable, "TempUnitList", "UserSiteUnitList"
    MsgBox "Your site unit list was successfully copied to the database " & RemoteDB.Name, vbInformation, "VPro"
        
MyExit:
    On Error Resume Next
    DoCmd.DeleteObject acTable, "TempUnitList"
    Set MyDB = Nothing
    Set RemoteDB = Nothing
    Exit Function

CantFindDatabase:
    Msg = "Couldn't find specified database" & Chr(13)
    Msg = Msg & "VPro will now create " & Trim(MyFile)
    MsgBox Msg, 64, "VPro"
    
    If Trim(MyFile) = MyDB.Name Then Exit Function
    DBEngine.Workspaces(0).CreateDatabase Trim(MyFile), DB_LANG_GENERAL
    Set TempDB = Nothing
    GoTo AfterCreation

errExportUserSiteUnits:
    If Err.Number = 3024 Then
        GoTo CantFindDatabase
    Else
        MsgBox Err.Number & Chr(13) & Err.Description
    End If
    Resume MyExit
End Function
