Private Sub WriteSpp2020(File2ExportTo As String)

'Jun 2, 2020
On Error GoTo errWriteSppFile
Dim MyDB As Database, MyRS As Recordset
Dim i As Integer, Text2Write As String
Set MyDB = CurrentDb
Set MyRS = MyDB.OpenRecordset("tmpUSysAllVegSpp", dbOpenDynaset)
MyRS.MoveLast
MyRS.MoveFirst
Close #1
Open File2ExportTo For Output As #1

For i = 1 To MyRS.RecordCount
    Text2Write = Format(MyRS!SpeciesNumber, "!@@@@@") & Format(MyRS!SppAndStrata, "!@@@@@@@@@@@") & MyRS!FirstOfScientificName
    Print #1, Text2Write
    MyRS.MoveNext
Next i

MyExit:
    On Error Resume Next
    Close #1
    MyRS.Close
    Set MyDB = Nothing
    Exit Sub
errWriteSppFile:
    MsgBox Err.Description
    Resume MyExit
End Sub

Private Sub WriteDatBreak2020(File2ExportTo As String)

'Jun 2, 2020
Dim MyDB As Database, Loop1RS As Recordset, Loop2RS As Recordset, Loop3RS As Recordset
Dim SqlText As String, Criteria As String, MyData As String
Dim MyUnit As String

Set MyDB = CurrentDb
Close #1
Open File2ExportTo For Output As #1

    SqlText = "SELECT BreakCode From USysBreakTable GROUP BY BreakCode;"
    Set Loop1RS = MyDB.OpenRecordset(SqlText)
    
    Do Until Loop1RS.EOF
        MyUnit = Loop1RS!BreakCode
        Print #1, "BREAK"
        Print #1, MyUnit
        SqlText = "SELECT PlotNumber From USysBreakTable WHERE (((BreakCode)='" & MyUnit & "'));"
        Set Loop2RS = MyDB.OpenRecordset(SqlText)
        
        Do Until Loop2RS.EOF 'list of units

            'get veg data
            SqlText = "SELECT [SpeciesNumber] & ',' & Format([SumOfSumOfCover],'0.0') AS Data"
            SqlText = SqlText & " From tmpUSysAllVegSum"
            SqlText = SqlText & " WHERE (((tmpUSysAllVegSum.PlotNumber)='" & Loop2RS!PlotNumber & "'));"
            Set Loop3RS = MyDB.OpenRecordset(SqlText)
            Print #1, Loop2RS!PlotNumber
            MyData = ""
            
            Do Until Loop3RS.EOF 'veg data
                MyData = MyData & Loop3RS!Data & " "
                Loop3RS.MoveNext
            Loop
            MyData = Trim(MyData) & "/"
            Print #1, MyData
        
        Loop2RS.MoveNext
        Loop
        
        
        Loop1RS.MoveNext
    Loop
    
MyExit:
    On Error Resume Next
    Close #1
    Loop1RS.Close
    Loop2RS.Close
    Loop3RS.Close
    Set MyDB = Nothing
    Exit Sub

End Sub

Public Function LumpCompactData()

'Apr. 28, 2020
'This is the main function for generating and exporting PC-ORD compact data
    Dim SuTable As String, SqlText As String, VegTable As String
    Dim MyDB As Database, MyTable As TableDef, MyQueryDef  As QueryDef

'get rid of old crap
    CleanUp

'reference the current su table
    SuTable = clsVProReg.CurrPlotlist

'use only plots in su to make it run faster
    MakeSmallProject SuTable
    
'if there is no lumping table then use all species else build sql for lump
    If sysLumpingTable = "" Or sysLumpingTable = "_Lump" Or sysLumpingTable = "CombinedSpp_Lump" Or sysLumpingTable = "None_Lump" Then
        CompactVegTable = clsVProReg.CurrProject & "_Veg"
        VegTable = clsVProReg.CurrProject & "_Veg"
        If SuTable <> "None" Then
            SqlText = "SELECT [" & VegTable & "].PlotNumber, Layer, Species, Cover1, Cover2, Cover3, Cover4, Cover5, Cover5a, Cover5b, Cover5c, Cover6, Cover7, TotalA, TotalB INTO TmpLumpDat"
            SqlText = SqlText & " FROM [" & SuTable & "_SU] LEFT JOIN [" & VegTable & "] ON [" & SuTable & "_SU].PlotNumber = [" & VegTable & "].PlotNumber;"
            DoCmd.RunSQL SqlText
        Else 'there is lumping
            SqlText = "SELECT PlotNumber, Layer, Species, Cover1, Cover2, Cover3, Cover4, Cover5, Cover5a, Cover5b, Cover5c, Cover6, Cover7, TotalA, TotalB INTO TmpLumpDat"
            SqlText = SqlText & " FROM [" & VegTable & "];"
            DoCmd.RunSQL SqlText
        End If
        CreateSppList
        Exit Function
    Else 'there is a lump table
        Set MyDB = CurrentDb()
        VegTable = clsVProReg.CurrProject & "_Veg"
        If SuTable <> "None" Then
            SuTable = SuTable & "_SU"
            SqlText = "SELECT [" & VegTable & "].PlotNumber, [" & VegTable & "].Layer, IIf(IsNull([Lumpcode]),[Species],[LumpCode]) AS Spp, [" & VegTable & "].Cover1, [" & VegTable & "].Cover2, [" & VegTable & "].Cover3, [" & VegTable & "].Cover4, [" & VegTable & "].Cover5, [" & VegTable & "].Cover5a, [" & VegTable & "].Cover5b, [" & VegTable & "].Cover5c, [" & VegTable & "].Cover6, [" & VegTable & "].Cover7, [" & VegTable & "].TotalA , [" & VegTable & "].TotalB INTO TmpLumpDat"
            SqlText = SqlText & " FROM [" & SuTable & "] LEFT JOIN ([" & VegTable & "] LEFT JOIN [" & sysLumpingTable & "] ON [" & VegTable & "].Species = [" & sysLumpingTable & "].SppCode) ON [" & SuTable & "].PlotNumber = [" & VegTable & "].PlotNumber;"
        Else
            SqlText = "SELECT PlotNumber, Layer, IIf(IsNull([Lumpcode]),[Species],[LumpCode]) AS Spp, Cover1, Cover2, Cover3, Cover4, Cover5, Cover5a, Cover5b, Cover5c, Cover6, Cover7, TotalA, TotalB INTO TmpLumpDat"
            SqlText = SqlText & " FROM [" & VegTable & "] LEFT JOIN [" & sysLumpingTable & "] ON [" & VegTable & "].Species = [" & sysLumpingTable & "].SppCode;"
        End If
    'MyDB.CreateQueryDef "herman", SqlText

'make table TmpLumpDat
        DoCmd.RunSQL SqlText
        Set MyTable = MyDB.TableDefs("TmpLumpDat")
        MyTable.Fields("Spp").Name = "Species"
        CompactVegTable = "TmpLumpDat"
        Set MyDB = Nothing
        GoTo MyExit
    End If

MyExit:
'Stop
    CreateSppList
    Exit Function
'should now have a query named "tmpUSysAllVeg"
'a table named "TmpLumpDat"

End Function

Private Sub CreateSppList()

'Apr. 28, 2020
    'On Error GoTo errCreateSppList
    Dim SqlText As String, VegTable As String, MySU As String
    
    'CompactVegTable = "tmpUSysAllVeg"
    
    VegTable = CompactVegTable
    DoCmd.SetWarnings False
    DoCmd.CopyObject , "SpeciesList", acTable, "USysExportSppList"

    MySU = clsVProReg.CurrPlotlist
    DoCmd.Hourglass True

'add new species records
    If MySU = "None" Then
        If clsRepOpt.PcOrdStrataAsSpp <> 0 Then
            'MsgBox "here1"
        Else
            SqlText = "INSERT INTO SpeciesList ( Code, ScientificName )"
            SqlText = SqlText & " SELECT DISTINCT [" & VegTable & "].Species, USysAllSpecs.ScientificName"
            SqlText = SqlText & " FROM [" & VegTable & "] LEFT JOIN USysAllSpecs ON [" & VegTable & "].Species = USysAllSpecs.Code"
            SqlText = SqlText & " WHERE (((USysAllSpecs.CodeType)='U')) OR (((USysAllSpecs.CodeType) Is Null));"
            DoCmd.RunSQL SqlText
        End If
    Else 'has SU table
        If clsRepOpt.PcOrdStrataAsSpp <> 0 Then
            'A strata
            SqlText = "INSERT INTO SpeciesList ( Code, CodePlusStrata, ScientificName )"
            SqlText = SqlText & " SELECT DISTINCT TmpLumpDat.Species, [Species] & 'A' AS StrataCode, USysAllSpecs.ScientificName"
            SqlText = SqlText & " FROM [" & MySU & "_SU] INNER JOIN (TmpLumpDat LEFT JOIN USysAllSpecs ON TmpLumpDat.Species = USysAllSpecs.Code) ON [" & MySU & "_SU].PlotNumber = TmpLumpDat.PlotNumber"
            SqlText = SqlText & " WHERE (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null) AND ((TmpLumpDat.Cover1)>0)) OR (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null) AND ((TmpLumpDat.Cover2)>0)) OR (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null) AND ((TmpLumpDat.Cover3)>0)) OR (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null) AND ((TmpLumpDat.TotalA)>0));"
            
            SqlText = "INSERT INTO SpeciesList ( Code, CodePlusStrata, ScientificName )"
            SqlText = SqlText & " SELECT DISTINCT First(TmpLumpDat.Species) AS FirstOfSpecies, [Species] & 'A' AS StrataCode, First(USysAllSpecs.ScientificName) AS FirstOfScientificName"
            SqlText = SqlText & " FROM [" & MySU & "_SU] INNER JOIN (TmpLumpDat LEFT JOIN USysAllSpecs ON TmpLumpDat.Species = USysAllSpecs.Code) ON [" & MySU & "_SU].PlotNumber = TmpLumpDat.PlotNumber"
            SqlText = SqlText & " WHERE (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null)) OR (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null)) OR (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null)) OR (((USysAllSpecs.CodeType)='U' Or (USysAllSpecs.CodeType) Is Null))"
            SqlText = SqlText & " GROUP BY [Species] & 'A'"
            SqlText = SqlText & " HAVING (((Sum(TmpLumpDat.Cover1))>0)) OR (((Sum(TmpLumpDat.Cover2))>0)) OR (((Sum(TmpLumpDat.Cover3))>0)) OR (((Sum(TmpLumpDat.TotalA))>0));"
            

            SqlText = "INSERT INTO SpeciesList ( Code, CodePlusStrata, ScientificName )"
            SqlText = SqlText & " SELECT DISTINCT TmpLumpDat.Species, Left([Species],7) & 'A' AS StrataCode, First(USysAllSpecs.ScientificName) AS FirstOfScientificName"
            SqlText = SqlText & " FROM [" & MySU & "_SU] INNER JOIN (TmpLumpDat LEFT JOIN USysAllSpecs ON TmpLumpDat.Species = USysAllSpecs.Code) ON [" & MySU & "_SU].PlotNumber = TmpLumpDat.PlotNumber"
            SqlText = SqlText & " WHERE (((USysAllSpecs.CodeType)='u' Or (USysAllSpecs.CodeType) Is Null))"
            SqlText = SqlText & " GROUP BY TmpLumpDat.Species, Left([Species],7) & 'A'"
            SqlText = SqlText & " HAVING (((Sum(TmpLumpDat.Cover1))>0)) OR (((Sum(TmpLumpDat.Cover2))>0)) OR (((Sum(TmpLumpDat.Cover3))>0)) OR (((Sum(TmpLumpDat.TotalA))>0));"


'CurrentDb.CreateQueryDef "Herman4b", SqlText
            DoCmd.RunSQL SqlText
            'B strata
            SqlText = "INSERT INTO SpeciesList ( Code, CodePlusStrata, ScientificName )"

Public Function JustTheFirstTwoWords(WordIn As Variant) As String

Dim Loc As Integer

If IsNull(WordIn) Then
    JustTheFirstTwoWords = ""
    Exit Function
End If

Loc = InStr(1, WordIn, " ")

Loc = InStr(Loc + 1, WordIn, " ")

If Loc = 0 Then
    JustTheFirstTwoWords = WordIn
Else
    JustTheFirstTwoWords = Left(WordIn, Loc - 1)
End If

End Function
