Public Sub ImportPdfXML(OverwriteOption As Integer, ImportVeg As Boolean, ImportHumus As Boolean, ImportMineral As Boolean)

Dim FileName As String

    FileName = GetFileName
    If XMLFileToUse = "" Then GoTo MyExit
    GetName XMLFileToUse
    MakePages
    Application.ImportXML PathToUse & NameToUse & ".xml", acAppendData
    ImportPg1All OverwriteOption
    ImportPg2All OverwriteOption
    ImportPg3All OverwriteOption
    If ImportVeg Then ImportVegData
    If ImportHumus Then ImportHumusData
    If ImportMineral Then ImportMineralData
    'DeletePages
    
MyExit:

End Sub
Private Sub ImportMineralData()

Dim MyDB As Database, VProRS As Recordset, XmlRS As Recordset
Dim SqlText As String, PltNumber As String, i As Integer, fldName As String
Set MyDB = CurrentDb
Set XmlRS = MyDB.OpenRecordset("Pg3")
Set VProRS = MyDB.OpenRecordset("UsysMineral")

'get plot number
    If XmlRS.RecordCount = 0 Then GoTo MyExit
    XmlRS.MoveFirst
    PltNumber = XmlRS!PlotNumber

'empty humus for the indicated plot
    DoCmd.SetWarnings False
    SqlText = "DELETE * From USysMineral WHERE (((PlotNumber)='" & PltNumber & "'));"
    DoCmd.RunSQL SqlText
    DoCmd.SetWarnings True

'import mineral (1-6)
    For i = 1 To 6
        fldName = "MineralHorizon0" & i
        If Not IsNull(XmlRS(fldName)) Then
            VProRS.AddNew
            VProRS!PlotNumber = PltNumber
            VProRS!Horizon = XmlRS("MineralHorizon0" & i)

            VProRS!LowerDepth = XmlRS("MineralLayerDepthLow0" & i)

            VProRS!PercentCoarseFragsStones = XmlRS("PercentCoarseFragsStones0" & i)
            VProRS!PercentCoarseFragsTotal = XmlRS("PercentCoarseFragsTotal0" & i)
            VProRS!PercentCoarseFragsShape = XmlRS("PercentCoarseFragsShape0" & i)
            VProRS!RootsAbundance = XmlRS("MineralRootsAbundance0" & i)
            VProRS!RootsSize = XmlRS("MineralRootsSize0" & i)

Private Sub ImportVegData()

'import herbs (1-18)

        fldName = "HE" & Format(i, "00")
        If Not IsNull(XmlRS(fldName)) Then
            VProRS.AddNew
            VProRS!PlotNumber = PltNumber

'import moss (1-9)
    For i = 1 To 9
        fldName = "MO0" & i
        If Not IsNull(XmlRS(fldName)) Then
            VProRS.AddNew
            VProRS!PlotNumber = PltNumber

            MsgBox "You are missing some layer information for " & XmlRS("OT0" & i) & ".  Please fix and try again.", vbInformation, "VPro"

Public Function ZLtoNull(ValIn As Variant)

    If ValIn = "" Then ValIn = Null
    ZLtoNull = ValIn
    
End Function

Private Function CheckForExistingPlot(MyDB As Database, PlotNumber As String) As Boolean

Dim MyRS As Recordset
Dim Criteria As String

    Set MyRS = MyDB.OpenRecordset("USysEnv")
    If MyRS.RecordCount > 0 Then
        MyRS.MoveLast
        MyRS.MoveFirst
    Else
        CheckForExistingPlot = False
        GoTo MyExit
    End If
    Criteria = "Plotnumber = '" & PlotNumber & "'"
    MyRS.FindFirst Criteria
    If MyRS.NoMatch Then
        CheckForExistingPlot = False
    Else
        CheckForExistingPlot = True
    End If
    
MyExit:
    On Error Resume Next
    MyRS.Close
    On Error GoTo 0
    
End Function

Private Sub DeletePages()

    On Error Resume Next
    DoCmd.DeleteObject acTable, "Pg1"
    DoCmd.DeleteObject acTable, "Pg2"
    DoCmd.DeleteObject acTable, "Pg3"
    On Error GoTo 0
    
End Sub

Private Sub MakePages()

    DeletePages
    DoCmd.CopyObject , "Pg1", acTable, "USysPg1"
    DoCmd.CopyObject , "Pg2", acTable, "USysPg2"
    DoCmd.CopyObject , "Pg3", acTable, "USysPg3"

End Sub


Private Sub ZLSBeGoneXML(TblName As String)

'removes zero-length strings
Dim MyDB As Database
Dim SqlText As String, i As Integer, X As Integer, fldName As String
Dim fldCount As Integer, tblCount As String
Set MyDB = CurrentDb
'TblName = "Sample_env"
tblCount = 0
fldCount = 0
DoCmd.SetWarnings False
    For i = 0 To MyDB.TableDefs("" & TblName & "").Fields.count - 1
        fldName = MyDB.TableDefs("" & TblName & "").Fields(i).Name
        If MyDB.TableDefs("" & TblName & "").Fields(i).type = 10 Then
            SqlText = "UPDATE [" & TblName & "] SET [" & TblName & "].[" & fldName & "] = Null WHERE ((([" & TblName & "].[" & fldName & "])=''));"
            DoCmd.RunSQL SqlText
        End If
        If MyDB.TableDefs("" & TblName & "").Fields(i).type = dbMemo Then
            SqlText = "UPDATE [" & TblName & "] SET [" & TblName & "].[" & fldName & "] = Null WHERE ((([" & TblName & "].[" & fldName & "])=''));"
            DoCmd.RunSQL SqlText
        End If
    Next i

Set MyDB = Nothing
DoCmd.SetWarnings True
End Sub

Private Function GetFileName()

    Dim FD As FileDialog
    Set FD = Application.FileDialog(msoFileDialogFilePicker)
    FD.AllowMultiSelect = False
    FD.filters.Add "XML File", "*.xml", 1
    FD.FilterIndex = 1
    XMLFileToUse = ""

    With FD
        If .Show = -1 Then
            XMLFileToUse = FD.SelectedItems.Item(1)
        End If
    End With
    
    If XMLFileToUse = "" Then DoStop = True
    Set FD = Nothing

End Function

Private Sub GetName(FileName As String)

On Error GoTo errGetName
Dim MyStr As String, MyPos As Integer, ExitLoop As Boolean, X As Integer, MaxNum As Integer
MyStr = FileName
MaxNum = 0

MyPos = InStr(1, MyStr, "\", vbTextCompare)

Do While Not ExitLoop
    X = InStr(MyPos + 1, MyStr, "\", vbTextCompare)
    If X = 0 Then
        MaxNum = MyPos
        Exit Do
    End If
    If X > MaxNum Then MaxNum = X
    MyPos = X
    If MyPos = 0 Then
        Exit Do
    End If
Loop

NameToUse = Mid(MyStr, MaxNum + 1, Len(MyStr) - MaxNum - 4)
PathToUse = Left(MyStr, Len(MyStr) - (Len(NameToUse) + 4))

MyExit:
    Exit Sub
    
errGetName:
    If Err.Number = 5 Then
        'do nothing
    Else
        MsgBox Err.Number & " - " & Err.Description
    End If
    DoStop = True
    Resume MyExit
End Sub
