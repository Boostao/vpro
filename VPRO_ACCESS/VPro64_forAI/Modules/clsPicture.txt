Function ShowPicture(PlotOrUnit As String, FormName As String, PictureControl As String, _
Optional FileName As Variant)
'copies the file from the db into a picture file and loads it into a picture control
'Jan. 5, 2005

If clsPicture.PictureLibrary = 0 Then
    'MsgBox "You haven't yet created a picture library.  Once you've dismissed this message box please use the button on the lower-right of the picture form to create a picture library.", vbInformation, "VPro"
    'OpenPicForm
    GoTo MyExit
End If

StartOver:
On Error GoTo errShowPicture
Dim MyDB As Database, MyRS As Recordset
Dim MyFile As String, BytesWritten As Variant, Criteria As String, SqlText As String
    Set MyDB = CurrentDb
    Set MyRS = MyDB.OpenRecordset("tblPictureBlob", dbOpenDynaset)
    OpenPicForm
    Forms(FormName).Controls("MyPlotNumber") = PlotOrUnit
    If clsPicture.nPictures(MyRS, PlotOrUnit) > 0 Then
        SqlText = "SELECT FileName FROM tblPictureBlob WHERE (((PlotOrUnit)='" & PlotOrUnit & "'));"
        Forms(FormName).Controls("cmbPicList").RowSource = SqlText
    End If
    MyRS.MoveFirst
    If Not IsMissing(FileName) Then
        Criteria = "PlotOrUnit = '" & PlotOrUnit & "' AND FileName = '" & FileName & "'"
    Else
        Criteria = "PlotOrUnit = '" & PlotOrUnit & "'"
    End If
    MyRS.FindFirst Criteria
    If Not MyRS.NoMatch Then
        MyFile = Left(CurrentDb.Name, Len(CurrentDb.Name) - Len(Dir(CurrentDb.Name)))
'        MyFile = MyFile & MyRS!FileName
        MyFile = MyFile & "xxx." & Right(MyRS!FileName, 3)
        If MyFile = "" Or IsNull(MyFile) Then GoTo MyExit
        Forms(FormName).Controls(PictureControl).Picture = ""
        Forms(FormName).Repaint
        'Kill MyFile
        BytesWritten = bWriteBLOB(MyRS, "Blob", MyFile)
        'need form name/open check here
        Forms(FormName).Controls(PictureControl).Picture = MyFile
        Forms(FormName).Repaint
        sysCurrPicture = MyRS!FileName
        Forms(FormName).Caption = sysCurrPicture
    Else
        Forms(FormName).Controls(PictureControl).Picture = ""
        sysCurrPicture = ""
        Forms(FormName).Caption = "No Picture"
        Forms(FormName).Repaint
    End If
    Forms(FormName).Controls("cmbPicList") = ""
MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
errShowPicture:
    SysCmd acSysCmdRemoveMeter
    Select Case Err.Number
        Case 3021
        'do nothing, there are no records in the picture db
        Case 70
            Forms(FormName).Controls(PictureControl).Picture = ""
            GoTo StartOver
        Case 3078
            MsgBox "VPro can't find your picture library.  Please locate an existing or create a new library.", vbInformation, "VPro"
            OpenPicForm
        Case Else
            MsgBox Err.Description & Chr(13) & Err.Number
        End Select
    Resume MyExit

End Function
Public Function OpenPictureLibrary() As Boolean

'Sep 22, 2008
Dim RemoteDB As Database, FoundFile As Boolean
Dim MyFile As String, i As Integer
    
    MyFile = GetFileName(clsVProReg.ProgramName, True)
    If MyFile = "" Or IsNull(MyFile) Then GoTo MyExit

Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(MyFile)

For i = 0 To RemoteDB.TableDefs.count - 1
    If RemoteDB.TableDefs(i).Name = "tblPictureBlob" Then
        DeletePictureDB
        DoCmd.TransferDatabase A_ATTACH, "Microsoft Access", MyFile, A_TABLE, "tblPictureBlob", "tblPictureBlob"
        clsPicture.PictureLibrary = MyFile
        FoundFile = True
        Exit For
    End If
Next i
    
    If FoundFile Then
        OpenPictureLibrary = True
    Else
        MsgBox MyFile & " did not contain a picture library.", vbInformation, "Sorry"
    End If
    
MyExit:
    On Error Resume Next
    Set RemoteDB = Nothing
    Exit Function
End Function

Public Function NewPictureDB()

'Sep 22, 2008
    On Error GoTo errNewPictureDB
    Dim MyDB As Database
    Dim MyTitle As String, DataLocationName As String
    Set MyDB = CurrentDb()
    
    MyTitle = ProgramName() & " - Create a new database"
    DataLocationName = GetFileName(MyTitle, True)
    If Trim(DataLocationName) = "" Then Exit Function
    If Trim(DataLocationName) = MyDB.Name Then
        MsgBox "You can't create tables in this database.", 64, ProgramName()
        Exit Function
    End If
    If Dir(DataLocationName) <> "" Then
        MsgBox "You must create a new database. (The picture database works differently than you may be used to.  With picture databases you can only have one table per database)"
        GoTo exitNewPictureDB
    Else
        DBEngine.Workspaces(0).CreateDatabase Trim(DataLocationName), DB_LANG_GENERAL
    End If
    
DoCmd.CopyObject DataLocationName, "tblPictureBlob", acTable, "USysPictureBlob"
DeletePictureDB
DoCmd.TransferDatabase A_ATTACH, "Microsoft Access", DataLocationName, A_TABLE, "tblPictureBlob", "tblPictureBlob"
clsPicture.PictureLibrary = DataLocationName

exitNewPictureDB:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function

    Resume exitNewPictureDB

Private Function bReadBLOB(Source As Variant, T As Recordset, _
sField As String)
    Dim NumBlocks As Integer, SourceFile As Integer, i As Integer
    Dim FileLength As Long, LeftOver As Long
    Dim FileData As String
    Dim RetVal As Variant

    On Error GoTo Err_ReadBLOB

    ' Open the source file.
    SourceFile = FreeFile
    Open Source For Binary Access Read As SourceFile

    ' Get the length of the file.
    FileLength = LOF(SourceFile)
    If FileLength = 0 Then
        bReadBLOB = 0
        Exit Function
    End If

    ' Calculate the number of blocks to read and leftover bytes.
    NumBlocks = FileLength \ BlockSize
    LeftOver = FileLength Mod BlockSize

    ' SysCmd is used to manipulate status bar meter.
    'RetVal = SysCmd(acSysCmdInitMeter, "Reading BLOB", _
             FileLength \ 1000)

    ' Put the record in edit mode.
    T.Edit

    ' Read the leftover data, writing it to the table.
    FileData = String$(LeftOver, 32)
    Get SourceFile, , FileData
    T(sField).AppendChunk (FileData)

    'RetVal = SysCmd(acSysCmdUpdateMeter, LeftOver / 1000)

    ' Read the remaining blocks of data, writing them to the table.
    FileData = String$(BlockSize, 32)
    For i = 1 To NumBlocks
        Get SourceFile, , FileData
        T(sField).AppendChunk (FileData)

        'RetVal = SysCmd(acSysCmdUpdateMeter, BlockSize * i / 1000)
    Next i

    ' Update the record and terminate function.
    T.Update
    'RetVal = SysCmd(acSysCmdRemoveMeter)
    Close SourceFile
    bReadBLOB = FileLength
    Exit Function

Err_ReadBLOB:
    SysCmd acSysCmdRemoveMeter
    MsgBox Err.Description
    bReadBLOB = -Err
    Exit Function

End Function

      Private Function bWriteBLOB(T As Recordset, sField As String, _
      Destination As String)
          Dim NumBlocks As Integer, DestFile As Integer, i As Integer
          Dim FileLength As Long, LeftOver As Long
          Dim FileData As String
          Dim RetVal As Variant

'          On Error GoTo Err_WriteBLOB

          ' Get the size of the field.
          FileLength = T(sField).FieldSize()
          If FileLength = 0 Then
              bWriteBLOB = 0
              Exit Function
          End If

          ' Calculate number of blocks to write and leftover bytes.
          NumBlocks = FileLength \ BlockSize
          LeftOver = FileLength Mod BlockSize

          ' Remove any existing destination file.
          DestFile = FreeFile
          Open Destination For Output As DestFile
          Close DestFile

          ' Open the destination file.
          Open Destination For Binary As DestFile

          ' SysCmd is used to manipulate the status bar meter.
'          RetVal = SysCmd(acSysCmdInitMeter, _
          "Writing BLOB", FileLength / 1000)

          ' Write the leftover data to the output file.
          FileData = T(sField).GetChunk(0, LeftOver)
          Put DestFile, , FileData

          ' Update the status bar meter.
'          RetVal = SysCmd(acSysCmdUpdateMeter, LeftOver / 1000)

          ' Write the remaining blocks of data to the output file.
          For i = 1 To NumBlocks
              ' Reads a chunk and writes it to output file.
              FileData = T(sField).GetChunk((i - 1) * BlockSize _
                 + LeftOver, BlockSize)
              Put DestFile, , FileData

'              RetVal = SysCmd(acSysCmdUpdateMeter, _
              ((i - 1) * BlockSize + LeftOver) / 1000)
          Next i

          ' Terminates function
'          RetVal = SysCmd(acSysCmdRemoveMeter)
          Close DestFile
          bWriteBLOB = FileLength
          Exit Function

Err_WriteBLOB:
    SysCmd acSysCmdRemoveMeter
    bWriteBLOB = -Err
    Err.Clear
    Exit Function

      End Function
