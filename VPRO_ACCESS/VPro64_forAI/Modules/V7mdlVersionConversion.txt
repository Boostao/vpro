Option Compare Database
Option Explicit

Public Function GetTableVersion(TestTable As String) As String
On Error GoTo errGetVersion
Dim MyDB As Database, MyTable As TableDef
Set MyDB = CurrentDb()
Set MyTable = MyDB.TableDefs(TestTable)

    GetTableVersion = MyTable.Properties(12).Value

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function
errGetVersion:
    If Err.Number = 3265 Then
        GetTableVersion = "Unknown"
        Resume MyExit
    End If
    MsgBox Err.Description
    Resume MyExit
End Function
