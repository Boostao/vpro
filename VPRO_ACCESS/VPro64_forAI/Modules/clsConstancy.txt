Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = False
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Compare Database
Option Explicit
Private Const strApp = "VPro64"

Public Property Get CurrConstancy()
    CurrConstancy = GetSetting(strApp, "Current", "CurrConstancy", "None")
End Property

Public Property Let CurrConstancy(CurrConstancy)
    SaveSetting strApp, "Current", "CurrConstancy", CurrConstancy
End Property
