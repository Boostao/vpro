        Case "sb01btn05"
            Set image = LoadPictureGDIP(PicDir & "Form6x4.png")
        Case "sb01cbtn05"
            Set image = LoadPictureGDIP(PicDir & "Form6x4.png")
        Case "sb01btn06"
            Set image = LoadPictureGDIP(PicDir & "frmSIVIsite.png")
        Case "sb01cbtn06"
            Set image = LoadPictureGDIP(PicDir & "frmSIVIsite.png")
        Case "sb01btn08"
            Set image = LoadPictureGDIP(PicDir & "CHARS.png")

End Sub
