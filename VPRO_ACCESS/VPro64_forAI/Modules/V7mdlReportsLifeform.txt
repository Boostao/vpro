Private Sub LoopUnits(SuTable As String)

Dim MyDB As Database, MyRS As Recordset
Set MyDB = CurrentDb
    SqlText = "SELECT SiteUnit, Count(PlotNumber) AS nPlots"
    SqlText = SqlText & " From " & SuTable
    SqlText = SqlText & " GROUP BY SiteUnit;"
    Set MyRS = MyDB.OpenRecordset(SqlText)

Do Until MyRS.EOF
    NumberOfPlots = MyRS!nPlots
    CleanUpQueries
    BuildQueries MyRS!SiteUnit
'Stop
    BuildReport MyRS!SiteUnit
    CountSummaryLabels
    CountSummaryValues
    AddLifeformData
    MyRS.MoveNext
Loop

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Sub
End Sub

Private Sub BuildReport(UnitName As String)

'-------------------------------------------------------------
'WARNING: IF THERE'S NO DATA OR NO BLANK CELLS, IT COULD CRASH
'-------------------------------------------------------------
Dim MyDB As Database, MyRS As Recordset, RepRow As Integer
Set MyDB = CurrentDb
xl.Sheets.Add
xl.ActiveSheet.Name = SheetNameCheck(UnitName, "-")
xl.Cells(1, 1) = "Project: " & clsVProReg.CurrProject
xl.Cells(2, 1) = "Site Unit Table: " & SuTable
xl.Cells(3, 1) = "Unit: " & SheetNameCheck(UnitName, "-")
xl.Cells(4, 1) = "nPlots: " & NumberOfPlots
xl.Range(xl.Cells(1, 1), xl.Cells(6, 1)).Font.Bold = True
RepRow = 8

'add number of unique spp
    Set MyRS = MyDB.OpenRecordset("qrySppCount")
    xl.Cells(5, 1) = "Number of unique species: " & MyRS.RecordCount

'add total species occurences
    Set MyRS = MyDB.OpenRecordset("qryNSpecies")
    MyRS.MoveLast
    xl.Cells(6, 1) = "Total species occurences : " & MyRS.RecordCount

'add SRank data
    If clsRepOpt.SppAttributeSRank Then
        Set MyRS = MyDB.OpenRecordset("qrySRank2")
        xl.Cells(RepRow + 1, 5).CopyFromRecordset MyRS
        xl.Cells(RepRow + 1, 5) = "nSRankDetail"
        AddHeader RepRow, 5, MyRS
        RepRow = RepRow + 4
    End If
    
'add wetland ind. data
    If clsRepOpt.SppAttributeWetland Then
        Set MyRS = MyDB.OpenRecordset("qryWetland2")
        xl.Cells(RepRow + 1, 5).CopyFromRecordset MyRS
        xl.Cells(RepRow + 1, 5) = "nWetland_Ind"
        AddHeader RepRow, 5, MyRS
        RepRow = RepRow + 4
    End If
    
'add weed status data
    If clsRepOpt.SppAttributeWeedStatus Then
        Set MyRS = MyDB.OpenRecordset("qryWeedStatus2")
        xl.Cells(RepRow + 1, 5).CopyFromRecordset MyRS
        xl.Cells(RepRow + 1, 5) = "nWeedStatusDetail"
        AddHeader RepRow, 5, MyRS
        RepRow = RepRow + 4
    End If
    
'add red blue list data
    If clsRepOpt.SppAttributeRedBlue Then
        Set MyRS = MyDB.OpenRecordset("qryRedBlueList2")
        xl.Cells(RepRow + 1, 5).CopyFromRecordset MyRS
        xl.Cells(RepRow + 1, 5) = "nRedBlueListDetail"
        AddHeader RepRow, 5, MyRS
        RepRow = RepRow + 4
    End If
    
'add ASMR list data
    If clsRepOpt.SppAttributeASMR Then
        Set MyRS = MyDB.OpenRecordset("qryASMR2")
        xl.Cells(RepRow + 1, 5).CopyFromRecordset MyRS
        xl.Cells(RepRow + 1, 5) = "nASMRDetail"
        AddHeader RepRow, 5, MyRS
        RepRow = RepRow + 4
    End If
    
'add Climate list data
    If clsRepOpt.SppAttributeClimate Then
        Set MyRS = MyDB.OpenRecordset("qryClimate2")
        xl.Cells(RepRow + 1, 5).CopyFromRecordset MyRS
        xl.Cells(RepRow + 1, 5) = "nClimateDetail"
        AddHeader RepRow, 5, MyRS
        RepRow = RepRow + 4
    End If
    
'format column widths
    xl.Columns("E:Y").EntireColumn.AutoFit
    xl.Columns("A:A").ColumnWidth = 15.14
    xl.Columns("B:B").ColumnWidth = 9.43
    xl.Columns("C:C").ColumnWidth = 13.71
    xl.Columns("D:D").ColumnWidth = 15

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Sub
End Sub
Private Sub AddHeader(Row As Integer, Col As Integer, MyRS As Recordset)

Dim i As Integer

'add names
    For i = 0 To MyRS.Fields.count - 1
        xl.Cells(Row, Col + i) = MyRS.Fields(i).Name
        xl.Cells(Row, Col + i).Font.Bold = True
    Next i
        xl.Cells(Row, Col) = "Type"
        
'shade cells
    With xl.Range(xl.Cells(Row, Col), xl.Cells(Row, Col + i - 1)).Interior
        .Pattern = xlSolid
        .PatternColorIndex = xlAutomatic
        .ThemeColor = xlThemeColorDark1
        .TintAndShade = -0.249977111117893
        .PatternTintAndShade = 0
    End With

'add borders
    On Error Resume Next
    xl.Range(xl.Cells(Row + 1, Col), xl.Cells(Row + 1, Col + i - 1)).SpecialCells(xlCellTypeBlanks).FormulaR1C1 = "0"
    On Error GoTo 0
    xl.Range(xl.Cells(Row, Col), xl.Cells(Row + 1, Col + i - 1)).BorderAround xlContinuous, xlThin, xlColorIndexAutomatic
    xl.Range(xl.Cells(Row, Col), xl.Cells(Row + 1, Col + i - 1)).Borders(xlInsideVertical).LineStyle = xlContinuous
    xl.Range(xl.Cells(Row, Col), xl.Cells(Row + 1, Col + i - 1)).Borders(xlInsideHorizontal).LineStyle = xlContinuous
    xl.Range(xl.Cells(Row, Col), xl.Cells(Row + 1, Col + i - 1)).HorizontalAlignment = xlCenter

'xl.Speech.Speak xl.Cells(3, 1)

End Sub

Private Function CreateSmallVeg(VegTable, SuTable)
' Apr 19, 2011
' creates a small vegetation table based on the site unit table
Dim SqlText As String
    DoCmd.SetWarnings False
    On Error Resume Next
    DoCmd.DeleteObject A_TABLE, "TempReportVegLF"
    On Error GoTo errCreateTempVeg
    
    'optimizes veg data
    DoCmd.CopyObject , "TempReportVegLF", acTable, "USysVegTable"
    SqlText = "INSERT INTO TempReportVegLF ( PlotNumber, Species, Cover1, Cover2, Cover3, Cover4, Cover5, Cover6, Cover7, Cover8, Cover9, Cover10, TotalA, TotalB )"
    SqlText = SqlText & " SELECT DISTINCTROW [" & VegTable & "].PlotNumber, Species, Max(Cover1) AS MaxOfCover1, Max(Cover2) AS MaxOfCover2, Max(Cover3) AS MaxOfCover3, Max(Cover4) AS MaxOfCover4, Max(Cover5) AS MaxOfCover5, Max(Cover6) AS MaxOfCover6, Max(Cover7) AS MaxOfCover7, Max(Cover8) AS MaxOfCover8, Max(Cover9) AS MaxOfCover9, Max(Cover10) AS MaxOfCover10, Max(TotalA) AS MaxOfTotalA, Max(TotalB) AS MaxOfTotalB"
    SqlText = SqlText & " FROM [" & VegTable & "] INNER JOIN [" & SuTable & "] ON [" & VegTable & "].PlotNumber = [" & SuTable & "].PlotNumber"
    SqlText = SqlText & " GROUP BY " & VegTable & ".PlotNumber, " & VegTable & ".Species;"
    DoCmd.RunSQL SqlText
MyExit:
    DoCmd.SetWarnings True
    Exit Function
errCreateTempVeg:
    DoCmd.SetWarnings True
    MsgBox Err.Description
    Resume MyExit
End Function

Private Sub BuildSRankQueries(Unit As String)

Dim SqlText As String
Dim MyVeg As String, MySU As String
MyVeg = clsVProReg.CurrProject & "_Veg"
'If UseBreaks Then MySU = "USysSuTableDynamic_SU" Else MySU = clsVProReg.CurrPlotlist & "_SU"
MySU = SuTable

SqlText = "SELECT " & MySU & ".SiteUnit, USysSppAttributes.SRank, Count(" & MySU & ".PlotNumber) AS CountOfPlotNumber"
SqlText = SqlText & " FROM (" & MySU & " INNER JOIN " & MyVeg & " ON " & MySU & ".PlotNumber = " & MyVeg & ".PlotNumber) INNER JOIN USysSppAttributes ON " & MyVeg & ".Species = USysSppAttributes.Code"
'SqlText = SqlText & " WHERE (((USysSppAttributes.Codetype)='u'))"

SqlText = "TRANSFORM Sum(qrySRank1.CountOfPlotNumber) AS SumOfCountOfPlotNumber"
SqlText = SqlText & " SELECT qrySRank1.SiteUnit"
SqlText = SqlText & " From qrySRank1"

SqlText = SqlText & " PIVOT qrySRank1.SRank In ('S1','S2','S2S3','S3','S3S4','S4','S4S5','S5','SE1','SE1SE2','SE2','SE3','SE3SE4','SE4','SE5','SEH','SEX','SH','SU','SX');"

SqlText = SqlText & " FROM (" & MySU & " INNER JOIN " & MyVeg & " ON " & MySU & ".PlotNumber = " & MyVeg & ".PlotNumber) INNER JOIN USysSppAttributes ON " & MyVeg & ".Species = USysSppAttributes.Code"
'SqlText = SqlText & " WHERE (((" & MySU & ".SiteUnit)='" & Unit & "') AND ((USysSppAttributes.SRank) Is Not Null) AND ((USysSppAttributes.Codetype)='u'))"
SqlText = SqlText & " WHERE (((" & MySU & ".SiteUnit)='" & Unit & "') AND ((USysSppAttributes.SRank) Is Not Null) AND ((USysSppAttributes.Codetype)<>'S'))"
SqlText = SqlText & " GROUP BY " & MyVeg & ".PlotNumber;"

SqlText = "SELECT Count(qrySRank3.PlotNumber) AS nPlots"
SqlText = SqlText & " FROM qrySRank3;"
CurrentDb.CreateQueryDef "qrySRank4", SqlText

SqlText = "SELECT Sum(qrySRank1.CountOfPlotNumber) AS nPlots"
SqlText = SqlText & " FROM qrySRank1;"

Private Sub BuildLifeformQuries(Unit As String)

SqlText = "SELECT EntryDatVegLF.Plotnumber, EntryDatVegLF.Layer, Count(EntryDatVegLF.Cover) AS CountOfCover, Sum(EntryDatVegLF.Cover) AS SumOfCover"
SqlText = SqlText & " FROM EntryDatVegLF INNER JOIN " & MySU & " ON EntryDatVegLF.Plotnumber = " & MySU & ".PlotNumber"

SqlText = SqlText & " GROUP BY LifeformCodes.Lifeform"
SqlText = SqlText & " HAVING (((LifeformCodes.Lifeform)<>13));"

SqlText = "SELECT EntryDatVegLF.Species"
SqlText = SqlText & " FROM EntryDatVegLF INNER JOIN " & MySU & " ON EntryDatVegLF.Plotnumber = " & MySU & ".PlotNumber"
SqlText = SqlText & " WHERE (((" & MySU & ".SiteUnit)='" & Unit & "'))"

Private Sub BuildASMRQueries(Unit As String)

Dim SqlText As String
Dim MyVeg As String, MySU As String
MyVeg = clsVProReg.CurrProject & "_Veg"
'If UseBreaks Then MySU = "USysSuTableDynamic_SU" Else MySU = clsVProReg.CurrPlotlist & "_SU"
MySU = SuTable

SqlText = "SELECT " & MySU & ".SiteUnit, Est_ASMR, Count(" & MySU & ".PlotNumber) AS CountOfPlotNumber"
SqlText = SqlText & " FROM (" & MySU & " INNER JOIN " & MyVeg & " ON " & MySU & ".PlotNumber = " & MyVeg & ".PlotNumber) INNER JOIN USysSppAttributes ON " & MyVeg & ".Species = USysSppAttributes.Code"
'SqlText = SqlText & " WHERE (((USysSppAttributes.Codetype)='u'))"
SqlText = SqlText & " WHERE (((USysSppAttributes.Codetype)<>'S'))"
SqlText = SqlText & " GROUP BY " & MySU & ".SiteUnit, USysSppAttributes.Est_ASMR"
SqlText = SqlText & " HAVING (((" & MySU & ".SiteUnit)='" & Unit & "') AND ((Est_ASMR) Is Not Null));"
CurrentDb.CreateQueryDef "qryASMR1", SqlText

SqlText = "TRANSFORM Sum(CountOfPlotNumber) AS SumOfCountOfPlotNumber"
SqlText = SqlText & " SELECT qryASMR1.SiteUnit"
SqlText = SqlText & " From qryASMR1"
SqlText = SqlText & " GROUP BY SiteUnit"
SqlText = SqlText & " PIVOT qryASMR1.Est_ASMR In (0,1,2,3,4,5,6);"
CurrentDb.CreateQueryDef "qryASMR2", SqlText

SqlText = "SELECT " & MyVeg & ".PlotNumber, Count(" & MySU & ".PlotNumber) AS CountOfPlotNumber"
SqlText = SqlText & " FROM (" & MySU & " INNER JOIN " & MyVeg & " ON " & MySU & ".PlotNumber = " & MyVeg & ".PlotNumber) INNER JOIN USysSppAttributes ON " & MyVeg & ".Species = USysSppAttributes.Code"
'SqlText = SqlText & " WHERE (((" & MySU & ".SiteUnit)='" & Unit & "') AND ((USysSppAttributes.Est_ASMR) Is Not Null) AND ((USysSppAttributes.Codetype)='u'))"
SqlText = SqlText & " WHERE (((" & MySU & ".SiteUnit)='" & Unit & "') AND ((USysSppAttributes.Est_ASMR) Is Not Null) AND ((USysSppAttributes.Codetype)<>'S'))"
SqlText = SqlText & " GROUP BY " & MyVeg & ".PlotNumber;"
CurrentDb.CreateQueryDef "qryASMR3", SqlText

SqlText = "SELECT Count(qryASMR3.PlotNumber) AS nPlots"
SqlText = SqlText & " FROM qryASMR3;"
CurrentDb.CreateQueryDef "qryASMR4", SqlText

SqlText = "SELECT Sum(CountOfPlotNumber) AS nPlots"
SqlText = SqlText & " FROM qryASMR1;"
CurrentDb.CreateQueryDef "qryASMR5", SqlText

End Sub
