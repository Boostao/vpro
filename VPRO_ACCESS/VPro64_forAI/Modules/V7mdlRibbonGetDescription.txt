Option Compare Database
Option Explicit
Private Function GetDescription(ControlName As String) As String

Dim MyRS As Recordset
Set MyRS = CurrentDb.OpenRecordset("USysRibbonDescriptions", dbOpenDynaset)

    MyRS.FindFirst "ControlName='" & ControlName & "'"
    If Not MyRS.NoMatch Then
        GetDescription = MyRS!Description
    Else
        GetDescription = "Unknown description"
    End If
    
    MyRS.Close
    Set MyRS = Nothing

End Function
