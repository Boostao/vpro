Option Compare Database

Public Sub AttributeLumping(optNullsAsZero As Boolean, optReportAllSpecies As Boolean)

Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String, AttName As String, AttShortName As String
On Error GoTo errAttributeLumping
Set MyDB = CurrentDb

DoCmd.SetWarnings False

On Error Resume Next
    DoCmd.DeleteObject acTable, "Attribute_Lump"
On Error GoTo errAttributeLumping
    
DoCmd.CopyObject , "Attribute_Lump", acTable, "USysLumpTable"

SqlText = "SELECT Attribute, ShortName From USysSppAttributeSummary WHERE (((Report)=Yes));"
Set MyRS = MyDB.OpenRecordset(SqlText)

Do Until MyRS.EOF
    AttName = MyRS!Attribute
    AttShortName = MyRS!ShortName
    If AttName = "Lifeform" Then
        AddLifeform
    Else
        SqlText = "INSERT INTO Attribute_Lump ( LumpCode, SppCode, Use )"
        'SqlText = SqlText & " SELECT '" & AttName & "-' + [" & AttName & "] AS Expr1, USysSppAttributes.Code, Yes AS Expr2"
        SqlText = SqlText & " SELECT '" & AttShortName & "' + [" & AttName & "] AS Expr1, USysSppAttributes.Code, Yes AS Expr2"
        SqlText = SqlText & " From USysSppAttributes"
        SqlText = SqlText & " WHERE ((([" & AttName & "]) Is Not Null));"
        DoCmd.RunSQL SqlText
        
        If optNullsAsZero Then
            SqlText = "INSERT INTO Attribute_Lump ( LumpCode, SppCode, Use )"
            'SqlText = SqlText & " SELECT '" & AttName & "-0' AS Expr1, Code, Yes AS Expr2"
            SqlText = SqlText & " SELECT '" & AttShortName & "Z' AS Expr1, Code, Yes AS Expr2"
            SqlText = SqlText & " From USysSppAttributes"
            SqlText = SqlText & " WHERE ((([" & AttName & "]) Is Null Or ([" & AttName & "])=''));"
            DoCmd.RunSQL SqlText
        End If
    End If
    MyRS.MoveNext
Loop

If optReportAllSpecies Then
    SqlText = "INSERT INTO Attribute_Lump ( LumpCode, SppCode, Use )"
    SqlText = SqlText & " SELECT USysSppAttributes.Code, USysSppAttributes.Code, Yes AS Expr2"
    SqlText = SqlText & " FROM USysSppAttributes;"
    DoCmd.RunSQL SqlText
End If

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    MyRS.Close
    Set MyDB = Nothing
    On Error GoTo 0
    Exit Sub
errAttributeLumping:
    MsgBox Err.Description, vbInformation, "VPro"
    Resume MyExit

End Sub

Private Sub AddLifeform()

Dim SqlText As String
    
    SqlText = "INSERT INTO Attribute_Lump ( LumpCode, Use, SppCode )"
    SqlText = SqlText & " SELECT LifeformCodes.ShortName, Yes AS Use, USysAllSpecs.Code"
    SqlText = SqlText & " FROM USysAllSpecs INNER JOIN LifeformCodes ON USysAllSpecs.Lifeform = LifeformCodes.Lifeform"
    SqlText = SqlText & " WHERE (((USysAllSpecs.Codetype)<>'S'))"
    SqlText = SqlText & " GROUP BY LifeformCodes.ShortName, USysAllSpecs.Code, Yes;"
    DoCmd.RunSQL SqlText

End Sub
