Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = False
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Compare Database
Option Explicit
Private Const strApp = "VPro64"

Public Property Get DontShowDataQualityMessage() As Boolean
    DontShowDataQualityMessage = GetSetting(strApp, "Message", "DontShowDataQualityMessage", False)
End Property

Public Property Let DontShowDataQualityMessage(DontShowDataQualityMessage As Boolean)
    SaveSetting strApp, "Message", "DontShowDataQualityMessage", DontShowDataQualityMessage
End Property

Public Property Get ShowWhatsNew() As Boolean
    ShowWhatsNew = GetSetting(strApp, "Message", "ShowWhatsNew", True)
End Property

Public Property Let ShowWhatsNew(ShowWhatsNew As Boolean)
    SaveSetting strApp, "Message", "ShowWhatsNew", ShowWhatsNew
End Property
