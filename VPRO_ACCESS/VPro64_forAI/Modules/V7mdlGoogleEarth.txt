Public Sub PlotPlotsInGE(OpenGE As Boolean)

'2012
On Error GoTo errPlotPlotsInGE
Dim ProgramLocation As String
GetPlotLocations
BuildKmlFile
WriteKML
ProgramLocation = clsVProReg.GEProgram
If OpenGE Then
    'Shell (CStr(["C:\Program Files\Google\Google Earth\client\googleearth.exe"]) & " " & GEFileName)
    shell (CStr([ProgramLocation]) & " " & GEFileName)
End If

MyExit:
    On Error Resume Next
    Exit Sub
errPlotPlotsInGE:
    If Err.Number = 462 Then
        MsgBox "This tool requires Google Earth to be installed first.", vbInformation, "VPro"
    Else
        MsgBox Err.Number & Chr(13) & Err.Description
    End If
    Resume MyExit
End Sub

Private Sub BuildKmlFile()

'Dec 19, 2008
KmlBody = "<?xml version=""1.0"" encoding=""UTF-8""?><kml xmlns=""http://earth.google.com/kml/2.1""><Document>   <name>" & clsRepOpt.GoogleEarthPlaceName & "</name>"

KmlBody = KmlBody & Chr(10) & "<Style id=""a""><IconStyle><Icon><href>http://maps.google.com/mapfiles/ms/icons/red-dot.png</href></Icon></IconStyle></Style>"

KmlBody = KmlBody & Chr(10) & PlotLocs

KmlBody = KmlBody & Chr(10) & "</Document></kml>"

End Sub

Private Sub GetPlotLocations()

'Dec 19, 2008
Dim SqlText As String, i As Long
Dim MyRS As Recordset, RunBefore As Boolean
PlotLocs = ""
RunBefore = False
i = 0

If clsVProReg.CurrPlotlist = "None" Then
    SqlText = "SELECT PlotNumber, Longitude, Latitude, " & clsRepOpt.GoogleEarthDescField
    SqlText = SqlText & " From [" & clsVProReg.CurrProject & "_Env]"
    SqlText = SqlText & " WHERE (((Longitude) Is Not Null) AND ((Latitude) Is Not Null));"
Else
    SqlText = "SELECT " & clsVProReg.CurrProject & "_Env.PlotNumber, Longitude, Latitude, " & clsRepOpt.GoogleEarthDescField
    SqlText = SqlText & " FROM [" & clsVProReg.CurrProject & "_Env] INNER JOIN [" & clsVProReg.CurrPlotlist & "_SU] ON [" & clsVProReg.CurrProject & "_Env].PlotNumber = [" & clsVProReg.CurrPlotlist & "_SU].PlotNumber"
    SqlText = SqlText & " WHERE (((Longitude) Is Not Null) AND ((Latitude) Is Not Null));"
End If
'CurrentDb.CreateQueryDef "fred", SqlText
Set MyRS = CurrentDb.OpenRecordset(SqlText)

Do Until MyRS.EOF
    If Not RunBefore Then
        PlotLocs = PlotLocs & "   <Placemark>       <name>" & MyRS!PlotNumber & "</name><styleUrl>#a</styleUrl>      <Point>         <coordinates>-" & MyRS!Longitude & ", " & MyRS!Latitude & ",0</coordinates>      </Point>      <description><![CDATA[" & MyRS(clsRepOpt.GoogleEarthDescField) & ".]]></description>   </Placemark>"
    Else
        'PlotLocs = PlotLocs & Chr(10) & "   <Placemark>       <name>" & MyRS!PlotNumber & "</name><styleUrl>#a</styleUrl>      <Point>         <coordinates>-" & MyRS!Longitude & ", " & MyRS!Latitude & ",0</coordinates>      </Point>      <description><![CDATA[" & MyRS(clsRepOpt.GoogleEarthDescField) & ".]]></description>   </Placemark>"
        PlotLocs = PlotLocs & Chr(10) & "   <Placemark>       <name>" & MyRS!PlotNumber & "</name><styleUrl>#a</styleUrl>      <Point>         <coordinates>-" & MyRS!Longitude & ", " & MyRS!Latitude & ",0</coordinates>      </Point>      <description><![CDATA[" & MyRS(3) & ".]]></description>   </Placemark>"
    End If
    RunBefore = True
    MyRS.MoveNext
    'i = i + 1
    'Debug.Print i
Loop
'Debug.Print "done"

MyRS.Close

End Sub
