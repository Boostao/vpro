Function ExposureSummary(Project As String, SuTable As String, SiteUnit As String)
'Oct. 12, 2002
'summarizes exposure by zone
On Error GoTo errBioGeoClimaticSummary
Dim MyDB As Database, MyLoopRS As Recordset, MyRS As Recordset, SqlText As String, Result As String
Dim i As Integer, BGCZone As String
Set MyDB = CurrentDb()
Result = ""

'create loop values
SqlText = "SELECT DISTINCT Exposure1"
SqlText = SqlText & " FROM [" & Project & "] INNER JOIN [" & SuTable & "] ON [" & Project & "].PlotNumber = [" & SuTable & "].PlotNumber"
SqlText = SqlText & " WHERE (((Exposure1) Is Not Null) And ((SiteUnit)='" & SiteUnit & "'))"
SqlText = SqlText & " ORDER BY Exposure1;"
Set MyLoopRS = MyDB.OpenRecordset(SqlText)
If MyLoopRS.RecordCount < 1 Then GoTo Jump
MyLoopRS.MoveLast
MyLoopRS.MoveFirst

'Exposure
For i = 1 To MyLoopRS.RecordCount
    BGCZone = MyLoopRS!Exposure1
    SqlText = "SELECT Exposure1"
    SqlText = SqlText & " FROM [" & Project & "] INNER JOIN [" & SuTable & "] ON [" & Project & "].PlotNumber = [" & SuTable & "].PlotNumber"
    SqlText = SqlText & " WHERE (((Exposure1)='" & BGCZone & "') AND ((SiteUnit)='" & SiteUnit & "'));"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount > 0 Then
        MyRS.MoveLast
        Result = Result & BGCZone & "(" & MyRS.RecordCount & ")  "
    End If
    MyLoopRS.MoveNext
Next i

Jump:
'get n null
    SqlText = "SELECT Count(SiteUnit) AS nNull"
    SqlText = SqlText & " FROM [" & Project & "] INNER JOIN [" & SuTable & "] ON [" & Project & "].PlotNumber = [" & SuTable & "].PlotNumber"
    SqlText = SqlText & " WHERE (((Exposure1) Is Null))"
    SqlText = SqlText & " GROUP BY SiteUnit"
    SqlText = SqlText & " HAVING (((SiteUnit)='" & SiteUnit & "'));"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount > 0 Then
        Result = Result & "(Null " & MyRS!nNull & ")"
    End If
    
    ExposureSummary = Result

MyExit:
    On Error Resume Next
    MyRS.Close
    MyLoopRS.Close
    Set MyDB = Nothing
    Exit Function
errBioGeoClimaticSummary:
    MsgBox Err.Description
    Resume MyExit
End Function

Function AspectSummary(Project As String, SuTable As String, SiteUnit As String)

Dim i As Integer, BGCZone As String

Function LifeformCoverSummary(Project As String, SuTable As String, SiteUnit As String, LifeForm As Integer, PlotsInUnit As Integer)
'Oct. 12, 2002
'summarizes cover by lifeform
'ignors records that can't be converted to values
On Error GoTo errLifeformCoverSummary
Dim MyDB As Database, MyRS As Recordset, SqlText As String, Result As String
Dim nMinRecords As Integer, nMinRecordsWithVeg As Integer
Set MyDB = CurrentDb()
Result = ""

'get min value
    'first get the number of plots in the unit
    SqlText = "SELECT Count(PlotNumber) AS CountOfPlotNumber"
    SqlText = SqlText & " FROM " & SuTable
    SqlText = SqlText & " GROUP BY SiteUnit"
    SqlText = SqlText & " HAVING (((SiteUnit)='" & SiteUnit & "'));"
    Set MyRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)
    If MyRS.RecordCount < 1 Then GoTo Jump
    nMinRecords = MyRS!CountOfPlotNumber

    'get the number of plots with veg in unit and lifeform
    SqlText = "SELECT DISTINCT SuDetailStep1.PlotNumber"
    SqlText = SqlText & " FROM (SuDetailStep1 INNER JOIN USysAllSpecies ON SuDetailStep1.Species = USysAllSpecies.Code) INNER JOIN [" & SuTable & "] ON SuDetailStep1.PlotNumber = [" & SuTable & "].PlotNumber"
    SqlText = SqlText & " WHERE (((USysAllSpecies.LifeForm)=" & LifeForm & ") AND (([" & SuTable & "].SiteUnit)='" & SiteUnit & "'));"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount < 1 Then GoTo Jump
    nMinRecordsWithVeg = MyRS.RecordCount

    If nMinRecordsWithVeg >= nMinRecords Then
        SqlText = "SELECT Format(Min(MyCover),'0.0') AS MinCover"
        SqlText = SqlText & " FROM (SuDetailStep1 INNER JOIN USysAllSpecies ON SuDetailStep1.Species = USysAllSpecies.Code) INNER JOIN [" & SuTable & "] ON SuDetailStep1.PlotNumber = [" & SuTable & "].PlotNumber"
        'SqlText = SqlText & " WHERE (((LifeForm)=" & LifeForm & ") AND ((CodeType)='U') AND ((SiteUnit)='" & SiteUnit & "'));"
        SqlText = SqlText & " WHERE (((LifeForm)=" & LifeForm & ") AND ((CodeType)<>'S') AND ((SiteUnit)='" & SiteUnit & "'));"
        Set MyRS = MyDB.OpenRecordset(SqlText)
        If MyRS.RecordCount < 1 Then GoTo Jump
        Result = MyRS!MinCover & "---"
    Else
        Result = "0---"
    End If
            
'get average value
    SqlText = "SELECT Format(sum(MyCover)/" & PlotsInUnit & ",'0.0') AS AvgCover"
    SqlText = SqlText & " FROM (SuDetailStep1 INNER JOIN USysAllSpecies ON SuDetailStep1.Species = USysAllSpecies.Code) INNER JOIN [" & SuTable & "] ON SuDetailStep1.PlotNumber = [" & SuTable & "].PlotNumber"
    'SqlText = SqlText & " WHERE (((LifeForm)=" & LifeForm & ") AND ((CodeType)='U') AND ((SiteUnit)='" & SiteUnit & "'));"
    SqlText = SqlText & " WHERE (((LifeForm)=" & LifeForm & ") AND ((CodeType)<>'S') AND ((SiteUnit)='" & SiteUnit & "'));"
Set MyRS = MyDB.OpenRecordset(SqlText)
If MyRS.RecordCount < 1 Then GoTo Jump

Function SiteUnitReport()
'Dec. 30, 2019
'creates a basic site unit report

On Error GoTo errSiteUnitReport
Dim Project As String, SuTable As String, EnvTable As String, Msg As String
Dim SqlText As String, Title As String
Project = CurrProject()
SuTable = CurrPlotlist() & "_SU"
'EnvTable = Project & "_Env"
EnvTable = "USysEnv"
Msg = "You must have an active Site Unit table to create this report"
Title = ProgramName()
If SuTable = "None_Su" Then MsgBox Msg, 64, Title
If SuTable = "None_Su" Then Exit Function

DoCmd.Echo False
DoCmd.SetWarnings False

    DoCmd.OpenReport "USysSiteUnitReportSub", A_DESIGN
    'SqlText = "SELECT DISTINCTROW " & SUtable & ".SiteUnit, Count(" & SUtable & ".PlotNumber) AS [#Plots], MasterSiteUnitList.UnitLongName"
    SqlText = "SELECT DISTINCTROW " & SuTable & ".SiteUnit, Count(" & SuTable & ".PlotNumber) AS [#Plots], MasterSiteUnitList.SiteSeriesLongName"
    'SqlText = SqlText & " FROM " & SUtable & " LEFT JOIN MasterSiteUnitList ON " & SUtable & ".SiteUnit = MasterSiteUnitList.Name"
    SqlText = SqlText & " FROM " & SuTable & " LEFT JOIN MasterSiteUnitList ON " & SuTable & ".SiteUnit = MasterSiteUnitList.SiteSeries"
    'SqlText = SqlText & " GROUP BY " & SUtable & ".SiteUnit, MasterSiteUnitList.UnitLongName;"
    SqlText = SqlText & " GROUP BY " & SuTable & ".SiteUnit, MasterSiteUnitList.SiteSeriesLongName;"
    Reports![USysSiteUnitReportSub].RecordSource = SqlText
    DoCmd.Close A_REPORT, "USysSiteUnitReportSub"

    DoCmd.OpenReport "USysSiteUnitReport", A_DESIGN
    SqlText = "SELECT DISTINCTROW " & SuTable & ".SiteUnit AS MySiteUnit, " & EnvTable & ".PlotNumber FROM " & SuTable & " RIGHT JOIN " & EnvTable & " ON " & SuTable & ".PlotNumber = " & EnvTable & ".PlotNumber WHERE ((Not " & SuTable & ".SiteUnit Is Null)) ORDER BY " & SuTable & ".SiteUnit;"
    Reports![USysSiteUnitReport].RecordSource = SqlText
    DoCmd.OpenReport "USysSiteUnitReport", A_PREVIEW

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    DoCmd.Echo True
    Exit Function
errSiteUnitReport:
    MsgBox Err.Description
    Resume MyExit
End Function
