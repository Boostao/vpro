Public Function CombineTerrain(Field1, Field2, Field3)

'Apr. 23, 2018
On Error GoTo errCombineTerrain
Dim NullSwitch As Boolean
If IsNull(Field1) Then
    CombineTerrain = Null
    NullSwitch = True
Else
    CombineTerrain = Field1
End If

If IsNull(Field2) Then
    CombineTerrain = CombineTerrain
    NullSwitch = True
Else
    CombineTerrain = CombineTerrain & Field2
End If

If IsNull(Field3) Then
    CombineTerrain = CombineTerrain
    NullSwitch = True
Else
    CombineTerrain = CombineTerrain & Field3
End If

If CombineTerrain = "" Then CombineTerrain = Null

MyExit:
    On Error Resume Next
    Exit Function
errCombineTerrain:
    CombineTerrain = Null
    GoTo MyExit
End Function
Public Function SplitTerrain(ValueIn As Variant, FieldIn As String, FieldNumOut As Integer) As String

'Apr. 23, 2018
On Error GoTo errSplitTerrain
Dim i As Integer
If IsNull(ValueIn) Then
    SplitTerrain = ""
    GoTo MyExit
End If
Select Case FieldIn
    Case "TerrainTextureSurf"
        SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
    Case "TerrainTextureSubSurf"
        SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
    
    Case "SurficialMaterialSurf"
        i = InStr(1, ValueIn, "FG")
        If i > 0 Then
            If FieldNumOut = i Then
                SplitTerrain = Mid(ValueIn, i, 2)
            Else
                If FieldNumOut > i Then
                    SplitTerrain = Mid(ValueIn, FieldNumOut + 1, 1)
                Else
                    SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
                End If
            End If
        Else
            SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
        End If
    Case "SurficialMaterialSubSurf"
        i = InStr(1, ValueIn, "FG")
        If i > 0 Then
            If FieldNumOut = i Then
                SplitTerrain = Mid(ValueIn, i, 2)
            Else
                If FieldNumOut > i Then
                    SplitTerrain = Mid(ValueIn, FieldNumOut + 1, 1)
                Else
                    SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
                End If
            End If
        Else
            SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
        End If
    Case "SurfaceExpSurf"
        SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
    Case "SurfaceExpSubSurf"
        SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
    Case "GeoMorProSurf"
        SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
    Case "GeoMorProSubSurf"
        SplitTerrain = Mid(ValueIn, FieldNumOut, 1)
    Case Else
        SplitTerrain = Null
End Select

MyExit:
    On Error Resume Next
    Exit Function
errSplitTerrain:
    SplitTerrain = ""
    GoTo MyExit
End Function

Public Function TestTerrainFieldSize(Optional Project) As Boolean 'returns false if old terrain (under 3)
'Nov 03, 2000
On Error GoTo errTestTerrainFieldSize
Dim MyDB As Database, MyTD As TableDef, MyFld As Field
Set MyDB = CurrentDb
If IsMissing(Project) Then Project = CurrProject
Project = Project & "_Env"
If Project = "Sample_Env" Then GoTo MyExit
Set MyTD = MyDB.TableDefs(Project)
Set MyFld = MyTD.Fields("SurfaceExpSurf")
If MyFld.Size = 3 Then
    TestTerrainFieldSize = True
End If

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function
errTestTerrainFieldSize:
    MsgBox Err.Description
    Resume MyExit
End Function
