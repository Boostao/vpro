Private Function SplitHorizon(Value2Parse As String, ReturnWhat As String) As Single
'Nov. 2, 2000
On Error GoTo errSplitHorizon
On Error GoTo 0
Dim X As Integer
Dim Val1 As Single, Val2 As Single

Select Case ReturnWhat
    Case "Low"
        X = InStr(1, Value2Parse, "-")
        If X = 0 Then
            SplitHorizon = 0
        Else
            Val1 = Left(Value2Parse, X - 1)
            Val2 = Right(Value2Parse, Len(Value2Parse) - X)
            If Val1 > Val2 Then
                SplitHorizon = Val2
            Else
                SplitHorizon = Val1
            End If
        End If
    Case "Up"
        X = InStr(1, Value2Parse, "-")
        If X = 0 Then
            SplitHorizon = Value2Parse
        Else
            Val1 = Left(Value2Parse, X - 1)
            Val2 = Right(Value2Parse, Len(Value2Parse) - X)
            If Val1 > Val2 Then
                SplitHorizon = Val1
            Else
                SplitHorizon = Val2
            End If
        End If
    Case Else
    
End Select

MyExit:
    On Error Resume Next
    Exit Function
errSplitHorizon:
    SplitHorizon = 0
    Resume MyExit
End Function

Private Function AddV42Project(db As Database) As Integer
'Nov. 4, 2000
On Error GoTo errAddV42Project
Dim MyRS As Recordset
Set MyRS = db.OpenRecordset("tblProjectManager", dbOpenDynaset)
    
    MyRS.AddNew
    MyRS!ProjectName = CurrProject
    MyRS.Update
    MyRS.MoveFirst
    AddV42Project = MyRS!ProjectID
    
MyExit:
    On Error Resume Next
    MyRS.Close
    Exit Function
errAddV42Project:
    MsgBox "Sorry, couldn't add project"
    AddV42Project = 0
    Err.Clear
    Resume MyExit
End Function

    Msg = "VPro has found duplicate plot numbers in this VENUS database.  Do you wish to see a list of these plots?"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef)    ' Get user response.
    If Response = IDYES Then
        Set MyRP = CreateReport
        ReportName = MyRP.Name
        MyRP.RecordSource = SqlText
        MyRP.Section("Detail").Height = 240
        Set MyCTL = CreateReportControl(ReportName, acTextBox)
        MyCTL.Properties("ControlSource") = "PlotNumber"
        Set MyCTL = CreateReportControl(ReportName, acTextBox, acDetail, , , 1700)
        MyCTL.Properties("ControlSource") = "ProjectName"
        DoCmd.Save acReport, MyRP.Name
        DoCmd.Close acReport, MyRP.Name
        DoCmd.Rename "Duplicate Plots", acReport, ReportName
        DoCmd.OpenReport "Duplicate Plots", acViewPreview
    End If
    Check4Duplicates = True
Else
    Check4Duplicates = False
End If

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
errCheck4Duplicates:
    MsgBox Err.Description
    Resume MyExit
End Function
