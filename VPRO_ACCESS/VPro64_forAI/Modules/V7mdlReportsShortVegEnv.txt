Public Function SVLifeformCover(SiteUnit As String, LifeForm As String)

'2014
'summarizes cover by lifeform
    On Error Resume Next
        DoCmd.DeleteObject acQuery, "LFsum"
    On Error GoTo errLifeformCoverSummary
    Dim MyDB As Database, MyRS As Recordset, SqlText As String
    Set MyDB = CurrentDb()

'build cover for each plot
    SqlText = "SELECT USysSuTableDynamic_SU.SiteUnit, QuickVeg.Plotnumber, Sum(QuickVeg.Cover) AS SumOfCover"
    SqlText = SqlText & " FROM (USysSuTableDynamic_SU INNER JOIN QuickVeg ON USysSuTableDynamic_SU.PlotNumber = QuickVeg.Plotnumber) INNER JOIN USysAllSpecs ON QuickVeg.Species = USysAllSpecs.Code"
    SqlText = SqlText & " WHERE (((USysAllSpecs.Lifeform)=" & LifeForm & ") AND ((USysAllSpecs.Codetype)<>'S'))"
    SqlText = SqlText & " GROUP BY USysSuTableDynamic_SU.SiteUnit, QuickVeg.Plotnumber"
    SqlText = SqlText & " HAVING (((USysSuTableDynamic_SU.SiteUnit)='" & SiteUnit & "'));"
    MyDB.CreateQueryDef "LFsum", SqlText

'get min, avg and max
    SqlText = "SELECT Format(Min([SumOfCover]),'0.0') & ' - ' & Format(Avg([SumOfCover]),'0.0') & ' - ' & Format(Max([SumOfCover]),'0.0') AS MyVal FROM LFsum;"
    Set MyRS = MyDB.OpenRecordset(SqlText)

'return value
    If MyRS.RecordCount = 0 Then
        SVLifeformCover = ""
    Else
        SVLifeformCover = MyRS!MyVal
    End If
    If SVLifeformCover = " -  - " Then SVLifeformCover = ""

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    DoCmd.DeleteObject acQuery, "LFsum"
    Exit Function
errLifeformCoverSummary:
    MsgBox Err.Description
    Resume MyExit
End Function

Dim MyDB As Database, MyLoopRS As Recordset, MyRS As Recordset, SqlText As String, Result As String
Dim i As Integer, BGCZone As String, Project As String, SuTable As String
Set MyDB = CurrentDb()

Project = clsVProReg.CurrProject & "_Env"
SuTable = "USysSuTableDynamic_SU"
Result = ""

'create loop values
SqlText = "SELECT DISTINCT Zone"
SqlText = SqlText & " FROM [" & Project & "] INNER JOIN [" & SuTable & "] ON [" & Project & "].PlotNumber = [" & SuTable & "].PlotNumber"
SqlText = SqlText & " WHERE (((Zone) Is Not Null) And ((SiteUnit)='" & SiteUnit & "'))"
SqlText = SqlText & " ORDER BY Zone;"

Set MyLoopRS = MyDB.OpenRecordset(SqlText)
If MyLoopRS.RecordCount < 1 Then GoTo Jump
MyLoopRS.MoveLast
MyLoopRS.MoveFirst

'BGC Zones
For i = 1 To MyLoopRS.RecordCount
    BGCZone = MyLoopRS!Zone
    SqlText = "SELECT Zone"
    SqlText = SqlText & " FROM [" & Project & "] INNER JOIN [" & SuTable & "] ON [" & Project & "].PlotNumber = [" & SuTable & "].PlotNumber"
    SqlText = SqlText & " WHERE (((Zone)='" & BGCZone & "') AND ((SiteUnit)='" & SiteUnit & "'));"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount > 0 Then
        MyRS.MoveLast
        Result = Result & BGCZone & "(" & MyRS.RecordCount & ")  "
    End If
    MyLoopRS.MoveNext
Next i

Jump:
'get n null
    SqlText = "SELECT Count(SiteUnit) AS nNull"
    SqlText = SqlText & " FROM [" & Project & "] INNER JOIN [" & SuTable & "] ON [" & Project & "].PlotNumber = [" & SuTable & "].PlotNumber"
    SqlText = SqlText & " WHERE (((Zone) Is Null))"
    SqlText = SqlText & " GROUP BY SiteUnit"
    SqlText = SqlText & " HAVING (((SiteUnit)='" & SiteUnit & "'));"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount > 0 Then
        Result = Result & "(Null " & MyRS!nNull & ")"
    End If
    
    SVBioGeoClimaticZone = Result
    
MyExit:
    On Error Resume Next
    MyRS.Close
    MyLoopRS.Close
    Set MyDB = Nothing
    Exit Function
errBioGeoClimaticSummary:
    MsgBox Err.Description
    Resume MyExit
End Function
