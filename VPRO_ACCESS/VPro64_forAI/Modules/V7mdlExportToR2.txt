Private Function NullToPeriod(DataIn As Variant) As Variant

    If IsNull(DataIn) Then
        NullToPeriod = "."
    Else
        NullToPeriod = DataIn
    End If
    
End Function

Private Sub CreateLumpQuery(LumpTable As String)

Dim SqlText As String

    If LumpTable = "none" Then
        SqlText = "SELECT Replace([UsysVeg]![PlotNumber],' ','') AS PlotNumber, Replace([UsysVeg]![Species],' ','') AS Spp, UsysVeg.Cover1 AS C1, UsysVeg.Cover2 AS C2, UsysVeg.Cover3 AS C3, UsysVeg.Cover4 AS C4, UsysVeg.Cover5 AS C5, UsysVeg.Cover5a AS C5a, UsysVeg.Cover5b AS C5b, UsysVeg.Cover5c AS C5c, UsysVeg.Cover6 AS C6, UsysVeg.Cover7 AS C7"
        SqlText = SqlText & " FROM USysEnv INNER JOIN UsysVeg ON USysEnv.PlotNumber = UsysVeg.PlotNumber;"
    Else
        SqlText = "SELECT UsysVeg.PlotNumber, Replace(IIf(IsNull([LumpCode]),[Species],[LumpCode]),' ','') AS Spp, Sum(UsysVeg.Cover1) AS C1, Sum(UsysVeg.Cover2) AS C2, Sum(UsysVeg.Cover3) AS C3, Sum(UsysVeg.Cover4) AS C4, Sum(UsysVeg.Cover5) AS C5, Sum(UsysVeg.Cover5a) AS C5a, Sum(UsysVeg.Cover5b) AS C5b, Sum(UsysVeg.Cover5c) AS C5c, Sum(UsysVeg.Cover6) AS C6, Sum(UsysVeg.Cover7) AS C7"
        SqlText = SqlText & " FROM (USysEnv INNER JOIN UsysVeg ON USysEnv.PlotNumber = UsysVeg.PlotNumber) LEFT JOIN " & LumpTable & "_Lump ON UsysVeg.Species = " & LumpTable & "_Lump.SppCode"
        SqlText = SqlText & " GROUP BY UsysVeg.PlotNumber, Replace(IIf(IsNull([LumpCode]),[Species],[LumpCode]),' ','');"
    End If

    CurrentDb.CreateQueryDef "RStep1", SqlText

End Sub

Private Function AddSlash(FileName As String) As String

Dim tmpStr As String, tmpChar, i As Integer
For i = 1 To Len(FileName)
    tmpChar = Mid(FileName, i, 1)
    If tmpChar = "\" Then
        tmpChar = "\\"
    End If
    tmpStr = tmpStr & tmpChar
Next i

AddSlash = tmpStr

End Function
