Function SaveAs(ObjectToSave As String)

'May 25, 2009
On Error GoTo errSaveAs
Dim FromDB As Database, ToDB As Database, MyDB As Database
Dim MyFile As String, MyTitle As String
Dim NewName As String, Msg As String

Set MyDB = CurrentDb

CleanUp MyDB 'removes all files that start with "USysSaveAsRK"

If clsVProReg.CurrPlotlist = "None" And ObjectToSave = "SU" Then
    MsgBox "You don't have a current site unit table"
    GoTo MyExit
End If

    NewName = NameToUse
    If NewName = "" Then GoTo MyExit
    
    MyTitle = ProgramName() & " - pick a database to store your list in"
    MyFile = GetFileName(MyTitle, False)
    If MyFile = "" Then GoTo MyExit

AfterCreation:
Set ToDB = DBEngine.Workspaces(0).OpenDatabase(MyFile)
        
If Check4Table(ToDB, NewName, ObjectToSave) Then
    MsgBox "There are already table(s) of that name.  Try a different database or a different name.", vbInformation, "VPro"
    GoTo MyExit
End If

Set FromDB = DBEngine.Workspaces(0).OpenDatabase(GetFromDB(ObjectToSave))

Select Case ObjectToSave
    Case "Project"
        ImportProjectTables FromDB
        MyDB.TableDefs.Refresh
        ExportProjectTables ToDB, MyDB, NewName
        CreateRelationship ToDB.Name, NewName, True
    Case "SU"
        ImportSUTable FromDB
        MyDB.TableDefs.Refresh
        ExportSUTable ToDB, MyDB, NewName
    Case "Hierarchy"
        ImportHierarchyTable FromDB
        MyDB.TableDefs.Refresh
        ExportHierarchyTable ToDB, MyDB, NewName
    Case "Lump"
    
    Case "Theme"
End Select
MsgBox NewName & " was saved in " & MyFile

MyExit:
    On Error Resume Next
    CleanUp MyDB 'removes all files that start with "USysSaveAsRK"
    Set MyDB = Nothing
    Set FromDB = Nothing
    Set ToDB = Nothing
    Exit Function
    
CantFindDatabase:
    Msg = "Couldn't find specified database" & Chr(13)
    Msg = Msg & "VPro will now create " & Trim(MyFile)
    MsgBox Msg, 64, "VPro"
    
    If Trim(MyFile) = MyDB.Name Then Exit Function
    DBEngine.Workspaces(0).CreateDatabase Trim(MyFile), DB_LANG_GENERAL
    GoTo AfterCreation
    
errSaveAs:
    If Err.Number = 3024 Then
        GoTo CantFindDatabase
    Else
        MsgBox Err.Number & Chr(13) & Err.Description
    End If
    Resume MyExit

End Function
Private Function GetFromDB(ObjectToSave As String) As String
'Oct. 12, 2002
Dim ConString As String

Select Case ObjectToSave
    Case "Project"
        ConString = CurrentDb.TableDefs(clsVProReg.CurrProject & "_Env").Connect
    Case "SU"
        ConString = CurrentDb.TableDefs(clsVProReg.CurrPlotlist & "_SU").Connect
    Case "Hierarchy"
        ConString = CurrentDb.TableDefs(clsVProReg.CurrHierarchy & "_Hierarchy").Connect
    Case "Lump"
    
    Case "Theme"
    
    Case Else
End Select

If ConString = "" Then
    ConString = CurrentDb.Name
Else
    ConString = Mid(ConString, 11)
End If
GetFromDB = ConString

MyExit:

End Function

Private Function GetToDB() As String

'Sep 22, 2008
Dim MyTitle$
    
    MyTitle$ = ProgramName() & " - pick a database to save your table(s) in"
    
If sysSaveTo = 1 Then
    GetToDB = GetFileName(MyTitle$, True)
End If

If sysSaveTo = 2 Then
    GetToDB = CreateNewDatabase()
End If

End Function

Private Function NameToUse() As String
'Oct. 12, 2002
Dim TempName As String, Msg As String, Title As String, DefValue As String

TryAgain:
    Msg = "Enter a name for your new table(s).  The name must start with a letter, contain no punctuation characters, preferably no spaces, and be less than 40 characters in length."
    Title = ProgramName() & " - New Name"
    DefValue = ""
    TempName = InputBox(Msg, Title, DefValue)
    If TempName = "Sample" Then
        MsgBox "You can't use the name Sample.  Try again."
        Resume TryAgain
        TempName = ""
    End If
    NameToUse = TempName
End Function
Private Sub ImportProjectTables(db As Database)
'Oct. 12, 2002
Dim i As Integer, TblName As String, tblToCheck As String, lenToCheck As Integer, FullName As String

tblToCheck = clsVProReg.CurrProject
lenToCheck = Len(tblToCheck)

For i = 0 To db.TableDefs.count - 1
    FullName = db.TableDefs(i).Name
    TblName = Left(FullName, lenToCheck)
    If TblName = tblToCheck Then
        If IsProjectTable(FullName) Then
            DoCmd.TransferDatabase acImport, "Microsoft Access", db.Name, acTable, FullName, "USysSaveAsRK" & Mid(FullName, lenToCheck + 1)
        End If
    End If
Next i

End Sub
Private Sub ImportSUTable(db As Database)
'Oct. 12, 2002

If db.Name = CurrentDb.Name Then
    DoCmd.CopyObject , "USysSaveAsRK_SU", acTable, clsVProReg.CurrPlotlist & "_SU"
Else
    DoCmd.TransferDatabase acImport, "Microsoft Access", db.Name, acTable, clsVProReg.CurrPlotlist & "_SU", "USysSaveAsRK_SU"
End If

MyExit:
End Sub

Private Sub ExportProjectTables(ToDB As Database, ThisDB As Database, NewName As String)
'Oct. 12, 2002
Dim i As Integer, TblName As String, tblToCheck As String, lenToCheck As Integer, FullName As String

tblToCheck = clsVProReg.CurrProject
lenToCheck = Len(tblToCheck)

For i = 0 To ThisDB.TableDefs.count - 1
    FullName = ThisDB.TableDefs(i).Name
    If Left(FullName, 13) = "USysSaveAsRK_" Then
        TblName = NewName & Mid(FullName, 13)
        DoCmd.TransferDatabase acExport, "Microsoft Access", ToDB.Name, acTable, FullName, TblName
    End If
Next i

End Sub

Private Sub CleanUp(db As Database)
'June 12, 2006
Dim i As Integer, TblName As String

For i = 0 To db.TableDefs.count - 1
    TblName = Left(db.TableDefs(i).Name, 12)
    If TblName = "USysSaveAsRK" Then
        DoCmd.DeleteObject acTable, db.TableDefs(i).Name
    End If
Next i

End Sub

Private Function IsProjectTable(TblName As String) As Boolean
'Nov. 22, 2021

If Right(TblName, 4) = "_Env" Then
    IsProjectTable = True
    GoTo MyExit
End If

If Right(TblName, 6) = "_Admin" Then
    IsProjectTable = True
    GoTo MyExit
End If

If Right(TblName, 4) = "_Veg" Then
    IsProjectTable = True
    GoTo MyExit
End If

If Right(TblName, 6) = "_Audit" Then
    IsProjectTable = True
    GoTo MyExit
End If

If Right(TblName, 6) = "_Other" Then
    IsProjectTable = True
    GoTo MyExit
End If

If Right(TblName, 6) = "_Humus" Then
    IsProjectTable = True
    GoTo MyExit
End If

If Right(TblName, 8) = "_Mineral" Then
    IsProjectTable = True
    GoTo MyExit
End If

If Right(TblName, 9) = "_Metadata" Then
    IsProjectTable = True
    GoTo MyExit
End If

MyExit:

End Function

Private Function Check4Table(db As Database, ByVal tbl As String, ObjectToSave As String) As Boolean
'Oct. 12, 2002
Dim TestTable As String, i As Integer
Check4Table = False

Select Case ObjectToSave
    Case "Project"
        tbl = tbl & "_Env"
    Case "SU"
        tbl = tbl & "_SU"
    Case "Hierarchy"
    
    Case "Lump"
    
    Case "Theme"
    
    Case Else
        Check4Table = True
        GoTo MyExit
End Select

For i = 0 To db.TableDefs.count - 1
    TestTable = db.TableDefs(i).Name
    If TestTable = tbl Then Check4Table = True
Next i

MyExit:
    On Error Resume Next
End Function
