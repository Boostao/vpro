Function QuickVegRecords(VegTable As String, SiteUnitList As String)

'Apr. 6, 2018
'creates a recordset of just the needed records
On Error Resume Next
    DoCmd.DeleteObject acTable, "QuickVeg"
On Error GoTo errQuickRecords
DoCmd.SetWarnings False
Dim SqlText As String, i As Integer
    
    DoCmd.CopyObject , "QuickVeg", acTable, "USysV2Veg"

For i = 1 To 9
    SqlText = "INSERT INTO QuickVeg ( Plotnumber, Layer, Species, Cover )"
    SqlText = SqlText & " SELECT [" & VegTable & "].PlotNumber, " & i & " AS Layer, [" & VegTable & "].Species, [" & VegTable & "].Cover" & i
    SqlText = SqlText & " FROM [" & VegTable & "] INNER JOIN [" & SiteUnitList & "] ON [" & VegTable & "].PlotNumber = [" & SiteUnitList & "].PlotNumber"
    SqlText = SqlText & " WHERE ((([" & VegTable & "].Cover" & i & ") Is Not Null));"
    DoCmd.RunSQL (SqlText)
Next i

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Exit Function
errQuickRecords:
    MsgBox Err.Description
    Resume MyExit
End Function
