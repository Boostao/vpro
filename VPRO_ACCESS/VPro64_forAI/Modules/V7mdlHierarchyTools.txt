Public Function GetParentName(rs As Recordset, ID As Long) As String
Dim Criteria As String, TempParent As String, BK As String

BK = rs.Bookmark
rs.MoveFirst
Criteria = "ID = " & ID
rs.FindFirst Criteria
    
If Not rs.NoMatch Then
    TempParent = rs!Name
Else
    TempParent = ""
End If
    
GetParentName = TempParent
rs.Bookmark = BK

End Function

Private Function DeleteNodeAndChildren(db As Database, Unit As Long, Optional MyCount As Integer)
'June 1, 2004
Dim MyRecords As Recordset, MyList As String, MyCriteria As String, MyParent As String
Dim NextUnit As Long, Shifty As Long, BK As String
Set MyRecords = db.OpenRecordset(CurrHierarchy(), dbOpenDynaset)
MyRecords.MoveFirst
MyCriteria = "ID = " & Unit
    MyRecords.FindFirst MyCriteria
    If MyRecords.NoMatch Then GoTo MyExit
    Shifty = MyRecords!ID
    
    If IsNull(MyRecords!Parent) Then
        WriteMasterAudit "Delete", MyRecords!Name, MyRecords!ID, "Delete", MyRecords!Name, "Null"
    Else
        MyParent = GetParentName(MyRecords, MyRecords!Parent)
        WriteMasterAudit "Delete", MyRecords!Name, MyRecords!ID, "Delete", MyRecords!Name, "Null", MyParent ' MyRecords!Parent
    End If
    MyRecords.Edit
    MyRecords!Name = "!*" & MyRecords!Name
    MyRecords.Update
    MyRecords.MoveFirst
    MyCriteria = "Parent = " & Shifty
    MyRecords.FindFirst MyCriteria

Do Until MyRecords.NoMatch
    BK = MyRecords.Bookmark
    NextUnit = MyRecords!ID
    DeleteNodeAndChildren db, NextUnit
    MyRecords.MoveNext
    MyRecords.Bookmark = BK
    MyRecords.FindNext MyCriteria
Loop

MyExit:
    MyRecords.Close

End Function

Public Sub CopyMyNodesBelowBreaks(TableName As String)

'May 27, 2009
On Error GoTo errCopyMyNodesBelowBreaks
Set db = CurrentDb()
Set NewRecords = db.OpenRecordset(TableName, dbOpenDynaset)
Set SearchRS = db.OpenRecordset(TableName, dbOpenDynaset)
Dim MyRS As Recordset
Dim SqlText As String, i As Integer, BK As String

SqlText = "SELECT Name, ID FROM [" & CurrHierarchy() & "] WHERE ((Left([Name],1)='~'));"
Set MyRS = db.OpenRecordset(SqlText)

MyRS.MoveLast
MyRS.MoveFirst

For i = 1 To MyRS.RecordCount
    BK = MyRS.Bookmark
    If Not CheckForTildeParents(MyRS!Name) Then
        Add2Hierarchy MyRS!Name '& "-"
        BuildList db, MyRS!ID, i
    End If
        MyRS.Bookmark = BK
    MyRS.MoveNext
Next i
SetLevels TableName

MyExit:
    NewRecords.Close
    SearchRS.Close
    MyRS.Close
    Set db = Nothing
    Exit Sub
errCopyMyNodesBelowBreaks:
    If Err.Number = 3021 Then
        Resume MyExit
    Else
        MsgBox Err.Description
        Resume MyExit
    End If
End Sub

Private Function CheckForTildeParents(Name As String) As Boolean
'Nov. 10, 1999
'returns false if no parents have tildies, or if is root
Dim MyRS2        As Recordset
Dim i           As Integer
Dim Criteria    As String
Dim SearchName  As String
Dim Parent      As Long
CheckForTildeParents = False
Set MyRS2 = db.OpenRecordset(CurrHierarchy(), dbOpenDynaset)
SearchName = Name

For i = 1 To 11
    MyRS2.MoveFirst
    Criteria = "Name = '" & SearchName & "'"
    MyRS2.FindFirst Criteria
    If IsNull(MyRS2!Parent) Then
        GoTo MyExit
    Else
        Parent = MyRS2!Parent
        Criteria = "ID = " & Parent
        MyRS2.MoveFirst
        MyRS2.FindFirst Criteria
        If Left(MyRS2!Name, 1) = "~" Then
            CheckForTildeParents = True
            GoTo MyExit
        End If
        SearchName = MyRS2!Name
    End If
Next i

CheckForTildeParents = True

MyExit:
    MyRS2.Close
    Exit Function
End Function

Private Sub Add2Hierarchy(Unit As String, Optional Parent As String)
'Nov. 10, 1999
NewRecords.AddNew

If Parent = "" Then 'must be a root item
    NewRecords!Name = Unit
Else
    NewRecords!Name = Unit
    NewRecords!Parent = FindParentID(Parent)
End If

NewRecords.Update

End Sub

Private Function BuildList(db As Database, Unit As Long, Optional MyCount As Integer)
'Nov. 10, 1999
Dim MyRecords As Recordset, MyList As String, MyCriteria As String
Dim NextUnit As Long, Shifty As Long, BK As String, Roz As String
Set MyRecords = db.OpenRecordset(CurrHierarchy(), dbOpenDynaset)
MyRecords.MoveFirst
MyCriteria = "ID = " & Unit
    MyRecords.FindFirst MyCriteria
    If MyRecords.NoMatch Then GoTo MyExit
    Shifty = MyRecords!ID
    Roz = MyRecords!Name
            
    MyRecords.MoveFirst
    MyCriteria = "Parent = " & Shifty
    MyRecords.FindFirst MyCriteria

Do Until MyRecords.NoMatch
    BK = MyRecords.Bookmark
    NextUnit = MyRecords!ID
    Add2Hierarchy MyRecords!Name, Roz
    BuildList db, NextUnit, MyCount
    MyRecords.Bookmark = BK
    MyRecords.FindNext MyCriteria
Loop

MyExit:
    MyRecords.Close
End Function

Private Sub SetLevels(TableName As String)
Dim MyRS As Recordset
Dim i As Integer
Set MyRS = db.OpenRecordset(TableName, dbOpenDynaset)
MyRS.MoveLast
MyRS.MoveFirst
Dim clnRS As Recordset
Set clnRS = MyRS.Clone

For i = 1 To MyRS.RecordCount
    If IsNull(MyRS!Parent) Then
        ChangeChildren MyRS!ID, clnRS, 2
        MyRS.Edit
        MyRS!Level = 1
        MyRS.Update
    End If
    MyRS.MoveNext
Next i

MyRS.Close
End Sub

Private Function FindParentID(Parent) As Long
'Nov. 10, 1999
Dim Criteria As String
Criteria = "Name = '" & Parent & "'"
SearchRS.Requery

    SearchRS.MoveLast
    SearchRS.MoveFirst

SearchRS.FindFirst Criteria
If Not SearchRS.NoMatch Then
    FindParentID = SearchRS!ID
End If

End Function

Private Function ChangeChildren(Unit As Long, ReadRS As Recordset, Level As Integer)
'Nov 10, 1999
'finds the last member in a hierarchy branch
Dim i As Integer, Criteria As String, MyID As Integer, MyTest As String, BK As String
Dim TempRS As Recordset
Set TempRS = ReadRS.Clone

ReadRS.MoveFirst
Criteria = "Parent = " & Unit
ReadRS.FindFirst Criteria

Do Until ReadRS.NoMatch

    BK = ReadRS.Bookmark
    
    If HasBrats(ReadRS!ID, TempRS) Then
        ReadRS.Edit
        ReadRS!Level = Level
        ReadRS.Update
        ReadRS.Bookmark = BK
        ChangeChildren ReadRS!ID, ReadRS, Level + 1
    Else
        ReadRS.Edit
        ReadRS!Level = 11
        ReadRS.Update
    End If
    
    ReadRS.Bookmark = BK
    ReadRS.FindNext Criteria
Loop
TempRS.Close
End Function
