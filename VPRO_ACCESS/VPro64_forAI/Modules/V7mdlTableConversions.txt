Private Function ExportUserSiteUnitNames(Optional SiteUnitList)

'Sep 22, 2008
'export long site unit names
On Error Resume Next
    DoCmd.DeleteObject acTable, "TempLongNames"
On Error GoTo errExportUserSiteUnitNames
Dim MyTitle As String, MyFile As String

If IsMissing(SiteUnitList) Then
        SiteUnitList = CurrPlotlist() & "_SU"
End If

    MyTitle = ProgramName()
    MyFile = GetFileName(MyTitle, True, "VPro User Files (*.accda)", "*.accda")
    If MyFile = "" Then Exit Function
    DoCmd.TransferDatabase acLink, "Microsoft Access", MyFile, acTable, "USysUserSpp", "UserSpecies"

MyExit:
    On Error Resume Next
    Exit Function
errExportUserSiteUnitNames:
    MsgBox Err.Description
    Resume MyExit
End Function

Public Function ConvertProVeg2V2(VegTableIn, VegTableOut)
'Sep 10, 2008
'converts VPro veg table to a one species per row table
On Error GoTo errConvertProVeg2V2
Dim VegTable As String, Project As String, SqlText As String
Dim i As Integer, LumpTable As String
Dim MyDB As Database, MyRS As Recordset
Set MyDB = CurrentDb

SqlText = "SELECT LayerCode.LayerText"
SqlText = SqlText & " From LayerCode"
SqlText = SqlText & " WHERE (((LayerCode.LayerText) Is Not Null));"
Set MyRS = MyDB.OpenRecordset(SqlText)

DoCmd.SetWarnings False

VegTable = VegTableIn
If CurrPlotlist() <> "None" Then VegTable = "TempReportVeg"
Project = CurrProject()
DoCmd.CopyObject , VegTableOut, A_TABLE, "USysV2Veg"

Do Until MyRS.EOF
    SqlText = "INSERT INTO [" & VegTableOut & "] ( Plotnumber, Species, Cover )"
    SqlText = SqlText & " SELECT DISTINCTROW [" & VegTable & "].PlotNumber, [" & VegTable & "].Species, [" & VegTable & "].Cover" & MyRS!layertext
    SqlText = SqlText & " FROM [" & VegTable & "]"
    SqlText = SqlText & " WHERE (([" & VegTable & "].Cover" & MyRS!layertext & " Is Not Null));"
    If MyRS!layertext = "5A" Then
        'CurrentDb.CreateQueryDef "fred1", SqlText
        'Stop
    End If
    'CurrentDb.CreateQueryDef "fred1", SqlText
    DoCmd.RunSQL SqlText

    SqlText = "UPDATE DISTINCTROW [" & VegTableOut & "] SET [" & VegTableOut & "].ProjectID = '" & Project & "', [" & VegTableOut & "].Layer = '" & MyRS!layertext & "'"
    SqlText = SqlText & " WHERE (([" & VegTableOut & "].Layer Is Null));"
    If MyRS!layertext = "5A" Then
        'Debug.Print SqlText
        'CurrentDb.CreateQueryDef "fred2", SqlText
    End If
    DoCmd.RunSQL SqlText
    MyRS.MoveNext
Loop


GoTo MyExit
For i = 1 To 7

    SqlText = "INSERT INTO [" & VegTableOut & "] ( Plotnumber, Species, Cover )"
    SqlText = SqlText & " SELECT DISTINCTROW [" & VegTable & "].PlotNumber, [" & VegTable & "].Species, [" & VegTable & "].Cover" & i
    SqlText = SqlText & " FROM [" & VegTable & "]"
    SqlText = SqlText & " WHERE (([" & VegTable & "].Cover" & i & " Is Not Null));"
    'CurrentDb.CreateQueryDef "fred1", SqlText
    DoCmd.RunSQL SqlText

    SqlText = "UPDATE DISTINCTROW [" & VegTableOut & "] SET [" & VegTableOut & "].ProjectID = '" & Project & "', [" & VegTableOut & "].Layer = " & i
    SqlText = SqlText & " WHERE (([" & VegTableOut & "].Layer Is Null));"
    'CurrentDb.CreateQueryDef "fred2", SqlText
    DoCmd.RunSQL SqlText
Next i

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Exit Function
errConvertProVeg2V2:
    DoCmd.SetWarnings True
    MsgBox Err.Description
    Resume MyExit
End Function
