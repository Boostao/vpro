Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = False
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Compare Database
Option Explicit
Private Const strApp = "VPro64"

Public Property Get VegBy() As Integer
    VegBy = GetSetting(strApp, "Export", "rVegBy", 1)
End Property

Public Property Let VegBy(VegBy As Integer)
    SaveSetting strApp, "Export", "rVegBy", VegBy
End Property

Public Property Get ExportUnits() As Integer
    ExportUnits = GetSetting(strApp, "Export", "rExportUnits", 1)
End Property

Public Property Let ExportUnits(ExportUnits As Integer)
    SaveSetting strApp, "Export", "rExportUnits", ExportUnits
End Property
