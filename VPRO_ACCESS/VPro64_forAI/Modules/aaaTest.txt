Private Sub RenameModules()

Dim i As Integer, tmpName As String, MyC As Container

On Error Resume Next
For Each MyC In CurrentDb.Containers

    If MyC.Name = "Modules" Then
        For i = 1 To MyC.Documents.count
            tmpName = MyC.Documents(i).Name
            If Left(tmpName, 7) = "V7xpmdl" Then
                DoCmd.CopyObject , "V7" & Right(tmpName, Len(tmpName) - 4), acModule, tmpName
                DoCmd.DeleteObject acModule, tmpName
            End If
        Next i
    End If
    
Next MyC
      
MsgBox "done"

MyExit:

End Sub
