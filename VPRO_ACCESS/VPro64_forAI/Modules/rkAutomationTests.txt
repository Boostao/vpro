Private Sub SendRsToExcel(MyRS1 As Recordset, SheetName1 As String, MyRS2 As Recordset, SheetName2 As String)

Dim xl As Excel.Application

CreateExcel xl
xl.Visible = True
xl.Workbooks.Add
RemoveSheets xl

xl.ActiveSheet.Name = SheetName1
xl.Cells(2, 1).CopyFromRecordset MyRS1
AddHeaders xl, MyRS1

xl.Sheets.Add
xl.ActiveSheet.Name = SheetName2
xl.Cells(2, 1).CopyFromRecordset MyRS2
AddHeaders xl, MyRS2

Set xl = Nothing

End Sub

Private Sub AddHeaders(xl As Excel.Application, MyRS As Recordset)

Dim i As Integer

For i = 0 To MyRS.Fields.count - 1
    xl.Cells(1, i + 1) = MyRS.Fields(i).Name
Next i

xl.Cells.EntireColumn.AutoFit

End Sub

Private Sub RemoveSheets(xl As Excel.Application)

Dim i As Integer

    xl.DisplayAlerts = False
    For i = 1 To xl.Sheets.count - 1
        xl.Sheets(i).Delete
    Next i
    xl.DisplayAlerts = True

End Sub

Private Sub GetExcel()
    Dim Myxl As Object, xl As Object  ' Variable to hold reference to Microsoft Excel.
    Dim MyDB As Database, MyDir As String, Msg, X
    Set MyDB = CurrentDb()
    MyDir = Left(MyDB.Name, Len(MyDB.Name) - 10)
    DoCmd.SetWarnings False

    On Error Resume Next    ' Defer error trapping.
    DoCmd.OpenForm "USysLoading"
    Forms![USysLoading]![Text0].Caption = "Working..."
    Forms![USysLoading].Repaint
    Set Myxl = GetObject(, "Excel.Application")  'try to get an instance of excel
    If Err.Number = 429 Then      'excel wasn't running
        Set Myxl = CreateObject("Excel.Application")  'create an excel object
    End If
    Set xl = Myxl.Application  'set a reference to excel's application object
    On Error GoTo 0   'reset error trapping
    Err.Clear   ' Clear Err object in case error occurred.
    xl.Workbooks.Open (MyDir & "HierarchyReport.xls")   'open an existing worksheet
    On Error Resume Next
    xl.Run ("InsertAddmissionInfo")

    DoCmd.Close acForm, "USysLoading"
    DoCmd.Echo True
    
    xl.Visible = True
    DoCmd.SetWarnings True
    Set Myxl = Nothing  ' Release reference to the
    Set xl = Nothing
End Sub
