Option Compare Database
Option Explicit

Function UnattachDupTables()
'May 16, 2006
'searchs for and destroys tables that have been attached twice
On Error GoTo errUnattachDupTables
DoCmd.SetWarnings False

'YYYYYYYYYYYYYYYYYY   START DELETE ATTACHED TABLES    YYYYYYYYYYYYYYYYYYYYYY
    Dim MyDB As Database, TestTable As String, i As Integer, TestConnect As String
    Set MyDB = CurrentDb()
    
        'ALWAYS USE I = 0 TO -1 BECAUSE OF THE WAY TABLEDEFS INDEX WORKS
    For i = 0 To MyDB.TableDefs.count - 1
        TestTable = MyDB.TableDefs(i).Name
        TestConnect = MyDB.TableDefs(i).Connect
        If TestConnect <> "" Then
            If Right(TestTable, 5) = "_Env1" Or Right(TestTable, 5) = "_Env2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 5) = "_Veg1" Or Right(TestTable, 5) = "_Veg2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 9) = "_Mineral1" Or Right(TestTable, 9) = "_Mineral2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 7) = "_Humus1" Or Right(TestTable, 7) = "_Humus2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 7) = "_Audit1" Or Right(TestTable, 7) = "_Audit2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 7) = "_Theme1" Or Right(TestTable, 7) = "_Theme2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 9) = "_Sorting1" Or Right(TestTable, 9) = "_Sorting2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 6) = "_Lump1" Or Right(TestTable, 6) = "_Lump2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 7) = "_Other1" Or Right(TestTable, 7) = "_Other2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 11) = "_Constancy1" Or Right(TestTable, 11) = "_Constancy2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 10) = "_TreePlot1" Or Right(TestTable, 10) = "_TreePlot2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 7) = "_Trees1" Or Right(TestTable, 7) = "_Trees2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 5) = "_CWD1" Or Right(TestTable, 5) = "_CWD2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 11) = "_CWDPiling1" Or Right(TestTable, 11) = "_CWDPiling2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
            If Right(TestTable, 11) = "_Herbarium1" Or Right(TestTable, 11) = "_Herbarium2" Then
                DoCmd.DeleteObject A_TABLE, TestTable
            End If
        End If
    Next i
    
MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Set MyDB = Nothing
    Exit Function
errUnattachDupTables:
    DoCmd.SetWarnings True
    MsgBox Err.Description
    Resume MyExit
End Function
