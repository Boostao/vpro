Public Function ConvertLongLatToDeg(Optional Deg As Variant, Optional Min As Variant, Optional Sec As Variant) As Variant

'Dec. 3, 2008
    If IsNull(Deg) Then
        ConvertLongLatToDeg = Null
    Else
        If IsMissing(Sec) Then Sec = 0
        If IsMissing(Min) Then Min = 0
        If IsNull(Sec) Then Sec = 0
        If IsNull(Min) Then Min = 0
        ConvertLongLatToDeg = Deg + ((Min + (Sec / 60)) / 60)
    End If

End Function

Public Function GetLongDMS(LongIn As Variant) As MyLong

Dim TempVal As Single, d As Integer, m As Integer, s As Single
If LongIn < 0 Then LongIn = LongIn * -1


If IsNull(LongIn) Then
    GetLongDMS.d = Null
    GetLongDMS.m = Null
    GetLongDMS.s = Null
    GoTo MyExit
Else
    d = Int(LongIn)
    GetLongDMS.d = d
End If

TempVal = LongIn - d
m = Int(TempVal * 60)
GetLongDMS.m = m

TempVal = (TempVal * 60) - m
s = TempVal * 60
GetLongDMS.s = s

MyExit:

End Function
