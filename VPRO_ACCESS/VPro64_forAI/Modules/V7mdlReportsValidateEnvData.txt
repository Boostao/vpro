Option Compare Database
Option Explicit
Private PlotCol As Integer, FieldCol As Integer
Private xl As Excel.Application

Public Sub ValidateEnvData()

'Jan 9, 2009
Dim ListRS As Recordset
Dim SqlText As String, i As Integer

CreateExcel xl
xl.Visible = True
xl.Workbooks.Add
PlotCol = 1
FieldCol = 2

'Build a list of fields to validate
SqlText = "SELECT ListName, FieldUsedIn, ValidateLoops"
SqlText = SqlText & " From USysTableOfLists"
SqlText = SqlText & " WHERE (((Validate) = Yes))"
SqlText = SqlText & " GROUP BY ListName, FieldUsedIn, ValidateLoops;"
Set ListRS = CurrentDb.OpenRecordset(SqlText)

Do Until ListRS.EOF
    If ListRS!ValidateLoops > 0 Then
        For i = 1 To ListRS!ValidateLoops
            ReportData ListRS!ListName, ListRS!FieldUsedIn & i
        Next i
    Else
        ReportData ListRS!ListName, ListRS!FieldUsedIn
    End If
    ListRS.MoveNext
Loop

xl.Cells.EntireColumn.AutoFit
xl.Cells(1, 1) = "Environment field code check for project: " & clsVProReg.CurrProject & ", and SU: " & clsVProReg.CurrPlotlist
If PlotCol = 1 Then xl.Cells(1, 2) = "No errors noted"

MyExit:
    On Error Resume Next
    ListRS.Close
    Set xl = Nothing
    Exit Sub

End Sub

Private Sub ReportData(ListName As String, FieldToValidate As String)

'Oct. 10, 2008
Dim SqlText As String, i As Integer
Dim MyDB As Database, ErrorRS As Recordset
Set MyDB = CurrentDb

'Build qryTableOfLists
    On Error Resume Next
    DoCmd.DeleteObject acQuery, "qryTableOfLists"
    SqlText = "SELECT ListName, Item, FieldUsedIn, ValidateLoops"
    SqlText = SqlText & " From USysTableOfLists"
    SqlText = SqlText & " WHERE (((ListName)='" & ListName & "'));"
    CurrentDb.CreateQueryDef "qryTableOfLists", SqlText
    On Error GoTo errReportData

'check data
    SqlText = "SELECT USysEnv.PlotNumber, USysEnv." & FieldToValidate & ""
    SqlText = SqlText & " FROM USysEnv LEFT JOIN qryTableOfLists ON USysEnv." & FieldToValidate & " = qryTableOfLists.Item"
    SqlText = SqlText & " WHERE (((USysEnv." & FieldToValidate & ") Is Not Null And (USysEnv." & FieldToValidate & ")<>'') AND ((qryTableOfLists.Item) Is Null));"
    Set ErrorRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)
    If ErrorRS.RecordCount = 0 Then GoTo MyExit

'Report data
    xl.Cells(2, PlotCol) = "Plot Number"
    xl.Cells(2, FieldCol) = FieldToValidate
    i = 3
    Do Until ErrorRS.EOF
        xl.Cells(i, PlotCol) = ErrorRS!PlotNumber
        xl.Cells(i, FieldCol) = ErrorRS(FieldToValidate)
        i = i + 1
        ErrorRS.MoveNext
    Loop

    PlotCol = PlotCol + 3
    FieldCol = FieldCol + 3


MyExit:
    On Error Resume Next
    ErrorRS.Close
    Set MyDB = Nothing
    Exit Sub
errReportData:
    MsgBox Err.Number & " - " & Err.Description, vbInformation, clsVProReg.ProgramName
    MsgBox "Field: " & FieldToValidate
    Resume MyExit

End Sub
