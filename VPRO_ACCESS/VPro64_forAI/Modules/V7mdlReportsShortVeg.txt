Public Sub LumpSummary(xl As Excel.Application, StartRow As Integer, StartCol As Integer)

'Mar. 9, 2016
Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String, LumpCode As String, i As Integer

Set MyDB = CurrentDb

    SqlText = "SELECT LumpCode, SppCode, Use"
    SqlText = SqlText & " FROM " & sysLumpingTable
    SqlText = SqlText & " WHERE (((Use) = Yes))"
    SqlText = SqlText & " ORDER BY LumpCode, SppCode;"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount = 0 Then GoTo MyExit

    MyRS.MoveLast
    MyRS.MoveFirst
    xl.Cells(StartRow, StartCol) = "Combined species codes in table " & Left(sysLumpingTable, Len(sysLumpingTable) - 5)
    xl.Cells(StartRow + 1, StartCol).CopyFromRecordset MyRS

    SqlText = "SELECT " & sysLumpingTable & ".LumpCode, USysSppAttributeDescription.Definition"
    SqlText = SqlText & " FROM " & sysLumpingTable & " INNER JOIN USysSppAttributeDescription ON " & sysLumpingTable & ".LumpCode = USysSppAttributeDescription.ShortName"
    SqlText = SqlText & " GROUP BY " & sysLumpingTable & ".LumpCode, USysSppAttributeDescription.Definition;"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount = 0 Then GoTo MyExit
    xl.Cells(5, 11).CopyFromRecordset MyRS
    xl.Cells(4, 11) = "Attribute lump code definitions"

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    On Error GoTo 0
    Exit Sub

End Sub

Private Function AddEnvFields(xl As Excel.Application, FirstRow As Integer) As Integer

'Mar. 9, 2016
On Error GoTo errAddEnvFields
Dim MyDB As Database, MyRS As Recordset, FunctionRS As Recordset
Dim CurrRow As Integer, CurrCol As Integer, SqlText As String, FirstCol As Integer
Dim fnStr As String, X As Integer, x2 As String

Set MyDB = CurrentDb
        
'list of functions and labels
    SqlText = "SELECT Order, Source, Label, Item, Report, FunctionName, FieldNumber"
    SqlText = SqlText & " From USysShortVegSummaryFieldList"
    SqlText = SqlText & " WHERE (((Report) = Yes))"
    SqlText = SqlText & " ORDER BY USysShortVegSummaryFieldList.Order;"
    Set FunctionRS = MyDB.OpenRecordset(SqlText)
    
'list of units
    Set MyRS = MyDB.OpenRecordset("SV1")
    
'set the initial row
    CurrRow = FirstRow

'set FirstCol
    If clsRepOpt.SVShowEnglishName Then
        FirstCol = 5
    Else
        FirstCol = 4
    End If

'add unit row
    CurrCol = FirstCol
    Do Until MyRS.EOF
        xl.Cells(CurrRow - 1, CurrCol) = MyRS!Name
        xl.Cells(CurrRow - 1, CurrCol).Font.Bold = True
        CurrCol = CurrCol + 1
        MyRS.MoveNext
    Loop
    MyRS.MoveFirst

'build quick veg
    BuildQuickVeg

'add labels and data
    Do Until FunctionRS.EOF
        If FunctionRS!Label = "n Plots" Then
            xl.Cells(CurrRow, 3) = FunctionRS!Label
        Else
            xl.Cells(CurrRow, 1) = FunctionRS!Source
            xl.Cells(CurrRow, 2) = FunctionRS!Label
        End If
        fnStr = FunctionRS!FunctionName
        CurrCol = FirstCol
        Do Until MyRS.EOF
            fnStr = FunctionRS!FunctionName
            If IsNull(FunctionRS!FieldNumber) Then
                xl.Cells(CurrRow, CurrCol) = Application.Run(fnStr, MyRS!Name)
            Else
                xl.Cells(CurrRow, CurrCol) = Application.Run(fnStr, MyRS!Name, FunctionRS!FieldNumber)
            End If
            xl.Cells(CurrRow, CurrCol).WrapText = True
            CurrCol = CurrCol + 1
            MyRS.MoveNext
        Loop
        MyRS.MoveFirst
        CurrRow = CurrRow + 1
        FunctionRS.MoveNext
    Loop
    
'call for formatting
    FormatEnvHeader xl, FirstRow, FirstCol, CurrCol - 1, CurrRow - 1

'set value to last row
    AddEnvFields = CurrRow + 2

MyExit:
    On Error Resume Next
    MyRS.Close
    FunctionRS.Close
    Set MyDB = Nothing
    DoCmd.DeleteObject acTable, "QuickVeg"
    On Error GoTo 0
    Exit Function

errAddEnvFields:
    MsgBox Err.Description
    Resume MyExit

End Function

Public Function NewShortVeg()

'Mar. 11, 2022

'checks for a current su table and exits if none
    If clsVProReg.CurrPlotlist = "None" Then
        MsgBox "You must select a site unit table in order to generate this report.", vbInformation, "VPro"
        Exit Function
    End If

'declare variables
Dim MyTime As Double, SuTable As String
Dim MyDB As Database, MyRS As Recordset
Dim GroupBy As Integer
Dim OrderBy As Integer
Dim DisplayValue As Integer, LumpList As String, CreateSummary As Integer, SpaceBetweenGroups As Integer, SqlText As String, ReturnHierarchy As String
Dim xl As Excel.Application
Dim T1 As Double, T2 As Double, T3 As Double, T4 As Double, T5 As Double, FirstRow As Integer, HasOrder As Boolean
Dim Doing7 As Boolean
Dim OriginalSuTable As String

Set MyDB = CurrentDb()
NewShortVeg = False

'because we'll be creating a temporary plot list, best to back up the current one and restore any previous run problems
    If clsVProReg.CurrPlotlist = "USysDeleteMe" Then
        clsVProReg.CurrPlotlist = clsVProReg.CurrPlotlistBak
    End If
    OriginalSuTable = clsVProReg.CurrPlotlist
    clsVProReg.CurrPlotlistBak = OriginalSuTable
    
'do some quality control work
    If QC = False Then
        OriginalSuTable = clsVProReg.CurrPlotlist
        'MsgBox OriginalSuTable
        GoTo MyExit
    End If

'set the first data row
    If clsRepOpt.SVIncludeEnvFields Then
        FirstRow = 5
    Else
        FirstRow = 4
    End If

'get options
    ReportTitle = clsRepOpt.SVReportTitle
    ShowCommon = clsRepOpt.SVShowEnglishName
    ValueLimit = clsRepOpt.SVShowValOpt
    GroupBy = clsRepOpt.SVGroupBy
    OrderBy = 10
    AddSpace = 0
    LumpList = clsVProReg.CurrLump & "_Lump"
        AddAttributeDescriptions
    If LumpList = "Attribute_Lump" Then
        
    End If
    CreateSummary = clsRepOpt.SVReportSummary

    optDiagnostic = 0 'disables vpro's diagnostics
    AverageType = clsRepOpt.SVAvgType
    DisplayValue = clsRepOpt.SVValue
    SuType = clsRepOpt.SVReportUnits
    sysSuType = SuType
    MyTime = Timer
    DoDiagnostic = False

'if breaks and SUs below breaks
    If SuType = 7 Then
        Doing7 = True
        SuType = 2
        HasOrder = True
    End If

'clean up previous mess
    On Error GoTo 0
    CleanUp
    
'option to use short guide names together with english names
'this will rename the field names in the USysAllSpecs table
    RenameGuideFieldNameBack
    If clsRepOpt.SVShowGuideName = -1 Then
        clsRepOpt.SVShowEnglishName = -1
        RenameGuidepFieldNameForward
    Else
        RenameGuideFieldNameBack
    End If
    
'check for too many columns
    If SuType = 1 Then
        If CheckForTooManyColumns Then
            MsgBox "There are too many units to report.  Try limiting the number of units (columns) to 250.", vbInformation, "VPro"
            GoTo MyExit
        End If
    End If

'build USysSuTableDynamic_SU for env data
    If SuType <> 3 Then
        DoCmd.SetWarnings False
        SqlText = "DELETE * FROM USysSuTableDynamic_SU;"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO USysSuTableDynamic_SU ( PlotNumber, SiteUnit )"
        SqlText = SqlText & " SELECT PlotNumber, SiteUnit"
        SqlText = SqlText & " FROM " & clsVProReg.CurrPlotlist & "_SU;"
        'MyDB.CreateQueryDef "fred", SqlText
        DoCmd.RunSQL SqlText
        DoCmd.SetWarnings True
    End If

'diagnostic options
    If SuType = 4 Then
        DoDiagnostic = True
        SuType = 2
    End If
    If SuType = 2 And Not DoDiagnostic And clsRepOpt.SVShowHierarchyHeader <> 0 Then
        ControlHierarchyOrder 'add order info to hierarchy
    End If

'set options for creating SV queries
    GroupAndOrder = GroupBy + OrderBy
    ReportOrder = OrderBy
    ReportGroup = GroupBy
    sysCurrSU = ""

'use breakpoints
    If SuType >= 5 And SuType < 7 Then
        If SuType = 6 Then
            CreateBreakTable "Hierarchy", 1
            ClipHierarchy 'create a lowest breakpoint hierarchy table
            DoDiagnostic = True
            SuType = 5
        Else
            ReturnHierarchy = clsVProReg.CurrHierarchy
            DoCmd.SetWarnings False
            SqlText = "delete * from USysLowestBreakpoints_Hierarchy"
            DoCmd.RunSQL SqlText
            DoCmd.SetWarnings True
            CopyMyNodesBelowBreaks "USysLowestBreakpoints_Hierarchy"
            clsVProReg.CurrHierarchy = "USysLowestBreakpoints"
            OrderHierarchyStep1 CurrentDb, CurrHierarchy
            CreateBreakTable "Hierarchy"
            clsVProReg.CurrHierarchy = ReturnHierarchy
            'CreateBreakTable "Hierarchy"
        End If

        DoCmd.SetWarnings False
        SqlText = "DELETE * FROM USysSuTableDynamic_SU;"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO USysSuTableDynamic_SU ( SiteUnit, PlotNumber )"
        SqlText = SqlText & " SELECT Mid([Reference],2) AS SiteUnit, USysBreakTable.PlotNumber"
        SqlText = SqlText & " FROM USysBreakTable;"
        DoCmd.RunSQL SqlText
        Set MyRS = MyDB.OpenRecordset("USysSuTableDynamic_SU")
        SuTable = "USysSuTableDynamic_SU"
        If MyRS.RecordCount = 0 Then
            MsgBox "No plots or break points.", vbCritical, "VPro"
            GoTo MyExit
        End If
        DoCmd.SetWarnings True
    End If

'set timers
    T1 = Timer - MyTime
    If SuType = 2 Then 'Or SuType = 4 Then    'create a su list of tildied members
        ReturnHierarchy = clsVProReg.CurrHierarchy
        DoCmd.SetWarnings False
        SqlText = "delete * from USysLowestBreakpoints_Hierarchy"
        DoCmd.RunSQL SqlText
        CopyMyNodesBelowBreaks "USysLowestBreakpoints_Hierarchy"
        clsVProReg.CurrHierarchy = "USysLowestBreakpoints"
        OrderHierarchyStep1 CurrentDb, CurrHierarchy
        SqlText = "UPDATE USysLowestBreakpoints_Hierarchy INNER JOIN " & ReturnHierarchy & "_Hierarchy ON USysLowestBreakpoints_Hierarchy.Name = " & ReturnHierarchy & "_Hierarchy.Name SET " & ReturnHierarchy & "_Hierarchy.MyOrder = [USysLowestBreakpoints_Hierarchy]![MyOrder];"

    T2 = Timer - MyTime
    SuTable = UnitType(SuType, MyDB, optDiagnostic)

'if su type really 7

        DoCmd.SetWarnings True

'if the option of user selected field is picked, run this code

'set T4 timer
    T4 = Timer - MyTime

'freeze top of report

        xl.ActiveWindow.FreezePanes = True

'timing code

    Msg = Msg & "T2: " & T2 & Chr(13)
    Msg = Msg & "T3: " & T3 & Chr(13)
    Msg = Msg & "T4: " & T4 & Chr(13)

    CleanupAddAttributeDescriptions

errNewShortVeg:
    clsVProReg.CurrPlotlist = OriginalSuTable
    If Err.Number = 440 Then

    End If
    If Err.Number = 1004 Then

        Resume MyExit
        
End Function

Public Function RkFormula(ReturnType As String, Optional CountCover, Optional SumCover, Optional Firstnplots, Optional AvgCover)

'Mar. 9, 2016
    
    If ReturnType = "nPlots" Then
        If CountCover > 0 Then
            RkFormula = Presence2Class(CountCover / Firstnplots) & " - " & vpRoundUp2(SumCover / Firstnplots)
        Else
            RkFormula = Null
        End If
    End If
    If ReturnType = "Characteristic" Then
        If CountCover > 0 Then
            RkFormula = Presence2Class(CountCover / Firstnplots) & " - " & vpRoundUp2(AvgCover)
        Else
            RkFormula = Null
        End If
    End If

End Function

'Mar. 9, 2016

    End If

    End If

End Function

    End If

End Function

Private Function SuFromBreaks() As Boolean

'Mar. 9, 2016
On Error GoTo myerr

Dim MyDB As Database, MyRS As Recordset, MySU As Recordset, MyBT As Recordset
Dim i As Integer, Criteria As String, MyTest As Boolean

    Set MyDB = CurrentDb()
    Set MyRS = MyDB.OpenRecordset(CurrPlotlist() & "_SU", dbOpenDynaset)
    
    sysBreakSU = "SuDiagnostic"
    
    Set MySU = MyDB.OpenRecordset(sysBreakSU, dbOpenDynaset)
    
    If Not CreateBreakTable("Hierarchy") Then
        SuFromBreaks = False
    End If
    
    Set MyBT = MyDB.OpenRecordset("USysBreakTable", dbOpenDynaset)
    
    MyBT.MoveLast
    MyBT.MoveFirst

    Do Until MyBT.EOF
        MyTest = True
        Criteria = "PlotNumber = '" & MyBT!PlotNumber & "'"
        If MySU.RecordCount > 0 Then
            MySU.MoveFirst
            MySU.FindFirst Criteria
        End If
        If MySU.NoMatch Or MySU.RecordCount = 0 Then
            MySU.AddNew
            MySU!PlotNumber = MyBT!PlotNumber
            MyRS.MoveFirst
            MyRS.FindFirst Criteria
            MySU!SiteUnit = MyRS!SiteUnit
            MySU.Update
        End If
        MyBT.MoveNext
    Loop

    SuFromBreaks = True
    
MyExit:
    On Error Resume Next
    MyBT.Close
    MyRS.Close
    MySU.Close
    Set MyDB = Nothing
    On Error GoTo 0
    Exit Function

myerr:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function FindUnits(Unit As String, ReadRS As Recordset, WriteRS As Recordset, Level As Integer, Parent As String)

'Mar. 9, 2016
'finds the last member in a hierarchy branch
Dim i As Integer, Criteria As String, MyID As Long, MyTest As String

    ReadRS.MoveFirst
    Criteria = "Name = '" & Unit & "'"
    ReadRS.FindFirst Criteria
    
    Do Until ReadRS.NoMatch
        If ReadRS!Level = 11 Then
            WriteUnits WriteRS, ReadRS!Name, Level, Parent
                'Debug.Print Level
        Else
            MyID = ReadRS!ID
            MyTest = "Parent = " & MyID
            ReadRS.MoveFirst
            ReadRS.FindFirst MyTest
            If ReadRS!Level = 11 Then
                WriteUnits WriteRS, ReadRS!Name, Level, Parent
    
            Else
                FindUnits ReadRS!Name, ReadRS, WriteRS, Level, Parent
            End If
        End If
        If MyTest = "" Then Exit Function
        ReadRS.FindNext MyTest
    Loop

End Function

Private Function OrderHierarchy(MyDB As Database, Optional Unit)

'Mar. 9, 2016

GroupsRS.MoveFirst

Private Function FindBreakPoints() As Boolean

'Mar. 9, 2016

Dim MyDB As Database, HierarchyRS As Recordset
Dim i As Integer, X As Integer, SqlText As String, ParentID As String

    FindBreakPoints = True
    Set MyDB = CurrentDb()

    OriginalParent = ""
    
    HierarchyRS.MoveLast
    HierarchyRS.MoveFirst

    DoEvents

    If Not CreateDiagnostic(MyDB) Then

    HierarchyRS.Close
    Set MyDB = Nothing
    On Error GoTo 0

Private Function CreateDiagnostic(db As Database) As Boolean

'Mar. 9, 2016

Dim MyDB As Database, MyRS As Recordset, MyQD As QueryDef, CountFieldsRS As Recordset, DiagnosticRS As Recordset
Dim SqlText As String, i As Integer

    CreateDiagnostic = True
    Set MyDB = CurrentDb()
    
    DoCmd.SetWarnings False
    SqlText = "Delete * from USysSuTableDynamic_SU"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO USysSuTableDynamic_SU ( [Group], SiteUnit, PlotNumber )"
    SqlText = SqlText & " SELECT USysBreakTable.Reference, USysBreakTable.BreakCode, USysBreakTable.PlotNumber"
    SqlText = SqlText & " FROM USysBreakTable;"
    DoCmd.RunSQL SqlText
    
    DoCmd.SetWarnings False
    SqlText = "DELETE * FROM DiagnosticGroup;"
    DoCmd.RunSQL SqlText
    DoCmd.SetWarnings True
    
    MakeDiagnosticData "SuTable"
    
    SqlText = "SELECT DISTINCT Reference FROM USysBreakTable;"
    Set MyRS = db.OpenRecordset(SqlText, dbOpenDynaset)
    If MyRS.RecordCount = 0 Then
        CreateDiagnostic = False
        GoTo MyExit
    End If
    MyRS.MoveLast
    MyRS.MoveFirst
    
    For i = 1 To MyRS.RecordCount
        On Error Resume Next
        DoCmd.DeleteObject acQuery, "MyRS5"
        
        If AverageType = 1 Then
            SqlText = "TRANSFORM First(DiagnosticPresenceClass([ncover]/[firstofnplots]) & ' - ' & SignifClass([TotalCover]/[FirstOfNPlots])) AS PresSignif"
        Else
            SqlText = "TRANSFORM First(DiagnosticPresenceClass([ncover]/[firstofnplots]) & ' - ' & SignifClass([TotalCover]/[nCover])) AS PresSignif"
        End If
            
        SqlText = SqlText & " SELECT MyRS4.Species"
        SqlText = SqlText & " FROM MyRS4"
        SqlText = SqlText & " WHERE (((MyRS4.Reference)='" & MyRS!Reference & "'))"
        SqlText = SqlText & " GROUP BY MyRS4.Species"
        SqlText = SqlText & " PIVOT MyRS4.SiteUnit;"
        Set MyQD = MyDB.CreateQueryDef("MyRS5", SqlText)
        Set CountFieldsRS = MyDB.OpenRecordset(SqlText)
        If CountFieldsRS.Fields.count < 3 Then GoTo Jump
    
        Diagnostic
    
        SqlText = "INSERT INTO DiagnosticGroup ( Species, Unit, Diagnosis, [Level])"
        SqlText = SqlText & " SELECT Diagnostic.Species, Diagnostic.Unit, Diagnostic.Diagnosis, '" & MyRS!Reference & "' AS Expr1"
        SqlText = SqlText & " FROM Diagnostic;"
        DoCmd.RunSQL SqlText
        
        AddEmptyUnit db, CountFieldsRS, MyRS!Reference
        CountFieldsRS.Close

Jump:
    
        MyRS.MoveNext

    Next i

' for each distinct Group in USysSuTableDynamic_SU run Diagnostic - rem to remove function MakeDiagnosticData
MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    MyRS.Close
    Set MyDB = Nothing
    On Error GoTo 0
    Exit Function
    
End Function

    End If

    On Error GoTo 0
    Exit Function
    
End Function

Private Function FindLowestMember(Unit As String, db As Database)

'Mar. 9, 2016
Dim MyRS As Recordset
Dim MyUnit As String
Dim i As Integer, Criteria As String, MyID As Integer, MyTest As String, BK As String
Dim TempRS As Recordset

    Set MyRS = db.OpenRecordset(CurrHierarchy(), dbOpenDynaset)
    
    Set TempRS = MyRS.Clone
    
    MyRS.MoveFirst
    Criteria = "Parent = " & Unit
    MyRS.FindFirst Criteria
    
    Do Until MyRS.NoMatch
    
        BK = MyRS.Bookmark
        
        If HasBrats(MyRS!ID, TempRS) Then
            MyRS.Bookmark = BK
            FindLowestMember MyRS!ID, db
        End If
        
        If MyRS!Level = 11 Then
            ReDim Preserve MyArray(UBound(MyArray) + 1)
            MyArray(UBound(MyArray)).Name = MyRS!Name
            MyArray(UBound(MyArray)).Parent = OriginalParent
        End If
        
        MyRS.Bookmark = BK
        MyRS.FindNext Criteria
    Loop
    
MyExit:
    On Error Resume Next
    TempRS.Close
    MyRS.Close
    On Error GoTo 0
    Exit Function
    
End Function
