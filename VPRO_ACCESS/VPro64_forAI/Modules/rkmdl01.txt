Private Sub CheckFonts(FormName As String)

Dim MyFrm As Form, Careful As Variant

Careful = MsgBox("Careful, this function will redesign the form USysAuditXL.  Continue?", vbYesNo, "VPro")
If Careful = vbNo Then Exit Sub

DoCmd.OpenForm FormName, acDesign
Set MyFrm = Application.Forms(FormName)
Dim ctl As control

On Error Resume Next

For Each ctl In MyFrm.Controls
    
    If ctl.FontName = "Arial" And ctl.FontSize = 7 Then
        Debug.Print ctl.Name
        ctl.FontName = "Arial"
        ctl.FontSize = 10
    End If

Next

DoCmd.Save acForm, MyFrm.Name
DoCmd.Close acForm, MyFrm.Name
MsgBox "Done."

End Sub
