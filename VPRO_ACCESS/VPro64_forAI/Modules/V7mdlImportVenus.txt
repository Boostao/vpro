Public Function ImportVENUS42()

'Dec 9, 2008
'this function imports VENUS >= 4.0
Dim MyDB As Database, RemoteDB As Database, VersionRecords As Recordset, i As Integer, ProjectRecords As Recordset
Dim Title As String, DataLocationName As String, TestTable As String, TestConnect As String
Dim nProjects As Integer, ProjectList As String, SqlText As String, MyQuery As QueryDef
Dim VersionPart1 As Variant, VersionPart2 As Variant, MyQD As QueryDef
Dim MyTest As Integer
On Error Resume Next
DoCmd.SetWarnings False

MyTest = WarnUser("Note that if plots already exist in the current project that have plot numbers the same as in the VENUS project your importing, the environment record will not be updated but the other tables will have duplicate records added.  You should always import data into an empty VPro project and then, if you wish to combine projects, use the VPro merge function.  Press Yes to continue or No to cancel.", "Import VENUS 4.2")
If MyTest = 1 Then GoTo MyExit

CleanUp42

On Error GoTo errImportVenus42

Set MyDB = CurrentDb()
sysVENUSProject = ""
    'get the location of the VENUS DB to be imported
    Title = "Pick a VENUS database to import"
    DataLocationName = GetFileName(Title, True)
    If Trim(DataLocationName) = "" Then GoTo MyExit
    If Trim(DataLocationName) = MyDB.Name Then MsgBox "You can't use projects from this database.", 64, ProgramName()
    If Trim(DataLocationName) = MyDB.Name Then Exit Function
    ZLSBeGone DataLocationName
    DoCmd.SetWarnings False
    
    'make sure that there aren't any conflicting tables
    For i = 0 To MyDB.TableDefs.count - 1
        TestTable = MyDB.TableDefs(i).Name
        TestConnect = MyDB.TableDefs(i).Connect
        If TestConnect <> "" Then
            If Right(TestTable, 11) = "USysVersion" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 17) = "tblProjectManager" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 6) = "tblEnv" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 14) = "tblMensuration" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 18) = "tblMineralHorizons" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 18) = "tblOrganicHorizons" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 7) = "tblSoil" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 25) = "tblTreeAttributesWildlife" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 10) = "tblVegMain" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 11) = "tblVegLayer" Then DoCmd.DeleteObject acTable, TestTable
            If Right(TestTable, 13) = "tblVegSpecies" Then DoCmd.DeleteObject acTable, TestTable
        End If
    Next i
    
    'link the VENUS tables
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "USysVersion", "USysVersion"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblProjectManager", "tblProjectManager"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblEnv", "tblEnv"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblMensuration", "tblMensuration"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblMineralHorizons", "tblMineralHorizons"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblOrganicHorizons", "tblOrganicHorizons"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblSoil", "tblSoil"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblTreeAttributesWildlife", "tblTreeAttributesWildlife"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblVegMain", "tblVegMain"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblVegLayer", "tblVegLayer"
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, "tblVegSpecies", "tblVegSpecies"

    'check the table version
    Set VersionRecords = MyDB.OpenRecordset("USysVersion")
    VersionRecords.MoveFirst
    VersionPart1 = VersionRecords("Version")
    VersionPart2 = VersionRecords("Beta")

    If VersionPart1 < 4 Then
        MsgBox "This database doesn't appear to be a VENUS 4.0"
        GoTo MyExit
    End If

    'build a list of projects
    Set ProjectRecords = MyDB.OpenRecordset("tblProjectManager")
    If Not ProjectRecords.BOF And Not ProjectRecords.EOF Then
        ProjectRecords.MoveLast
        nProjects = ProjectRecords.RecordCount
        If nProjects > 0 Then
            ProjectRecords.MoveFirst
            ProjectList = ""
            For i = 1 To nProjects
                If i > 1 Then ProjectList = ProjectList & ";" & ProjectRecords("ProjectName")
                If i = 1 Then ProjectList = ProjectRecords("ProjectName")
                ProjectRecords.MoveNext
            Next i
        End If
    End If
    
'user picks a project
    DoCmd.OpenForm "USysVENUSPickProject"
    Forms![USysVENUSPickProject]![ProjectList].RowSource = ProjectList
    Do Until sysVENUSProject > ""
        DoEvents
    Loop
    DoCmd.Close acForm, "USysVENUSPickProject"

    If sysVENUSProject = "None" Then GoTo MyExit
    
'import env data into current project
    SqlText = "INSERT INTO [" & CurrProject & "_ENV] ( [Date], ProjectID, PlotNumber, SiteSurveyor, Location, FSRegionDistrict, NtsMapSheet, UTMZone, UTMNorthing, UTMEasting, Latitude, Longitude, AirPhotoNum, XCoord, YCoord, MapUnit, PlotRepresenting, [Zone], SubZone, SiteSeries, TransDistrib, Ecosection, MoistureRegime, MoistureRegimeSub, NutrientRegime, SuccessionalStatus, StructuralStage, RealmClass, SiteDisturbance1, SiteDisturbance2, Elevation, SlopeGradient, Aspect, MesoSlopePosition, SurfaceShape, SurfaceTopographySize, SurfaceTopographyType, Exposure1, Photo, SubstrateDecWood, SubstrateBedRock, SubstrateRocks, SubstrateMineralSoil, SubstrateOrganicMatter, SubstrateWater, SiteNotes, UserSiteUnit, StrataCoverHerb, StrataCoverMoss, StrataCoverShrub, StrataCoverTree, VegSurveyor, VegNotes, BedrockGeology1, BedrockGeology2, BedrockGeology3, CoarseFragLith1, CoarseFragLith2, CoarseFragLith3, SoilSurveyor, SoilClassGroup, SoilClassSubGroup, HumusForm, HydroGeoSystem,"
    SqlText = SqlText & " RootingDepth , RootZoneParticleSize, RootRestrictingDepth, RootRestrictingType, WaterSource, SeepageDepth, SoilDrainage, FloodingRegimeFreq, pHMethodCodeOrganic, pHMethodCodeMineral, SoilNotes, HumusThickness, TerrainTextureSurf, TerrainTextureSubSurf, SurficialMaterialSurf, SurficialMaterialSubSurf, SurfaceExpSurf, SurfaceExpSubSurf, GeoMorProSurf, GeoMorProSubSurf, SitePlotQuality )"
    SqlText = SqlText & " SELECT tblEnv.Date, tblProjectManager.ProjectName, tblEnv.PlotNumber, tblEnv.SiteSurveyor, tblEnv.Location, tblEnv.Region, tblEnv.NtsMapSheet, tblEnv.UTMZone, tblEnv.UTMNorthing, tblEnv.UTMEasting, ConvertLongLatToDeg([LatitudeDegree],[LatitudeMinute],[LatitudeSecond]) AS Latitude, ConvertLongLatToDeg([LongitudeDegree],[LongitudeMinute],[LongitudeSecond]) AS Longitude, tblEnv.AirPhotoNum, tblEnv.XCoord, tblEnv.YCoord, tblEnv.MapUnit, tblEnv.PlotRepresenting, tblEnv.Zone, tblEnv.SubZone, tblEnv.SiteSeries, tblEnv.TransDistrib, tblEnv.Ecosection, tblEnv.MoistureRegime, tblEnv.MoistureRegimeSub, tblEnv.NutrientRegime, tblEnv.SuccessionalStatus, tblEnv.StructuralStage, tblEnv.RealmClass, tblEnv.SiteDisturbance1, tblEnv.SiteDisturbance2, tblEnv.Elevation, tblEnv.SlopeGradient, tblEnv.Aspect, tblEnv.MesoSlopePosition, tblEnv.SurfaceTopography, tblEnv.SurfaceSize, tblEnv.SurfaceType, tblEnv.Exposure, tblEnv.PhotoRollNum, tblEnv.SubstrateDecWood,"
    SqlText = SqlText & " tblEnv.SubstrateBedRock , tblEnv.SubstrateRocks, tblEnv.SubstrateMineralSoil, tblEnv.SubstrateOrganicMatter, tblEnv.SubstrateWater, tblEnv.SiteNotes, tblEnv.AssignedBECUnit, tblVegMain.StrataCoverHerb, tblVegMain.StrataCoverMoss, tblVegMain.StrataCoverShrub, tblVegMain.StrataCoverTree, tblVegMain.VegSurveyor, tblVegMain.VegNotes, tblSoil.BedrockGeology, tblSoil.BedrockGeology2, tblSoil.BedrockGeology3, tblSoil.CoarseFragLith, tblSoil.CoarseFragLith2, tblSoil.CoarseFragLith3, tblSoil.SoilSurveyor, tblSoil.SoilClassGroup, tblSoil.SoilClassSubGroup, tblSoil.HumusForm, tblSoil.HydroGeo, tblSoil.RootingDepth, tblSoil.RootZoneParticleSize, tblSoil.RootRestrictingDepth, tblSoil.RootRestrictingType, tblSoil.WaterSource, tblSoil.SeepageDepth, tblSoil.SoilDrainage, tblSoil.FloodingRegime, tblSoil.pHMethodCodeOrganic, tblSoil.pHMethodCodeMineral, tblSoil.SoilNotes, tblSoil.HumusThickness,"
    SqlText = SqlText & " CombineTerrain([TerrainSTexture1],[TerrainSTexture2],[TerrainSTexture3]) AS TerrainTextureSurf, CombineTerrain([TerrainUTexture1],[TerrainUTexture2],[TerrainUTexture3]) AS TerrainTextureSubSurf, CombineTerrain([TerrainSMaterial1],[TerrainSMaterial2],[TerrainSMaterial3]) AS SurficialMaterialSurf, CombineTerrain([TerrainUMaterial1],[TerrainUMaterial2],[TerrainUMaterial3]) AS SurficialMaterialSubSurf, CombineTerrain([TerrainSExpression1],[TerrainSExpression2],[TerrainSExpression3]) AS SurfaceExpSurf, CombineTerrain([TerrainUExpression1],[TerrainUExpression2],[TerrainUExpression3]) AS SurfaceExpSubSurf, CombineTerrain([TerrainSGeo1],[TerrainSGeo2],[TerrainSGeo3]) AS GeoMorProSurf, CombineTerrain([TerrainUGeo1],[TerrainUGeo2],[TerrainUGeo3]) AS GeoMorProSubSurf, tblProjectManager.PlotQuality"
    SqlText = SqlText & " FROM ((tblProjectManager INNER JOIN tblEnv ON tblProjectManager.ProjectID = tblEnv.ProjectID) INNER JOIN tblSoil ON tblEnv.PlotID = tblSoil.PlotID) INNER JOIN tblVegMain ON tblEnv.PlotID = tblVegMain.PlotID"
    SqlText = SqlText & " WHERE (((tblProjectManager.ProjectName)='" & sysVENUSProject & "'));"
    'Set MyQD = MyDB.CreateQueryDef("Q01-4", SqlText)
    DoCmd.RunSQL SqlText

'import humus records
    SqlText = "INSERT INTO [" & CurrProject & "_Humus] ( PlotNumber, Horizon, UpperDepth, LowerDepth, HumusStructureKind, HumusStructureDegree, vonPost, MycelAbundance, FecalAbundance, RootsAbundance, RootsSize, HumusFormpH, Comment )"
    SqlText = SqlText & " SELECT tblEnv.PlotNumber, tblOrganicHorizons.Horizon, tblOrganicHorizons.DepthUpper, tblOrganicHorizons.DepthLower, tblOrganicHorizons.StructureKind, tblOrganicHorizons.StructureDegree, tblOrganicHorizons.vonPost, tblOrganicHorizons.MycelAbundance, tblOrganicHorizons.FecalAbundance, tblOrganicHorizons.RootsAbundance, tblOrganicHorizons.RootsSize, tblOrganicHorizons.pHOrganic, tblOrganicHorizons.Comment"
    SqlText = SqlText & " FROM (tblProjectManager INNER JOIN tblEnv ON tblProjectManager.ProjectID = tblEnv.ProjectID) INNER JOIN tblOrganicHorizons ON tblEnv.PlotID = tblOrganicHorizons.PlotID"
    SqlText = SqlText & " WHERE (((tblProjectManager.ProjectName)='" & sysVENUSProject & "'));"
    'Set MyQD = MyDB.CreateQueryDef("Q02", SqlText)
    DoCmd.RunSQL SqlText

'import mineral data
    SqlText = "INSERT INTO [" & CurrProject & "_Mineral] ( PlotNumber, Horizon, UpperDepth, LowerDepth, Colour, ASP, Texture, PercentCoarseFragsGravel, PercentCoarseFragsCobbles, PercentCoarseFragsStones, PercentCoarseFragsTotal, PercentCoarseFragsShape, RootsAbundance, RootsSize, MineralStructureClass, MineralStructureKind, MineralFormpH, Comments )"
    SqlText = SqlText & " SELECT tblEnv.PlotNumber, tblMineralHorizons.Horizon, tblMineralHorizons.DepthUpper, tblMineralHorizons.DepthLower, tblMineralHorizons.Colour, tblMineralHorizons.ASP, tblMineralHorizons.Texture, tblMineralHorizons.PercentCoarseFragsGravel, tblMineralHorizons.PercentCoarseFragsCobbles, tblMineralHorizons.PercentCoarseFragsStones, tblMineralHorizons.PercentCoarseFragsTotal, tblMineralHorizons.PercentCoarseFragsShape, tblMineralHorizons.RootsAbundance, tblMineralHorizons.RootsSize, tblMineralHorizons.StructureClass, tblMineralHorizons.StructureKind, tblMineralHorizons.pHMineral, tblMineralHorizons.Comments"
    SqlText = SqlText & " FROM (tblProjectManager INNER JOIN tblEnv ON tblProjectManager.ProjectID = tblEnv.ProjectID) INNER JOIN tblMineralHorizons ON tblEnv.PlotID = tblMineralHorizons.PlotID"
    SqlText = SqlText & " WHERE (((tblProjectManager.ProjectName)='" & sysVENUSProject & "'));"
    'Set MyQD = MyDB.CreateQueryDef("Q03", SqlText)
    DoCmd.RunSQL SqlText

'import veg data - step 1
    SqlText = "TRANSFORM Sum(tblVegSpecies.Cover) AS SumOfCover"
    SqlText = SqlText & " SELECT tblEnv.PlotNumber, tblVegSpecies.Species"
    SqlText = SqlText & " FROM (tblProjectManager INNER JOIN (tblEnv INNER JOIN tblVegLayer ON tblEnv.PlotID = tblVegLayer.PlotID) ON tblProjectManager.ProjectID = tblEnv.ProjectID) INNER JOIN tblVegSpecies ON tblVegLayer.VegLayerID = tblVegSpecies.VegLayerID"
    SqlText = SqlText & " WHERE (((tblProjectManager.ProjectName) = '" & sysVENUSProject & "'))"
    SqlText = SqlText & " GROUP BY tblEnv.PlotNumber, tblVegSpecies.Species"
    SqlText = SqlText & " PIVOT tblVegLayer.Layer In (1,2,3,4,5,6,7,'A','B');"
    Set MyQuery = MyDB.CreateQueryDef("TempQuery", SqlText)

'import veg data - step 2
    SqlText = "INSERT INTO [" & CurrProject & "_Veg] ( PlotNumber, Species, Cover1, Cover2, Cover3, Cover4, Cover5, Cover6, Cover7, TotalA, TotalB )"
    SqlText = SqlText & " SELECT TempQuery.PlotNumber, TempQuery.Species, TempQuery.[1] AS cover1, TempQuery.[2] AS Cover2, TempQuery.[3] AS Cover3, TempQuery.[4] AS Cover4, TempQuery.[5] AS Cover5, TempQuery.[6] AS Cover6, TempQuery.[7] AS Cover7, TempQuery.A, TempQuery.B"
    SqlText = SqlText & " FROM TempQuery;"
    'Set MyQD = MyDB.CreateQueryDef("Q04", SqlText)
    DoCmd.RunSQL SqlText

LogImportedVenusPlots "4.x", sysVENUSProject

MsgBox "The VENUS project '" & sysVENUSProject & "' has been imported into the VPro project '" & CurrProject & "'.", vbInformation, "VPro"

MyExit:
    DoCmd.SetWarnings True
    On Error Resume Next
    VersionRecords.Close
    Set MyDB = Nothing
    CleanUp42
    Exit Function
errImportVenus42:
    DoCmd.SetWarnings True
    MsgBox Err.Description
    Resume MyExit
End Function

Private Sub LogImportedVenusPlots(VenusVer As String, vProjectName As String)

'Dec. 9, 2008
On Error GoTo errLogImportedPlots
On Error GoTo 0

Dim MyDB As Database, MyRS As Recordset, LogRS As Recordset, MyQD As QueryDef
Dim SqlText As String, ProjectName As String, MyProject As String, myUser As String
Dim i As Integer
Set MyDB = CurrentDb()
MyProject = CurrProject()
ProjectName = MyProject & "_Env"
myUser = CurrentUser()
Set LogRS = MyDB.OpenRecordset("USysAuditTrail", dbOpenDynaset)

If MyVersion = 5 Then
    SqlText = "SELECT PlotNumber, ProjectName"
    SqlText = SqlText & " FROM tblProjectManager INNER JOIN tblPlotData ON tblProjectManager.ProjectID = tblPlotData.ProjectID"
    SqlText = SqlText & " WHERE (((tblProjectManager.ProjectName)='" & vProjectName & "'));"
Else
    SqlText = "SELECT tblEnv.PlotNumber, tblProjectManager.ProjectName"
    SqlText = SqlText & " FROM tblProjectManager INNER JOIN tblEnv ON tblProjectManager.ProjectID = tblEnv.ProjectID"
    SqlText = SqlText & " WHERE (((tblProjectManager.ProjectName)='" & vProjectName & "'));"
End If
    'Set MyQD = MyDB.CreateQueryDef("AFred2", SqlText)
    Set MyRS = MyDB.OpenRecordset(SqlText)
    Debug.Print SqlText
    If MyRS.RecordCount = 0 Then GoTo MyExit
    MyRS.MoveLast
    MyRS.MoveFirst
    
    For i = 1 To MyRS.RecordCount
        LogRS.AddNew
        LogRS!Project = MyProject
        LogRS!User = myUser
        LogRS!PlotNumber = MyRS!PlotNumber
        LogRS!Table = "Environment"
        LogRS!EditField = "Imported VENUS ver. " & VenusVer & " record.  Project: " & vProjectName
        LogRS!EditWhen = Now()
        LogRS.Update
        MyRS.MoveNext
    Next i

MyExit:
    On Error Resume Next
    MyRS.Close
    LogRS.Close
    Set MyDB = Nothing
    Exit Sub
errLogImportedPlots:
    MsgBox Err.Description
    Resume MyExit
End Sub
