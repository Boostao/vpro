Function AddField(tblDef As TableDef, FieldName As String, FieldType As String)
'Sept. 26, 2003
If FieldType = "Text" Then
    tblDef.Fields.Append tblDef.CreateField(FieldName, dbText, 255)
End If
If FieldType = "Integer" Then
    tblDef.Fields.Append tblDef.CreateField(FieldName, dbInteger)
End If
If FieldType = "Long" Then
    tblDef.Fields.Append tblDef.CreateField(FieldName, dbLong)
End If
If FieldType = "Memo" Then
    tblDef.Fields.Append tblDef.CreateField(FieldName, dbMemo)
    tblDef.Fields(FieldName).AllowZeroLength = True
End If

End Function

Function CreateTableSet(Optional Extended As Boolean, Optional ProjectName As String, Optional RemoteDB As String)

'30 Sep 2019
'this code will create a VP08 version of VPro's data tables
Dim Msg As String, Title As String, DefValue As String, LumpTableName As String, LumpTable As String, i As Integer, TestTable As String
Dim MyTitle As String, DataLocationName As String
Dim MyTable As Recordset, MyDB As Database, TempDB As Database
Dim EnvTableName As String, VegTableName As String, OtherTableName As String, SoilHumusTableName As String, SoilMineralTableName As String, AuditTableName As String, MetaDataTableName As String, AdminTableName As String
Dim Answer As Variant
Set MyDB = CurrentDb()
    
'get a project name
    If ProjectName = "" Then
        Msg = "Enter a name for your new project.  Preferably, the name must start with a letter, contain no punctuation characters, preferably no spaces, and be less than 31 characters in length"
        Title = "New Project Name"
        DefValue = ""
        ProjectName = InputBox(Msg, Title, DefValue)
        If ProjectName = "" Then GoTo CreateTableSetWasCanceled
        If ProjectName = "" Then Exit Function
        If ProjectName = "Sample" Then Exit Function
    End If
    If TestForTableInVPro(ProjectName & "_Env") Then
        sysExportProject = "None999"
        MsgBox "A project of that name already exists.  Try a different name.", vbInformation, "VPro"
        GoTo MyExit
    End If

'test for too many characters in name
    If Len(ProjectName) > 31 Then
        Answer = MsgBox("Your project name is longer than Excel can use in reports, maximum 31 characters.  Do you wish to cancel?", vbYesNo, "VPro")
        If Answer = vbYes Then GoTo CreateTableSetWasCanceled
    End If

'get a db name
    sysRemoteProject = ProjectName
    If RemoteDB = "" Then
        MyTitle$ = "Pick a database to save your project in"
        DataLocationName = GetFileName(MyTitle$, False)
        If DataLocationName = "" Then GoTo CreateTableSetWasCanceled  'DeleteTables
        If Trim(DataLocationName) = MyDB.Name Then MsgBox "You can't create tables in this database.", 64, "VPro"
        If Trim(DataLocationName) = MyDB.Name Then Exit Function
        If Not FileExists(Trim(DataLocationName)) Then GoTo CantFindDatabase
AfterCreation:
        Set TempDB = DBEngine.Workspaces(0).OpenDatabase(DataLocationName)
        If Check4Table(TempDB, ProjectName & "_Env") Then
            sysExportProject = "None999"
            MsgBox "There is already a project of that name.  Try a different database or rename project.", vbInformation, "VPro"
            GoTo MyExit
        End If
        Set TempDB = Nothing
    Else
        DataLocationName = RemoteDB
    End If
    DoCmd.Hourglass True
    On Error GoTo 0
    
'set name variables
    EnvTableName = ProjectName & "_Env"
    VegTableName = ProjectName & "_Veg"
    OtherTableName = ProjectName & "_Other"
    SoilHumusTableName = ProjectName & "_Humus"
    SoilMineralTableName = ProjectName & "_Mineral"
    AuditTableName = ProjectName & "_Audit"
    MetaDataTableName = ProjectName & "_Metadata"
    AdminTableName = ProjectName & "_Admin"
    
'create tables
    DoCmd.CopyObject , EnvTableName, acTable, "USysEnvTable"
    DoCmd.CopyObject , VegTableName, acTable, "USysVegTable"
    DoCmd.CopyObject , OtherTableName, acTable, "USysOtherTable"
    DoCmd.CopyObject , SoilHumusTableName, acTable, "USysSoilHumusTable"
    DoCmd.CopyObject , SoilMineralTableName, acTable, "USysSoilMineralTable"
    DoCmd.CopyObject , AuditTableName, acTable, "USysAudit"
    DoCmd.CopyObject , MetaDataTableName, acTable, "USysMetadataTable"
    DoCmd.CopyObject , AdminTableName, acTable, "USysAdminTable"

    RefreshDatabaseWindow
'transfer tables to user's db
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, EnvTableName, EnvTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, VegTableName, VegTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, OtherTableName, OtherTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, SoilHumusTableName, SoilHumusTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, SoilMineralTableName, SoilMineralTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, AuditTableName, AuditTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, MetaDataTableName, MetaDataTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", DataLocationName, acTable, AdminTableName, AdminTableName

'delete temporary tables
    DoCmd.DeleteObject acTable, EnvTableName
    DoCmd.DeleteObject acTable, VegTableName
    DoCmd.DeleteObject acTable, OtherTableName
    DoCmd.DeleteObject acTable, SoilHumusTableName
    DoCmd.DeleteObject acTable, SoilMineralTableName
    DoCmd.DeleteObject acTable, AuditTableName
    DoCmd.DeleteObject acTable, MetaDataTableName
    DoCmd.DeleteObject acTable, AdminTableName

'link to new tables
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, EnvTableName, EnvTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, VegTableName, VegTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, OtherTableName, OtherTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, SoilHumusTableName, SoilHumusTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, SoilMineralTableName, SoilMineralTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, AuditTableName, AuditTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, MetaDataTableName, MetaDataTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", DataLocationName, acTable, AdminTableName, AdminTableName

'log the project
    LogNewProject ProjectName

'create relationships
    CreateRelationship DataLocationName, ProjectName, False

'refresh main menu float
    If IsLoaded("frmMainMenuFloat") Then
        Forms![frmMainMenuFloat].btnRefresh_Click
    End If

MyExit:
    On Error Resume Next
    DoCmd.Hourglass False
    DoCmd.Echo True
    Set MyDB = Nothing
    Set TempDB = Nothing
    Exit Function

CantFindDatabase:
Dim DefaultWorkspace As Workspace, NewDB As Database
Set DefaultWorkspace = DBEngine.Workspaces(0)

    Msg = ""
    Msg = "Couldn't find specified database" & Chr(13)
    Msg = Msg & "VPro will now create " & Trim(DataLocationName)
    MsgBox Msg, 64, "VPro"
    
    If Trim(DataLocationName) = MyDB.Name Then Exit Function
    DBEngine.Workspaces(0).CreateDatabase Trim(DataLocationName), DB_LANG_GENERAL
    
DataLocationName = Trim(DataLocationName)
Set TempDB = Nothing
GoTo AfterCreation

Set MyDB = Nothing
Exit Function

DeleteTables:
DoCmd.DeleteObject acTable, EnvTableName
DoCmd.DeleteObject acTable, VegTableName
DoCmd.DeleteObject acTable, OtherTableName
DoCmd.DeleteObject acTable, SoilHumusTableName
DoCmd.DeleteObject acTable, SoilMineralTableName
DoCmd.DeleteObject acTable, AuditTableName
DoCmd.DeleteObject acTable, MetaDataTableName
DoCmd.Echo True

Set MyDB = Nothing
Set TempDB = Nothing
Exit Function

CreateTableSetWasCanceled:
DoCmd.Echo True
Set MyDB = Nothing
Set TempDB = Nothing
Exit Function

End Function

Function TestForTableInVPro(TableName As String) As Boolean
'Oct. 6, 1998
'tests the current DB for a table
TestForTableInVPro = False
Dim MyDB As Database, i As Integer, strTestTable As String
Set MyDB = CurrentDb()
    
    For i = 0 To MyDB.TableDefs.count - 1
        strTestTable = MyDB.TableDefs(i).Name
        If strTestTable = TableName Then
            TestForTableInVPro = True
        End If
    Next i
    
Set MyDB = Nothing
End Function

Function CreateHerbariumTable()

'June 11, 2009
    On Error GoTo errCreateHerbariumTable
    Dim MyDB As Database, TestName As Integer
    Dim Msg As String, Title As String, DefValue As String, HerbariumTableName As String, HerbariumTable As String, i As Integer, TestTable As String
    Dim MyTitle As String, DataLocationName As String
    Set MyDB = CurrentDb()
    
    sysLumpingTable = ""
    Msg = "Enter a name for your new herbarium table.  The name must start with a letter, contain no punctuation characters, preferably no spaces, and be less than 40 characters in length"
    Title = "New Herbarium Table"
    DefValue = ""
    HerbariumTableName = InputBox(Msg, Title, DefValue)
    If HerbariumTableName = "" Then Exit Function
    If IsNull(HerbariumTableName) Then Exit Function
    TestName = InStr(HerbariumTableName, "_")
    If TestName > 0 Then Exit Function
    'If HerbariumTableName = "Sample" Then Exit Function
    HerbariumTable = HerbariumTableName & "_Herbarium"
    
    For i = 0 To MyDB.TableDefs.count - 1
        TestTable = MyDB.TableDefs(i).Name
        If TestTable = HerbariumTable Then
            MsgBox "Name already exists"
            Exit Function
        End If
    Next i

    MyTitle$ = ProgramName() & " - pick a database to save your table in"
    DataLocationName = GetFileName(MyTitle$, True)
    If Trim(DataLocationName) = "" Then Exit Function
    If Trim(DataLocationName) = MyDB.Name Then MsgBox "You can't create tables in this database.", 64, ProgramName()
    If Trim(DataLocationName) = MyDB.Name Then Exit Function
Jump:
DoCmd.CopyObject DataLocationName, HerbariumTable, acTable, "USysHerbarium"
DoCmd.TransferDatabase A_ATTACH, "Microsoft Access", DataLocationName, A_TABLE, HerbariumTable, HerbariumTable

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function
errCreateHerbariumTable:
    If Err.Number = 2501 Then
        Resume MyExit
    End If
    If Err.Number = 3024 Or Err.Number = 2766 Then
        DBEngine.Workspaces(0).CreateDatabase DataLocationName, dbLangGeneral
        Resume Jump
    End If
    MsgBox Err.Description & " - " & Err.Number
    Resume MyExit
End Function
