Public Sub AttachProject()

'July 8, 2015
On Error GoTo errAttachEnv

    Dim MyDB As Database, RemoteDB As Database ', MyRS As Recordset
    Dim RemoteDbName As String, ProjectList As String
    Dim MyList As Variant, i As Integer
    Dim MyProject As String
   
    Set MyDB = CurrentDb()
    ReDim sysSelectedItems(0)
    
    RemoteDbName = GetRemoteDb()
    
    If RemoteDbName = "" Then GoTo MyExit
    
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(RemoteDbName)
    
    ProjectList = BuildProjectPickList(RemoteDB, MyDB)

'test for items in list
    If ProjectList = "" Then
        MsgBox "No projects found or one with the same name may already be attached.", vbInformation, clsVProReg.ProgramName
        GoTo MyExit
    End If

'call the pick projects function (old method)
    Call PickProjects(ProjectList)

'do events until something to do
    Do
        DoEvents
    Loop Until UBound(sysSelectedItems) > 0
    
'now work, may as well quit
    If UBound(sysSelectedItems) = 999 Then GoTo MyExit
    
'go through list and attach projects
    For i = 1 To UBound(sysSelectedItems)
        MyProject = sysSelectedItems(i)
        AttachProjectTables MyProject, RemoteDB
    Next i

'reset array
    ReDim sysSelectedItems(0)
        
MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Set RemoteDB = Nothing
    Exit Sub
errAttachEnv:
    If Err.Number = 3343 Then
        MsgBox "This isn't Access 2007 database format"
        MsgBox Err.Description
        Resume Next
        Resume MyExit
    Else
        MsgBox Err.Description
        Resume MyExit
    End If
End Sub

Private Function GetRemoteDb() As String

'Apr. 14, 2015
Dim RemoteDB As Database
Dim MyFile As String, MyTitle As String
sysPickTable = "None"

'get file name
    MyTitle = clsVProReg.ProgramName
    MyFile = GetFileName(MyTitle, True)

'no file, quit
    If MyFile = "" Then Exit Function

'we now have a file name
    GetRemoteDb = MyFile

End Function

Private Function BuildProjectPickList(RemoteDB As Database, LocalDB As Database) As String

'Apr. 14, 2015
Dim i As Integer, strTestTable As String, strList As String, ListLen As Integer

'loop through the tables and make a list of projects
    For i = 0 To RemoteDB.TableDefs.count - 1
        strTestTable = RemoteDB.TableDefs(i).Name
        If InStr(1, strTestTable, "_Env") > 0 Then
            If Not TestForTableInVPro(LocalDB, strTestTable) Then
                strList = strList & Left(strTestTable, Len(strTestTable) - Len("_Env")) & ";"
            End If
        End If
    Next i

'return the list
    ListLen = Len(strList)
    If ListLen > 0 Then
        BuildProjectPickList = Left(strList, ListLen - 1)
    Else
        BuildProjectPickList = ""
    End If

End Function

Private Function TestForTableInVPro(MyDB As Database, TableName As String) As Boolean

'Apr. 14, 2015
'tests the current DB for a table
TestForTableInVPro = False
Dim i As Integer, strTestTable As String
    
    For i = 0 To MyDB.TableDefs.count - 1
        strTestTable = MyDB.TableDefs(i).Name
        If strTestTable = TableName Then
            TestForTableInVPro = True
        End If
    Next i
    
End Function

Private Function GetVProTableVersion(TestTable As String, MyDB As Database) As String

'Apr. 14, 2015
On Error GoTo errGetVersion
Dim MyTable As TableDef

Set MyTable = MyDB.TableDefs(TestTable)

'reade the description property
    GetVProTableVersion = MyTable.Properties("Description").Value

MyExit:
    On Error Resume Next
    Exit Function
errGetVersion:
    If Err.Number = 3265 Then
        GetVProTableVersion = "Unknown"
        Resume MyExit
    End If
    MsgBox Err.Description
    Resume MyExit
End Function
