Option Compare Database
Option Explicit

Function CreateBackUp()

'Mar. 12, 2021
'backs up the current project in a new DB
Dim MyDB As Database, NewDB As Database, Prompt As String, NewDBName As String, Msg As String
Dim ProjectName As String, EnvTableName As String, VegTableName As String, OtherTableName As String, AdminTableName As String
Dim SoilHumusTableName As String, SoilMineralTableName As String, AuditTableName As String, SortingTableName As String, MetaDataTableName As String
Dim SqlText As String
DoCmd.SetWarnings False
Set MyDB = CurrentDb()

    Prompt = "Create New Database"
    NewDBName = GetFileName(Prompt, False)
    If NewDBName = "" Then GoTo MyExit
    If NewDBName = MyDB.Name Then GoTo errCreateBackUp

    On Error GoTo errCreateBackUp
        
    Set NewDB = DBEngine.Workspaces(0).CreateDatabase(Trim(NewDBName), DB_LANG_GENERAL)
   
'set table names
    ProjectName = "BAK" & CurrProject()
    EnvTableName = ProjectName & "_Env"
    AdminTableName = ProjectName & "_Admin"
    VegTableName = ProjectName & "_Veg"
    OtherTableName = ProjectName & "_Other"
    SoilHumusTableName = ProjectName & "_Humus"
    SoilMineralTableName = ProjectName & "_Mineral"
    AuditTableName = ProjectName & "_Audit"
    MetaDataTableName = ProjectName & "_MetaData"

'transfer new tables to new db
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysEnvTable", EnvTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysAdminTable", AdminTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysVegTable", VegTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysOtherTable", OtherTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysSoilHumusTable", SoilHumusTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysSoilMineralTable", SoilMineralTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysAudit", AuditTableName
    DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysMetadataTable", MetaDataTableName

'make a link to the transfered tables
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, EnvTableName, EnvTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, AdminTableName, AdminTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, VegTableName, VegTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, OtherTableName, OtherTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, SoilHumusTableName, SoilHumusTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, SoilMineralTableName, SoilMineralTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, AuditTableName, AuditTableName
    DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, MetaDataTableName, MetaDataTableName

'time to merge
    SqlText = "INSERT INTO [" & EnvTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_Env].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_Env];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & AdminTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_Admin].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_Admin];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & VegTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_Veg].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_Veg];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & OtherTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_Other].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_Other];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & SoilHumusTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_Humus].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_Humus];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & SoilMineralTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_Mineral].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_Mineral];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & AuditTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_Audit].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_Audit];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & MetaDataTableName & "]"
    SqlText = SqlText & " SELECT [" & CurrProject() & "_MetaData].*"
    SqlText = SqlText & " FROM [" & CurrProject() & "_MetaData];"
    DoCmd.RunSQL SqlText

'remove links
    DoCmd.DeleteObject acTable, EnvTableName
    DoCmd.DeleteObject acTable, AdminTableName
    DoCmd.DeleteObject acTable, VegTableName
    DoCmd.DeleteObject acTable, OtherTableName
    DoCmd.DeleteObject acTable, SoilHumusTableName
    DoCmd.DeleteObject acTable, SoilMineralTableName
    DoCmd.DeleteObject acTable, AuditTableName
    DoCmd.DeleteObject acTable, MetaDataTableName
        
    MsgBox "Done!", vbInformation, "VPro"

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Set MyDB = Nothing
    Set NewDB = Nothing
    Exit Function
errCreateBackUp:
    DoCmd.SetWarnings True
    MsgBox Err.Description
    Resume MyExit

End Function


Function CreateHierarchyAndSuBackUp(BackUpType As String) 'SU, Hierarchy, Both

'May 27, 2009
'backs up the current hierarchy in a new DB
Dim MyDB As Database, NewDB As Database, Prompt As String, NewDBName As String, Msg As String
Dim HierarchyName As String, HierarchyBackName As String, SUName As String, SuBackName As String, Results As Integer
Dim SqlText As String
Results = 0
If CurrPlotlist = "None" And BackUpType = "Both" Then
    Results = MsgBox("VPro can backup both the current hierarchy and site unit list at once.  You don't have a current site unit list.  Click OK to continue or Cancel to stop.", vbOKCancel, "Hierarchy and Site Unit Table Backup")
    If Results = 2 Then
        GoTo MyExit
    Else
        BackUpType = "Hierarchy"
    End If
End If

DoCmd.SetWarnings False
Set MyDB = CurrentDb()

    Prompt = "Create New Database"
    NewDBName = GetFileName(Prompt, False)
    If NewDBName = "" Then GoTo MyExit
    If NewDBName = MyDB.Name Then GoTo errCreateBackUp

    On Error GoTo errCreateBackUp
    
    Set NewDB = DBEngine.Workspaces(0).CreateDatabase(Trim(NewDBName), DB_LANG_GENERAL)
   
    If BackUpType = "Both" Or BackUpType = "Hierarchy" Then
        HierarchyBackName = "BAK" & CurrHierarchy
        HierarchyName = CurrHierarchy
        DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysHierarchyTable", HierarchyBackName
        DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, HierarchyBackName, HierarchyBackName
        SqlText = "DELETE * FROM [" & HierarchyBackName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO [" & HierarchyBackName & "]"
        SqlText = SqlText & " SELECT [" & HierarchyName & "].*"
        SqlText = SqlText & " FROM [" & HierarchyName & "];"
        DoCmd.RunSQL SqlText
        DoCmd.DeleteObject acTable, HierarchyBackName
    End If

    If BackUpType = "Both" Or BackUpType = "SU" Then
        SuBackName = "BAK" & CurrPlotlist & "_SU"
        SUName = CurrPlotlist & "_SU"
        DoCmd.TransferDatabase acExport, "Microsoft Access", NewDBName, acTable, "USysSuTable", SuBackName
        DoCmd.TransferDatabase acLink, "Microsoft Access", NewDBName, acTable, SuBackName, SuBackName
        SqlText = "INSERT INTO [" & SuBackName & "]"
        SqlText = SqlText & " SELECT [" & SUName & "].*"
        SqlText = SqlText & " FROM [" & SUName & "];"
        DoCmd.RunSQL SqlText
        DoCmd.DeleteObject acTable, SuBackName
    End If
    
    MsgBox "Done!", vbInformation, ProgramName()

MyExit:
    On Error Resume Next
    DoCmd.DeleteObject acTable, SuBackName
    DoCmd.DeleteObject acTable, HierarchyBackName
    DoCmd.SetWarnings True
    Set MyDB = Nothing
    Set NewDB = Nothing
    Exit Function
errCreateBackUp:
    DoCmd.SetWarnings True
    MsgBox Err.Description
    Resume MyExit

End Function
