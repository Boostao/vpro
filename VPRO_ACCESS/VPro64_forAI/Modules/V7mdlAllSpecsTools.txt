Public Function ChangeAllSpecsTable()

    Dim MyDB As Database, RemoteDB As Database ', MyRS As Recordset
    Dim RemoteDbName As String, Success As Boolean, Question As Variant
    Dim MyList As Variant, i As Integer
    Dim MyProject As String
    
    Question = MsgBox("There are dangers involved in not using the 'Standard' version of USysAllSpecs.  If you understand these dangers and wish to proceed, click Yes.", vbYesNo, "VPro")
    If Question = vbNo Then GoTo MyExit
    Success = False
    Set MyDB = CurrentDb()
    RemoteDbName = GetRemoteDb()
    If RemoteDbName = "" Then GoTo MyExit
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(RemoteDbName)

    For i = 0 To RemoteDB.TableDefs.count - 1
        If RemoteDB.TableDefs(i).Name = "USysAllSpecs" Then
            DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, "USysAllSpecs", "USysAllSpecsRK"
            DoCmd.DeleteObject acTable, "USysAllSpecs"
            DoCmd.Rename "USysAllSpecs", acTable, "USysAllSpecsRK"
            Success = True
            Exit For
        End If
    Next i

If Not Success Then
    MsgBox "VPro could not find a USysAllSpecs table in " & RemoteDB.Name & ".  Your species table has not been changed.", vbInformation, "VPro"
Else
    MsgBox "The USysAllSpecs table from '" & RemoteDB.Name & "' has been" & Chr(13) & "attached.  It has the version name '" & AllSpecsVersion & "'.", vbInformation, "VPro"
    LogProjectOut
    LogProjectIn
End If

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Set RemoteDB = Nothing
    Exit Function

End Function


Public Function AllSpecsVersion() As String

'June 7, 2006
On Error GoTo errAllSpecsVersion
Dim MyDB As Database, MyTable As TableDef
Dim TestTable As String
Set MyDB = CurrentDb

TestTable = "USysAllSpecs"
Set MyTable = MyDB.TableDefs(TestTable)

AllSpecsVersion = MyTable.Properties("Description").Value

MyExit:
    On Error Resume Next
    Set MyTable = Nothing
    Set MyDB = Nothing
    Exit Function
errAllSpecsVersion:
    AllSpecsVersion = "Unknown"
    Resume MyExit
End Function

Function ProjectVersion()
'June 8, 2006
Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String
ProjectVersion = ""

Set MyDB = CurrentDb
SqlText = "SELECT Table, EditWhen, AfterEdit"
SqlText = SqlText & " FROM [" & clsVProReg.CurrProject & "_Audit]"
SqlText = SqlText & " WHERE (((Table)='USysAllSpecs'))"
SqlText = SqlText & " ORDER BY EditWhen DESC;"
Set MyRS = MyDB.OpenRecordset(SqlText)

If MyRS.RecordCount = 0 Then
    ProjectVersion = "Unknown"
    GoTo MyExit
End If

If IsNull(MyRS!AfterEdit) Or MyRS!AfterEdit = "" Then
    ProjectVersion = "Unknown"
Else
    ProjectVersion = MyRS!AfterEdit
End If

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
End Function
