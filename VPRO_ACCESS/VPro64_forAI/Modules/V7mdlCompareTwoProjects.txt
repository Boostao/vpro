Private Function TestBeforeRunning(Proj1 As String, Proj2 As String) As Boolean

'Apr. 13, 2015
On Error GoTo errTestBeforeRunning
Dim CurrProj As String

'close forms
    If IsLoaded("FS882-6x4") Then DoCmd.Close acForm, "FS882-6x4"
    If IsLoaded("FS882-8x6") Then DoCmd.Close acForm, "FS882-8x6"
    If IsLoaded("FS882-1x1") Then DoCmd.Close acForm, "FS882-1x1"

'add indexes
    CurrProj = Proj1 & "_Env"
    If Not HasPrimary(Proj1 & "_Env") Then AddIndex Proj1 & "_Env"
    CurrProj = Proj2 & "_Env"
    If Not HasPrimary(Proj2 & "_Env") Then AddIndex Proj2 & "_Env"
    CurrProj = Proj1 & "_Veg"
    If Not HasPrimary(Proj1 & "_Veg") Then AddIndex Proj1 & "_Veg"
    CurrProj = Proj2 & "_Veg"
    If Not HasPrimary(Proj2 & "_Veg") Then AddIndex Proj2 & "_Veg"
    CurrProj = Proj1 & "_Humus"
    If Not HasPrimary(Proj1 & "_Humus") Then AddIndex Proj1 & "_Humus"
    CurrProj = Proj2 & "_Humus"
    If Not HasPrimary(Proj2 & "_Humus") Then AddIndex Proj2 & "_Humus"
    CurrProj = Proj1 & "_Mineral"
    If Not HasPrimary(Proj1 & "_Mineral") Then AddIndex Proj1 & "_Mineral"
    CurrProj = Proj2 & "_Mineral"
    If Not HasPrimary(Proj2 & "_Mineral") Then AddIndex Proj2 & "_Mineral"
    CurrProj = Proj1 & "_Metadata"
    If Not HasPrimary(Proj1 & "_Metadata") Then AddIndex Proj1 & "_Metadata"
    CurrProj = Proj2 & "_Mineral"
    If Not HasPrimary(Proj2 & "_Metadata") Then AddIndex Proj2 & "_Metadata"
    
    TestBeforeRunning = True
    
MyExit:
    Exit Function
errTestBeforeRunning:
    TestBeforeRunning = False
    CleanUp Proj1, Proj2
    MsgBox "Unable to create a necessary index on table " & CurrProj & ".  Please fix and try again.", vbInformation, "VPro"
    Resume MyExit

End Function

Private Sub MakeDataTable()

'Apr. 13, 2015
Dim MyTbl As TableDef, MyDB As Database
Set MyDB = CurrentDb

    On Error Resume Next
    DoCmd.DeleteObject acTable, "tblCompare"
    On Error GoTo 0
    
'define table
    Set MyTbl = MyDB.CreateTableDef("tblCompare")
    MyTbl.Fields.Append MyTbl.CreateField("Table", dbText, 10)
    MyTbl.Fields.Append MyTbl.CreateField("PlotNumber", dbText, 10)
    MyTbl.Fields.Append MyTbl.CreateField("FieldName", dbText, 64)
    MyTbl.Fields.Append MyTbl.CreateField("SeekData1", dbText, 255)
    MyTbl.Fields.Append MyTbl.CreateField("SeekData2", dbText, 255)
    MyTbl.Fields.Append MyTbl.CreateField("Comment", dbText, 255)
    
'create table
    MyDB.TableDefs.Append MyTbl

MyExit:
    Set MyTbl = Nothing
    Set MyDB = Nothing

End Sub
Private Function GetDbName(Project As String) As String

'Apr. 13, 2015
Dim MyDB As Database, MyTD As TableDef
Set MyDB = CurrentDb
Set MyTD = MyDB.TableDefs(Project)
    
    GetDbName = Mid(MyTD.Connect, 11)

    Set MyDB = Nothing
    Set MyTD = Nothing

End Function

Private Sub AddIndex(VegTable As String)

'Apr. 13, 2015
Dim MyDB As Database, MyTD As TableDef, MyIndex As index, RemoteDB As Database, RemoteName As String

Set MyDB = CurrentDb
RemoteName = GetDbName(VegTable)
Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(RemoteName)
Set MyTD = RemoteDB.TableDefs(VegTable)
Set MyIndex = MyTD.CreateIndex("rkPrimaryKey")

    MyIndex.Unique = True
    
    If Right(VegTable, 3) = "Veg" Then
        MyIndex.Fields.Append MyIndex.CreateField("PlotNumber")
        MyIndex.Fields.Append MyIndex.CreateField("Species")
    End If

    If Right(VegTable, 3) = "Env" Then
        MyIndex.Fields.Append MyIndex.CreateField("PlotNumber")
    End If

    If Right(VegTable, 9) = "_Metadata" Then
        MyIndex.Fields.Append MyIndex.CreateField("ProjectID")
    End If

    If Right(VegTable, 6) = "_Humus" Then
        MyIndex.Fields.Append MyIndex.CreateField("PlotNumber")
        MyIndex.Fields.Append MyIndex.CreateField("Horizon")
    End If

    If Right(VegTable, 8) = "_Mineral" Then
        MyIndex.Fields.Append MyIndex.CreateField("PlotNumber")
        MyIndex.Fields.Append MyIndex.CreateField("Horizon")
    End If

MyTD.Indexes.Append MyIndex

End Sub

Private Sub RemoveIndexes(TableName As String)

'Apr. 13, 2015
On Error GoTo errRemoveIndexes
Dim MyTD As TableDef, RemoteDB As Database, RemoteName As String

    RemoteName = GetDbName(TableName)
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(RemoteName)
    
    RemoteDB.TableDefs(TableName).Indexes.Delete "rkPrimaryKey"

MyExit:
    Set MyTD = Nothing
    Set RemoteDB = Nothing
    Exit Sub
errRemoveIndexes:
    If Err.Number = 3265 Then
        'do nothing
    Else
        MsgBox Err.Number & Chr(13) & Err.Description, vbInformation, "VPro"
    End If
    Resume MyExit
    
End Sub
Private Function HasPrimary(VegTable As String) As Boolean

'Apr. 13, 2015
On Error GoTo errHasPrimary
Dim MyDB As Database, MyTD As TableDef
Set MyDB = CurrentDb
Set MyTD = MyDB.TableDefs(VegTable)

    If MyTD.Indexes("rkPrimaryKey").Fields <> "" Then
        HasPrimary = True
    End If

MyExit:
    Set MyTD = Nothing
    Set MyDB = Nothing
    Exit Function
errHasPrimary:
    HasPrimary = False
    Resume MyExit
End Function

Public Sub CompareProjectData(Proj1 As String, Proj2 As String, optMissingPlots As Boolean)

'Apr. 13, 2015
Dim MyFld As Field, i As Integer
Dim MyDB As Database, MyRS1 As Recordset, MyRS2 As Recordset, MyRS As Recordset
Dim Remote1 As String, Remote2 As String
Dim RemoteDB1 As Database, RemoteDB2 As Database
Dim Criteria As String
Dim MyTime As New clsStopWatch, MyTime2 As New clsStopWatch
'On Error GoTo errCompareProjectData
DoCmd.Hourglass True

    MyTime2.StartTimer
    MyTime.StartTimer
    Set MyDB = CurrentDb

'add and test indexes
    If Not TestBeforeRunning(Proj1, Proj2) Then
        GoTo MyExit
    End If
   
'create references to each project's database
    Remote1 = GetDbName(Proj1 & "_Env")
    Remote2 = GetDbName(Proj2 & "_Env")
    If Remote1 = "" Then
        Set RemoteDB1 = CurrentDb
    Else
        Set RemoteDB1 = DBEngine.Workspaces(0).OpenDatabase(Remote1)
    End If
    If Remote2 = "" Then
        Set RemoteDB2 = CurrentDb
    Else
        Set RemoteDB2 = DBEngine.Workspaces(0).OpenDatabase(Remote2)
    End If

'create a table and a connection to store the data in
    MakeDataTable
    Set MyRS = MyDB.OpenRecordset("tblCompare")
    Forms("frmCompareTwoProjects").lblStatus.Caption = "Creating references and preparing data"
    Forms("frmCompareTwoProjects").Repaint

'Extract Env differences
    Set MyRS1 = RemoteDB1.OpenRecordset(Proj1 & "_Env") ', dbOpenTable)
    Set MyRS2 = RemoteDB2.OpenRecordset(Proj2 & "_Env", dbOpenTable)
    Do Until MyRS1.EOF
        MyRS2.index = "rkPrimaryKey"
        MyRS2.Seek "=", MyRS1!PlotNumber
        If Not MyRS2.NoMatch Then
            For Each MyFld In MyRS1.Fields
                If Nz(MyRS1(MyFld.Name).Value) = Nz(MyRS2(MyFld.Name).Value) Then
                    'do nothing
                Else
                    MyRS.AddNew
                    MyRS!Table = "Env"
                    MyRS!PlotNumber = MyRS1!PlotNumber
                    MyRS!FieldName = MyFld.Name
                    MyRS!SeekData1 = MyRS1(MyFld.Name).Value
                    MyRS!SeekData2 = MyRS2(MyFld.Name).Value
                    MyRS.Update
                End If
            Next MyFld
        Else
            MyRS.AddNew
            MyRS!Table = "Env"
            MyRS!PlotNumber = MyRS1!PlotNumber
            MyRS!Comment = "Plot """ & MyRS1!PlotNumber & """ not found in """ & Proj2 & """"
            MyRS.Update
        End If
        MyRS1.MoveNext
    Loop
    DoEvents
    Forms("frmCompareTwoProjects").lblStatus.Caption = "Completed env comparision and starting veg (" & MyTime.EndTimer / 1000 & " sec.)"
    Forms("frmCompareTwoProjects").Repaint
    DoEvents
    Debug.Print "Env: " & MyTime.EndTimer
    MyTime.StartTimer

'Extract Veg differences
    Set MyRS1 = RemoteDB1.OpenRecordset(Proj1 & "_Veg") ', dbOpenTable)
    Set MyRS2 = RemoteDB2.OpenRecordset(Proj2 & "_veg", dbOpenTable)
    Do Until MyRS1.EOF
        MyRS2.index = "rkPrimaryKey"
        MyRS2.Seek "=", MyRS1!PlotNumber, MyRS1!Species
        If Not MyRS2.NoMatch Then
            For Each MyFld In MyRS1.Fields
                If MyFld.Name <> "ID" Then
                    If Nz(MyRS1(MyFld.Name).Value) = Nz(MyRS2(MyFld.Name).Value) Then
                        'do nothing
                    Else
                        MyRS.AddNew
                        MyRS!Table = "Veg"
                        MyRS!PlotNumber = MyRS1!PlotNumber
                        MyRS!FieldName = MyFld.Name
                        MyRS!SeekData1 = MyRS1(MyFld.Name).Value
                        MyRS!SeekData2 = MyRS2(MyFld.Name).Value
                        MyRS!Comment = "For species: " & MyRS1!Species
                        MyRS.Update
                    End If
                End If
            Next MyFld
        Else
            MyRS.AddNew
            MyRS!Table = "Veg"
            MyRS!PlotNumber = MyRS1!PlotNumber
            MyRS!Comment = "Species """ & MyRS1!Species & """ not found in """ & Proj2 & """"
            MyRS.Update
        End If
        MyRS1.MoveNext
    Loop
    DoEvents
    Forms("frmCompareTwoProjects").lblStatus.Caption = "Completed veg comparision and starting humus (" & MyTime.EndTimer / 1000 & " sec.)"
    Forms("frmCompareTwoProjects").Repaint
    DoEvents
    Debug.Print "Veg: " & MyTime.EndTimer
    MyTime.StartTimer

'Extract Humus differences
    Set MyRS1 = RemoteDB1.OpenRecordset(Proj1 & "_Humus")
    Set MyRS2 = RemoteDB2.OpenRecordset(Proj2 & "_Humus", dbOpenTable)
    Do Until MyRS1.EOF
        MyRS2.index = "rkPrimaryKey"
        MyRS2.Seek "=", MyRS1!PlotNumber, MyRS1!Horizon
        If Not MyRS2.NoMatch Then
            For Each MyFld In MyRS1.Fields
                If Nz(MyRS1(MyFld.Name).Value) = Nz(MyRS2(MyFld.Name).Value) Or MyFld.Name = "ID" Then
                    'do nothing
                Else
                    MyRS.AddNew
                    MyRS!Table = "Humus"
                    MyRS!PlotNumber = MyRS1!PlotNumber
                    MyRS!FieldName = MyFld.Name
                    MyRS!SeekData1 = MyRS1(MyFld.Name).Value
                    MyRS!SeekData2 = MyRS2(MyFld.Name).Value
                    MyRS!Comment = "For horizon: " & MyRS1!Horizon
                    MyRS.Update
                End If
            Next MyFld
        Else
            MyRS.AddNew
            MyRS!Table = "Humus"
            MyRS!PlotNumber = MyRS1!PlotNumber
            MyRS!Comment = "Horizon """ & MyRS1!Horizon & """ not found in """ & Proj2 & """"
            MyRS.Update
        End If
        MyRS1.MoveNext
    Loop
    DoEvents
    Forms("frmCompareTwoProjects").lblStatus.Caption = "Completed humus comparision and starting mineral (" & MyTime.EndTimer / 1000 & " sec.)"
    Forms("frmCompareTwoProjects").Repaint
    DoEvents
    Debug.Print "Humus: " & MyTime.EndTimer
    MyTime.StartTimer

'Extract Mineral differences
    Set MyRS1 = RemoteDB1.OpenRecordset(Proj1 & "_Mineral")
    Set MyRS2 = RemoteDB2.OpenRecordset(Proj2 & "_Mineral", dbOpenTable)
    Do Until MyRS1.EOF
        MyRS2.index = "rkPrimaryKey"
        MyRS2.Seek "=", MyRS1!PlotNumber, MyRS1!Horizon
        If Not MyRS2.NoMatch Then
            For Each MyFld In MyRS1.Fields
                If Nz(MyRS1(MyFld.Name).Value) = Nz(MyRS2(MyFld.Name).Value) Or MyFld.Name = "ID" Then
                    'do nothing
                Else
                    MyRS.AddNew
                    MyRS!Table = "Mineral"
                    MyRS!PlotNumber = MyRS1!PlotNumber
                    MyRS!FieldName = MyFld.Name
                    MyRS!SeekData1 = MyRS1(MyFld.Name).Value
                    MyRS!SeekData2 = MyRS2(MyFld.Name).Value
                    MyRS!Comment = "For horizon: " & MyRS1!Horizon
                    MyRS.Update
                End If
            Next MyFld
        Else
            MyRS.AddNew
            MyRS!Table = "Mineral"
            MyRS!PlotNumber = MyRS1!PlotNumber
            MyRS!Comment = "Horizon """ & MyRS1!Horizon & """ not found in """ & Proj2 & """"
            MyRS.Update
        End If
        MyRS1.MoveNext
    Loop
    DoEvents
    Forms("frmCompareTwoProjects").lblStatus.Caption = "Completed mineral comparision and starting metadata (" & MyTime.EndTimer / 1000 & " sec.)"
    Forms("frmCompareTwoProjects").Repaint
    DoEvents
    Debug.Print "Mineral: " & MyTime.EndTimer
    MyTime.StartTimer

'Extract Metadata differences
    Set MyRS1 = RemoteDB1.OpenRecordset(Proj1 & "_Metadata")
    Set MyRS2 = RemoteDB2.OpenRecordset(Proj2 & "_Metadata", dbOpenTable)
    Do Until MyRS1.EOF
        MyRS2.index = "rkPrimaryKey"
        MyRS2.Seek "=", MyRS1!ProjectID
        If Not MyRS2.NoMatch Then
            For Each MyFld In MyRS1.Fields
                If Nz(MyRS1(MyFld.Name).Value) = Nz(MyRS2(MyFld.Name).Value) Or MyFld.Name = "ID" Then
                    'do nothing
                Else
                    MyRS.AddNew
                    MyRS!Table = "Metadata"
                    MyRS!PlotNumber = MyRS1!ProjectID
                    MyRS!FieldName = MyFld.Name
                    MyRS!SeekData1 = MyRS1(MyFld.Name).Value
                    MyRS!SeekData2 = MyRS2(MyFld.Name).Value
                    MyRS!Comment = "For project: " & MyRS1!ProjectID
                    MyRS.Update
                End If
            Next MyFld
        Else
            MyRS.AddNew
            MyRS!Table = "Metadata"
            MyRS!PlotNumber = MyRS1!ProjectID
            MyRS!Comment = "Project """ & MyRS1!ProjectID & """ not found in """ & Proj2 & """"
            MyRS.Update
        End If
        MyRS1.MoveNext
    Loop
    DoEvents
    Forms("frmCompareTwoProjects").lblStatus.Caption = "Completed metadata comparision and starting optional tasks (" & MyTime.EndTimer / 1000 & " sec.)"
    Forms("frmCompareTwoProjects").Repaint
    DoEvents
    Debug.Print "Metadata: " & MyTime.EndTimer
    MyTime.StartTimer

    If optMissingPlots Then
        FindMissingPlots Proj1, Proj2, MyRS
    End If

'all done comparing
    Debug.Print "Done collecting data: " & MyTime.EndTimer
    Forms("frmCompareTwoProjects").lblStatus.Caption = "Completed all comparisions and starting to report (" & MyTime.EndTimer / 1000 & " sec.)"
    Forms("frmCompareTwoProjects").Repaint
    MyTime.StartTimer
    
'send report to Excel
    MyRS.Close
    Set MyRS = MyDB.OpenRecordset("tblCompare")
    If MyRS.RecordCount > 0 Then
        BuildReport Proj1, Proj2, MyRS
    Else
        MsgBox "Nothing to do.  I quit!", vbInformation, "VPro"
    End If

'debug.print the times and reset timers
    Debug.Print "Done generating report: " & MyTime.EndTimer
    Debug.Print "Total Time: " & MyTime2.EndTimer
    Set MyTime = Nothing
    Set MyTime2 = Nothing

MyExit:
    On Error Resume Next
    DoCmd.Hourglass False
    Forms("frmCompareTwoProjects").lblStatus.Caption = ""
    Forms("frmCompareTwoProjects").Repaint
    MyRS.Close
    MyRS1.Close
    MyRS2.Close
    RemoteDB1.Close
    RemoteDB2.Close
    Set RemoteDB1 = Nothing
    Set RemoteDB2 = Nothing
    Set MyDB = Nothing
    DoCmd.Close acTable, "tblCompare"
    CleanUp Proj1, Proj2
    Exit Sub
errCompareProjectData:
    MsgBox Err.Description, vbInformation, "VPro"
    Resume MyExit

End Sub

Private Sub BuildReport(Proj1 As String, Proj2 As String, MyRS As Recordset)

'Apr. 13, 2015
Dim xl As Excel.Application, i As Long
CreateExcel xl
xl.Visible = True
xl.Workbooks.Add

    xl.Cells(3, 1) = "Table"
    xl.Cells(3, 2) = "Plot Number"
    xl.Cells(3, 3) = "Field Name"
    xl.Cells(3, 4) = Proj1
    xl.Cells(3, 5) = Proj2
    xl.Cells(3, 6) = "Comment"
    
    xl.Cells(4, 1).CopyFromRecordset MyRS

    FormatReport xl

    xl.Cells(1, 1) = "This is a report of differences found between project """ & Proj1 & """ and project """ & Proj2 & """"
    xl.Cells(2, 1) = "Created: " & Now()

MyExit:
End Sub

Private Sub FindMissingPlots(Proj1 As String, Proj2 As String, MyRS2 As Recordset)

'Apr. 13, 2015
Dim SqlText As String, MyRS As Recordset
On Error GoTo errFindMissingPlots

    SqlText = "SELECT [" & Proj2 & "_Env].PlotNumber"
    SqlText = SqlText & " FROM [" & Proj2 & "_Env] LEFT JOIN [" & Proj1 & "_Env] ON [" & Proj2 & "_Env].PlotNumber = [" & Proj1 & "_Env].PlotNumber"
    SqlText = SqlText & " WHERE ((([" & Proj1 & "_Env].PlotNumber) Is Null));"
    Set MyRS = CurrentDb.OpenRecordset(SqlText)
    
    Do Until MyRS.EOF
        MyRS2.AddNew
        MyRS2!Table = "Env"
        MyRS2!PlotNumber = MyRS!PlotNumber
        MyRS2!Comment = "Plot """ & MyRS!PlotNumber & """ not found in """ & Proj1 & """"
        MyRS2.Update
        MyRS.MoveNext
    Loop

MyExit:
    On Error Resume Next
    MyRS.Close
    Exit Sub
errFindMissingPlots:
    MsgBox Err.Description
    Resume MyExit

End Sub

Private Sub FormatReport(xl As Excel.Application)

'Apr. 13, 2015
    xl.Range("A3:F3").Select

    With xl.Selection.Interior
        .Pattern = xlSolid
        .PatternColorIndex = xlAutomatic
        .ThemeColor = xlThemeColorDark1
        .TintAndShade = -0.249977111117893
        .PatternTintAndShade = 0
    End With
    With xl.Selection.Borders(xlEdgeLeft)
        .LineStyle = xlContinuous
        .ColorIndex = xlAutomatic
        .TintAndShade = 0
        .Weight = xlThin
    End With
    With xl.Selection.Borders(xlEdgeTop)
        .LineStyle = xlContinuous
        .ColorIndex = xlAutomatic
        .TintAndShade = 0
        .Weight = xlThin
    End With
    With xl.Selection.Borders(xlEdgeBottom)
        .LineStyle = xlContinuous
        .ColorIndex = xlAutomatic
        .TintAndShade = 0
        .Weight = xlThin
    End With
    With xl.Selection.Borders(xlEdgeRight)
        .LineStyle = xlContinuous
        .ColorIndex = xlAutomatic
        .TintAndShade = 0
        .Weight = xlThin
    End With
    With xl.Selection.Borders(xlInsideVertical)
        .LineStyle = xlContinuous
        .ColorIndex = xlAutomatic
        .TintAndShade = 0
        .Weight = xlThin
    End With

    xl.Columns.AutoFit

End Sub
