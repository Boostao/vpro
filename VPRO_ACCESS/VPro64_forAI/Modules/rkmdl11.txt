Option Compare Database

Private Sub FlatHierarchyAndPlots()

BuildTable

Dim SqlText As String
DoCmd.SetWarnings False
SqlText = "INSERT INTO tblFlatHierarchyAndPlots ( PlotNumber, Level11 )"
SqlText = SqlText & " SELECT Sample_SU.PlotNumber, Sample_SU.SiteUnit"
SqlText = SqlText & " FROM Sample_SU;"
DoCmd.RunSQL SqlText

For i = 11 To 2 Step -1

    SqlText = "UPDATE (tblFlatHierarchyAndPlots INNER JOIN Sample_Hierarchy ON tblFlatHierarchyAndPlots.Level" & i & " = Sample_Hierarchy.Name) INNER JOIN Sample_Hierarchy AS Sample_Hierarchy_1 ON Sample_Hierarchy.Parent = Sample_Hierarchy_1.ID SET tblFlatHierarchyAndPlots.Level" & i - 1 & " = [Sample_Hierarchy_1]![Name];"
    DoCmd.RunSQL SqlText

Next i

DoCmd.SetWarnings True
MsgBox "Done", vbInformation, "VPro"

End Sub

Private Sub BuildTable()

On Error Resume Next
DoCmd.DeleteObject acTable, "tblFlatHierarchyAndPlots"
On Error GoTo 0

Dim MyDB As Database, MyTable As TableDef, MyFld As Field
Dim i As Integer
Set MyDB = CurrentDb()

Set MyTable = MyDB.CreateTableDef("tblFlatHierarchyAndPlots")
MyTable.Fields.Append MyTable.CreateField("PlotNumber", dbText, 8)

For i = 11 To 1 Step -1
    MyTable.Fields.Append MyTable.CreateField("Level" & i, dbText, 255)
Next i

MyDB.TableDefs.Append MyTable

MyExit:
    Set MyDB = Nothing

End Sub
