Public Const dbRelationCascadeNull As Long = &H2000

Function DeleteRelationship(RemoteDB As String, Project As String)
'Sept. 30, 2019

On Error GoTo errDeleteRelationship
Dim MyDB As Database, MyRel As Relation, MyFld As Field, RelationName As String
Dim VegTable As String, HumusTable As String, MineralTable As String, MetadataTable As String, AdminTable As String
Dim AuditTable As String, OtherTable As String, i As Integer
Dim MyRelName As String, X As Integer
Set MyDB = CurrentDb()
Dim EnvTable As String, db As Database ', MyWS As Workspace
EnvTable = Project
    
Set db = DBEngine.Workspaces(0).OpenDatabase(RemoteDB)

'set variables
    RelationName = EnvTable
    EnvTable = RelationName & "_Env"
    VegTable = RelationName & "_Veg"
    HumusTable = RelationName & "_Humus"
    MineralTable = RelationName & "_Mineral"
    OtherTable = RelationName & "_Other"
    AuditTable = RelationName & "_Audit"
    MetadataTable = RelationName & "_Metadata"
    AdminTable = RelationName & "_Admin"

'bail if there's no relation
    If FindRelation(RelationName) = True Then GoTo MyExit

'delete old relationships
    If db.Relations.count > 0 Then
        On Error Resume Next
        For i = 0 To 7
            db.Relations.Delete (Project & i)
        Next i
    End If

MyExit:
    On Error Resume Next
    Set db = Nothing
    Set MyDB = Nothing
    Exit Function
errDeleteRelationship:
    MsgBox Err.Description
    Resume MyExit
End Function

Function CreateRelationship(RemoteDB As String, Project As String, Optional IsExtended As Boolean)
'Sept. 30, 2019

On Error GoTo errCreateRelationship
Dim MyDB As Database, MyRel As Relation, MyFld As Field, RelationName As String
Dim VegTable As String, HumusTable As String, MineralTable As String, MetadataTable As String, AdminTable As String
Dim AuditTable As String, OtherTable As String, i As Integer
Dim TreePlotTable As String, TreesTable As String, CwdTable As String, CwdPileTable As String
Dim EnvTable As String, db As Database
Set MyDB = CurrentDb()

EnvTable = Project
Set db = DBEngine.Workspaces(0).OpenDatabase(RemoteDB)

'set variables
    RelationName = EnvTable
    EnvTable = RelationName & "_Env"
    VegTable = RelationName & "_Veg"
    HumusTable = RelationName & "_Humus"
    MineralTable = RelationName & "_Mineral"
    OtherTable = RelationName & "_Other"
    AuditTable = RelationName & "_Audit"
    MetadataTable = RelationName & "_Metadata"
    AdminTable = RelationName & "_Admin"

'bail if no relation
    If FindRelation(RelationName) = True Then GoTo MyExit

'delete old relationships
    If db.Relations.count > 0 Then
        On Error Resume Next
        For i = 0 To db.Relations.count - 1
            db.Relations.Delete (RelationName & i)
        Next i
    End If

'set veg relationship
    Set MyRel = db.CreateRelation(RelationName & "1", EnvTable, VegTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set humus relations
    Set MyRel = db.CreateRelation(RelationName & "2", EnvTable, HumusTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set mineral relations
    Set MyRel = db.CreateRelation(RelationName & "3", EnvTable, MineralTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set other relations
    Set MyRel = db.CreateRelation(RelationName & "4", EnvTable, OtherTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set audit relations
    Set MyRel = db.CreateRelation(RelationName & "5", EnvTable, AuditTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

'set metadata relations
    Set MyRel = db.CreateRelation(RelationName & "6", EnvTable, MetadataTable)
    Set MyFld = MyRel.CreateField("ProjectID")
    MyFld.ForeignName = "ProjectID"
    MyRel.Attributes = 2
    MyRel.Fields.Append MyFld
    
    Set MyFld = MyRel.CreateField("StartDate")
    MyFld.ForeignName = "StartDate"
    MyRel.Attributes = 2
    MyRel.Fields.Append MyFld
    
    db.Relations.Append MyRel

'set admin relations
    Set MyRel = db.CreateRelation(RelationName & "7", EnvTable, AdminTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "Plot"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade + 1
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

IsExtended = False
On Error GoTo 0
If IsExtended Then
    'set Tree plot relations
    Set MyRel = db.CreateRelation(RelationName & "6", EnvTable, TreePlotTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

    'set Trees relations
    Set MyRel = db.CreateRelation(RelationName & "7", EnvTable, TreesTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel

    'set CWD relations
    Set MyRel = db.CreateRelation(RelationName & "8", EnvTable, CwdTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel
    
    'set CWD Piling relations
    Set MyRel = db.CreateRelation(RelationName & "9", EnvTable, CwdPileTable)
    Set MyFld = MyRel.CreateField("PlotNumber")
    MyFld.ForeignName = "PlotNumber"
    MyRel.Attributes = dbRelationUpdateCascade + dbRelationDeleteCascade
    MyRel.Fields.Append MyFld
    db.Relations.Append MyRel
End If

MyExit:
    On Error Resume Next
    Set db = Nothing
    Set MyDB = Nothing
    Exit Function
errCreateRelationship:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function FindRelation(RelationName As String) As Boolean
'Sept. 30, 2019

Dim MyDB As Database, MyRel As Relation
Set MyDB = CurrentDb()
FindRelation = False

'look for relation
    For Each MyRel In MyDB.Relations
        If MyRel.Table = RelationName Then
            FindRelation = True
        End If
    Next

'clean up
    Set MyDB = Nothing

End Function
