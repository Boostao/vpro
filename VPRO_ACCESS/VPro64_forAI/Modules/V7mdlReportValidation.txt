Private Function Report4SuUnitsWoHierarchyUnits(Optional WriteReport As Variant) As Boolean
'Oct. 8, 2002
'checks for unit 11's in the hierarchy table that don't have a match in the site unit table
On Error GoTo errReport4SuUnitsWoHierarchyUnits
Dim MyDB As Database, MyRS As Recordset
Dim xl As Object, i As Integer
Dim SqlText As String, CurrSU As String, CurrH As String
Set MyDB = CurrentDb
CurrSU = CurrPlotlist & "_SU"
CurrH = CurrHierarchy

SqlText = "SELECT DISTINCT [" & CurrSU & "].SiteUnit"
SqlText = SqlText & " FROM [" & CurrSU & "] LEFT JOIN [" & CurrH & "] ON [" & CurrSU & "].SiteUnit = [" & CurrH & "].Name"
SqlText = SqlText & " WHERE ((([" & CurrH & "].Name) Is Null));"

Set MyRS = MyDB.OpenRecordset(SqlText)
If MyRS.RecordCount = 0 Then
    GoTo MyExit
Else
    Report4SuUnitsWoHierarchyUnits = True
End If

If Not IsMissing(WriteReport) Then
    Set xl = CreateObject("Excel.Application")
        xl.Visible = True
        xl.Cursor = xlWait
        xl.Workbooks.Add
        MyRS.MoveLast
        MyRS.MoveFirst
        xl.Cells(1, 1) = "Unit"
            For i = 1 To MyRS.RecordCount
                xl.Cells(i + 1, 1) = MyRS!SiteUnit
                MyRS.MoveNext
            Next i
        MyRS.Close
End If

MyExit:
    On Error Resume Next
    xl.Cursor = xlDefault
    MyRS.Close
    Set MyDB = Nothing
    Set xl = Nothing
    Exit Function
errReport4SuUnitsWoHierarchyUnits:
    MsgBox Err.Description
    Resume MyExit
End Function

Function FixOne2ManyCheck()

'Oct. 6, 2008
'deletes records without matching Env plotnumbers
On Error GoTo errFixOne2ManyCheck
Dim Msg As String, dgdef As Integer, Response As Integer, Title As String, OtherTable As String
Dim EnvTable As String, SuTable As String, SqlText As String, HumusTable As String
Dim MineralTable As String, VegTable As String, AuditTable As String

Title = "Fix One-To-Many Relationship"
Msg = "If you continue: all records in this project's vegetation, mineral, humus, other, and audit tables "
Msg = Msg & "where there isn't a matching plotnumber in the environment table will be deleted.  If you "
Msg = Msg & "don't have backup of this project, it is suggested that you click No and create one now!"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef, Title)    ' Get user response.
    If Response = 6 Then
        DoCmd.SetWarnings False
        EnvTable = clsVProReg.CurrProject & "_Env"
        HumusTable = clsVProReg.CurrProject & "_Humus"
        OtherTable = clsVProReg.CurrProject & "_Other"
        MineralTable = clsVProReg.CurrProject & "_Mineral"
        VegTable = clsVProReg.CurrProject & "_Veg"
        AuditTable = clsVProReg.CurrProject & "_Audit"
    
        'delete humus records
        SqlText = "DELETE [" & HumusTable & "].*, [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " FROM [" & HumusTable & "] LEFT JOIN [" & EnvTable & "] ON [" & HumusTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " WHERE ((([" & EnvTable & "].PlotNumber) Is Null));"
        DoCmd.RunSQL (SqlText)
        
        'delete Other records
        SqlText = "DELETE [" & OtherTable & "].*, [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " FROM [" & OtherTable & "] LEFT JOIN [" & EnvTable & "] ON [" & OtherTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " WHERE ((([" & EnvTable & "].PlotNumber) Is Null));"
        DoCmd.RunSQL (SqlText)
        
        'delete mineral records
        SqlText = "DELETE [" & MineralTable & "].*, [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " FROM [" & MineralTable & "] LEFT JOIN [" & EnvTable & "] ON [" & MineralTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " WHERE ((([" & EnvTable & "].PlotNumber) Is Null));"
        DoCmd.RunSQL (SqlText)
        
        'delete veg records
        SqlText = "DELETE [" & VegTable & "].*, [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " FROM [" & VegTable & "] LEFT JOIN [" & EnvTable & "] ON [" & VegTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " WHERE ((([" & EnvTable & "].PlotNumber) Is Null));"
        DoCmd.RunSQL (SqlText)
            
        'delete audit records
        SqlText = "DELETE [" & AuditTable & "].*, [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " FROM [" & AuditTable & "] LEFT JOIN [" & EnvTable & "] ON [" & AuditTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"
        SqlText = SqlText & " WHERE ((([" & EnvTable & "].PlotNumber) Is Null));"
        DoCmd.RunSQL (SqlText)
    End If

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Exit Function
errFixOne2ManyCheck:
    MsgBox Err.Description
    Resume MyExit
End Function
Function One2ManyCheck()

'Oct. 6, 2008
'Checks if there are any plot numbers in subordinate tables that don't have a plot number in the Env table
On Error GoTo errOne2ManyCheck

Dim EnvTable As String, SuTable As String, SqlText As String, HumusTable As String
Dim MineralTable As String, VegTable As String, AuditTable As String
Set MyDB = CurrentDb()
PlotCol = 1
EnvTable = clsVProReg.CurrProject & "_Env"
HumusTable = clsVProReg.CurrProject & "_Humus"
OtherTable = clsVProReg.CurrProject & "_Other"
MineralTable = clsVProReg.CurrProject & "_Mineral"
VegTable = clsVProReg.CurrProject & "_Veg"
AuditTable = clsVProReg.CurrProject & "_Audit"

Dim xl As Excel.Application, i As Integer
    CreateExcel xl
    xl.Workbooks.Add
    xl.Visible = True

'find bad humus records
SqlText = "SELECT DISTINCT [" & HumusTable & "].PlotNumber"
SqlText = SqlText & " FROM [" & HumusTable & "] LEFT JOIN [" & EnvTable & "] ON [" & HumusTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"
SqlText = SqlText & " WHERE ((([" & HumusTable & "].PlotNumber) Is Not Null) AND (([" & EnvTable & "].PlotNumber) Is Null));"
Set MyRS = MyDB.OpenRecordset(SqlText)
If MyRS.RecordCount >= 1 Then
    xl.Cells(2, PlotCol) = "Humus"
    xl.Cells(3, PlotCol) = "Plot Number"
        For i = 1 To MyRS.RecordCount
            xl.Cells(i + 3, PlotCol) = "'" & MyRS!PlotNumber
            MyRS.MoveNext
        Next i
    MyRS.Close
    PlotCol = PlotCol + 2
End If

'find bad other records

SqlText = SqlText & " FROM [" & OtherTable & "] LEFT JOIN [" & EnvTable & "] ON [" & OtherTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"

    xl.Cells(3, PlotCol) = "Plot Number"
        For i = 1 To MyRS.RecordCount
            xl.Cells(i + 3, PlotCol) = "'" & MyRS!PlotNumber
            MyRS.MoveNext
        Next i
    MyRS.Close
    PlotCol = PlotCol + 2

'find bad veg records
SqlText = "SELECT DISTINCT [" & VegTable & "].PlotNumber"
SqlText = SqlText & " FROM [" & VegTable & "] LEFT JOIN [" & EnvTable & "] ON [" & VegTable & "].PlotNumber = [" & EnvTable & "].PlotNumber"

Function SuRecordsWoEnvRecords()

'Oct. 6, 2008
'Checks for plots in the site unit table that don't exist in the Env table
On Error GoTo errSuRecordsWoEnvRecords
Dim MyDB As Database, MyRS As Recordset
Dim EnvTable As String, SuTable As String, SqlText As String
Set MyDB = CurrentDb()
EnvTable = clsVProReg.CurrProject & "_Env"
SuTable = clsVProReg.CurrPlotlist & "_SU"

SqlText = "SELECT DISTINCTROW [" & SuTable & "].[PlotNumber], [" & SuTable & "].[SiteUnit]"
SqlText = SqlText & " FROM [" & SuTable & "] LEFT JOIN [" & EnvTable & "] ON [" & SuTable & "].[PlotNumber] = [" & EnvTable & "].[PlotNumber]"
SqlText = SqlText & " WHERE ([" & EnvTable & "].[PlotNumber] Is Null);"
Set MyRS = MyDB.OpenRecordset(SqlText)

If MyRS.RecordCount >= 1 Then
    Dim xl As Excel.Application, i As Integer
    CreateExcel xl
    xl.Visible = True
    xl.Workbooks.Add
    xl.Cells(3, 1) = "Plot Number"
    xl.Cells(3, 2) = "Site Unit"
        For i = 1 To MyRS.RecordCount
            xl.Cells(i + 3, 1) = "'" & MyRS!PlotNumber
            xl.Cells(i + 3, 2) = "'" & MyRS!SiteUnit
            MyRS.MoveNext
        Next i
    xl.ActiveSheet.Columns().AutoFit
    xl.Cells(1, 1) = "Plots in site unit table: " & clsVProReg.CurrPlotlist & " that don't exist in the project: " & clsVProReg.CurrProject
Else
    MsgBox "There were no unmatched records!"
End If

MyExit:
    On Error Resume Next
    MyRS.Close
    Set xl = Nothing
    Set MyDB = Nothing
    Exit Function
errSuRecordsWoEnvRecords:
    MsgBox Err.Description
    Resume MyExit
End Function

Function ValidateFieldWithPick(DataField As String, PickField As String, ListTable As String, Optional ListField As Variant)

'Oct. 6, 2008
'creates a recordset of errors
On Error GoTo errValidateFieldWithPick
Dim MyDB As Database, SqlText As String, MyQD As QueryDef, ListFilter As String
Set MyDB = CurrentDb()
ListFilter = ListField

CreateTableOfListsQuery ListFilter

SqlText = "SELECT PlotNumber, " & DataField
SqlText = SqlText & " FROM USysEnv LEFT JOIN qryTableOfLists ON USysEnv." & DataField & " = qryTableOfLists.Item"
SqlText = SqlText & " WHERE (((USysEnv." & DataField & ") Is Not Null And (USysEnv." & DataField & ")<>'') AND ((qryTableOfLists.Item) Is Null))"
SqlText = SqlText & " ORDER BY UsysEnv." & DataField & ";"
'Debug.Print "here"
'Set MyQD = MyDB.CreateQueryDef("herman", SqlText)
'If PickField = "subzone" Then Stop
Set ActiveRS = MyDB.OpenRecordset(SqlText)

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function
errValidateFieldWithPick:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Sub CreateTableOfListsQuery(FilterBy As String)

'Oct. 6, 2008
Dim SqlText As String
On Error Resume Next
DoCmd.DeleteObject acQuery, "qryTableOfLists"
On Error GoTo 0
    
    SqlText = "SELECT ListName, Item"
    SqlText = SqlText & " From USysTableOfLists"

    CurrentDb.CreateQueryDef "qryTableOfLists", SqlText

End Sub

Function ValidateOtherFieldWithPick(DataField As String, PickField As String, ListTable As String, TableQuery As String)
'Oct. 8, 2002
'creates a recordset
On Error GoTo errValidateFieldWithPick

SqlText = SqlText & " FROM " & TableQuery & " LEFT JOIN " & ListTable & " ON " & TableQuery & "." & DataField & " = " & ListTable & "." & PickField

SqlText = SqlText & " ORDER BY " & TableQuery & "." & DataField & ";"
Set ActiveRS = MyDB.OpenRecordset(SqlText)

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function

End Function

Function PlotRecordsWoVegRecords(CalledBy As String) As Boolean

'2019
'checks for plots that contain no veg data
On Error GoTo errPlotRecordsWoVegRecords
Dim MyDB As Database, MyRS As Recordset
Dim EnvTable As String, VegTable As String, SuTable As String, SqlText As String
Set MyDB = CurrentDb()
EnvTable = clsVProReg.CurrProject & "_Env"
VegTable = clsVProReg.CurrProject & "_Veg"
SuTable = clsVProReg.CurrPlotlist & "_SU"

If clsVProReg.CurrPlotlist = "None" Then
    SqlText = "SELECT DISTINCTROW [" & EnvTable & "].[PlotNumber]"
    SqlText = SqlText & " FROM [" & EnvTable & "] LEFT JOIN [" & VegTable & "] ON [" & EnvTable & "].[PlotNumber] = [" & VegTable & "].[PlotNumber]"
    SqlText = SqlText & " WHERE ([" & VegTable & "].[PlotNumber] Is Null);"
Else
    SqlText = "SELECT DISTINCTROW [" & EnvTable & "].PlotNumber"
    SqlText = SqlText & " FROM ([" & EnvTable & "] LEFT JOIN [" & VegTable & "] ON [" & EnvTable & "].[PlotNumber] = [" & VegTable & "].[PlotNumber]) INNER JOIN [" & SuTable & "] ON [" & EnvTable & "].PlotNumber = [" & SuTable & "].PlotNumber"
    SqlText = SqlText & " WHERE ((([" & VegTable & "].PlotNumber) Is Null));"
End If

Set MyRS = MyDB.OpenRecordset(SqlText)

If MyRS.RecordCount >= 1 Then
    PlotRecordsWoVegRecords = True
    If CalledBy = "Validate" Then
        Dim xl As Excel.Application, i As Integer
        CreateExcel xl
        xl.Visible = True
        xl.Workbooks.Add
        xl.Cells(3, 1) = "Plot Number"
            For i = 1 To MyRS.RecordCount
                xl.Cells(i + 3, 1) = "'" & MyRS!PlotNumber
                MyRS.MoveNext
            Next i
        xl.ActiveSheet.Columns().AutoFit
        xl.Cells(1, 1) = "Plots without any vegetation records in the project: " & clsVProReg.CurrProject
    Else
        GoTo MyExit
    End If
Else
    PlotRecordsWoVegRecords = False
    If CalledBy = "Validate" Then
        MsgBox "There were no plots without vegetation records!", vbOKOnly, "VPro"
    End If
End If

MyExit:
    On Error Resume Next
    MyRS.Close
    Set xl = Nothing
    Set MyDB = Nothing
    Exit Function
errPlotRecordsWoVegRecords:
    MsgBox Err.Description
    Resume MyExit
End Function
