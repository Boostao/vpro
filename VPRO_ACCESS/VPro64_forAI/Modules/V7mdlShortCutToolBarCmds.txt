Function DeleteAttachedProject(Optional ProjectName As String)

'Oct. 5, 2016
On Error GoTo errDeleteAttachedProject
Dim Title As String, Msg As String, Response As Integer, dgdef As String
Dim MyDB As Database, TestProject As String, i As Integer, MyForm As Form
Dim TestTable As String, ProjectDB As Database, ProjectDbLocation As String
Dim ProjectRecords As Recordset
Dim Criteria As String
Set MyDB = CurrentDb()

'close any open FS882 forms
    If clsFormInfo.IsDataFormLoaded Then
        MsgBox "VPro must close some items in order for you to be able to perform this task.  Please close this message box and try again.", vbInformation, "VPro"
        On Error Resume Next
        DoCmd.Close acForm, "FS882-6x4XL"
        DoCmd.Close acForm, "FS882-8x6XL"
        DoCmd.Close acForm, "FS882-1x1"
        On Error GoTo errDeleteAttachedProject
        GoTo MyExit
    End If

'create form object
    Set MyForm = Screen.ActiveForm
    
'set TestProject variable
    If ProjectName > "" Then
        TestProject = ProjectName
    Else
        TestProject = MyForm.ActiveControl
    End If
    
'warn the user
    Title = "VPro"
    Msg = "WARNING!  This function will permanently delete project: '" & TestProject & "'.  There is no recovery from this action!"
    Msg = Msg & Chr(13) & "Proceed?"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef, Title)    ' Get user response.
    If Response <> 6 Then GoTo MyExit
        
'get the link info
    For i = 0 To MyDB.TableDefs.count - 1
        TestTable = MyDB.TableDefs(i).Name
        If TestTable = TestProject & "_Env" Then
            ProjectDbLocation = MyDB.TableDefs(i).Connect
            If Not ProjectDbLocation = "" Then ProjectDbLocation = Right(ProjectDbLocation, Len(ProjectDbLocation) - 10) 'this - 10 okay
            If ProjectDbLocation = "" Then
                MsgBox "You can't delete this project!", vbInformation, "VPro"
                GoTo MyExit
            End If
        End If
    Next i
    Set ProjectDB = DBEngine.Workspaces(0).OpenDatabase(ProjectDbLocation)
    
'delete local links
    DoCmd.DeleteObject acTable, TestProject & "_Env"
    DoCmd.DeleteObject acTable, TestProject & "_Admin"
    DoCmd.DeleteObject acTable, TestProject & "_Humus"
    DoCmd.DeleteObject acTable, TestProject & "_Other"
    DoCmd.DeleteObject acTable, TestProject & "_Mineral"
    DoCmd.DeleteObject acTable, TestProject & "_Audit"
    DoCmd.DeleteObject acTable, TestProject & "_Veg"
    DoCmd.DeleteObject acTable, TestProject & "_Metadata"

'delete old relationships
    DeleteRelationship ProjectDB.Name, TestProject

'delete remote tables
    ProjectDB.TableDefs.Delete TestProject & "_Humus"
    ProjectDB.TableDefs.Delete TestProject & "_Other"
    ProjectDB.TableDefs.Delete TestProject & "_Mineral"
    ProjectDB.TableDefs.Delete TestProject & "_Audit"
    ProjectDB.TableDefs.Delete TestProject & "_Veg"
    ProjectDB.TableDefs.Delete TestProject & "_Metadata"
    ProjectDB.TableDefs.Delete TestProject & "_Admin"
    ProjectDB.TableDefs.Delete TestProject & "_Env"
    
    SetCurrentProject "Sample"

MyExit:
    On Error Resume Next
    MyForm.Recalc
    ProjectRecords.Close
    Set MyForm = Nothing
    Set MyDB = Nothing
    Set ProjectDB = Nothing
    Exit Function
errDeleteAttachedProject:
    MsgBox Err.Description
    Resume MyExit
End Function

Function SetLumpsTo(SetTo As String, Optional IsCombo)
'Oct. 12, 2002
On Error GoTo errSetLumpsToYes
Dim LumpTable As String, Species As String, SqlText As String, xTree As TreeView
Dim MyForm As Form
If IsMissing(IsCombo) Then
    Set MyForm = Forms![USysLumpMaster]
    Set xTree = Forms![USysLumpMaster]!LumpTree.Object
    LumpTable = clsVProReg.CurrLump & "_Lump"
    If LumpTable = "None_Lump" Then GoTo MyExit
    Species = xTree.SelectedItem
    SqlText = "UPDATE [" & LumpTable & "] SET [" & LumpTable & "].Use = " & SetTo
    SqlText = SqlText & " WHERE ((([" & LumpTable & "].LumpCode)='" & Species & "'));"
    DoCmd.SetWarnings False
    DoCmd.RunSQL SqlText
Else
    Set MyForm = Forms![USysComboLumpMaster]
    Set xTree = Forms![USysComboLumpMaster]!LumpTree.Object
    LumpTable = Forms![USysComboLumpMaster]![cmbLumpTable] & "_Theme"
    Species = xTree.SelectedItem
    SqlText = "UPDATE [" & LumpTable & "] SET [" & LumpTable & "].Use = " & SetTo
    SqlText = SqlText & " WHERE ((([" & LumpTable & "].LumpCode)='" & Species & "'));"
    DoCmd.SetWarnings False
    DoCmd.RunSQL SqlText
End If
    MyForm.Refresh
    DoCmd.SetWarnings True

MyExit:
    On Error Resume Next
    Exit Function
errSetLumpsToYes:
    MsgBox Err.Description
    Resume MyExit
End Function

Function DeleteSiteUnit(Optional SiteUnit As String)
On Error GoTo errDeleteSiteUnit

Dim i As Integer, ItemX As ListItem
Set MyDB = CurrentDb()
Set MyRS = MyDB.OpenRecordset("USysUserSiteUnitList", dbOpenDynaset)

    Criteria = "Name = '" & MyList.SelectedItem & "'"

    GoTo MyExit

    MyRS.FindFirst Criteria
    MyRS.Delete
Next i
    With MyList
        .ListItems.Clear
        .HideColumnHeaders = False
    End With
    MyRS.MoveFirst
    If MyRS.RecordCount > 0 Then
        While Not MyRS.EOF
            Set ItemX = MyList.ListItems.Add()
            If Not IsNull(MyRS!Name) Then ItemX.Text = MyRS!Name
            If Not IsNull(MyRS!UnitLongName) Then ItemX.SubItems(1) = MyRS!UnitLongName
            If Not IsNull(MyRS!Zone) Then ItemX.SubItems(2) = MyRS!Zone
            If Not IsNull(MyRS!SubZone) Then ItemX.SubItems(3) = MyRS!SubZone
            If Not IsNull(MyRS!PlotVariant) Then ItemX.SubItems(4) = MyRS!PlotVariant
            If Not IsNull(MyRS!RegionCode) Then ItemX.SubItems(5) = MyRS!RegionCode

Function ChangeSetLevel()
'Aug 3, 2000
'create groups of plots based on the hierarchy and su table
'On Error GoTo errChangeSetLevel
Dim MyDB As Database, ReadRS As Recordset, WriteRS As Recordset, HierarchyRS As Recordset
Dim SqlText As String, Criteria As String, i As Integer, X As Long, MyTest As Long, Cascade As Variant
Dim MyTree As TreeView, NewLevel As Integer, OldLevel As Long
Set MyTree = Screen.ActiveControl.Object
Set MyDB = CurrentDb
Set HierarchyRS = MyDB.OpenRecordset(CurrHierarchy(), dbOpenDynaset)
Set ReadRS = MyDB.OpenRecordset(CurrHierarchy(), dbOpenDynaset)
Set WriteRS = MyDB.OpenRecordset("USysBreakFilter", dbOpenDynaset)

ReadRS.MoveLast
ReadRS.MoveFirst
HierarchyRS.MoveLast
HierarchyRS.MoveFirst

Criteria = "Name = '" & MyTree.SelectedItem.Text & "'"
ReadRS.FindFirst Criteria
X = ReadRS!ID
NewLevel = Val(InputBox("What level do you want the selected item to be placed at?" & Chr(13) & "(1 to 10 for hierarchy members and 11 for units)", "VPro"))
If NewLevel < 1 Then GoTo MyExit
DoCmd.Hourglass True

For i = 1 To HierarchyRS.RecordCount
    If HierarchyRS!ID = X Then
        If IsNull(ReadRS!Level) Then
            ReadRS.Edit
            ReadRS!Level = NewLevel
            ReadRS.Update
        End If
        OldLevel = ReadRS!Level
        ReadRS.Edit
        ReadRS!Level = NewLevel
        ReadRS.Update
        ChangeScreenLevel ReadRS!Name, NewLevel, MyTree
        If Not IsNull(ReadRS!Parent) Then
            WriteMasterAudit "Change", MyTree.SelectedItem.Text, X, "Level", OldLevel, NewLevel, GetParentName(ReadRS, ReadRS!Parent)
        Else
            WriteMasterAudit "Change", MyTree.SelectedItem.Text, X, "Level", OldLevel, NewLevel
        End If

        Cascade = MsgBox("Do you want this change to cascade through the children?", vbYesNo, "VPro")
        If Cascade = vbYes Then
            ChangeChildren HierarchyRS!ID, ReadRS, WriteRS, NewLevel + 1, HierarchyRS!ID, MyTree
        End If
    End If
    HierarchyRS.MoveNext
Next i

MyExit:
    On Error Resume Next
    DoCmd.Hourglass False
    DoCmd.SetWarnings True
    ReadRS.Close
    WriteRS.Close
    HierarchyRS.Close
    Set MyDB = Nothing
    Set MyTree = Nothing
    Exit Function
errChangeSetLevel:
    Resume MyExit
    MsgBox Err.Description
    Resume MyExit
End Function

Function ChangeChildren(Unit As Long, ReadRS As Recordset, WriteRS As Recordset, Level As Integer, Parent As Long, MyTree As TreeView)
'May 01, 2000
'finds the last member in a hierarchy branch
Dim i As Integer, Criteria As String, MyID As Long, MyTest As String, BK As String
Dim TempRS As Recordset
Set TempRS = ReadRS.Clone

ReadRS.MoveFirst
Criteria = "Parent = " & Unit
ReadRS.FindFirst Criteria

Do Until ReadRS.NoMatch

    BK = ReadRS.Bookmark
    
    If HasBrats(ReadRS!ID, TempRS) Then
        ReadRS.Edit
        ReadRS!Level = Level
        ReadRS.Update
        WriteMasterAudit "Change", ReadRS!Name, ReadRS!ID, "Level", Level - 1, Level, GetParentName(ReadRS, ReadRS!Parent)
        ChangeScreenLevel ReadRS!Name, Level, MyTree
        ReadRS.Bookmark = BK
        ChangeChildren ReadRS!ID, ReadRS, WriteRS, Level + 1, ReadRS!ID, MyTree
    Else
        If ReadRS!Level <> 11 Or IsNull(ReadRS!Level) Then
            WriteMasterAudit "Change", ReadRS!Name, ReadRS!ID, "Level", ReadRS!Level, 11, GetParentName(ReadRS, ReadRS!Parent)
            ReadRS.Edit
            ReadRS!Level = 11
            ReadRS.Update
            ChangeScreenLevel ReadRS!Name, 11, MyTree
        End If
    End If
    
    ReadRS.Bookmark = BK
    ReadRS.FindNext Criteria
Loop
TempRS.Close
End Function
Function ChangeScreenLevel(UnitText As String, Level As Integer, MyTree As TreeView)
Dim i As Integer

For i = 1 To MyTree.Nodes.count
    If MyTree.Nodes(i).Text = UnitText And Level < 13 Then
        MyTree.Nodes(i).ExpandedImage = Level
        MyTree.Nodes(i).SelectedImage = Level
        MyTree.Nodes(i).image = Level
    End If
Next i

End Function
