Public Sub FillRemovedBy()

'Apr. 12, 2022
Dim MyDB As Database, MyTbl As TableDef, MyFld As Field, MyRS As Recordset
Dim SqlText As String, AdminTbl As String, SuTbl As String
Dim bSite As Boolean, bVeg As Boolean, bSoil As Boolean, bBEC As Boolean, MySum As Integer

'set the admin and su table variables
    AdminTbl = clsVProReg.CurrProject
    SuTbl = clsVProReg.CurrPlotlistBak ' clsVProReg.CurrPlotlist
    
'make db object
    Set MyDB = CurrentDb

'clear last run
    On Error Resume Next
    DoCmd.DeleteObject acTable, "tblRemovedBy"
    On Error GoTo 0

'set level variables
    SetLevels

'create table
    Set MyTbl = MyDB.CreateTableDef("tblRemovedBy")
    
    With MyTbl
        .Fields.Append .CreateField("PlotNumber", dbText)
        .Fields.Append .CreateField("RemovedBy", dbText)
        .Fields.Append .CreateField("Site", dbText)
        .Fields.Append .CreateField("Veg", dbText)
        .Fields.Append .CreateField("Soil", dbText)
        .Fields.Append .CreateField("BEC", dbText)
    End With
    
    MyDB.TableDefs.Append MyTbl

'fill table
    SqlText = "INSERT INTO tblRemovedBy ( PlotNumber, Site, Veg, Soil, BEC )"
    SqlText = SqlText & " SELECT " & SuTbl & "_SU.PlotNumber, " & AdminTbl & "_Admin.SitePlotQuality, " & AdminTbl & "_Admin.VegPlotQuality, " & AdminTbl & "_Admin.SoilPlotQuality, " & AdminTbl & "_Admin.BEC_Use"
    SqlText = SqlText & " FROM (" & SuTbl & "_SU LEFT JOIN USysSuTableDynamic_SU ON " & SuTbl & "_SU.PlotNumber = USysSuTableDynamic_SU.PlotNumber) INNER JOIN " & AdminTbl & "_Admin ON " & SuTbl & "_SU.PlotNumber = " & AdminTbl & "_Admin.Plot"
    SqlText = SqlText & " WHERE (((USysSuTableDynamic_SU.PlotNumber) Is Null));"
    'MyDB.CreateQueryDef "fred", SqlText
    DoCmd.SetWarnings False
    DoCmd.RunSQL SqlText
    DoCmd.SetWarnings True
'Stop
'create table recordset
    Set MyRS = MyDB.OpenRecordset("tblRemovedBy")

'loop through table and find which criteria removed the plot
    Do Until MyRS.EOF
        MySum = 0
        bSite = False
        bVeg = False
        bSoil = False
        bBEC = False
        If Not IsNull(MyRS!Site) Then
            If LevelAsNumber(MyRS!Site) < lvSite Then
                MySum = MySum + 1
                MyRS.Edit
                MyRS!RemovedBy = "Site"
                MyRS.Update
            End If
        End If
        
        If Not IsNull(MyRS!Veg) Then
            If LevelAsNumber(MyRS!Veg) < lvVeg Then
                MySum = MySum + 1
                MyRS.Edit
                MyRS!RemovedBy = "Veg"
                MyRS.Update
            End If
        End If
        
        If Not IsNull(MyRS!Soil) Then
            If LevelAsNumber(MyRS!Soil) < lvSoil Then
                MySum = MySum + 1
                MyRS.Edit
                MyRS!RemovedBy = "Soil"
                MyRS.Update
            End If
        End If
        
        If Not IsNull(MyRS!BEC) Then 'first get the trhreshold and see if it is higher
            If MyRS!BEC < lvBEC Then
                MySum = MySum + 1
                MyRS.Edit
                MyRS!RemovedBy = "BEC"
                MyRS.Update
            End If
        End If
        
        If MySum > 1 Then
            MyRS.Edit
            MyRS!RemovedBy = "Mixed"
            MyRS.Update
        End If
    
        MyRS.MoveNext
    Loop

MyExit:
    MyRS.Close
    Set MyDB = Nothing

End Sub

Private Sub SetLevels()

'Apr. 12, 2022
Dim Site As String, Veg As String, Soil As String

'look up the quality settings in the registry
    Site = clsRepOpt.DataQualityFilterSite
    lvSite = LevelAsNumber(Site)
    
    Veg = clsRepOpt.DataQualityFilterVeg
    lvVeg = LevelAsNumber(Veg)

    Soil = clsRepOpt.DataQualityFilterSoil
    lvSoil = LevelAsNumber(Soil)

    lvBEC = clsRepOpt.DataQualityFilterBECUseLV

End Sub

Private Function LevelAsNumber(LevelAsText As String) As Integer

'Apr. 12, 2022
'convert the quality text to a number
    If LevelAsText = "Poor" Then LevelAsNumber = 1
    If LevelAsText = "Fair" Then LevelAsNumber = 2
    If LevelAsText = "Good" Then LevelAsNumber = 3
    If LevelAsText = "Excellent" Then LevelAsNumber = 4

End Function
