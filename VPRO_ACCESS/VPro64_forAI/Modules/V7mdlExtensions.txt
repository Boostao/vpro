Private Sub AddMenuWithPopup(MyCmdBar As String, MyCmdBarControl1 As String, _
MyCmdBarPopup As String, MyCmdBarControl2 As String, MyCommand As String)
  Dim cmdBar As CommandBar
  Dim toolsMenu As CommandBarControl
  Dim myMenu As CommandBarPopup
  Dim subMenu As CommandBarControl
  
  Set cmdBar = Application.CommandBars(MyCmdBar)

  Set toolsMenu = cmdBar.Controls(MyCmdBarControl1)

  Set myMenu = toolsMenu.Controls.Add(type:=msoControlPopup)
  
  Set subMenu = myMenu.Controls.Add
  
  With myMenu
    .Caption = MyCmdBarPopup
    .BeginGroup = True
    With subMenu
      .Caption = MyCmdBarControl2
      .BeginGroup = True
      .OnAction = MyCommand
    End With
  End With

End Sub

Private Sub AddPopupItem(MyCmdBar As String, MyCmdBarControl1 As String, _
MyCmdBarPopup As String, MyCmdBarControl2 As String, MyCommand As String)
  Dim cmdBar As CommandBar
  Dim toolsMenu As CommandBarControl
  Dim myMenu As CommandBarPopup
  Dim subMenu As CommandBarControl
  
  Set cmdBar = Application.CommandBars(MyCmdBar)

  Set toolsMenu = cmdBar.Controls(MyCmdBarControl1)

  Set myMenu = toolsMenu.Controls(MyCmdBarPopup)
  
  Set subMenu = myMenu.Controls.Add
  
    With subMenu
      .Caption = MyCmdBarControl2
      .BeginGroup = False
      .OnAction = MyCommand
    End With

End Sub
