Public Function ImportSortingInst()

'Sep 22, 2008
Dim MyDB As Database, RemoteDB As Database  'declare variables
Dim CurrSU As String, SuTable As String, SqlText As String, SortingInst As String
Dim TableToSave As String, MyTitle As String, FileToOpen As String
Dim MyGroup As Integer, Counter1 As Integer, Counter2 As Integer, NumberOfPlots As Integer
Dim MyTable As Recordset, Line2 As String, Line3 As String, z As String, DestDB As String
Dim Title As String, Msg As String, dgdef As Integer, Response As Integer, Criteria As String, SqlTxt As String
Dim TextData As String, ThisGroup As Integer, ShortName As String, LongName As String
Dim PlotStart As Integer, PlotCount As Integer, ThisPlot As String, TestCriteria As String
Dim MyQuery As QueryDef 'declare variable
Set MyDB = CurrentDb()                      'create object MyDB
CurrSU = CurrPlotlist()                     'assign current site unit table name to variable CurrSU
'program checks for a current site unit list and code ends if none
If CurrSU = "None" Then MsgBox "VPro couldn't find a current site unit table. Please attach a site unit table and try again.", 64, ProgramName()
If CurrSU = "None" Then Exit Function

    SuTable = CurrSU & "_SU"   'set variable for the site unit table

'delete any records from USysSortingTable
    DoCmd.SetWarnings False
    SqlText = "DELETE DISTINCTROW USysSortingTable.*"
    SqlText = SqlText & " FROM USysSortingTable;"
    DoCmd.RunSQL SqlText
    SortingInst = "USysSortingTable"  'assign a variable to USysSortingTable
    DoCmd.SetWarnings True

'close #1 in case the program is recovering from a crash or power failure
Close #1
    TableToSave = CurrProject() & "_Env"  'assign a variable to the current env table
    MyTitle = ProgramName() & " - choose a VTAB sorting instruction file"
    'FileToOpen = GetFileName(MyTitle$, MyFilter, 2, "Open")
    FileToOpen = GetFileName(MyTitle$, True, "Sorting (*.st*)", "*.st*") 'get a sorting file name to import & assign to variable
    If FileToOpen = "" Then GoTo MyExit
    
    'DoCmd.Hourglass True
    DoCmd.Hourglass False

    Open FileToOpen For Input As #1   'open the vtab sorting file
    If LOF(1) = 0 Then GoTo ExitSorting  'exit procedure if file is empty

    Set MyTable = MyDB.OpenRecordset(SortingInst, DB_OPEN_DYNASET)     'create the object MyTable
MyGroup = 0                                            'set counter variable MyGroup to 0

Home:
Do While Not EOF(1)   'start first do loop
MyGroup = MyGroup + 1                       'set counter variable MyGroup to 0
Counter1 = 0                                'set counter variable Counter1 to 0
Counter2 = 0                                'set counter variable Counter2 to 0
NumberOfPlots = 0                           'set variable to 0

Line Input #1, TextData                     'read line 1 of sorting file
    ThisGroup = Format(MyGroup, "00") 'Mid$(TextData, 9, 2) 'sets ThisGroup variable to equal MyGroup variable
    ShortName = Mid$(TextData, 1, 16)        'assign columns 1 - 16 to ShortName variable
    ShortName = Trim(ShortName)
    LongName = Mid$(TextData, 17, 80)        'assign columns 17 - 80 to LongName variable
GoTo NextLine

NextLine:
Line2 = ""            'sets variable to ""
Line Input #1, Line2  'reads in second line
Counter2 = 0           'set counter variable Counter2 to 0

If Counter1 = 0 Then PlotStart = 4 Else PlotStart = 1   'if counter1 = 0 program knows that first plotnumber starts in the fourth column
If NumberOfPlots = 0 Then NumberOfPlots = Val(Mid$(Line2, 1, 3))  'reads in the number of plots in unit

PlotCount = 0      'set counter variable PlotCount to 0

'start second do loop
    Do
        ThisPlot = Mid$(Line2, PlotStart, 7)  'read plotnumber in variable ThisPlot
        ThisPlot = Trim(ThisPlot)             'remove leading and trailing blanks
'Looks for existing plotnumber. This would happen if the sorting instructions had the same plot in more than one unit
    If MyTable.RecordCount > 0 Then
        Criteria = "[PlotNumber] = '" & ThisPlot & "'"
        MyTable.MoveFirst
        MyTable.FindFirst Criteria
        If MyTable.NoMatch Then
            z = ""
        Else
            z = "xxxxxxx"
        End If
    End If

'If there are no matching plotnumbers
            If IsNull(z) Or z = "" Then
                MyTable.AddNew                           'add new record to object MyTable
                MyTable("Group") = Trim(ThisGroup)       'add data to field Group
                Debug.Print ShortName & " - " & ThisPlot
                MyTable("PlotNumber") = Trim(ThisPlot)       'add data to field PlotNumber
                MyTable("GroupShortName") = Trim(ShortName) 'Trim(ShortName)  'add data to field Name
                If LongName > "" Then
                    MyTable("GroupLongName") = LongName 'Trim(LongName)    'add data to field UnitLongName
                End If
                MyTable.Update                               'update record
                PlotStart = PlotStart + 7                    'add 7 to PlotStart variable
            Else

'If there are matching plot numbers ask user what action
    Const MB_OK = 0, MB_OKCANCEL = 1    ' Define buttons.
    Const MB_YESNOCANCEL = 3, MB_YESNO = 4
    Const MB_ICONSTOP = 16, MB_ICONQUESTION = 32   ' Define icons.
    Const MB_ICONEXCLAMATION = 48, MB_ICONINFORMATION = 64
    Const MB_DEFBUTTON2 = 256, IDYES = 6, IDNO = 7   ' Define other.
    Title = "Import Sorting"
    ' Put together a message box with all the proper components.
    Msg = "This plot number exists more than once in the VTAB sorting file and was already written to your site unit table."
    Msg = Msg & Chr(13) & "Do you wish to replace plot number: " & ThisPlot & "?"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef, Title)    ' Get user response.
'Response = IDYES
'If the user wants to replace record
    If Response = IDYES Then    ' Evaluate response
        Criteria = "[PlotNumber] = '" & ThisPlot & "'"  'create a criteria string with the current plot number
        MyTable.MoveFirst                               'move to first record
        MyTable.FindFirst Criteria                      'search for plot
        MyTable.Delete                                  'delete record
        MyTable.AddNew                                  'add new record
        MyTable("PlotNumber") = Trim(ThisPlot)          'add data to field PlotNumber
        MyTable("Name") = Trim(ShortName)     'add data to field Name
        If LongName > "" Then
            MyTable("UnitLongName") = LongName 'Trim(LongName)    'add data to field UnitLongName
        End If
        
        MyTable.Update                                  'update record
        PlotStart = PlotStart + 7                       'add 7 to PlotStart variable
    End If
'If the user wants to leave existing records
    If Response = IDNO Then                            'no action
    PlotStart = PlotStart + 7                          'add 7 to PlotStart variable
    ThisPlot = ""                                      'set ThisPlot variable to ""
    End If
    End If

'procedure has now finnished reading in first plot number
Counter1 = Counter1 + 1      'add 1 to Counter1
Counter2 = Counter2 + 1      'add 1 to Counter2
If Counter1 >= NumberOfPlots Then GoTo Home    'if all plots for current unit have been read in
If Counter2 >= 11 Then GoTo NextLine           'read next line if there are more plots to read
Loop                                           'end inner loop

Loop                                           'loop until EOF
GoTo UpDateEnv                                 'move to code UpDateEnv
'Close #1
'Exit Sub

SortingErrorHandler:
    Close #1                                        'close sorting file
    MsgBox "Progran stopped at SortingErrorHandler" 'tell user where program stopped
    GoTo MyExit
    Exit Function

ExitSorting:
    Close #1                                        'close sorting file
    MsgBox "File didn't exist or contained zero records"  'tell the user that there weren't any reords
    GoTo MyExit
    Exit Function

UpDateEnv:
'ask the user if the program should overwite existing plot numbers
    Title = ProgramName()
    ' Put together a message box with all the proper components.
    Msg = "Where plot numbers already exist in your site unit table, overwrite with imported site units?"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef, Title)    ' Get user response.

DoCmd.SetWarnings False

'If the user wants to replace plotnumbers
If Response = IDYES Then    ' Evaluate response
    SqlText = "DELETE DISTINCTROW USysSortingTable.PlotNumber, " & SuTable & ".PlotNumber, " & SuTable & ".*"
    SqlText = SqlText & " FROM USysSortingTable LEFT JOIN " & SuTable & " ON USysSortingTable.PlotNumber = " & SuTable & ".PlotNumber"
    SqlText = SqlText & " WHERE ((" & SuTable & ".PlotNumber Is Not Null));"
    DoCmd.RunSQL SqlText        'run the sql string
End If

'ask the user if the program should overwite existing plot numbers
    Title = ProgramName()
    ' Put together a message box with all the proper components.
    Msg = "Do you want VPro to add the imported site units to your personal list of site units?"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef, Title)    ' Get user response.
If Response = IDYES Then    ' Evaluate response
'create a sql string that inserts records from the sorting instructions into the site name table
    SqlText = "INSERT INTO USysUserSiteUnitList ( Name, UnitLongName )"
    SqlText = SqlText & " SELECT DISTINCT USysSortingTable.Name, USysSortingTable.UnitLongName"
    SqlText = SqlText & " FROM USysSortingTable;"
    DoCmd.RunSQL SqlText        'run the sql string
End If

'create a sql string that inserts records from the sorting instructions into the site unit table
'SqlText = "INSERT INTO " & SuTable & " ( PlotNumber, SiteUnit )"
'SqlText = SqlText & " SELECT DISTINCTROW USysSortingTable.PlotNumber, USysSortingTable.Name"
'SqlText = SqlText & " FROM USysSortingTable, " & SuTable & ";"
SqlText = "INSERT INTO [" & SuTable & "] ( PlotNumber, SiteUnit )"
SqlText = SqlText & " SELECT DISTINCTROW USysSortingTable.PlotNumber, USysSortingTable.GroupShortName"
SqlText = SqlText & " FROM USysSortingTable;"
'CurrentDb.CreateQueryDef "Frank2", SqlText
DoCmd.RunSQL SqlText         'run the sql string
    
    DoCmd.SetWarnings True                   'turn system warnings on
    DoCmd.Hourglass False                    'turn off the hour glass
    Close #1                                 'close the sorting file
    GoTo MyExit
    Exit Function                            'end procedure
    
'XXXXXXXXXXXXXXXXXX  THE FOLLOWING CODE INSERTS MISSING PLOT NUMBERS INTO THE CURRENT ENV TABLE   XXXXXXXXXXX
    On Error Resume Next
    DoCmd.SetWarnings False

    SqlTxt = "INSERT INTO " & TableToSave & " ( PlotNumber ) SELECT DISTINCTROW " & SortingInst & ".PlotNumber FROM " & SortingInst & " LEFT JOIN " & TableToSave & " ON " & SortingInst & ".PlotNumber = " & TableToSave & ".PlotNumber;"
    DoCmd.RunSQL SqlTxt
    
    DoCmd.DeleteObject A_TABLE, "ThisTempSorting"
    DoCmd.CopyObject , "ThisTempSorting", A_TABLE, SortingInst
    
    SqlTxt = "DELETE DISTINCTROW " & SortingInst & ".* FROM " & SortingInst & ";"
    DoCmd.RunSQL SqlTxt
    
    SqlTxt = "INSERT INTO " & SortingInst & " SELECT DISTINCT ThisTempSorting.* FROM ThisTempSorting;"
    DoCmd.RunSQL SqlTxt

    DoCmd.Hourglass True
    
    Close #1
    GoTo MyExit
    Exit Function

WasDupTable:

Msg = "That name is already in use in " & Trim(DestDB)
Title = ProgramName()
MsgBox Msg, 64, Title

MyExit:
    'delete any records from USysSortingTable
    On Error Resume Next
    DoCmd.SetWarnings False
    MsgBox nMatchingPlots("Sorting") & " plots were imported."
    SqlText = "DELETE DISTINCTROW USysSortingTable.*"
    SqlText = SqlText & " FROM USysSortingTable;"
    DoCmd.RunSQL SqlText
    SortingInst = "USysSortingTable"  'assign a variable to USysSortingTable
    DoCmd.SetWarnings True
    Set MyDB = Nothing
    Exit Function

End Function

Public Function ImportVtabVeg()

'Sep 22, 2008
On Error Resume Next
    DoCmd.Hourglass False
    Close #1
    DoCmd.DeleteObject acTable, "tmpImportVegData"
On Error GoTo errImportVtabVeg

Dim MyDB As Database, MyRS As Recordset, MyPlots As Recordset, ImportFile As String
Dim Title As String, VegTable As String, EnvTable As String
Dim TextData As String, MyCriteria As String, Char As String
Dim Layer As String, SqlString As String, SqlText As String
Dim Msg As String, dgdef As Variant, Response As Integer
Dim ThisPlot As String, DeletedPlot As String, z As Integer
Dim PlotsAdded2Env As Integer
Set MyDB = CurrentDb()
PlotsAdded2Env = 0
z = 1
' Put together a message box with all the proper components.
    Title = ProgramName()
    Msg = "The proper order for importing VTAB files is Env first and Veg second.  Continue?"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef, Title)    ' Get user response.
    If Response <> 6 Then GoTo MyExit
    DoCmd.CopyObject , "tmpImportVegData", acTable, "USysVegTable"
    
    Response = Forms![USysImportVtab]![optVegImportOption]
            
    Title = ProgramName() & " - select a vegetation file to import into current project"
    ImportFile = GetFileName(Title, True, "Vegetation (*.veg)", "*.veg")
    If ImportFile = "" Then GoTo MyExit
    Screen.ActiveForm.Repaint

    VegTable = "tmpImportVegData"
    EnvTable = CurrProject() & "_Env"
    Open ImportFile For Input As #1
    If LOF(1) = 0 Then
        MsgBox "This file contains no records"
        GoTo MyExit
    End If

    Set MyRS = MyDB.OpenRecordset(VegTable, dbOpenDynaset)
    Set MyPlots = MyDB.OpenRecordset(EnvTable, dbOpenDynaset)
        
    DoCmd.Hourglass True
    Do While Not EOF(1)
       
    Do
        Char = Input(1, #1)
        TextData = TextData & Char
    Loop Until Char = Chr(10)
        
        ThisPlot = Trim(Mid$(TextData, 1, 7))

        If MyPlots.RecordCount = 0 Then
            MyPlots.AddNew
            MyPlots!PlotNumber = ThisPlot
            MyPlots.Update
            If Forms![USysImportVtab]![optLogVegPlotsImported] Then LogImportedPlots "Veg", ThisPlot, CurrProject(), "Environment"
            PlotsAdded2Env = PlotsAdded2Env + 1
        End If
        If MyPlots.RecordCount >= 1 Then
            MyPlots.MoveLast
            MyPlots.MoveFirst
        End If
        MyCriteria = "PlotNumber = '" & ThisPlot & "'"
        MyPlots.FindFirst MyCriteria
        If MyPlots.NoMatch Then
            MyPlots.AddNew
            MyPlots!PlotNumber = ThisPlot
            MyPlots.Update
            If Forms![USysImportVtab]![optLogVegPlotsImported] Then LogImportedPlots "Veg", ThisPlot, CurrProject(), "Environment"
            PlotsAdded2Env = PlotsAdded2Env + 1
        End If

    If Len(TextData) > 16 Then
        Layer = Mid$(TextData, 8, 1)
        MyRS.AddNew
        MyRS("PlotNumber") = Trim(Mid$(TextData, 1, 7))
'        MyRS("Layer") = Trim(Mid$(TextData, 8, 1))
        MyRS("Species") = Trim(Mid$(TextData, 9, 8))
        MyRS("Cover" & Layer) = Trim(Mid$(TextData, 17, 2))
        If IsNumeric(Trim(Mid$(TextData, 19, 1))) Then MyRS("Dist" & Layer) = Trim(Mid$(TextData, 19, 1))
        If IsNumeric(Trim(Mid$(TextData, 20, 1))) Then MyRS("Vig" & Layer) = Trim(Mid$(TextData, 20, 1))
        MyRS.Update
    End If
    If Len(TextData) > 28 Then
        MyRS.AddNew
        MyRS("PlotNumber") = Trim(Mid$(TextData, 1, 7))
'        MyRS("Layer") = Trim(Mid$(TextData, 8, 1))
        MyRS("Species") = Trim(Mid$(TextData, 21, 8))
        MyRS("Cover" & Layer) = Trim(Mid$(TextData, 29, 2))
        If IsNumeric(Trim(Mid$(TextData, 31, 1))) Then MyRS("Dist" & Layer) = Trim(Mid$(TextData, 31, 1))
        If IsNumeric(Trim(Mid$(TextData, 32, 1))) Then MyRS("Vig" & Layer) = Trim(Mid$(TextData, 32, 1))
        MyRS.Update
    End If
    If Len(TextData) > 40 Then
        MyRS.AddNew
        MyRS("PlotNumber") = Trim(Mid$(TextData, 1, 7))
'        MyRS("Layer") = Trim(Mid$(TextData, 8, 1))
        MyRS("Species") = Trim(Mid$(TextData, 33, 8))
        MyRS("Cover" & Layer) = Trim(Mid$(TextData, 41, 2))
        If IsNumeric(Trim(Mid$(TextData, 43, 1))) Then MyRS("Dist" & Layer) = Trim(Mid$(TextData, 43, 1))
        If IsNumeric(Trim(Mid$(TextData, 44, 1))) Then MyRS("Vig" & Layer) = Trim(Mid$(TextData, 44, 1))
        MyRS.Update
    End If
    If Len(TextData) > 52 Then
        MyRS.AddNew
        MyRS("PlotNumber") = Trim(Mid$(TextData, 1, 7))
'        MyRS("Layer") = Trim(Mid$(TextData, 8, 1))
        MyRS("Species") = Trim(Mid$(TextData, 45, 8))
        MyRS("Cover" & Layer) = Trim(Mid$(TextData, 53, 2))
        If IsNumeric(Trim(Mid$(TextData, 55, 1))) Then MyRS("Dist" & Layer) = Trim(Mid$(TextData, 55, 1))
        If IsNumeric(Trim(Mid$(TextData, 56, 1))) Then MyRS("Vig" & Layer) = Trim(Mid$(TextData, 56, 1))
        MyRS.Update
    End If
    If Len(TextData) > 64 Then
        MyRS.AddNew
        MyRS("PlotNumber") = Trim(Mid$(TextData, 1, 7))
'        MyRS("Layer") = Trim(Mid$(TextData, 8, 1))
        MyRS("Species") = Trim(Mid$(TextData, 57, 8))
        MyRS("Cover" & Layer) = Trim(Mid$(TextData, 65, 2))

        If IsNumeric(Trim(Mid$(TextData, 68, 1))) Then MyRS("Vig" & Layer) = Trim(Mid$(TextData, 68, 1))
        MyRS.Update
    End If
    If Len(TextData) > 76 Then
        MyRS.AddNew
        MyRS("PlotNumber") = Trim(Mid$(TextData, 1, 7))
'        MyRS("Layer") = Trim(Mid$(TextData, 8, 1))
        MyRS("Species") = Trim(Mid$(TextData, 69, 8))
        MyRS("Cover" & Layer) = Trim(Mid$(TextData, 77, 2))
        If IsNumeric(Trim(Mid$(TextData, 79, 1))) Then MyRS("Dist" & Layer) = Trim(Mid$(TextData, 79, 1))
        If IsNumeric(Trim(Mid$(TextData, 80, 1))) Then MyRS("Vig" & Layer) = Trim(Mid$(TextData, 80, 1))
        MyRS.Update
    End If
       
ReLoop:
     TextData = ""
    z = z + 1
Loop

    If Forms![USysImportVtab]![optVegImportOption] = 1 Then
        AppendMissingPlots "Veg"
    Else
        MakePlotList
        DeleteMatchingPlots "Veg"
        AppendMissingPlots "Veg"
    End If
    
If Forms![USysImportVtab]![optLogVegPlotsImported] Then LogImportedPlots "AllVeg", ThisPlot, CurrProject(), "Vegetation", sysFileTitle
MsgBox PlotsAdded2Env & " new plots were added to your environment table and " & nMatchingPlots("Veg") & " records were added to your vegetation table.", vbInformation, ProgramName()

MyExit:
    On Error Resume Next
    Close #1
    MyRS.Close
    ClearPlotList
    DoCmd.DeleteObject acTable, "tmpImportVegData"
    DoCmd.Close acForm, "USysImportVtab"
    DoCmd.Close A_FORM, FormSet()

    DoCmd.Hourglass False
    Set MyDB = Nothing
    Exit Function
errImportVtabVeg:
    DoCmd.Hourglass False
    MsgBox "The following problem happened while importing line: " & z & "  " & Err.Description
    Resume MyExit
End Function

End Function

Private Function nMatchingPlots(ImportType As String) As Integer
On Error GoTo errnMatchingPlots
Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String, ProjectName As String
Set MyDB = CurrentDb()
ProjectName = CurrProject() & "_Env"

Select Case ImportType
    Case "Env"
        SqlText = "SELECT DISTINCTROW MergeEnv.PlotNumber"
        SqlText = SqlText & " FROM MergeEnv LEFT JOIN " & ProjectName & " ON MergeEnv.PlotNumber = " & ProjectName & ".PlotNumber"
        SqlText = SqlText & " WHERE (((" & ProjectName & ".PlotNumber) Is Null));"
        Set MyRS = MyDB.OpenRecordset(SqlText)
        nMatchingPlots = MyRS.RecordCount
    Case "Veg"
        SqlText = "SELECT PlotNumber FROM tmpImportVegData;"
        Set MyRS = MyDB.OpenRecordset(SqlText)
        MyRS.MoveLast
        nMatchingPlots = MyRS.RecordCount
    Case "Sorting"
        SqlText = "SELECT DISTINCT PlotNumber FROM USysSortingTable;"
        Set MyRS = MyDB.OpenRecordset(SqlText)
        MyRS.MoveLast
        nMatchingPlots = MyRS.RecordCount
    Case Else
        GoTo MyExit
End Select

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
errnMatchingPlots:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Sub AppendMissingPlots(TableType As String)
On Error GoTo errAppendMissingPlots
Dim SqlText As String, ProjectName As String, VegTable As String

Select Case TableType
    Case "Env"
        ProjectName = CurrProject() & "_Env"
        SqlText = "INSERT INTO " & ProjectName
        SqlText = SqlText & " SELECT MergeEnv.*"
        SqlText = SqlText & " FROM MergeEnv LEFT JOIN " & ProjectName & " ON MergeEnv.PlotNumber = " & ProjectName & ".PlotNumber"
        SqlText = SqlText & " WHERE (((" & ProjectName & ".PlotNumber) Is Null));"
    Case "Veg"
        ProjectName = CurrProject() & "_Env"
        VegTable = CurrProject() & "_Veg"
        SqlText = "INSERT INTO " & VegTable

DoCmd.SetWarnings False
DoCmd.RunSQL SqlText

    DoCmd.SetWarnings True
    Exit Sub

    Resume MyExit

Private Sub LogImportedPlots(LogFrom As String, Optional PlotNumber As String, Optional TableName As String, Optional TableType As String, Optional FileName As Variant)
'Apr. 10,, 2002
On Error GoTo errLogImportedPlots
Dim MyDB As Database, MyRS As Recordset, LogRS As Recordset
Dim SqlText As String, ProjectName As String, MyProject As String, myUser As String
Dim i As Integer
Set MyDB = CurrentDb()
MyProject = CurrProject()
ProjectName = MyProject & "_Env"
myUser = CurrentUser()
Set LogRS = MyDB.OpenRecordset("USysAuditTrail", dbOpenDynaset)
If IsMissing(FileName) Then FileName = "" Else FileName = " - File: " & FileName

Select Case LogFrom
    Case "Env"
        SqlText = "Select * FROM AuditList"
        Set MyRS = MyDB.OpenRecordset(SqlText)
        If MyRS.RecordCount = 0 Then GoTo MyExit
        MyRS.MoveLast
        MyRS.MoveFirst
        
        For i = 1 To MyRS.RecordCount
            LogRS.AddNew
            LogRS!Project = MyProject
            LogRS!User = myUser
            LogRS!PlotNumber = MyRS!PlotNumber
            LogRS!Table = "Environment"
            LogRS!EditField = "Imported VTAB record" & FileName
            LogRS!EditWhen = Now()
            LogRS.Update
            MyRS.MoveNext
        Next i
    Case "Veg"
            LogRS.AddNew
            LogRS!Project = MyProject
            LogRS!User = myUser
            LogRS!PlotNumber = PlotNumber
            LogRS!Table = TableType
            LogRS!EditField = "Imported VTAB record" & FileName
            LogRS!EditWhen = Now()
            LogRS.Update
    Case "AllVeg"
        SqlText = "SELECT DISTINCT PlotNumber FROM tmpImportVegData;"
        Set MyRS = MyDB.OpenRecordset(SqlText)
        If MyRS.RecordCount = 0 Then GoTo MyExit
        MyRS.MoveLast
        MyRS.MoveFirst
        For i = 1 To MyRS.RecordCount
            LogRS.AddNew
            LogRS!Project = MyProject
            LogRS!User = myUser
            LogRS!PlotNumber = MyRS!PlotNumber
            LogRS!Table = TableType
            LogRS!EditField = "Imported VTAB record" & FileName
            LogRS!EditWhen = Now()
            LogRS.Update
            MyRS.MoveNext
        Next i
    Case "Sorting"
        SqlText = "SELECT DISTINCT PlotNumber FROM USysSortingTable;"
        Set MyRS = MyDB.OpenRecordset(SqlText)
        If MyRS.RecordCount = 0 Then GoTo MyExit
        MyProject = CurrPlotlist()
        MyRS.MoveLast
        MyRS.MoveFirst
        For i = 1 To MyRS.RecordCount
            LogRS.AddNew
            LogRS!Project = MyProject
            LogRS!User = myUser
            LogRS!PlotNumber = MyRS!PlotNumber
            LogRS!Table = "Sorting"
            LogRS!EditField = "Imported VTAB record" & FileName
            LogRS!EditWhen = Now()
            LogRS.Update
            MyRS.MoveNext
        Next i
    Case Else
        GoTo MyExit
End Select

MyExit:
    On Error Resume Next
    MyRS.Close
    LogRS.Close
    Set MyDB = Nothing
    Exit Sub
errLogImportedPlots:
    MsgBox Err.Description
    Resume MyExit
End Sub
