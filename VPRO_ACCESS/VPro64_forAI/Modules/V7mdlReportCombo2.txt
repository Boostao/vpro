Private Function GetComboSppColourCode(Species As String, rs As Recordset) As Integer
'not in use
'Apr. 7, 1999
Dim Criteria As String
rs.MoveFirst
Criteria = "ScientificName = '" & Species & "'"
rs.FindFirst Criteria

If Not rs.NoMatch Then
    GetComboSppColourCode = rs!ColourCode
Else
    GetComboSppColourCode = 999
End If

End Function
Private Function GetFormatCode(FormatObject As String, Species As String, rs As Recordset) As Integer
'Sep 10, 2008
Dim Criteria As String
rs.MoveFirst
Criteria = "ScientificName = '" & Species & "'"
rs.FindFirst Criteria

Select Case FormatObject
    Case "CellColour"
        If Not rs.NoMatch Then
            GetFormatCode = rs!ColourCode
        Else
            GetFormatCode = 999
        End If
    Case "CellPattern"
        If Not rs.NoMatch Then
            GetFormatCode = rs!PatternCode
        Else
            GetFormatCode = 999
        End If
    Case "FontColour"
        If Not rs.NoMatch Then
            GetFormatCode = rs!FontColour
        Else
            GetFormatCode = 999
        End If
    Case Else
        GetFormatCode = 999
End Select
End Function

Private Function FormatComboReport() '(FileName As String)

'Sep 10, 2008
Dim X As Long, Y As Long, cell As Variant, GreaterX As Single, GreaterY As Single, MyRow As Variant, ThemeTable As String
Dim xlSheet As Object, CellColour As Integer, MyAddress As String, FontColour As Integer, PatternCode As Integer
Dim GreaterYtext As String
Dim AddX As Integer
Dim MyDB As Database, MyRS As Recordset
Set MyDB = CurrentDb()
ThemeTable = Forms![USysComboReportOptions]![cmbThemeTable] & "_Theme"
If ThemeTable <> "_Theme" Then
    Set MyRS = MyDB.OpenRecordset(ThemeTable, dbOpenDynaset)
    MyRS.MoveLast
End If
Set xlSheet = xl.ActiveSheet ' GetObject(FileName)
GreaterX = Forms![USysComboReportOptions]![optX]
GreaterY = Forms![USysComboReportOptions]![optY]
GreaterYtext = "'" & Forms![USysComboReportOptions]![optY] & "'"
xl.Visible = True

With xl.Sheets("ComboReport")
    .Rows("1:5").Columns.AutoFit
    .Columns("A:A").EntireColumn.AutoFit
    .Rows("6:6").Orientation = 90
    .Rows("6:6").WrapText = True
    .Rows("6:6").Font.Name = "Arial"
    .Rows("6:6").Font.Size = 8
    .Rows("6:6").RowHeight = 100
    .Range("A1") = "Plotnumber"
    Y = xl.CountA(.Range("1:1"))
    X = xl.CountA(.Range("A:A"))
    AddX = X
    .Names.Add "VegData", .Range(.Cells(33, 1), .Cells(X, Y))
    X = .Range("VegData").Cells.count
    .Range("D32") = "1"
    .Range("D32").Copy
    'MsgBox .Name
    'MyAddress = .Range("VegData").Address
    MyAddress = xl.Range(.Cells(33, 1), .Cells(AddX, Y)).Address
    'MsgBox MyAddress
    .Range(MyAddress).SpecialCells(2, 23).PasteSpecial 1, 4
    .Range("D32") = Null
    .Range(MyAddress).NumberFormat = "0.0"
    Y = 0
    
If Forms![USysComboReportOptions]![optConditionalMethod] = 1 Then
    For Each MyRow In .Range(MyAddress).Rows
        Y = Y + 1
        CellColour = GetFormatCode("CellColour", MyRow.Columns(1).Value, MyRS)
        FontColour = GetFormatCode("FontColour", MyRow.Columns(1).Value, MyRS)
        PatternCode = GetFormatCode("CellPattern", MyRow.Columns(1).Value, MyRS)
             If CellColour <> 999 Then
                .Rows(MyRow.Row).FormatConditions.Add 1, 5, GreaterY
                .Rows(MyRow.Row).FormatConditions(1).Interior.ColorIndex = CellColour
                .Rows(MyRow.Row).FormatConditions(1).Interior.Pattern = PatternCode
                .Rows(MyRow.Row).FormatConditions(1).Font.ColorIndex = FontColour
                .Rows(MyRow.Row).FormatConditions(1).Font.Bold = True
                .Rows(MyRow.Row).FormatConditions.Add 1, 1, GreaterX, 101
                .Rows(MyRow.Row).FormatConditions(2).Interior.ColorIndex = 15
            Else
                .Rows(MyRow.Row).FormatConditions.Add 1, 1, GreaterX, 101
                .Rows(MyRow.Row).FormatConditions(1).Interior.ColorIndex = 15
            End If
    Next MyRow
End If
If Forms![USysComboReportOptions]![optConditionalMethod] = 2 Then
    For Each MyRow In .Range("VegData").Rows 'MyAddress).Rows
        Y = Y + 1
        CellColour = GetFormatCode("CellColour", MyRow.Columns(1).Value, MyRS)
        FontColour = GetFormatCode("FontColour", MyRow.Columns(1).Value, MyRS)
        PatternCode = GetFormatCode("CellPattern", MyRow.Columns(1).Value, MyRS)
             If CellColour <> 999 Then
                For Each cell In MyRow.Cells '.Rows(MyRow.Row)
                    '.Cells(1, 1) = MyRow.Address
                    If Not IsNull(cell.Value) And (Val(cell.Value) > GreaterY Or cell.Value > GreaterYtext) Then 'Val(cell.Value) > 5 Then
                        With cell
                            .Interior.ColorIndex = CellColour
                            .Interior.Pattern = PatternCode
                            .Font.ColorIndex = FontColour
                            .Font.Bold = True
                        End With
                    End If
                Next cell
            Else
                For Each cell In MyRow.Cells
                    If Not IsNull(cell.Value) And Val(cell.Value) > GreaterX Then 'Val(cell.Value) > 5 Then
                        With cell
                            .Interior.ColorIndex = 15
                            .Font.Bold = True
                        End With
                    End If
                Next cell
            End If
    Next MyRow
End If

End With

Jump:
'    xlSheet.Save
    xl.Cursor = xlDefault
    
MyExit:
    On Error Resume Next
    Set xl = Nothing
'    Set xlSheet = Nothing
    MyRS.Close
    Set MyDB = Nothing
    Application.SysCmd acSysCmdClearStatus
    Exit Function

End Function
Private Function OpenReportInExcel(FileName As String)

'Sep 10, 2008
'On Error GoTo errOpenReportInExcel
Dim MyBook As Workbook, i As Integer
CreateExcel xl
xl.Workbooks.Add
xl.Cursor = xlWait
        
    xl.Workbooks.Open FileName
    For i = 1 To xl.Workbooks.count
        If Not xl.Workbooks(i).Name = "Personal.xls" Then
               xl.Windows(xl.Workbooks(i).Name).Visible = True
        End If
    Next i
    
MyExit:
    On Error Resume Next
    Exit Function
    
errOpenReportInExcel:
    If Err.Number = 9 Then
        xl.Visible = True
        Resume MyExit
    Else
        MsgBox Err.Description
        xl.Visible = True
        Resume MyExit
    End If
End Function

Private Function ExportComboReport2Excel()

'Sep 22, 2008
'On Error GoTo errSendComboReport2Excel
Dim MyFile As String, i As Integer, MyTitle As String

    MyTitle = "Save combo report to"
    'MyFile = Trim(GetFileName(MyTitle, Filter, 23, "save", "*.xls"))
    MyFile = GetFileName(MyTitle, False, "Excel (*.xls)", "*.xls")
    If MyFile = "" Then GoTo MyExit

On Error Resume Next
Kill MyFile
On Error GoTo errSendComboReport2Excel
    DoCmd.TransferSpreadsheet acExport, 5, "ComboReport", MyFile
    
    OpenReportInExcel MyFile

'format report
Screen.ActiveForm.Repaint
FormatComboReport MyFile
DeleteOldComboParts

MyExit:
    On Error Resume Next
    DoCmd.Close acForm, "USysComboReportOptions"
    Exit Function
errSendComboReport2Excel:
    If Err.Number = 70 Or Err.Number = 75 Then
        MsgBox "A report is already open in Excel.  Close that report and try again"
        Resume MyExit
    Else
        If Err.Number = 53 Or Err.Number = 432 Then Resume Next Else
            MsgBox Err.Description
        Resume MyExit
    End If
End Function

Private Sub AddCover(Spp As String, Row As Integer, db As Database, Strata As String)
Dim i As Integer, Plot As Variant
Dim SqlText As String, Criteria As String
Dim MyRS As Recordset

SqlText = "SELECT Spp, Strata, PlotNumber, SumOfCover"
SqlText = SqlText & " FROM CmbVegData"
SqlText = SqlText & " WHERE (((Spp)='" & Spp & "') AND ((Strata)='" & Strata & "'))"
SqlText = SqlText & " ORDER BY Spp, Strata, PlotNumber;"
Set MyRS = db.OpenRecordset(SqlText)
MyRS.MoveLast


For i = 2 To xl.CountA(Range("1:1"))
    MyRS.MoveFirst
    Plot = xl.Cells(1, i)
    Criteria = "PlotNumber = '" & Plot & "'"
    MyRS.FindFirst Criteria
    If MyRS.NoMatch Then
        'do nothing
    Else
        xl.Cells(Row, i) = MyRS!SumOfCover
    End If
Next i

MyRS.Close

End Sub

    ReportRS![Env Field] = "Strata Cover Moss(%)"
    For i = 1 To EnvRS.RecordCount
        ReportRS.Fields(EnvRS!PlotNumber).Value = EnvRS.Fields("StrataCoverMoss").Value
        EnvRS.MoveNext
    Next i
        ReportRS.Update
        EnvRS.MoveFirst

'add species list for Strata D
    Application.SysCmd acSysCmdUpdateMeter, 8
    SqlText = "SELECT DISTINCT CmbVegData.Strata, CmbVegData.Spp FROM CmbVegData WHERE (((CmbVegData.Strata) = 'D')) ORDER BY CmbVegData.Spp;"
    Set SppRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)
    For i = 1 To SppRS.RecordCount
        ReportRS.AddNew
        ReportRS![Env Field] = "D-" & SppRS!Spp
        ReportRS.Update
        SppRS.MoveNext
    Next i
        SppRS.Close
    Application.SysCmd acSysCmdUpdateMeter, 9

'add cover data
SqlText = "TRANSFORM Sum(CmbVegData.SumOfCover) AS SumOfSumOfCover"
SqlText = SqlText & " SELECT CmbVegData.Strata, CmbVegData.Spp"
SqlText = SqlText & " FROM CmbVegData"
SqlText = SqlText & " GROUP BY CmbVegData.Strata, CmbVegData.Spp"
SqlText = SqlText & " PIVOT CmbVegData.PlotNumber;"
Set VegRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)
VegRS.MoveLast
VegRS.MoveFirst
Application.SysCmd acSysCmdInitMeter, "Adding vegetation data", VegRS.RecordCount

For i = 1 To VegRS.RecordCount
    Application.SysCmd acSysCmdUpdateMeter, i
    Criteria = "[Env Field] = '" & VegRS!Strata & "-" & VegRS!Spp & "'"
    ReportRS.MoveFirst
    ReportRS.FindFirst Criteria
    ReportRS.Edit
    For X = 2 To VegRS.Fields.count - 1
        MyData = VegRS.Fields(X).Name
            ReportRS.Fields(MyData).Value = VegRS.Fields(MyData)
    Next X
    ReportRS.Update
    VegRS.MoveNext
Next i

'clean up species labels
SppClean ReportRS
ExportComboReport2Excel

MyExit:
    On Error Resume Next
    DoCmd.Hourglass False
    Application.SysCmd acSysCmdClearStatus
    FieldsRS.Close
    SppRS.Close
    ReportRS.Close
    EnvRS.Close
    VegRS.Close
    Set MyDB = Nothing
    DeleteOldComboParts
    DoCmd.SetWarnings True
    Exit Function
errComboReport:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function SppClean(rs As Recordset)
Dim Criteria As String
    rs.MoveFirst
    Criteria = "left([Env Field],2) = 'A-'"
    rs.FindFirst Criteria
    Application.SysCmd acSysCmdInitMeter, "Formatting species labels", rs.RecordCount
    Do Until rs.NoMatch
        Application.SysCmd acSysCmdUpdateMeter, rs.AbsolutePosition
        rs.Edit
        rs![Env Field] = Mid(rs![Env Field], 3)
        rs.Update
        rs.FindNext Criteria
    Loop

    rs.MoveFirst
    Criteria = "left([Env Field],2) = 'B-'"
    rs.FindFirst Criteria
    Do Until rs.NoMatch
        Application.SysCmd acSysCmdUpdateMeter, rs.AbsolutePosition
        rs.Edit
        rs![Env Field] = Mid(rs![Env Field], 3)
        rs.Update
        rs.FindNext Criteria
    Loop

    rs.MoveFirst
    Criteria = "left([Env Field],2) = 'C-'"
    rs.FindFirst Criteria
    Do Until rs.NoMatch
        Application.SysCmd acSysCmdUpdateMeter, rs.AbsolutePosition
        rs.Edit
        rs![Env Field] = Mid(rs![Env Field], 3)
        rs.Update
        rs.FindNext Criteria
    Loop

    rs.MoveFirst
    Criteria = "left([Env Field],2) = 'D-'"
    rs.FindFirst Criteria
    Do Until rs.NoMatch
        Application.SysCmd acSysCmdUpdateMeter, rs.AbsolutePosition
        rs.Edit
        rs![Env Field] = Mid(rs![Env Field], 3)
        rs.Update
        rs.FindNext Criteria
    Loop

End Function

Public Function CheckForMissingComboCodes()
'Apr. 6, 1999
'On Error GoTo errCheckForMissingCodes
Dim MyDB As Database, MyRS As Recordset, WriteRS As Recordset, ThemeTable As String
Dim SqlText As String, i As Integer, Criteria As String
Set MyDB = CurrentDb()
ThemeTable = Forms![USysComboLumpMaster]![cmbLumpTable] & "_Theme"
'ThemeTable = Forms![USysComboReportOptions]![cmbThemeTable] & "_Theme"
Set WriteRS = MyDB.OpenRecordset(ThemeTable, dbOpenDynaset) '"USysComboColourTheme"
SqlText = "SELECT DISTINCT LumpCode FROM " & ThemeTable & " ORDER BY LumpCode;"
Set MyRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)
MyRS.MoveLast

If MyRS.RecordCount < 10 Then
    For i = 1 To 10
        Criteria = "LumpCode = 'Group" & Format(i, "00") & "'"
        MyRS.MoveFirst
        MyRS.FindFirst Criteria
        If MyRS.NoMatch Then
            WriteRS.AddNew
            WriteRS!LumpCode = "Group" & Format(i, "00")
            WriteRS.Update
        End If
    Next i
End If

MyExit:
    On Error Resume Next
    MyRS.Close
    WriteRS.Close
    Set MyDB = Nothing
    Exit Function
errCheckForMissingCodes:
    MsgBox Err.Description
    Resume MyExit
End Function
