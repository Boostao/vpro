Public Function HierarchyDiagram()
'Dec. 30, 2019

Dim db As Database
Set db = CurrentDb()
Dim MyRS As Recordset, MyQD As QueryDef
Dim SqlText As String, i As Integer
Dim BK As String
Dim CutOffLevel As Integer
CutOffLevel = 13

On Error Resume Next
CutOffLevel = InputBox("What is the lowest level you wish to display?", "VPro", 11)
If CutOffLevel = 13 Then GoTo MyExit
On Error GoTo 0

Set LnRS = db.OpenRecordset("MasterSiteUnitList", dbOpenDynaset)
SqlText = "SELECT ID, Name"
SqlText = SqlText & " FROM [" & CurrHierarchy() & "]"
SqlText = SqlText & " WHERE (((Parent) Is Null));"
Set MyRS = db.OpenRecordset(SqlText)

MyRS.MoveLast
MyRS.MoveFirst
MyRow = 1
MyCol = 1
NameNum = 1
NameStr = "Name"
ShapeName = NameStr & NameNum

Set xl = CreateObject("Excel.Application")
xl.Visible = True
xl.Workbooks.Add
xl.ActiveWindow.DisplayGridlines = False
xl.Cursor = xlWait
xl.ScreenUpdating = False

For i = 1 To MyRS.RecordCount
    BK = MyRS.Bookmark
    If MyCol = 0 Then MyCol = 1
    xl.Cells(1, 1) = MyRS!Name & " : " & GetLongName(MyRS!Name)
    AddShape ShapeName 'MyRS!Name 'draw a box
    MyRow = MyRow + 1
    MyCol = MyCol + 2
    
    BuildListInXl db, MyRS!ID, xl, CutOffLevel, ShapeName 'MyRS!Name
    MyRS.Bookmark = BK
    MyRS.MoveNext
Next i

    xl.Cells.Columns.AutoFit
    FormatPage
    xl.ScreenUpdating = True
    xl.Cursor = xlDefault

MyExit:
    Set xl = Nothing
    MyRS.Close
'----------------------------------------------------------------------------------------------------
'May 1, 2007
    'HlRS.Close
'----------------------------------------------------------------------------------------------------
    Set db = Nothing

End Function

Private Sub FormatPage()
'Oct. 11, 2002
    With xl.ActiveSheet.PageSetup
        .LeftFooter = CurrHierarchy
        .LeftMargin = xl.Application.InchesToPoints(0.25)
        .RightMargin = xl.Application.InchesToPoints(0.25)
        .TopMargin = xl.Application.InchesToPoints(0.75)
        .BottomMargin = xl.Application.InchesToPoints(0.75)
        .PrintGridlines = False
        .CenterHorizontally = True
        .Orientation = xlLandscape
        .FitToPagesWide = 1
        .FitToPagesTall = False
    End With
End Sub
Private Sub AddShape(ShapeName As String, Optional ShapeParent As Variant)
'Oct. 11, 2002
    With xl.Cells(MyRow, MyCol)
        xl.ActiveSheet.Shapes.AddShape(msoShapeRectangle, .Left, .Top, .Width, .Height).Select
        xl.Selection.ShapeRange.Fill.Visible = msoFalse
        xl.Selection.Name = ShapeName
        .VerticalAlignment = xlTop
        .WrapText = True
        .ColumnWidth = 14
        .Rows.AutoFit
        .Font.Name = "Small Fonts"
        .Font.FontStyle = "Regular"
        .Font.Size = 7
    End With
    If Left(xl.Cells(MyRow, MyCol), 1) = "~" Then
        With xl.Cells(MyRow, MyCol).Interior
            .ColorIndex = 6
            .Pattern = xlSolid
        End With
    End If
End Sub
Private Sub IsLowestTildeMember(UnitID As Integer, MyRecords As Recordset)
'Oct. 11, 2002
Dim MyList As String, MyCriteria As String
Dim NextUnit As Integer, Shifty As Integer, BK As String, ParentShapeName As String

MyRecords.MoveFirst

    MyCriteria = "ID = " & UnitID
    MyRecords.FindFirst MyCriteria
    If MyRecords.NoMatch Then GoTo MyExit
    
    Shifty = MyRecords!ID
    MyRecords.MoveFirst
    MyCriteria = "Parent = " & Shifty
    MyRecords.FindFirst MyCriteria
Do Until MyRecords.NoMatch
    If Left(MyRecords!Name, 1) = "~" Then
        WasLowest = False
    End If
    BK = MyRecords.Bookmark
    NextUnit = MyRecords!ID
    IsLowestTildeMember NextUnit, MyRecords
    MyRecords.Bookmark = BK
    MyRecords.FindNext MyCriteria
Loop

MyExit:
    Exit Sub

End Sub
Private Sub AddArrow(ShapeName As String, ShapeParent As String)
'Oct. 11, 2002
With xl
    .ActiveSheet.Shapes.AddConnector(msoConnectorElbow, 77.25, 19.5, 45.75, 17.25). _
        Select
    .Selection.ShapeRange.Line.EndArrowheadStyle = msoArrowheadTriangle
    .Selection.ShapeRange.Flip msoFlipHorizontal
    .Selection.ShapeRange.Flip msoFlipVertical
    .Selection.ShapeRange.ConnectorFormat.BeginConnect .ActiveSheet.Shapes(ShapeParent), 4
    .Selection.ShapeRange.ConnectorFormat.EndConnect .ActiveSheet.Shapes(ShapeName), 2
End With

End Sub

Private Function BuildListInXl(db As Database, Unit As Long, xl As Object, CutOffLevel As Integer, Optional ParentName As Variant)
'Oct. 11, 2002
Dim MyRecords As Recordset, MyList As String, MyCriteria As String
Dim NextUnit As Long, Shifty As Long, BK As String, ParentShapeName As String
Dim SqlText As String, CutLevel As Integer

CutLevel = CutOffLevel
SqlText = "SELECT * FROM [" & CurrHierarchy() & "] WHERE (((Level)<=" & CutLevel & "));"

Set MyRecords = db.OpenRecordset(SqlText, DB_OPEN_DYNASET)
MyRecords.MoveFirst
MyCriteria = "ID = " & Unit
    MyRecords.FindFirst MyCriteria
    If MyRecords.NoMatch Then GoTo MyExit
    Shifty = MyRecords!ID
        
    MyRecords.MoveFirst
    MyCriteria = "Parent = " & Shifty
    MyRecords.FindFirst MyCriteria
    ParentShapeName = ParentName

Do Until MyRecords.NoMatch
    BK = MyRecords.Bookmark
    NextUnit = MyRecords!ID
    If MyCol = 0 Then MyCol = 1
    If Right(MyRecords!Name, 1) = "?" Then
        GoTo MyExit
    End If
    xl.Cells(MyRow, MyCol) = MyRecords!Name & " : " & GetLongName(MyRecords!Name)

'----------------------------------------------------------------------------------------------------
'May 1, 2007
If MyRecords!Tag = "Yellow" Then
    xl.Cells(MyRow, MyCol).Interior.ColorIndex = 6
End If
If MyRecords!Tag = "Blue" Then
    xl.Cells(MyRow, MyCol).Interior.ColorIndex = 33
End If
If MyRecords!Tag = "Green" Then
    xl.Cells(MyRow, MyCol).Interior.ColorIndex = 4
End If
If MyRecords!Tag = "Red" Then
    xl.Cells(MyRow, MyCol).Interior.ColorIndex = 3
End If
If MyRecords!Tag = "Orange" Then
    xl.Cells(MyRow, MyCol).Interior.ColorIndex = 45
End If

    'If GetHighlight(MyRecords!Name) And MyRecords!Level = 11 Then
    '    xl.Cells(MyRow, MyCol).Interior.ColorIndex = 6
    '    xl.Cells(MyRow, MyCol).AddComment
    '    xl.Cells(MyRow, MyCol).Comment.Text Text:=nPlots 'ActiveCell.Value
    'End If
'----------------------------------------------------------------------------------------------------
    NameNum = NameNum + 1
    ShapeName = NameStr & NameNum

    AddShape ShapeName 'draw a box
    AddArrow ShapeName, ParentShapeName

    MyRow = MyRow + 1
    MyCol = MyCol + 2

Jump:
    BuildListInXl db, NextUnit, xl, CutLevel, ShapeName
    MyRecords.Bookmark = BK
    MyRecords.FindNext MyCriteria
Loop

MyExit:
    MyRecords.Close
    MyCol = MyCol - 2
End Function
