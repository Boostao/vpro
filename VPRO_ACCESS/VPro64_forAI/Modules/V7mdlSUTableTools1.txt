Private Function CreateSuFromFields()
'NO LONGER IN USE
'Oct. 25, 2002
Dim MyDB As Database, ListRS As Recordset, EnvRS As Recordset
Dim Criteria As String, i As Integer, SqlText As String
Dim SuList As String, MyProject As String
Set MyDB = CurrentDb()
SuList = CurrPlotlist & "_SU"
MyProject = "USysEnv"

If CurrPlotlist <> "None_SU" Then
    SqlText = "SELECT [" & MyProject & "].*, *"
    SqlText = SqlText & " FROM [" & SuList & "] LEFT JOIN [" & MyProject & "] ON [" & SuList & "].PlotNumber = [" & MyProject & "].PlotNumber;"
Else
    SqlText = "SELECT * FROM [" & MyProject & "];"
End If

Set ListRS = MyDB.OpenRecordset("MasterSiteUnitList", dbOpenDynaset)
Set EnvRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)

EnvRS.MoveLast
EnvRS.MoveFirst

For i = 1 To EnvRS.RecordCount
    Criteria = "Zone = '" & EnvRS!Zone & "' And SubZone = '" & EnvRS!SubZone & "' And SiteSeries = '" & EnvRS!SiteSeries & "'"

    ListRS.FindFirst Criteria

    If Not ListRS.NoMatch Then
        MsgBox ListRS!Name
    
    End If
    EnvRS.MoveNext
Next i

MyExit:
    On Error Resume Next
    ListRS.Close
    EnvRS.Close
    Set MyDB = Nothing
    Exit Function

End Function

Public Function InsertEnvSuIntoSu()
'Dec. 30, 2019
'inserts the assigned siteunit field into the current sute unit table
On Error GoTo errInsertEnvSuIntoSu
Dim MyDB As Database, MyRS As Recordset
Dim EnvTable As String, SuTable As String, SqlText As String, i As Long
Set MyDB = CurrentDb()
    'EnvTable = CurrProject() & "_Env"
    EnvTable = "USysEnv"
    SuTable = CurrPlotlist() & "_SU"
    If SuTable = "None_SU" Then MsgBox "You must have a current site unit table", 64, "VPro"
    If SuTable = "None_SU" Then Exit Function
    If WarnUser("This action will modify the current site unit table.  Do you wish to continue?", "VPro") = 0 Then
        SqlText = "UPDATE DISTINCTROW [" & EnvTable & "] INNER JOIN [" & SuTable & "] ON [" & EnvTable & "].PlotNumber = [" & SuTable & "].PlotNumber SET [" & SuTable & "].SiteUnit = [" & EnvTable & "]![UserSiteUnit];"
        DoCmd.RunSQL SqlText
    Else
        GoTo MyExit
    End If
    'add units to personal list
    'SqlText = "SELECT UserSiteUnit, SiteUnitLongName, Name"
    SqlText = "SELECT UserSiteUnit, SiteUnitLongName, MasterSiteUnitList.SiteSeries"
    'SqlText = SqlText & " FROM [" & EnvTable & "] LEFT JOIN MasterSiteUnitList ON [" & EnvTable & "].UserSiteUnit = MasterSiteUnitList.Name"
    SqlText = SqlText & " FROM [" & EnvTable & "] LEFT JOIN MasterSiteUnitList ON [" & EnvTable & "].UserSiteUnit = MasterSiteUnitList.SiteSeries"
    SqlText = SqlText & " WHERE ((Not (UserSiteUnit) Is Null) AND (Not (SiteUnitLongName) Is Null) AND ((MasterSiteUnitList.SiteSeries) Is Null));"
    Set MyRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)

    If MyRS.RecordCount > 0 Then
        If WarnUser("VPro has found site units that are defined in your environmental table but do not exist in the provincial or your personal site unit list.  Do you want VPro to add these units to your personal list?", ProgramName()) = 0 Then
            'SqlText = "INSERT INTO USysUserSiteUnitList ( Name, UnitLongName )"
            SqlText = "INSERT INTO USysUserSiteUnitList ( SiteSeries, SiteSeriesLongName )"
            SqlText = SqlText & " SELECT UserSiteUnit, SiteUnitLongName"
            'SqlText = SqlText & " FROM [" & EnvTable & "] LEFT JOIN MasterSiteUnitList ON [" & EnvTable & "].UserSiteUnit = MasterSiteUnitList.Name"
            SqlText = SqlText & " FROM [" & EnvTable & "] LEFT JOIN MasterSiteUnitList ON [" & EnvTable & "].UserSiteUnit = MasterSiteUnitList.SiteSeries"
            'SqlText = SqlText & " WHERE ((Not (UserSiteUnit) Is Null) AND (Not (SiteUnitLongName) Is Null) AND ((Name) Is Null));"
            SqlText = SqlText & " WHERE ((Not (UserSiteUnit) Is Null) AND (Not (SiteUnitLongName) Is Null) AND ((MasterSiteUnitList.SiteSeries) Is Null));"
            DoCmd.SetWarnings False
            DoCmd.RunSQL SqlText
            DoCmd.SetWarnings True
            'SqlText = "SELECT USysUserSiteUnitList.* FROM USysUserSiteUnitList WHERE (((USysUserSiteUnitList.UnitLongName) Is Null)) OR (((USysUserSiteUnitList.Level) Is Null));"
            SqlText = "SELECT USysUserSiteUnitList.* FROM USysUserSiteUnitList WHERE (((USysUserSiteUnitList.SiteSeriesLongName) Is Null)) OR (((USysUserSiteUnitList.Level) Is Null));"
            Set MyRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)
            If MyRS.RecordCount > 0 Then
                If WarnUser("VPro has located site units in your personal list that don't have long names or don't have set levels.  Would you like to edit these records now?", "VPro") = 0 Then
                    DoCmd.OpenForm "USysUserSiteUnits"
                    Forms!USysUserSiteUnits.RecordSource = SqlText
                    Forms!USysUserSiteUnits.Caption = "VPro - Site units without long names"
                End If
            End If
        End If
    End If

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    MyRS.Close
    Set MyDB = Nothing
    Screen.ActiveForm.Refresh
    Exit Function
errInsertEnvSuIntoSu:
    MsgBox Err.Description
    Resume MyExit

End Function

Public Function GetFilteredFormRecords()

'Dec 12, 2019
'creates a site unit table from a form filter
'contains a good example of DoEvents
On Error GoTo errGetFormRecords
Dim MyDB As Database, SuRecords As Recordset, MyCriteria As String, ActiveForm As Form
Dim MyForm As Form, MyRecords As Recordset, X As Variant, i As Integer, Con As String, SuFrom As String, SuTo As String
Dim SqlText As String, BreakSU As String
Set ActiveForm = Screen.ActiveForm

If Not (ActiveForm.Name = "FS882-8x6XL" Or ActiveForm.Name = "FS882-6x4" Or ActiveForm.Name = "FS882-8x6XL-CHARS" Or ActiveForm.Name = "FS882-1x1" Or ActiveForm.Name = "FS882-6x4XL" Or ActiveForm.Name = "FS882-6x4XL-CHARS") Then
    MsgBox "Either data form FS882(4 Page) or FS882(2 Page) must be the active form for this tool to work", vbInformation, "VPro"
    GoTo MyExit
End If

Set MyDB = CurrentDb()

On Error Resume Next
    DoCmd.DeleteObject acTable, "FilterQuerySu"
On Error GoTo errGetFormRecords

Set MyForm = Forms(Screen.ActiveForm.Name)
Set MyRecords = MyForm.RecordsetClone

sysBreakSU = "bla"
sysSaveFilter2 = 0
DoCmd.OpenForm "USysFilter2SuList"
Do Until sysSaveFilter2 > 0
    DoEvents
Loop

If sysIncludeExisting And clsVProReg.CurrPlotlist <> "None" Then
    Con = Mid(CurrentDb.TableDefs(clsVProReg.CurrPlotlist & "_SU").Connect, 11)
    If Con <> "" Then
        DoCmd.TransferDatabase acImport, "Microsoft Access", Con, acTable, clsVProReg.CurrPlotlist & "_SU", "FilterQuerySu"
    Else
        DoCmd.CopyObject , "FilterQuerySu", acTable, clsVProReg.CurrPlotlist & "_SU"
    End If
End If

MyContinue:

    If sysSaveFilter2 = 2 Then
        CreateSuTable
        If SuErr = "Dup" Then GoTo MyExit
        If sysBreakSU = "bla2" Then GoTo MyExit
        If sysBreakSU = Empty Then GoTo MyExit
        MyRecords.MoveLast
        MyRecords.MoveFirst
        
        Set SuRecords = MyDB.OpenRecordset(sysBreakSU, dbOpenDynaset)
        For i = 1 To MyRecords.RecordCount
            SuRecords.AddNew
            SuRecords!PlotNumber = MyRecords!PlotNumber
            SuRecords.Update
            MyRecords.MoveNext
        Next i
        'sysBreakSU hold the new table name
    End If
    If sysSaveFilter2 = 1 Then
        If CurrPlotlist = "None" Then
            MsgBox "You must have a current site unit list if you want to modify it."
            GoTo MyExit
        End If
        Set SuRecords = MyDB.OpenRecordset(clsVProReg.CurrPlotlist & "_SU", dbOpenDynaset)
        SuRecords.MoveLast
        SuRecords.MoveFirst
        'delete non matching plots
        For i = 1 To SuRecords.RecordCount
            MyCriteria = "PlotNumber = '" & SuRecords!PlotNumber & "'"
            MyRecords.MoveFirst
            MyRecords.FindFirst MyCriteria
                If MyRecords.NoMatch Then
                    SuRecords.Delete
                End If
            SuRecords.MoveNext
        Next i
        'add non matching plots
        MyRecords.MoveLast
        MyRecords.MoveFirst
        For i = 1 To MyRecords.RecordCount
            MyCriteria = "PlotNumber = '" & MyRecords!PlotNumber & "'"
            SuRecords.MoveFirst
            SuRecords.FindFirst MyCriteria
                If SuRecords.NoMatch Then
                    SuRecords.AddNew
                    SuRecords!PlotNumber = MyRecords!PlotNumber
                    SuRecords.Update
                End If
            MyRecords.MoveNext
        Next i
    End If
    If sysSaveFilter2 = 3 Then
        GoTo MyExit
    End If
    
    DoCmd.SetWarnings False
    'Debug.Print sysBreakSU
        
    If sysIncludeExisting And sysBreakSU <> "bla" And clsVProReg.CurrPlotlist <> "None" Then
    SuFrom = clsVProReg.CurrPlotlist & "_SU"
    
        SqlText = "UPDATE " & sysBreakSU & " INNER JOIN FilterQuerySu ON " & sysBreakSU & ".PlotNumber = FilterQuerySu.PlotNumber SET " & sysBreakSU & ".SiteUnit = [FilterQuerySu]![SiteUnit];"
        'SqlText = "UPDATE testing_SU INNER JOIN Sample_SU ON testing_SU.PlotNumber = Sample_SU.PlotNumber SET testing_SU.SiteUnit = [Sample_SU]![SiteUnit];"

        'CurrentDb.CreateQueryDef "fred", SqlText
        DoCmd.RunSQL SqlText
    End If
    
MsgBox i - 1 & " records were filtered into the Site Unit list.", vbInformation, "VPro"
If IsLoaded("frmMainMenuFloat") Then
    Forms("frmMainMenuFloat").cmbCurrProject.SetFocus
End If

MyExit:
    On Error Resume Next
    SuRecords.Close
    MyRecords.Close
    Set MyForm = Nothing
    Set MyDB = Nothing
    DoCmd.DeleteObject acTable, "FilterQuerySu"
    DoCmd.SetWarnings True
    Exit Function

errGetFormRecords:
    If Err.Number = 2475 Then
        MsgBox "Either data form FS882(4 Page) or FS882(2 Page) must be the active form for this tool to work", vbInformation, "VPro"
        Resume MyExit
    Else
        MsgBox Err.Description
        Resume MyExit
    End If

End Function
