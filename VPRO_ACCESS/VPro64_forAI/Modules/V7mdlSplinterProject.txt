Option Compare Database
Option Explicit

Function ExportSplinter()

'02 Oct 2019
'lots of variables - let's get started
Dim DefaultWorkspace As Workspace, MyDB As Database, ThatDB As Database, Msg As String
Dim Title As String, Default As String, ExProjectName As String, ProjectName As String
Dim EnvTableName As String, VegTableName As String, AuditTableName As String
Dim SoilHumusTableName As String, SoilMineralTableName As String, OtherTableName As String, AdminTableName As String
Dim MetaDataTableName As String

Dim Ex2EnvTableName As String, Ex2VegTableName As String, Ex2AuditTableName As String
Dim Ex2SoilHumusTableName As String, Ex2SoilMineralTableName As String, Ex2OtherTableName As String
Dim Ex2MetaDataTableName As String, Ex2AdminTableName As String

Dim ExEnvTableName As String, ExVegTableName As String, ExAuditTableName As String
Dim ExSoilHumusTableName As String, ExSoilMineralTableName As String, ExOtherTableName As String, ExAdminTableName As String
Dim ExMetaDataTableName As String

Dim SuList As String, SqlText As String, Filter As String, MyTitle As String, NewDBName As String
Dim ThatProject As String, Msg2 As String, TestLen As Integer, i As Integer, TestTable As String

'is there a current su table
    If clsVProReg.CurrPlotlist = "None" Then
        MsgBox "You need an SU table for this function.", vbInformation, "VPro"
        Exit Function
    End If

'create a workspace for the new db and a reference to the current db
    Set DefaultWorkspace = DBEngine.Workspaces(0)
    Set MyDB = CurrentDb()

    DoCmd.SetWarnings False

'make sure there isn't any garbage from previous runs
    CleanUp
    
    On Error GoTo errExportProject
    On Error GoTo 0

'again, check for su table
    If clsVProReg.CurrPlotlist <> "None" Then
        SuList = clsVProReg.CurrPlotlist & "_SU"
    Else
        SuList = "None"
    End If

'get a name for the splinter project
    Msg = "Name for exported project:"
    Title = "VPro"
    Default = ProjectName
    ExProjectName = InputBox(Msg, Title, Default)
    If ExProjectName = "" Then Exit Function
    If IsNull(ExProjectName) Then Exit Function

'fill variables with table names
    ProjectName = clsVProReg.CurrProject
    EnvTableName = ProjectName & "_Env"
    VegTableName = ProjectName & "_Veg"
    AuditTableName = ProjectName & "_Audit"
    SoilHumusTableName = ProjectName & "_Humus"
    SoilMineralTableName = ProjectName & "_Mineral"
    OtherTableName = ProjectName & "_Other"
    MetaDataTableName = ProjectName & "_MetaData"
    AdminTableName = ProjectName & "_Admin"

    Ex2EnvTableName = ExProjectName & "_Env"
    Ex2VegTableName = ExProjectName & "_Veg"
    Ex2AuditTableName = ExProjectName & "_Audit"
    Ex2SoilHumusTableName = ExProjectName & "_Humus"
    Ex2SoilMineralTableName = ExProjectName & "_Mineral"
    Ex2OtherTableName = ExProjectName & "_Other"
    Ex2MetaDataTableName = ExProjectName & "_MetaData"
    Ex2AdminTableName = ExProjectName & "_Admin"
    
    ExEnvTableName = EnvTableName
    ExVegTableName = VegTableName
    ExAuditTableName = AuditTableName
    ExSoilHumusTableName = SoilHumusTableName
    ExSoilMineralTableName = SoilMineralTableName
    ExOtherTableName = OtherTableName
    ExMetaDataTableName = MetaDataTableName
    ExAdminTableName = AdminTableName
    
'Create tables
    DoCmd.CopyObject CurrentDb.Name, "RkEnv", acTable, "USysEnvTable"
    DoCmd.CopyObject CurrentDb.Name, "RkAdmin", acTable, "USysAdminTable"
    DoCmd.CopyObject CurrentDb.Name, "RkVeg", acTable, "USysVegTable"
    DoCmd.CopyObject CurrentDb.Name, "RkAudit", acTable, "USysAudit"
    DoCmd.CopyObject CurrentDb.Name, "RkHumus", acTable, "USysSoilHumusTable"
    DoCmd.CopyObject CurrentDb.Name, "RkMineral", acTable, "USysSoilMineralTable"
    DoCmd.CopyObject CurrentDb.Name, "RkOther", acTable, "USysOtherTable"
    DoCmd.CopyObject CurrentDb.Name, "RkMetaData", acTable, "USysMetadataTable"
    
    
'Add data to tables
    If SuList = "None" Then
        SqlText = "INSERT INTO RkEnv SELECT * FROM [" & EnvTableName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkAdmin SELECT * FROM [" & AdminTableName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkVeg SELECT * FROM [" & VegTableName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkAudit SELECT * FROM [" & AuditTableName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkHumus SELECT * FROM [" & SoilHumusTableName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkMineral SELECT * FROM [" & SoilMineralTableName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkOther SELECT * FROM [" & OtherTableName & "];"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkMetaData SELECT * FROM [" & MetaDataTableName & "];"
        DoCmd.RunSQL SqlText
    Else
        SqlText = "INSERT INTO RkEnv SELECT [" & EnvTableName & "].* FROM [" & EnvTableName & "] INNER JOIN [" & SuList & "] ON [" & EnvTableName & "].PlotNumber = [" & SuList & "].PlotNumber;"
        DoCmd.RunSQL SqlText
        
        SqlText = "INSERT INTO RkAdmin SELECT [" & AdminTableName & "].* FROM [" & AdminTableName & "] INNER JOIN [" & SuList & "] ON [" & AdminTableName & "].Plot = [" & SuList & "].PlotNumber;"
        DoCmd.RunSQL SqlText
        
        SqlText = "INSERT INTO RkVeg SELECT [" & VegTableName & "].* FROM [" & VegTableName & "] INNER JOIN [" & SuList & "] ON [" & VegTableName & "].PlotNumber = [" & SuList & "].PlotNumber;"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkAudit SELECT [" & AuditTableName & "].* FROM [" & AuditTableName & "] INNER JOIN [" & SuList & "] ON [" & AuditTableName & "].PlotNumber = [" & SuList & "].PlotNumber;"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkHumus SELECT [" & SoilHumusTableName & "].* FROM [" & SoilHumusTableName & "] INNER JOIN [" & SuList & "] ON [" & SoilHumusTableName & "].PlotNumber = [" & SuList & "].PlotNumber;"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkMineral SELECT [" & SoilMineralTableName & "].* FROM [" & SoilMineralTableName & "] INNER JOIN [" & SuList & "] ON [" & SoilMineralTableName & "].PlotNumber = [" & SuList & "].PlotNumber;"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkOther SELECT [" & OtherTableName & "].* FROM [" & OtherTableName & "] INNER JOIN [" & SuList & "] ON [" & OtherTableName & "].PlotNumber = [" & SuList & "].PlotNumber;"
        DoCmd.RunSQL SqlText
        SqlText = "INSERT INTO RkMetaData"
        SqlText = SqlText & " SELECT DISTINCTROW [" & MetaDataTableName & "].*"
        SqlText = SqlText & " FROM [" & SuList & "] INNER JOIN ([" & MetaDataTableName & "] INNER JOIN [" & EnvTableName & "] ON [" & MetaDataTableName & "].ProjectID = [" & EnvTableName & "].ProjectID) ON [" & SuList & "].PlotNumber = [" & EnvTableName & "].PlotNumber;"
        DoCmd.RunSQL SqlText
    End If

'fill variables
    ExEnvTableName = "RkEnv"
    ExAdminTableName = "RkAdmin"
    ExVegTableName = "RkVeg"
    ExAuditTableName = "RkAudit"
    ExSoilHumusTableName = "RkHumus"
    ExSoilMineralTableName = "RkMineral"
    ExOtherTableName = "RkOther"
    ExMetaDataTableName = "RkMetadata"

'get a file name
    MyTitle$ = "Export Current Project To"
    NewDBName$ = Trim(GetFileName(MyTitle$, True))
    If NewDBName$ = "" Then Exit Function

'make sure that the db isn't VPro
    If NewDBName$ = MyDB.Name Then
        Msg = "You can't export using that database."
        Title = "VPro"
        MsgBox Msg, 64, Title
        GoTo MyExit
    End If

'create a new db if needed
    If Dir(NewDBName$) = "" Then
        Set ThatDB = DefaultWorkspace.CreateDatabase(NewDBName$, DB_LANG_GENERAL)
    Else
        Set ThatDB = DefaultWorkspace.OpenDatabase(NewDBName$)
    End If
    NewDBName$ = ThatDB.Name

'fill a variable
    ThatProject = ExProjectName

'check to ensure there isn't a project of that name in the db
    Msg2 = "There's already a project named " & ThatProject & " in database " & NewDBName$
    Msg2 = Msg2 & Chr(13) & "Try another database or create a new database"
    TestLen = Len(ExProjectName) + 4
    For i = 0 To ThatDB.TableDefs.count - 1
        TestTable = ThatDB.TableDefs(i).Name
        If TestTable = "" Then Exit Function
        If Left(TestTable, TestLen) = ExProjectName & "_Env" Then MsgBox Msg2, 64, Title
        If Left(TestTable, TestLen) = ExProjectName & "_Env" Then Exit Function
    Next i

'transfer the tables
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExEnvTableName, Ex2EnvTableName
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExAdminTableName, Ex2AdminTableName
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExVegTableName, Ex2VegTableName
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExAuditTableName, Ex2AuditTableName
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExSoilHumusTableName, Ex2SoilHumusTableName
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExSoilMineralTableName, Ex2SoilMineralTableName
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExOtherTableName, Ex2OtherTableName
    DoCmd.TransferDatabase A_EXPORT, "Microsoft Access", NewDBName$, A_TABLE, ExMetaDataTableName, Ex2MetaDataTableName

'get rid of all the junk created
    CleanUp
 
'all done
    MsgBox nPlotsInSplinter & " of " & nPlotsInOriginal & " plots from project " & clsVProReg.CurrProject & " have been exported to " & NewDBName$ & " as project " & ExProjectName, vbInformation, "VPro"

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Set MyDB = Nothing
    Set ThatDB = Nothing
    Exit Function
errExportProject:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Sub CleanUp()

    On Error Resume Next
    DoCmd.DeleteObject A_TABLE, "RkEnv"
    DoCmd.DeleteObject A_TABLE, "RkAdmin"
    DoCmd.DeleteObject A_TABLE, "RkVeg"
    DoCmd.DeleteObject A_TABLE, "RkAudit"
    DoCmd.DeleteObject A_TABLE, "RkHumus"
    DoCmd.DeleteObject A_TABLE, "RkMineral"
    DoCmd.DeleteObject A_TABLE, "RkOther"
    DoCmd.DeleteObject A_TABLE, "RkMetaData"
    DoCmd.DeleteObject A_TABLE, "RkEnv1"
    DoCmd.DeleteObject A_TABLE, "RkAdmin1"
    DoCmd.DeleteObject A_TABLE, "RkVeg1"
    DoCmd.DeleteObject A_TABLE, "RkAudit1"
    DoCmd.DeleteObject A_TABLE, "RkHumus1"
    DoCmd.DeleteObject A_TABLE, "RkMineral1"
    DoCmd.DeleteObject A_TABLE, "RkOther1"
    DoCmd.DeleteObject A_TABLE, "RkMetaData1"
    On Error GoTo 0
    
End Sub

Private Sub test()

    Debug.Print nPlotsInSplinter

End Sub

Private Function nPlotsInOriginal() As Long

Dim MyDB As Database, MyRS As Recordset, SqlText As String
Set MyDB = CurrentDb

'create a recordset and populate it with the plots from the current project
    SqlText = "Select PlotNumber from [" & clsVProReg.CurrProject & "_Env];"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount > 0 Then
        MyRS.MoveLast
'return number of plots
        nPlotsInOriginal = MyRS.RecordCount
    Else
        nPlotsInOriginal = 0
    End If
    
MyExit:
    MyRS.Close
    Set MyDB = Nothing

End Function

Private Function nPlotsInSplinter() As Long

Dim MyDB As Database, MyRS As Recordset, SqlText As String
Set MyDB = CurrentDb

'create a recordset and populate it with the plots from the current SU table
    SqlText = "Select PlotNumber from [" & clsVProReg.CurrPlotlist & "_SU];"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount > 0 Then
        MyRS.MoveLast
    'return number of plots
        nPlotsInSplinter = MyRS.RecordCount
    Else
        nPlotsInSplinter = 0
    End If
MyExit:
    MyRS.Close
    Set MyDB = Nothing

End Function
