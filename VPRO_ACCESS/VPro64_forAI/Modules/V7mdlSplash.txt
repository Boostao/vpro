Private Sub CheckForBrokeRefs()
'Dec. 30, 2015
Dim Ref As Reference

If Not Application.IsCompiled Then
    For Each Ref In Application.References
        If Ref.IsBroken Then
            MsgBox "VPro relies on functions that are stored in your system directory.  A reference to " & Ref.Name & " is missing or broken.  VPro will try to remove the problem reference and reestablish the connection.  You may have to exit and restart VPro in order for this to be accomplished."
            Application.References.Remove Ref
        End If
    'Debug.Print Ref.Name
    Next Ref
    
End If

MyExit:

End Sub
Public Function ReferenceFromFile(strFilename As String) As Boolean
    Dim Ref As Reference

    On Error GoTo Error_ReferenceFromFile
    ' Create new reference.
    Set Ref = References.AddFromFile(strFilename)
    ReferenceFromFile = True

Exit_ReferenceFromFile:
    Exit Function

Error_ReferenceFromFile:
    If Err.Number = 32813 Then
        ReferenceFromFile = True
        Resume Exit_ReferenceFromFile
    Else
        MsgBox Err & ": " & Err.Description
        ReferenceFromFile = False
        Resume Exit_ReferenceFromFile
    End If
End Function

Private Sub CloseAllModules()

Dim n As Integer
    Dim mods As Modules
    
    Set mods = Application.Modules
    
    ' loop backwards through all open modules
    ' and close and save each
    For n = mods.count - 1 To 0 Step -1
        DoCmd.Close acModule, mods(n).Name, acSaveYes
    Next n

Public Sub ShowSplash()

'14 May 2020
Dim TestCompilation As Boolean, Ref As Reference, IveBeenMoved As Boolean
Dim OfficeDir As String, SystemDir As String, ctlName As String, ExtDir As String, NewLoc As String
Dim CommonDir As String

'close all modules so that vpro opens faster
    CloseAllModules

'set some references if computer is server
    If Environ$("computername") = "OFFLINE" Then
        OfficeDir = "E:\Program Files (x86)\Microsoft Office\Office12\"
        SystemDir = "C:\Windows\System32\"
        
        ctlName = OfficeDir & "EXCEL.exe"
        If Not ReferenceFromFile(ctlName) Then
            MsgBox "Reference to " & ctlName & " not created successfully."
        End If
        
        ctlName = SystemDir & "MSCOMCTL.OCX"
        If Not ReferenceFromFile(ctlName) Then
            MsgBox "Reference to " & ctlName & " not created successfully."
        End If
        
        If Not Application.IsCompiled Then
            Call SysCmd(504, 16483)
        End If
            
    Else
        'do normal stuff
        OfficeDir = Application.SysCmd(acSysCmdAccessDir)
        CommonDir = "C:\Program Files (x86)\Common Files\microsoft shared\"
        SystemDir = GetSystemDir()
    End If
    
    Debug.Print SystemDir
    Debug.Print OfficeDir
    Debug.Print CommonDir
    Debug.Print "Access version: " & Application.SysCmd(acSysCmdAccessVer)

'compile if not compiled
    If Not Application.IsCompiled Then
        Call SysCmd(504, 16483)
    End If

'set program name in registry
    clsVProReg.FullProgramName = CurrentDb.Name
    CurrentDb.Properties("AppIcon").Value = CurrentProject.path & "\VPro19.ico"

'set location of vpro in registy
    If clsVProReg.Location = "" Then
        clsVProReg.Location = Left(CurrentDb.Name, Len(CurrentDb.Name) - Len(Dir(CurrentDb.Name)))
    End If

'if vpro has moved then freak out
    If clsVProReg.Location <> Left(CurrentDb.Name, Len(CurrentDb.Name) - Len(Dir(CurrentDb.Name))) Then
        IveBeenMoved = True
        'NewLoc = Left(CurrentDb.Name, Len(CurrentDb.Name) - (Len(Dir(CurrentDb.Name)) + 11))
        NewLoc = Left(CurrentDb.Name, Len(CurrentDb.Name) - Len(Dir(CurrentDb.Name)))
        MsgBox "VPro here.  Looks like I've been moved.  I used to live at " & clsVProReg.Location & ". It's a little dark in here, but now it looks like I'm at " & NewLoc & ".  I'm going to assume that you moved all my sub-directories when you moved me, but just to be sure, please double-check by looking at Directories in User-setup.  Thanks, and have a nice day.", vbInformation, "VPro"
        clsVProReg.Location = Left(CurrentDb.Name, Len(CurrentDb.Name) - Len(Dir(CurrentDb.Name)))
        UnAttachSupportTables
        AttachSupportTables
        SetappIcon
    End If

'reset location of vpro support in registry
    If clsVProReg.RLocation = "" Or IveBeenMoved Then
        clsVProReg.RLocation = clsVProReg.Location & "R\"
    End If
    
    If clsVProReg.GELocation = "" Or IveBeenMoved Then
        clsVProReg.GELocation = clsVProReg.Location & "GoogleEarth\"
    End If
    
    If clsVProReg.PictureDir = "" Or IveBeenMoved Then
        clsVProReg.PictureDir = clsVProReg.Location & "PlotPictures\"
    End If
    
    If clsVProReg.MetadataLocation = "" Or IveBeenMoved Then
        clsVProReg.MetadataLocation = clsVProReg.Location & "VMetaData.accda"
    End If
    
    If clsVProReg.RibbonPicLocation = "" Or IveBeenMoved Then
        clsVProReg.RibbonPicLocation = clsVProReg.Location & "Ribbon\Icons\"
    End If
    
'get USysAllSpecs and USysTableOfLists version
    Dim RemoteDB As Database
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(clsVProReg.Location & "VLists.accda")
    If RemoteDB.Containers("Tables").Documents("USysAllSpecs").Properties("Description") <> clsVProReg.USysAllSpecsVer Then
        clsVProReg.USysAllSpecsVer = RemoteDB.Containers("Tables").Documents("USysAllSpecs").Properties("Description")
    End If
    If RemoteDB.Containers("Tables").Documents("USysTableOfLists").Properties("Description") <> clsVProReg.USysTableOfListsVer Then
        clsVProReg.USysTableOfListsVer = RemoteDB.Containers("Tables").Documents("USysTableOfLists").Properties("Description")
    End If
    Set RemoteDB = Nothing
    
    If NeverRanBefore Then
        DoCmd.Close acForm, "Splash"
        DoCmd.OpenForm "frmRunOnce"
        Forms![frmRunOnce]![btnCancel].Visible = False
        SetappIcon
        If Environ$("computername") <> "OFFLINE" Then
            CheckForBrokeRefs
        End If
        Exit Sub
    Else
        SpCheck
    End If

'set user name and log in
    clsVProReg.UserName = LocalUserName
    LogIn

'check to see if someone else has your copy of vpro open
    CheckForMultiUsers

'check that your current project is current
    CompareRegToCurrent

'set toolbars
    Application.MenuBar = ""
    DoCmd.ShowToolbar "VENUSPro"
    DoCmd.ShowToolbar "Menu Bar"
    
'set system variables
    If Environ$("computername") <> "OFFLINE" Then
        OfficeDir = Application.SysCmd(acSysCmdAccessDir)
        SystemDir = GetSystemDir()
    End If

'CheckForBrokeRefs 'do initial error check

If Application.IsCompiled Then GoTo Jump

'reference components if not server
    On Error GoTo errShowSplash
    If Environ$("computername") <> "OFFLINE" Then
        ReferenceComponents
    End If
        
Jump:
  AttachSupportTables
  CheckMessageBoard
  
'in case of crash or power failure, rename the fields in USysAllSpecs
  RenameGuideFieldNameBack

MyExit:
    On Error Resume Next
    DoCmd.Echo True
    DoCmd.Close A_FORM, "Splash"
    DoCmd.OpenForm "frmMainMenuFloat"
    Forms("frmMainMenuFloat").Repaint
    Exit Sub
    
errShowSplash:
    MsgBox Err.Description
    Resume MyExit

End Sub

        DoCmd.TransferDatabase acLink, "Microsoft Access", ListDB, acTable, "USysSiteSeriesNames", "USysSiteSeriesNames"
    End If

    If PicTable = False Then
        On Error Resume Next
        DoCmd.TransferDatabase acLink, "Microsoft Access", PicDB, acTable, "tblVPics", "tblVPics"
    End If

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function
errAttachSupportTables:
    MsgBox Err.Description
    Resume MyExit

End Function

Private Function GetRefVersion(RefName As String)
'Nov 23, 1999
Dim i As Integer, MyVersion As String

For i = 1 To References.count
    With References(i)
        MyVersion = .Major & "." & .Minor
        If .Name = RefName Then
        GetRefVersion = MyVersion
        End If
    End With
Next i

End Function

Private Function IsNewMessage() As Boolean

Dim MyRS As Recordset, SqlText As String
SqlText = "SELECT tblMessageBoard.*"
SqlText = SqlText & " FROM tblMessageBoard INNER JOIN tblMessageList ON tblMessageBoard.ID=tblMessageList.MessageID"

Set MyRS = CurrentDb.OpenRecordset(SqlText)

Else

Private Sub UpdateMessageList()

Dim SqlText As String

SqlText = "INSERT INTO tblMessageList ( MessageID )"
SqlText = SqlText & " SELECT tblMessageBoard.ID"
SqlText = SqlText & " FROM tblMessageBoard LEFT JOIN tblMessageList ON tblMessageBoard.ID = tblMessageList.MessageID"

DoCmd.SetWarnings True

End Sub
