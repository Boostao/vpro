Public Function mnuEditSiteUnits()
'Jan 9, 2009
On Error GoTo errbtnEditSiteUnits_Click

sysHierarchyType = "SuTable"
sysCurrSU = clsVProReg.CurrPlotlist & "_SU" ' CurrPlotlist() & "_SU"
If sysCurrSU = "None_SU" Or sysCurrSU = "" Then
    MsgBox "You don't have a current site unit table.  Please select a table and try again."
    GoTo MyExit
End If

If CurrentDb().OpenRecordset(sysCurrSU).RecordCount = 0 Then
    MsgBox "There are no plots in your site unit table.  This form requires plots to assign to units.  Please add plots using a different method such as the filter query."
    GoTo MyExit
End If
    DoCmd.OpenForm "USysAssignSU", , , , , acHidden
    DoCmd.Echo False
    Forms![usysassignsu].CallAddSUsAndPlotsOnly_Click
    DoCmd.OpenForm "USysAssignSU", , , , , acWindowNormal
    DoCmd.Echo True
    
MyExit:
    On Error Resume Next
    Exit Function
errbtnEditSiteUnits_Click:
    MsgBox Err.Description
    Resume MyExit
End Function

Public Function mnuEditHierarchy()
'Oct. 11, 2002
On Error GoTo Err_btnEditHierarchy_Click
sysCurrSU = clsVProReg.CurrPlotlist & "_SU"
sysHierarchyType = "Hierarchy"
sysCurrHierarchy = clsVProReg.CurrHierarchy & "_Hierarchy"

If (sysCurrSU = "None_SU" Or sysCurrSU = "") And clsVProReg.CurrHierarchy <> "MasterSiteUnitList" Then
    MsgBox "You don't have a current site unit table.  Please select a table and try again."
    GoTo Exit_btnEditHierarchy_Click
End If

If CurrentDb().OpenRecordset(sysCurrSU).RecordCount = 0 Then
    MsgBox "There are no plots in your site unit table.  Please add plots using a different method such as the filter query."
    GoTo Exit_btnEditHierarchy_Click
End If

    Dim stDocName As String
    Dim stLinkCriteria As String
    
    stDocName = "USysAssignSU"
    DoCmd.OpenForm stDocName, , , stLinkCriteria

Exit_btnEditHierarchy_Click:
    On Error Resume Next
    Exit Function

Err_btnEditHierarchy_Click:
    MsgBox Err.Description
    Resume Exit_btnEditHierarchy_Click
End Function
