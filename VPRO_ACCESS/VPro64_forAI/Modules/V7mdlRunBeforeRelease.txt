Private Sub DeleteAllAttachedTables()

Dim MyTD As TableDef, X As Integer

For Each MyTD In CurrentDb.TableDefs

    If MyTD.Connect <> "" Then
        DoCmd.DeleteObject acTable, MyTD.Name
        X = X + 1
    End If
    
Next MyTD

Debug.Print X

End Sub
