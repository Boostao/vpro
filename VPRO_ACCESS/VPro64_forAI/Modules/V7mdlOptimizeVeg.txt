Option Compare Database
Option Explicit

Function OptimizeVeg(Optional VegTable As Variant, Optional Silent As Variant)

'Apr. 13, 2015
'optimzes vegetation tables by flattening out data
'NOTE: WORKS ON ENTIRE PROJECT
On Error GoTo errOptimizeVeg
DoCmd.SetWarnings False

Dim MyDB As Database, MyQD As QueryDef
Dim SqlText As String
Set MyDB = CurrentDb()

    If IsMissing(VegTable) Then VegTable = clsVProReg.CurrProject & "_Veg"
    
    On Error Resume Next
    DoCmd.DeleteObject acQuery, "xxxqryTempVeg"
    DoCmd.DeleteObject acTable, "xxxtblTempVeg"
    On Error GoTo errOptimizeVeg
    On Error GoTo 0

    If Check4DuplicatedSpecies() Then
        DoCmd.CopyObject , "xxxtblTempVeg", acTable, "USysVegTable"

'create a flat dataset
        SqlText = "SELECT PlotNumber, Species, Max(Cover1) AS MaxOfCover1, Max(Height1) AS MaxOfHeight1, Max(Cover2) AS MaxOfCover2, Max(Height2) AS MaxOfHeight2, Max(Cover3) AS MaxOfCover3, Max(Height3) AS MaxOfHeight3, Max(Cover4) AS MaxOfCover4, Max(Height4) AS MaxOfHeight4, Max(Cover5) AS MaxOfCover5, Max(Height5) AS MaxOfHeight5, Max(Cover5a) AS MaxOfCover5a, Max(Height5a) AS MaxOfHeight5a, Max(Cover5b) AS MaxOfCover5b, Max(Height5b) AS MaxOfHeight5b, Max(Cover5c) AS MaxOfCover5c, Max(Height5c) AS MaxOfHeight5c, Max(Cover6) AS MaxOfCover6, Max(Height6) AS MaxOfHeight6, Max(Cover7) AS MaxOfCover7, Max(Cover8) AS MaxOfCover8, Max(Cover9) AS MaxOfCover9, Max(Cover10) AS MaxOfCover10, Max(TotalA) AS MaxOfTotalA, Max(TotalB) AS MaxOfTotalB,"
        SqlText = SqlText & " First(Collected) AS FirstOfCollected, First(Flag) AS FirstOfFlag, First(LL) AS FirstOfLL, First(AF) AS FirstOfAF, First(DC) AS FirstOfDC, First(UT) AS FirstOfUT, First(VI) AS FirstOfVI, First(PV) AS FirstOfPV, First(PG) AS FirstOfPG, First(FFA) AS FirstOfFFA, First(Cultural1) AS FirstOfCultural1, First(Cultural2) AS FirstOfCultural2, First(Other1) AS FirstOfOther1, First(Other2) AS FirstOfOther2"
        SqlText = SqlText & " FROM [" & VegTable & "]"
        SqlText = SqlText & " GROUP BY PlotNumber, Species"
        SqlText = SqlText & " ORDER BY PlotNumber, Species;"
        Set MyQD = MyDB.CreateQueryDef("xxxqryTempVeg", SqlText)
    
'merge the flat records into a blank table
        SqlText = "INSERT INTO xxxtblTempVeg ( PlotNumber, Species, Cover1, Height1, Cover2, Height2, Cover3, Height3, Cover4, Height4, Cover5, Height5, Cover5a, Height5a, Cover5b, Height5b, Cover5c, Height5c, Cover6, Height6, Cover7, Cover8, Cover9, Cover10, TotalA, TotalB, Collected, Flag, LL, AF, DC, UT, VI, PV, PG, FFA, Cultural1, Cultural2, Other1, Other2 )"
        SqlText = SqlText & " SELECT PlotNumber, Species, MaxOfCover1, MaxOfHeight1, MaxOfCover2, MaxOfHeight2, MaxOfCover3, MaxOfHeight3, MaxOfCover4, MaxOfHeight4, MaxOfCover5, MaxOfHeight5, MaxOfCover5a, MaxOfHeight5a, MaxOfCover5b, MaxOfHeight5b, MaxOfCover5c, MaxOfHeight5c, MaxOfCover6, MaxOfHeight6, MaxOfCover7, MaxOfCover8, MaxOfCover9, MaxOfCover10, MaxOfTotalA, MaxOfTotalB, FirstOfCollected, FirstOfFlag, FirstOfLL, FirstOfAF, FirstOfDC, FirstOfUT, FirstOfVI, FirstOfPV, FirstOfPG, FirstOfFFA, FirstOfCultural1, xxxqryTempVeg.FirstOfCultural2 , FirstOfOther1, FirstOfOther2"
        SqlText = SqlText & " From xxxqryTempVeg"
        SqlText = SqlText & " ORDER BY PlotNumber, Species;"
        DoCmd.RunSQL SqlText
    
'delete existing veg records
        SqlText = "DELETE * FROM [" & VegTable & "]"
        DoCmd.RunSQL SqlText
    
'merge flat records into project
        SqlText = "INSERT INTO [" & VegTable & "] SELECT xxxtblTempVeg.* FROM xxxtblTempVeg;"
        DoCmd.RunSQL SqlText
    
        MsgBox "VPro has optimized your vegetation records.  If you have a data form open, it may be necessary to close and reopen the form to show the changes.", , "VPro"
    Else
        MsgBox "Once you've fixed your vegetation records, try again.", , "VPro"
    End If

MyExit:
    DoCmd.SetWarnings True
    On Error Resume Next
    DoCmd.DeleteObject acQuery, "xxxqryTempVeg"
    DoCmd.DeleteObject acTable, "xxxtblTempVeg"
    Set MyDB = Nothing
    Exit Function
errOptimizeVeg:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function Check4DuplicatedSpecies() As Boolean

'Apr. 13, 2015
'checks for veg duplication errors
On Error GoTo errCheck4DuplicatedSpecies

Check4DuplicatedSpecies = True
Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String, Msg As String, Response As Integer, Title As String, dgdef As Variant
Set MyDB = CurrentDb()

    SqlText = "SELECT USysAllVeg.PlotNumber, USysAllVeg.Species, USysAllVeg.MyLayer AS Layer, Count(USysAllVeg.Cover) AS [n of Spp]"
    SqlText = SqlText & " FROM USysAllVeg"
    SqlText = SqlText & " GROUP BY USysAllVeg.PlotNumber, USysAllVeg.Species, USysAllVeg.MyLayer"
    SqlText = SqlText & " HAVING (((Count(USysAllVeg.Cover))>1));"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount = 0 Then GoTo MyExit
    MyRS.MoveLast
    MyRS.MoveFirst
    If MyRS.RecordCount > 0 Then
        Check4DuplicatedSpecies = False
        Msg = "There are some problems in your vegetation data.  Would you like VPro to generate an error report?"
        dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
        Title = "VPro Vegetation Check"
        Response = MsgBox(Msg, dgdef, Title)    ' Get user response.
        If Response = 6 Then
            Dim xl As Object, i As Integer
            Set xl = CreateObject("Excel.Application")
            xl.Visible = True
            xl.Cursor = xlWait
            xl.Workbooks.Add
            xl.Caption = "VPro Vegetation Error Report"
            
            xl.Cells(1, 1) = "Below are the problems VPro has identified with this test"
            xl.Cells(2, 1) = "Correct these errors before you use this project"
            xl.Cells(4, 1) = "Plot Number"
            xl.Cells(4, 2) = "Species"
            xl.Cells(4, 3) = "Layer"
            xl.Cells(4, 4) = "n of Spp"
            For i = 1 To MyRS.RecordCount
                xl.Cells(i + 4, 1) = "'" & MyRS!PlotNumber
                xl.Cells(i + 4, 2) = "'" & MyRS!Species
                xl.Cells(i + 4, 3) = "'" & MyRS!Layer
                xl.Cells(i + 4, 4) = "'" & MyRS![n of spp]
                MyRS.MoveNext
            Next i
            xl.Range("A1").ColumnWidth = 14
            xl.Range("B1").ColumnWidth = 10
            xl.Range("C1").ColumnWidth = 5
            xl.Range("d1").ColumnWidth = 8
            xl.Range("A4:D4").Font.Bold = True
            xl.Cursor = xlDefault
            Set xl = Nothing
        Else
            MsgBox "These errors should be fixed before using this data."
        End If
        
    Else
        Check4DuplicatedSpecies = True
    End If

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
errCheck4DuplicatedSpecies:
    MsgBox Err.Description
    Check4DuplicatedSpecies = False
    Resume MyExit
End Function
