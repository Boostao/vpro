Function AttachStep1(AttachmentType As String)

'Sep 22, 2008
On Error GoTo errAttachStep1
Dim MyDB As Database, RemoteDB As Database, strList As String, i As Integer
Dim MyFile As String, MyTitle As String, MyProject As String, strTestTable As String, X As Variant
sysPickTable = "None"


    Set MyDB = CurrentDb()
    MyFile = ""
    MyTitle = ProgramName()
    MyFile = GetFileName(MyTitle, True)
    If MyFile = "" Then Exit Function
    sysRemoteDB = MyFile
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(MyFile)

        strList = ""
        DoCmd.Hourglass True
        'ALWAYS USE I = 0 TO -1 BECAUSE OF THE WAY TABLEDEFS INDEX WORKS
        DoCmd.OpenForm "USysPickTable", A_NORMAL, , , , A_NORMAL
        For i = 0 To RemoteDB.TableDefs.count - 1
            strTestTable = RemoteDB.TableDefs(i).Name
            If InStr(1, strTestTable, AttachmentType) > 0 Then
                If Not TestForTableInVPro(strTestTable) Then
                    strList = strList & Left(strTestTable, Len(strTestTable) - Len(AttachmentType)) & ";"
                End If
            End If
        Next i
'        DoCmd.OpenForm "USysPickTable", A_NORMAL, , , , A_NORMAL
        Forms![USysPickTable]![PickList].RowSource = strList
        DoCmd.Hourglass False

        Do
            DoEvents
        Loop Until Screen.ActiveForm.Name <> "UsysPickTable"
MsgBox sysPickTable
If sysPickTable <> "None" Then
    Select Case AttachmentType
        Case "_Env"
            X = AttachEnv(RemoteDB)
        Case "_SU"
            X = AttachSu(RemoteDB)
        Case "_Hierarchy"
            X = AttachHierarchy(RemoteDB)
        Case "_Lump"
            X = AttachLump(RemoteDB)
        Case "_Profile"
            X = AttachProfile(RemoteDB)
        Case "_Theme"
            X = AttachTheme(RemoteDB)
    End Select
End If

MyExit:
    On Error Resume Next
    Set RemoteDB = Nothing
    Exit Function
errAttachStep1:
    MsgBox Err.Description
    Resume MyExit
End Function

Function AttachSu(RemoteDB As Database)
'Jan. 22, 2018
'attaches a SU table of the name found in variable sysPickTable
On Error GoTo errAttachSu
Dim MyDB As Database, MyRS As Recordset, X As Variant
Dim MySU As String, TestTable As String
Set MyDB = CurrentDb()

TestTable = sysPickTable & "_SU"
DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, TestTable, TestTable

    Resume MyExit
End Function
Private Function AttachEnv(RemoteDB As Database)
'Oct. 7, 2002
'attaches a project of the name found in variable sysPickTable

    Dim MyDB As Database, MyRS As Recordset
    Dim TableVersion As String, RemoteTable As String, MyProject As String
    Dim RelationTable As String, TestTable As String, Msg As String, dgdef As Integer
    Dim Response As Integer, X As Variant
    Set MyDB = CurrentDb()

'MsgBox "1"

On Error GoTo errAttachEnv
        Do
            DoEvents
        Loop Until Screen.ActiveForm.Name <> "UsysPickTable"
'MsgBox "2"

    MyProject = sysPickTable
    RelationTable = sysPickTable

TestTable = MyProject & "_ENV"
DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, TestTable, TestTable
            
TestTable = MyProject & "_Humus"
DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, TestTable, TestTable
            
TestTable = MyProject & "_Audit"
DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, TestTable, TestTable

TestTable = MyProject & "_Mineral"
DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, TestTable, TestTable
            
TestTable = MyProject & "_Veg"
DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, TestTable, TestTable

TestTable = MyProject & "_Other"
DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, TestTable, TestTable

TableVersion = GetTableVersion(MyProject & "_ENV")

'MsgBox "3"
If TableVersion <> "VP05" Then
    
    Msg = "The project you just attached isn't compatible with this version of VPro. Would you like VPro to convert this project?"
    Msg = Msg & Chr(13) & "WARNING: You should backup your data before attempting any conversion process."
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef)    ' Get user response.

    If Response = IDYES Then 'If the user wants to convert the project
        Select Case TableVersion
            Case "VP04"
                MsgBox "Sorry, this version of database is not supported yet."
            Case "VP03"
                X = ConvertVP03ToVP04(RemoteDB.Name, Left(TestTable, Len(TestTable) - 9))
            Case "VP02"
                X = ConvertVENUSPro2VPro(RemoteDB.Name, Left(TestTable, Len(TestTable) - 9))
            Case Else 'must be blank
                X = ConvertVENUSPro2VPro(RemoteDB.Name, Left(TestTable, Len(TestTable) - 9))
        End Select
        TableVersion = "VP04"
    End If
    
    If Response = IDNO Then 'If the user does not want to convert the project
        DoCmd.DeleteObject A_TABLE, MyProject & "_ENV"
        DoCmd.DeleteObject A_TABLE, MyProject & "_Humus"
        DoCmd.DeleteObject A_TABLE, MyProject & "_Audit"
        DoCmd.DeleteObject A_TABLE, MyProject & "_Mineral"
        DoCmd.DeleteObject A_TABLE, MyProject & "_Veg"
        DoCmd.DeleteObject A_TABLE, MyProject & "_Other"
        GoTo MyExit
    End If
End If
    
    If TableVersion = "VP04" Then
        'Set MyRS = MyDB.OpenRecordset("ProjectManager")
        'MyRS.AddNew
        'MyRS("ProjectName") = MyProject
        'MyRS.Update
        'MyRS.Close
    End If
Jump:
    X = CreateRelationship(RemoteDB.Name, RelationTable)
    
    If Screen.ActiveForm.Name = "MainMenu" Then
        Forms![MainMenu].grpListDisplay_AfterUpdate
    End If
        
MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Function
errAttachEnv:
    MsgBox Err.Description
    Resume MyExit
End Function
