Private Function TotalTreeLayer(Cover1 As Variant, Cover2 As Variant, Cover3 As Variant) As Single

Dim xl As New Excel.Application
Dim MaxCvr As Single
Dim AvgCvr As Single
Dim TmpCvr As Single

Cover1 = Nz(Cover1)
Cover2 = Nz(Cover2)
Cover3 = Nz(Cover3)

MaxCvr = xl.Max(Cover1, Cover2, Cover3)
AvgCvr = xl.Average(Cover1, Cover2, Cover3)

TmpCvr = AvgCvr + (MaxCvr / 2)

If MaxCvr = 0 Then
    TotalTreeLayer = 0
Else
    TotalTreeLayer = TmpCvr
End If
    
End Function

Private Function MatchSiteUnits2PlotFields()
'Mar. 11, 1999
'matches zone, subzone, and site series in the env table to those fields in the master list
'when matches are found the function enters the Name from the master list into
'the assigned site unit field of the env table

'Dec. 28, 2019 - FUNCTION NO LONGER IN USE
Dim MyDB As Database, MasterRS As Recordset, EnvRS As Recordset
Dim Criteria As String, SqlText As String, i As Integer, MyProject As String, MyUnit As String
Set MyDB = CurrentDb()
MyProject = CurrProject() & "_Env"
Set MasterRS = MyDB.OpenRecordset("USysMasterSiteUnitList")
Set EnvRS = MyDB.OpenRecordset(MyProject, dbOpenDynaset)
EnvRS.MoveLast
EnvRS.MoveFirst

DoCmd.SetWarnings False
'SqlText = "UPDATE " & MyProject & " SET " & MyProject & ".AssignedSiteUnit = Null, " & MyProject & ".SiteUnitShortName = Null, " & MyProject & ".SiteUnitLongName = Null;"
SqlText = "UPDATE " & MyProject & " SET " & MyProject & ".UserSiteUnit = Null, " & MyProject & ".SiteUnitShortName = Null, " & MyProject & ".SiteUnitLongName = Null;"
DoCmd.RunSQL SqlText
DoCmd.SetWarnings True

For i = 1 To EnvRS.RecordCount
    MasterRS.MoveFirst
    If Not IsNull(EnvRS!Zone) And Not IsNull(EnvRS!SubZone) And Not IsNull(EnvRS!SiteSeries) Then
        Criteria = "Zone = '" & EnvRS!Zone & "' And Subzone = '" & EnvRS!SubZone & "' And SiteSeries = '" & EnvRS!SiteSeries & "'"
        MasterRS.FindFirst Criteria
        If Not MasterRS.NoMatch Then
            'MsgBox MasterRS!Name
            EnvRS.Edit
            EnvRS!UserSiteUnit = MasterRS!Name
            EnvRS!SiteUnitShortName = MasterRS!Name
            EnvRS!SiteUnitLongName = MasterRS!UnitLongName
            EnvRS.Update
        End If
        If MasterRS.NoMatch Then
            MyUnit = Format(EnvRS!Zone, "!@@@@") & Format(EnvRS!SubZone, "!@@@@@") & "/" & EnvRS!SiteSeries
            EnvRS.Edit
            MsgBox EnvRS!PlotNumber
            EnvRS!UserSiteUnit = MyUnit
            EnvRS.Update
        End If
    End If
    EnvRS.MoveNext
Next i

MyExit:
    On Error Resume Next
    MasterRS.Close
    EnvRS.Close
    Set MyDB = Nothing
    Exit Function

End Function

Private Function ListUnits()
On Error GoTo errListUnits  'set a place to go when an error happens
'declarations
Dim MyDB As Database, MyLongNameRS As Recordset, MyShortNameRS As Recordset
Dim SqlText As String, i As Integer, nRecords As Integer, X As Integer
Dim xl As Object
Dim StartTime As Double, ElapsedTime As Double
StartTime = Time    'start timing the function
Set MyDB = CurrentDb()  'CurrentDB() is a Access function
SqlText = "SELECT DISTINCT UnitLongName"
SqlText = SqlText & " FROM [USysUserSiteUnitList]"
SqlText = SqlText & " WHERE (((UnitLongName) Is Not Null))"
SqlText = SqlText & " ORDER BY UnitLongName;"
Set MyLongNameRS = MyDB.OpenRecordset(SqlText)      'create a recordset from an SQL statement

CreateExcel xl
'Set xl = CreateObject("Excel.Application")  'create XL object
xl.Visible = True                           'make XL visible
xl.Workbooks.Add                            'add a workbook
xl.Cells(1, 1) = "Title"                    'add a title

MyLongNameRS.MoveLast   'populate object
MyLongNameRS.MoveFirst  'return to first record
nRecords = MyLongNameRS.RecordCount 'count the number of records in the recordset

'start a loop that runs once for each unique longname
For i = 1 To nRecords
    SqlText = "SELECT DISTINCT Name"
    SqlText = SqlText & " FROM [USysUserSiteUnitList]"
    SqlText = SqlText & " WHERE (((UnitLongName)=""" & MyLongNameRS!UnitLongName & """));"
    Set MyShortNameRS = MyDB.OpenRecordset(SqlText)
    MyShortNameRS.MoveLast  'populate the recordset
    MyShortNameRS.MoveFirst 'return to first record
    xl.Cells(i + 2, 1) = MyLongNameRS!UnitLongName 'enter the longname
    'start an inner loop that runs once for each associated shortname
    For X = 1 To MyShortNameRS.RecordCount
        xl.Cells(i + 2, X + 1) = MyShortNameRS!Name   'enter the shortname
        MyShortNameRS.MoveNext  'move to the next record
    Next X              'end of x loop
MyLongNameRS.MoveNext   'move to the next record
Next i                  'end of i loop
xl.ActiveSheet.Columns().AutoFit    'size the cells on the worksheet
xl.Cells(2, 1) = "Time to create report: " & Format(Time - StartTime, "s.sssss") & " seconds" 'stop timing
    
MyExit:
    Set xl = Nothing        'destroy the object
    MyShortNameRS.Close     'close the recordset
    MyLongNameRS.Close      'close the recordset
    Set MyDB = Nothing      'destroy the object
    Exit Function           'exit function
errListUnits:
    MsgBox Err.Description  'post the error description
    Resume MyExit           'resume at MyExit
End Function
