Option Compare Database

Public Sub ImportFileMakerGo()

Dim MyTitle As String, FileToImport As String

'get the file
    MyTitle = "Select a Go file"
    FileToImport = GetFileName(MyTitle, True, "FileMaker Go (FMEcoPlots_env*.xlsx)", "FMEcoPlots_env*.xlsx")
    If FileToImport = "" Then GoTo MyExit
    CleanUp
    
'check for a complete table set
    If Not CheckForMissingFile(GetPath(FileToImport), GetFile(FileToImport)) Then
        MsgBox "You do not have a full set of Go tables.  Function cannot proceed.", vbInformation, "VPro"
        GoTo MyExit
    End If

'link the excel tables
    LinkTables GetPath(FileToImport), GetFile(FileToImport)

'check for existing plots
    If Not CheckForExistingData Then GoTo MyExit
    
'append data
    AppendData
    MsgBox "Completed import", vbInformation, "VPro"
    
MyExit:
    CleanUp

End Sub

Private Function CheckForExistingData() As Boolean

Dim MyDB As Database, MyRS As Recordset, Answer As Variant
Dim SqlText As String
Set MyDB = CurrentDb

    SqlText = "SELECT UsysEnv.* FROM UsysEnv INNER JOIN xMyEnv ON UsysEnv.PlotNumber = xMyEnv.PlotNumber;"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    If MyRS.RecordCount > 0 Then MyRS.MoveLast
    
    If MyRS.RecordCount > 0 Then
        Answer = MsgBox(MyRS.RecordCount & " of the plots you are attemting to import already exist in the current project.  Click Yes to delete those plots from your current project and proceed with the import or click No to cancel this operation.", vbYesNo, "VPro")
        If Answer = vbYes Then
            DoCmd.SetWarnings False
            SqlText = "DELETE DISTINCTROW UsysEnv.* FROM UsysEnv INNER JOIN xMyEnv ON UsysEnv.PlotNumber = xMyEnv.PlotNumber;"
            DoCmd.RunSQL SqlText
            DoCmd.SetWarnings True
            CheckForExistingData = True
        Else
            CheckForExistingData = False
            GoTo MyExit
        End If
    Else
        CheckForExistingData = True
        GoTo MyExit
    End If
    
MyExit:
    MyRS.Close
    Set MyDB = Nothing

End Function

Private Function GetFile(PathAndFile As String) As String

    GetFile = Dir(PathAndFile)
   
End Function

Private Function GetPath(PathAndFile As String) As String

    GetPath = Left(PathAndFile, Len(PathAndFile) - Len(Dir(PathAndFile)))
    
End Function

Private Sub LinkTables(sPath As String, sFile As String)
    
    Dim FileBit As String, MyFile As String
    
    If Len(sFile) > 19 Then
        FileBit = Mid(sFile, 15, Len(sFile) - 19)
    Else
        FileBit = ""
    End If
    
    MyFile = sPath & "FMEcoPlots_env" & FileBit & ".xlsx"
    DoCmd.TransferSpreadsheet acLink, acSpreadsheetTypeExcel12, "xMyEnv", MyFile, True
    
    MyFile = sPath & "FMEcoPlots_Humus" & FileBit & ".xlsx"
    DoCmd.TransferSpreadsheet acLink, acSpreadsheetTypeExcel12, "xMyHumus", MyFile, True
    
    MyFile = sPath & "FMEcoPlots_Mineral" & FileBit & ".xlsx"
    DoCmd.TransferSpreadsheet acLink, acSpreadsheetTypeExcel12, "xMyMineral", MyFile, True
    
    MyFile = sPath & "FMEcoPlots_VEG_herb" & FileBit & ".xlsx"
    DoCmd.TransferSpreadsheet acLink, acSpreadsheetTypeExcel12, "xMyHerb", MyFile, True
    
    MyFile = sPath & "FMEcoPlots_VEG_moss" & FileBit & ".xlsx"
    DoCmd.TransferSpreadsheet acLink, acSpreadsheetTypeExcel12, "xMyMoss", MyFile, True
    
    MyFile = sPath & "FMEcoPlots_VEG_tree" & FileBit & ".xlsx"
    DoCmd.TransferSpreadsheet acLink, acSpreadsheetTypeExcel12, "xMyTree", MyFile, True

MyExit:

End Sub

Private Function CleanUp()

On Error Resume Next
    DoCmd.DeleteObject acTable, "xMyEnv"
    DoCmd.DeleteObject acTable, "xMyHumus"
    DoCmd.DeleteObject acTable, "xMyMineral"
    DoCmd.DeleteObject acTable, "xMyHerb"
    DoCmd.DeleteObject acTable, "xMyMoss"
    DoCmd.DeleteObject acTable, "xMyTree"
On Error GoTo 0

End Function

Private Function CheckForMissingFile(sPath As String, sFile As String) As Boolean
    
    Dim FileBit As String
    CheckForMissingFile = True
    If Len(sFile) > 19 Then
        FileBit = Mid(sFile, 15, Len(sFile) - 19)
    Else
        FileBit = ""
    End If
    
    If Dir(sPath & "FMEcoPlots_env" & FileBit & ".xlsx", vbDirectory) = vbNullString Then
        CheckForMissingFile = False
        GoTo MyExit
    End If
    
    If Dir(sPath & "FMEcoPlots_Humus" & FileBit & ".xlsx", vbDirectory) = vbNullString Then
        CheckForMissingFile = False
        GoTo MyExit
    End If
    
    If Dir(sPath & "FMEcoPlots_Mineral" & FileBit & ".xlsx", vbDirectory) = vbNullString Then
        CheckForMissingFile = False
        GoTo MyExit
    End If
    
    If Dir(sPath & "FMEcoPlots_VEG_herb" & FileBit & ".xlsx", vbDirectory) = vbNullString Then
        CheckForMissingFile = False
        GoTo MyExit
    End If
    
    If Dir(sPath & "FMEcoPlots_VEG_moss" & FileBit & ".xlsx", vbDirectory) = vbNullString Then
        CheckForMissingFile = False
        GoTo MyExit
    End If
    
    If Dir(sPath & "FMEcoPlots_VEG_tree" & FileBit & ".xlsx", vbDirectory) = vbNullString Then
        CheckForMissingFile = False
        GoTo MyExit
    End If

MyExit:

End Function
