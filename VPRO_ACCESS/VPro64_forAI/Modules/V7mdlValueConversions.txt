Public Function Presence2Class(Presence As Double)
'June 16, 1998
'used in reports

On Error GoTo errPresence2Class
    If Presence >= 0 And Presence <= 0.2 Then
        Presence2Class = "I"
    End If
    If Presence > 0.2 And Presence <= 0.4 Then
        Presence2Class = "II"
    End If
    If Presence > 0.4 And Presence <= 0.6 Then
        Presence2Class = "III"
    End If
    If Presence > 0.6 And Presence <= 0.8 Then
        Presence2Class = "IV"
    End If
    If Presence > 0.8 And Presence <= 1 Then
        Presence2Class = "V"
    End If
    'If Presence > 1 Then
    '    Presence2Class = "VI"
    'End If
MyExit:
    On Error Resume Next
    Exit Function
errPresence2Class:
    MsgBox Err.Description
    Resume MyExit
End Function

Public Function ProminenceClass(MeanCover, Presence)
'Nov. 3, 2004
'used in reports for converting data into prominence classes

Dim TempVal As Single, Msg As String, Title As String
    On Error GoTo errProminenceClass
    TempVal = vpRoundUp(((MeanCover) * 10) * Sqr(Presence))
    If clsRepOpt.SVShowClass = 20 Then
        ProminenceClass = TempVal
        GoTo MyExit
    End If
    If TempVal >= 0 And TempVal <= 15 Then
        ProminenceClass = 1
    End If
    If TempVal > 15 And TempVal <= 50 Then
        ProminenceClass = 2
    End If
    If TempVal > 50 And TempVal <= 100 Then
        ProminenceClass = 3
    End If
    If TempVal > 100 And TempVal <= 200 Then
        ProminenceClass = 4
    End If
    If TempVal > 200 Then
        ProminenceClass = 5
    End If
    Exit Function

MyExit:
    On Error Resume Next
    Exit Function
errProminenceClass:
    MsgBox Err.Description
    Resume MyExit
End Function

Public Function GoldstreamClass(MeanCover, Presence)
'Nov. 3, 2004
'used in reports for converting data into prominence classes
'classes by Will MacKenzie

Dim TempVal As Single, Msg As String, Title As String
    On Error GoTo errProminenceClass
    'equation for goldstream value
    TempVal = vpRoundUp((Presence * 100) * Sqr(MeanCover))
   ' rare species
    If clsRepOpt.SVShowClass = 20 Then
        GoldstreamClass = TempVal
        GoTo MyExit
    End If
'V. rare
    If TempVal >= 0 And TempVal <= 5 Then
        GoldstreamClass = 0
    End If
'rare
    If TempVal > 5 And TempVal <= 25 Then
        GoldstreamClass = 1
    End If
    'uncommon species
    If TempVal > 25 And TempVal <= 75 Then
        GoldstreamClass = 2
    End If
   'common species
    If TempVal > 75 And TempVal <= 150 Then
        GoldstreamClass = 3
    End If
    'abundant species
    If TempVal > 150 And TempVal <= 300 Then
        GoldstreamClass = 4
    End If
   ' very abundant species
    If TempVal > 300 And TempVal <= 500 Then
        GoldstreamClass = 5
    End If
    ' dominant species
    If TempVal > 500 Then
        GoldstreamClass = 6
    End If
MyExit:
    On Error Resume Next
    Exit Function
errProminenceClass:
    MsgBox Err.Description
    Resume MyExit
End Function

Public Function SignifClass(Signif)
'June 16, 1998
'used in reports

On Error GoTo errSignifClass
    If Signif > -1 And Signif <= 0.3 Then
        SignifClass = "+"
    End If
    If Signif > 0.3 And Signif <= 1 Then
        SignifClass = 1
    End If
    If Signif > 1 And Signif <= 2.2 Then
        SignifClass = 2
    End If
    If Signif > 2.2 And Signif <= 5 Then
        SignifClass = 3
    End If
    If Signif > 5 And Signif <= 10 Then
        SignifClass = 4
    End If
    If Signif > 10 And Signif <= 20 Then
        SignifClass = 5
    End If
    If Signif > 20 And Signif <= 33 Then
        SignifClass = 6
    End If
    If Signif > 33 And Signif <= 50 Then
        SignifClass = 7
    End If
    If Signif > 50 And Signif <= 75 Then
        SignifClass = 8
    End If
    If Signif > 75 Then
        SignifClass = 9
    End If
    
MyExit:
    On Error Resume Next
    Exit Function
errSignifClass:
    MsgBox Err.Description
    Resume MyExit
End Function


Public Function vpRoundUp(ValIn As Variant) As Variant
'round value < .1 to .1
'formats values to 0.0
If IsNull(ValIn) Then
    vpRoundUp = Null
    Exit Function
End If
    
If Not ValIn < 0.1 Then
    vpRoundUp = Val(Format(ValIn, "0.0"))
    vpRoundUp = ValIn
Else
    vpRoundUp = 0.1
    vpRoundUp = ValIn
End If
    
End Function

Public Function vpRoundUp2(ValIn As Variant) As Variant
'round value < .1 to .1
'formats values to 0.0
If IsNull(ValIn) Then
    vpRoundUp2 = Null
    Exit Function
End If
    
If Not ValIn < 0.01 Then
    vpRoundUp2 = Format(ValIn, "0.00")
Else
    vpRoundUp2 = 0.01
End If
    
End Function
