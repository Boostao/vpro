Option Compare Database
Option Explicit

Private XMLFileToUse As String, ProjectName As String, NameToUse As String, PathToUse As String
Private DoStop As Boolean

Public Sub ImportXMLProject()

XMLFileToUse = ""
DoStop = False

If Not DoStop Then
    If PlotsInProject() Then 'CHECK FOR PLOTS IN THE CURRENT PROJECT, IF SO ADVISE THE RULES
        MsgBox "Sorry, the project " & clsVProReg.CurrProject & " already contains plots.  XML files can only be imported into empty VPro projects.  Please create an empty VPro project and try again.", vbInformation, "VPro"
        GoTo MyExit
    End If
    
    GetFileName
    
    If Not DoStop Then
        GetName XMLFileToUse
    Else
        GoTo MyExit
    End If
Else
    GoTo MyExit
End If

ImportXMLData

MergeData

CleanUp

DoCmd.SetWarnings True

MsgBox "Done", vbInformation, "VPro"

Exit Sub

MyExit:


End Sub

Private Sub CleanUp()

    DoCmd.DeleteObject acTable, NameToUse & "_Env"
    DoCmd.DeleteObject acTable, NameToUse & "_Veg"
    DoCmd.DeleteObject acTable, NameToUse & "_Humus"
    DoCmd.DeleteObject acTable, NameToUse & "_Mineral"
    DoCmd.DeleteObject acTable, NameToUse & "_Other"
    DoCmd.DeleteObject acTable, NameToUse & "_Audit"

End Sub

Private Sub ImportXMLData()

On Error GoTo errImportXML

    DoCmd.SetWarnings True
    Application.ImportXML PathToUse & NameToUse & ".xml", acAppendData

MyExit:
    Exit Sub
errImportXML:
    If Err.Number = 31550 Then
        MsgBox "The XML file you selected uses an internal name that is also in use by one of your attached projects.  Please check the xml file and unattach the project in conflict.", vbInformation, "VPro"
        DoStop = True
    Else
        MsgBox Err.Description
        DoStop = True
    End If
    Resume MyExit
    
End Sub

Private Sub GetFileName()

    Dim FD As FileDialog
    Set FD = Application.FileDialog(msoFileDialogFilePicker)
    FD.AllowMultiSelect = False
    FD.filters.Add "XML File", "*.xml", 1
    FD.FilterIndex = 1
    XMLFileToUse = ""

    With FD
        If .Show = -1 Then
            XMLFileToUse = FD.SelectedItems.Item(1)
        End If
    End With
    
    If XMLFileToUse = "" Then DoStop = True
    Set FD = Nothing

End Sub

Private Sub GetName(FileName As String)

On Error GoTo errGetName
Dim MyStr As String, MyPos As Integer, ExitLoop As Boolean, X As Integer, MaxNum As Integer
MyStr = FileName
MaxNum = 0

MyPos = InStr(1, MyStr, "\", vbTextCompare)

Do While Not ExitLoop
    X = InStr(MyPos + 1, MyStr, "\", vbTextCompare)
    If X = 0 Then
        MaxNum = MyPos
        Exit Do
    End If
    If X > MaxNum Then MaxNum = X
    MyPos = X
    If MyPos = 0 Then
        Exit Do
    End If
Loop

NameToUse = Mid(MyStr, MaxNum + 1, Len(MyStr) - MaxNum - 4)
PathToUse = Left(MyStr, Len(MyStr) - (Len(NameToUse) + 4))

MyExit:
    Exit Sub
    
errGetName:
    If Err.Number = 5 Then
        'do nothing
    Else
        MsgBox Err.Number & " - " & Err.Description
    End If
    DoStop = True
    Resume MyExit
End Sub

Private Function PlotsInProject() As Boolean

Dim MyDB As Database, MyRS As Recordset
Set MyDB = CurrentDb
Set MyRS = MyDB.OpenRecordset(clsVProReg.CurrProject & "_ENV")

If MyRS.RecordCount > 0 Then
    PlotsInProject = True
Else
    PlotsInProject = False
End If

MyRS.Close
Set MyDB = Nothing

End Function

Private Sub MergeData()

Dim SqlText As String

'time to merge
    DoCmd.SetWarnings False
    SqlText = "INSERT INTO [" & CurrProject() & "_Env]"
    SqlText = SqlText & " SELECT [" & NameToUse & "_Env].*"
    SqlText = SqlText & " FROM [" & NameToUse & "_Env];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & CurrProject() & "_Veg]"
    SqlText = SqlText & " SELECT [" & NameToUse & "_Veg].*"
    SqlText = SqlText & " FROM [" & NameToUse & "_Veg];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & CurrProject() & "_Other]"
    SqlText = SqlText & " SELECT [" & NameToUse & "_Other].*"
    SqlText = SqlText & " FROM [" & NameToUse & "_Other];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & CurrProject() & "_Humus]"
    SqlText = SqlText & " SELECT [" & NameToUse & "_Humus].*"
    SqlText = SqlText & " FROM [" & NameToUse & "_Humus];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & CurrProject() & "_Mineral]"
    SqlText = SqlText & " SELECT [" & NameToUse & "_Mineral].*"
    SqlText = SqlText & " FROM [" & NameToUse & "_Mineral];"
    DoCmd.RunSQL SqlText
    
    SqlText = "INSERT INTO [" & CurrProject() & "_Audit]"
    SqlText = SqlText & " SELECT [" & NameToUse & "_Audit].*"
    SqlText = SqlText & " FROM [" & NameToUse & "_Audit];"
    DoCmd.RunSQL SqlText
    DoCmd.SetWarnings True

End Sub
