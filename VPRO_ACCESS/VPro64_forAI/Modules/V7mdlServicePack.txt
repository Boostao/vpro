Public Sub SpCheck()

Dim MyDB As Database, MyRS As Recordset, Answer As Variant
Dim LastCheckDate As Date
Set MyDB = CurrentDb
On Error GoTo errSpCheck

Set MyRS = MyDB.OpenRecordset("USysServicePackLastCheck")

    If clsVProReg.DoUpdateCheck Then
        If Now - MyRS!LastCheck > clsVProReg.DoUpdateCheckInterval Then
            Answer = MsgBox("It has been " & Int(Now - MyRS!LastCheck) & " days since VPro last checked for updates.  Check now?", vbYesNo, "VPro")
            If Answer = vbYes Then
                If DownloadPackList = True Then
                    AttachList
                    SpStatus
                    UnattachList
                    MyRS.Edit
                    MyRS!LastCheck = Now
                    MyRS.Update
                Else
                    MsgBox "Unable to check for service packs", vbInformation, "VPro"
                End If
            End If
        End If
    End If
    
MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    On Error GoTo 0
    Exit Sub
errSpCheck:
    MsgBox Err.Description, vbInformation, "VPro"
    Resume MyExit

End Sub

Private Sub SpStatus()

Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String, Msg As String
Set MyDB = CurrentDb
On Error GoTo errSpStatus

    SqlText = "SELECT USysServicePackList.ServicePackName, USysServicePackList.DateCreated, USysServicePackList.Description"
    SqlText = SqlText & " FROM USysServicePackList LEFT JOIN USysServicePacks ON USysServicePackList.ServicePackName = USysServicePacks.ServicePackName"
    SqlText = SqlText & " WHERE (((USysServicePacks.ServicePackName) Is Null));"
    Set MyRS = MyDB.OpenRecordset(SqlText)
    
    If MyRS.RecordCount > 0 Then
        Msg = "Service pack(s) that may be of interest to you:" & Chr(13)
        Do Until MyRS.EOF
            Msg = Msg & Chr(13) & MyRS!DateCreated & " - " & MyRS!ServicePackName & ".  " & MyRS!Description & Chr(13)
            MyRS.MoveNext
        Loop
        MsgBox Msg, vbInformation, "VPro"
    Else
        MsgBox "No new service packs found.", vbInformation, "VPro"
    End If
    
MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    On Error GoTo 0
    Exit Sub
errSpStatus:
    MsgBox Err.Description, vbInformation, "VPro"
    Resume MyExit
    
End Sub

MyExit:

    On Error GoTo 0
    Exit Sub
    
errAttachList:

End Sub

Private Function DownloadFilefromWeb(URL As String, strSavePath As String) As Boolean

     End If
     
    Exit Function
myerr:

End Function
