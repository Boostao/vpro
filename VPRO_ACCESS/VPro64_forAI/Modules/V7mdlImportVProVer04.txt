Private Sub BuildReport(xl As Object, MyRow As Integer, MyCol As Integer, FieldName As String, TableName As String)
Dim i As Integer

CheckCodes TableName, FieldName

xl.Cells(MyRow, MyCol) = TableName
xl.Cells(MyRow + 1, MyCol) = FieldName
MyRow = MyRow + 2
xl.Cells(MyRow, MyCol) = "Plot Number"
xl.Cells(MyRow, MyCol + 1) = "Item"

For i = 1 To UBound(MyArray)
    xl.Cells(MyRow + i, MyCol) = MyArray(i).Plot
    xl.Cells(MyRow + i, MyCol + 1) = MyArray(i).Item
Next i

xl.Range(xl.Cells(1, MyCol), xl.Cells(1, MyCol + 1)).Merge
xl.Range(xl.Cells(2, MyCol), xl.Cells(2, MyCol + 1)).Merge
xl.Range(xl.Cells(1, MyCol), xl.Cells(2, MyCol + 1)).HorizontalAlignment = xlCenter

End Sub

Private Sub CheckCodes(Project As String, FieldName As String)

'Oct. 30, 2002
Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String
Set MyDB = CurrentDb
ReDim MyArray(0)

SqlText = "SELECT PlotNumber, " & FieldName
SqlText = SqlText & " FROM " & Project
SqlText = SqlText & " WHERE (((" & FieldName & ") Is Not Null) AND ((IsNumeric([" & FieldName & "]))=0));"
Set MyRS = MyDB.OpenRecordset(SqlText, dbOpenForwardOnly)

If MyRS.RecordCount > 0 Then
    Do Until MyRS.EOF
        ReDim Preserve MyArray(UBound(MyArray) + 1)
        MyArray(UBound(MyArray)).Plot = MyRS!PlotNumber
        MyArray(UBound(MyArray)).Item = MyRS(FieldName)
        MyRS.MoveNext
    Loop
    NoErrors = False
End If

MyExit:
    On Error Resume Next
    Set MyDB = Nothing
    Exit Sub
errCheckCodes:
    MsgBox Err.Description
    Resume MyExit
End Sub

Private Function GetLongMin(LongIn As Variant)

If IsNull(LongIn) Or LongIn = "" Then
    GetLongMin = Null
    GoTo MyExit
End If

If IsNumeric(Mid(LongIn, 4, 1)) Then
    GetLongMin = Mid(LongIn, 4, 2)
Else
    GetLongMin = Mid(LongIn, 5, 2)
End If

MyExit:

End Function

Private Function GetLongSec(LongIn As Variant)

If IsNull(LongIn) Or LongIn = "" Then
    GetLongSec = Null
    GoTo MyExit
End If

If IsNumeric(Right(LongIn, 1)) Then
    GetLongSec = Right(LongIn, 2)
Else
    GetLongSec = Mid(LongIn, Len(LongIn) - 2, 2)
End If

MyExit:

End Function

Private Function GetLatMin(LatIn As Variant)

If IsNull(LatIn) Or LatIn = "" Then
    GetLatMin = Null
    GoTo MyExit
End If

If IsNumeric(Mid(LatIn, 3, 1)) Then
    GetLatMin = Mid(LatIn, 3, 2)
Else
    GetLatMin = Mid(LatIn, 4, 2)
End If

MyExit:

End Function

Private Function GetLatSec(LatIn As Variant)

If IsNull(LatIn) Or LatIn = "" Then
    GetLatSec = Null
    GoTo MyExit
End If

If IsNumeric(Right(LatIn, 1)) Then
    GetLatSec = Right(LatIn, 2)
Else
    GetLatSec = Mid(LatIn, Len(LatIn) - 2, 2)
End If

MyExit:

End Function
