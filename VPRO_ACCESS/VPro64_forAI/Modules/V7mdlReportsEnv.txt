Private Sub SortSheets(xl As Excel.Application)

Dim lCount As Long, lCounted As Long
Dim lShtLast As Long
lShtLast = xl.Sheets.count

    For lCount = 1 To lShtLast

        For lCount2 = lCount To lShtLast

            If UCase(xl.Sheets(lCount2).Name) < UCase(xl.Sheets(lCount).Name) Then

                xl.Sheets(lCount2).Move Before:=xl.Sheets(lCount)

            End If

        Next lCount2

    Next lCount

End Sub

Private Sub BuildDynamicList()

Dim SqlText As String, SuTable As String
DoCmd.SetWarnings False
SuTable = CurrPlotlist & "_SU"

DoCmd.RunSQL SqlText

SqlText = "INSERT INTO USysSuTableDynamic_SU ( PlotNumber, SiteUnit )"
SqlText = SqlText & " SELECT PlotNumber, SiteUnit"
SqlText = SqlText & " FROM " & SuTable & ";"
DoCmd.RunSQL SqlText

DoCmd.SetWarnings True

End Sub

Private Function Integer2Strata(StrataNumber As Integer)
'June 23, 1998
    Select Case StrataNumber
    Case 1
        Integer2Strata = "A"
    Case 2
        Integer2Strata = "B"
    Case 3
        Integer2Strata = "C"
    Case 4
        Integer2Strata = "D"
    Case Else
        Integer2Strata = "N/A"
    End Select
End Function

Private Function CreateBreakPointUnits()

'Apr. 6, 2018
'creates site units from break points
Dim SqlText As String, X
On Error GoTo errCreateBreakPointUnits

    X = CreateBreakTable("Hierarchy")
    SqlText = "DELETE * FROM USysSuTableDynamic_SU;"
    DoCmd.SetWarnings False
    DoCmd.RunSQL (SqlText)
    SqlText = "INSERT INTO USysSuTableDynamic_SU ( SiteUnit, PlotNumber )"
    SqlText = SqlText & " SELECT Mid([Reference],2) AS SiteUnit, USysBreakTable.PlotNumber"
    SqlText = SqlText & " FROM USysBreakTable;"
    DoCmd.RunSQL (SqlText)
    DoCmd.SetWarnings True
    
MyExit:
    On Error Resume Next
    Exit Function
errCreateBreakPointUnits:
    MsgBox Err.Description
    Resume MyExit
End Function
