Private Sub OrderHierarchyStep2(MyDB As Database, Unit As Long, TableName As String, Optional MyCount As Integer)
Dim MyRecords As Recordset, MyCriteria As String
Dim NextUnit As Long, Shifty As Long, BK As String

If IsMissing(MyCount) Then MyCount = 1
If UnitOrder = 0 Then
    UnitOrder = 1
Else
    'UnitOrder = UnitOrder + 1
End If

Set MyRecords = MyDB.OpenRecordset(TableName, dbOpenDynaset)
'MyRecords.MoveLast
'MyRecords.MoveFirst
MyCriteria = "ID = " & Unit
    MyRecords.FindFirst MyCriteria
    If IsNull(MyRecords!Parent) Then
        MyRecords.Edit
        MyRecords!MyOrder = Format(MasterCount, "00000")
        MyRecords.Update
    End If
    If MyRecords.NoMatch Then GoTo MyExit
    Shifty = MyRecords!ID
        
    MyRecords.MoveFirst
    MyCriteria = "Parent = " & Shifty
    MyRecords.FindFirst MyCriteria

MyCount = MyCount + 1
MasterCount = MyCount

Do Until MyRecords.NoMatch
    BK = MyRecords.Bookmark
    NextUnit = MyRecords!ID
        MyRecords.Edit
        If MyRecords!Level = 11 Then
            MyRecords!ChildID = UnitOrder
            UnitOrder = UnitOrder + 1
        End If
        MyRecords!MyOrder = Format(MyCount, "00000")
        MyRecords.Update
    OrderHierarchyStep2 MyDB, NextUnit, TableName, MyCount
    MyRecords.Bookmark = BK
    MyRecords.FindNext MyCriteria
Loop

MyExit:
    MyRecords.Close
End Sub

Private Function Check4AllLevel11(db As Database, TableName As String) As Boolean
Dim MyRS As Recordset
Dim SqlText As String
SqlText = "SELECT [Level]"
SqlText = SqlText & " From " & TableName
SqlText = SqlText & " WHERE ((([Level])<>11));"
Set MyRS = db.OpenRecordset(SqlText)

If MyRS.RecordCount > 0 Then
    Check4AllLevel11 = False
Else
    Check4AllLevel11 = True
End If

MyExit:
    On Error Resume Next
    MyRS.Close
    
End Function

Private Function AddRoot(db As Database, TableName As String)
Dim MyRS As Recordset, RootID As Long, i As Integer, MyFld As Field

Set MyRS = db.OpenRecordset(TableName, dbOpenDynaset)
MyRS.AddNew
MyRS!Name = "Root"
RootID = MyRS!ID
MyRS.Update

Set MyFld = MyRS.Fields("Parent")
MyFld.Properties("RowSource") = TableName

MyRS.MoveLast
MyRS.MoveFirst

For i = 1 To MyRS.RecordCount
    If MyRS!ID <> RootID Then
        MyRS.Edit
        MyRS!Parent = RootID
        MyRS.Update
    End If
    MyRS.MoveNext
Next i

MyExit:
    On Error Resume Next
    MyRS.Close
    
End Function
