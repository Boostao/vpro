Private Sub ListFieldInfo(TableName As String)

Dim MyDB As Database, MyTD As TableDef
Dim i As Integer, xl As New Excel.Application
Set MyDB = CurrentDb
Set MyTD = MyDB.TableDefs(TableName)

xl.Visible = True
xl.Workbooks.Add

    With xl.ActiveSheet
    For i = 0 To MyTD.Fields.count - 1
        
        .Cells(i + 1, 1) = MyTD.Fields(i).Name
        .Cells(i + 1, 2) = MyTD.Fields(i).type
        .Cells(i + 1, 3) = MyTD.Fields(i).Size
    
    'MsgBox dbText
    Next i
    End With

Set MyDB = Nothing

End Sub
