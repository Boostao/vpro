Option Compare Database
Option Explicit

Public Sub ZLSBeGone(dbName As Variant)
'Oct. 25, 2002
'removes zero-length strings
Dim MyDB As Database
Dim SqlText As String, i As Integer, X As Integer, TblName As String, fldName As String
Dim DefaultWorkspace As Workspace
Dim fldCount As Integer, tblCount As String
Set DefaultWorkspace = DBEngine.Workspaces(0)
Set MyDB = DefaultWorkspace.OpenDatabase(dbName)
TblName = "Sample_env"
tblCount = 0
fldCount = 0
DoCmd.SetWarnings False
For X = 0 To MyDB.TableDefs.count - 1
    TblName = MyDB.TableDefs(X).Name
    If TableExclude(TblName, MyDB) Then GoTo Jump
    DoCmd.TransferDatabase acLink, "Microsoft Access", dbName, acTable, TblName, TblName
    For i = 0 To MyDB.TableDefs("" & TblName & "").Fields.count - 1
        fldName = MyDB.TableDefs("" & TblName & "").Fields(i).Name
        Application.SysCmd acSysCmdSetStatus, "Cleaning table: " & TblName & "   field: " & fldName
        If MyDB.TableDefs("" & TblName & "").Fields(i).type = 10 Then
            SqlText = "UPDATE [" & TblName & "] SET [" & TblName & "].[" & fldName & "] = Null WHERE ((([" & TblName & "].[" & fldName & "])=''));"
            DoCmd.RunSQL SqlText
        End If
        If MyDB.TableDefs("" & TblName & "").Fields(i).type = dbMemo Then
            SqlText = "UPDATE [" & TblName & "] SET [" & TblName & "].[" & fldName & "] = Null WHERE ((([" & TblName & "].[" & fldName & "])=''));"
            DoCmd.RunSQL SqlText
        End If
    Next i
    DoCmd.DeleteObject acTable, TblName
    fldCount = fldCount + i
Jump:
Next X
tblCount = X

'MsgBox Application.hWndAccessApp

Application.SysCmd acSysCmdClearStatus

Set MyDB = Nothing

DoCmd.SetWarnings True
'MsgBox fldCount & " fields were cleaned in " & tblCount & " tables."
'Forms![frmControl]![lblStatus].Caption = fldCount & " fields were cleaned in " & tblCount & " tables."
End Sub

Private Function TableExclude(TblName As String, MyDB As Database) As Boolean

If Left(TblName, 4) = "MSys" Then TableExclude = True

If Left(TblName, 4) = "USys" Then TableExclude = True

If MyDB.TableDefs(TblName).Connect <> "" Then TableExclude = True

Jump:


End Function
