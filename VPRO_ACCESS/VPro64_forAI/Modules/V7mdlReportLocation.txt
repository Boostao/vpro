Private Type eData
    PlotNumber As String
    Zone As String
    SubZone As String
    SiteSeries As String
    LocationAccuracy As String
    Long As Single
    Lat As Single
End Type
Public Function ReportLocation()

'Dec. 19, 2019
On Error GoTo errReportLocation
Dim MyDB As Database, MyRS As Recordset
Dim xl As Excel.Application
Dim SqlText As String, i As Integer ', MyData() As eData
Set MyDB = CurrentDb

Dim MyTime As New clsStopWatch
MyTime.StartTimer

'SqlText = "SELECT USysEnv.PlotNumber, USysEnv.Zone, USysEnv.SubZone, USysEnv.SiteSeries, USysEnv.LocationAccuracy, Longitude, Latitude, Elevation"
SqlText = "SELECT USysEnv.PlotNumber, USysEnv.Zone, USysEnv.SubZone, USysEnv.SiteSeries, USysEnv.LocationAccuracy, [Longitude]*-1 AS [Long], USysEnv.Latitude, USysEnv.Elevation"
SqlText = SqlText & " FROM USysEnv"
SqlText = SqlText & " WHERE (((USysEnv.Latitude) Is Not Null) AND ((USysEnv.Longitude) Is Not Null));"
'MyDB.CreateQueryDef "fred", SqlText
Set MyRS = MyDB.OpenRecordset(SqlText)
If MyRS.RecordCount = 0 Then
    MsgBox "These plots contain no lat/longs", vbInformation, "VPro"
    GoTo MyExit
End If

i = 3
On Error Resume Next
Set xl = GetObject(, "Excel.Application")
If Err.Number <> 0 Then
    Set xl = CreateObject("Excel.Application")
    Err.Clear
End If
On Error GoTo errReportLocation
xl.Workbooks.Add
xl.Visible = True

On Error GoTo 0

MyRS.MoveLast
MyRS.MoveFirst
Dim MyData As Variant, MySheet As Worksheet
Set MySheet = xl.ActiveSheet
ReDim MyData(1 To MyRS.RecordCount, 1 To 8)
Dim MyRow As Long
MyRow = 1

Do Until MyRS.EOF
    MyData(MyRow, 1) = "'" & MyRS!PlotNumber
    MyData(MyRow, 2) = MyRS!Zone
    MyData(MyRow, 3) = MyRS!SubZone
    MyData(MyRow, 4) = "'" & MyRS!SiteSeries
    MyData(MyRow, 5) = MyRS!LocationAccuracy
    MyData(MyRow, 6) = MyRS!Latitude
    MyData(MyRow, 7) = MyRS!Long
    MyData(MyRow, 8) = MyRS!Elevation
    MyRow = MyRow + 1
    MyRS.MoveNext
Loop

MySheet.Range(xl.Cells(3, 1), xl.Cells(MyRow + 1, 8)).Value = MyData

xl.Cells(2, 1) = "Plot Number"
xl.Cells(2, 2) = "Zone"
xl.Cells(2, 3) = "Subzone"
xl.Cells(2, 4) = "Site Series"
xl.Cells(2, 5) = "Accuracy"
xl.Cells(2, 6) = "Latitude"
xl.Cells(2, 7) = "Longitude"
xl.Cells(2, 8) = "Elevation"
xl.Range("A1:H2").Font.Bold = True

xl.Columns.AutoFit
xl.Cells(1, 1) = "Project: " & clsVProReg.CurrProject & "  /  SU: " & clsVProReg.CurrPlotlist

xl.Range(xl.Cells(3, 1), xl.Cells(MyRow + 1, 8)).Select
    With xl.Selection.Borders(xlEdgeLeft)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .ColorIndex = xlAutomatic
    End With
    With xl.Selection.Borders(xlEdgeTop)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .ColorIndex = xlAutomatic
    End With
    With xl.Selection.Borders(xlEdgeBottom)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .ColorIndex = xlAutomatic
    End With
    With xl.Selection.Borders(xlEdgeRight)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .ColorIndex = xlAutomatic
    End With
    With xl.Selection.Borders(xlInsideVertical)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .ColorIndex = xlAutomatic
    End With
    With xl.Selection.Borders(xlInsideHorizontal)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .ColorIndex = xlAutomatic
    End With
    xl.Range("A2:H2").Select
    xl.Selection.Borders(xlEdgeLeft).LineStyle = xlNone
    With xl.Selection.Borders(xlEdgeTop)
        .LineStyle = xlContinuous
        .Weight = xlMedium
        .ColorIndex = xlAutomatic
    End With
    With xl.Selection.Borders(xlEdgeBottom)
        .LineStyle = xlContinuous
        .Weight = xlMedium
        .ColorIndex = xlAutomatic
    End With
    xl.Selection.Interior.ColorIndex = 40

xl.Cells(1, 8).Select

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Set xl = Nothing
    Exit Function
errReportLocation:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function ConvertDegreeDecimalMinutesToDecimalDegrees(Degrees As Integer, DecimalMinutes As Single) As Single

Dim Minutes As Single

Minutes = DecimalMinutes / 60
ConvertDegreeDecimalMinutesToDecimalDegrees = Degrees + Minutes

End Function

Private Function ConvertDecimalDegreesToDegreeDecimalMinutes(DecimalDegrees As Single) As String

Dim Degrees As Integer, Minutes As Single

Degrees = Int(DecimalDegrees)
Minutes = ((DecimalDegrees - Degrees) * 60)
ConvertDecimalDegreesToDegreeDecimalMinutes = Degrees & " - " & Minutes

End Function

Private Function ConvertDecimalDegreesToLongLat(DecimalDegrees As Single) As String

Dim Degrees As Integer, Minutes As Single, Seconds As Single

Degrees = Int(DecimalDegrees)
Minutes = ((DecimalDegrees - Degrees) * 60)
Seconds = (Minutes - Int(Minutes)) * 60
ConvertDecimalDegreesToLongLat = Degrees & " - " & Int(Minutes) & " - " & Seconds

End Function
