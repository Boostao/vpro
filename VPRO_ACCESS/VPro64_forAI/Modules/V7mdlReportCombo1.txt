Private Sub NewShadeCells(FR As Integer, FC As Integer, LR As Integer, LC As Integer)
'Apr. 1, 2015

Dim xl As Excel.Application
Dim ColourGreater As Single, GrayGreater As Single
Dim MyDB As DAO.Database, MyRS As Recordset
Dim SqlText As String, ThemeTable As String, Criteria As String, CurrRow As Integer, CurrSpp As String
Dim c As Variant, MyTest As Variant, ThemeTest As Boolean
clsStopWatch.StartTimer

CreateExcel xl
xl.Range(xl.Cells(FR, FC), xl.Cells(LR, LC)).Select

Set MyDB = CurrentDb
ThemeTable = GetSetting("VPro64", "ReportOptions", "cmbThemeTable")
If ThemeTable = "None" Then GoTo MyExit
SqlText = "SELECT ScientificName, ColourCode, PatternCode, FontColour FROM " & ThemeTable & "_Theme WHERE (((Use)=Yes));"
Set MyRS = MyDB.OpenRecordset(SqlText)
ColourGreater = GetSetting("VPro64", "ReportOptions", "cmbColourGreater")

NoColour:
GrayGreater = GetSetting("VPro64", "ReportOptions", "cmbgrayGreater")
CurrRow = xl.ActiveCell.Row
CurrSpp = xl.Cells(CurrRow, 2)

If ThemeTable = "None" Then GoTo Jump1
ThemeTest = LoadSppInfo(CurrSpp, MyRS)
Jump1:

On Error Resume Next
For Each c In xl.Selection
    If c.Row <> CurrRow Then
        CurrRow = c.Row
        CurrSpp = xl.Cells(CurrRow, 2)
        If ThemeTable = "None" Then GoTo Jump2
        ThemeTest = LoadSppInfo(CurrSpp, MyRS)
Jump2:
    End If
    'MyTest = c.Value
    'If c.Value > 0 Then
        MyTest = c.Value
    'Else
    '    GoTo Jump
    'End If
    'MyTest = GetCover(c.Value)
    If ThemeTable = "None" Then GoTo Jump
    If ThemeTest And MyTest > ColourGreater Then
        With xl.Range(c.Address)
            .Interior.ColorIndex = Shade.ColourCode
            .Interior.Pattern = Shade.PatternCode
            .Font.ColorIndex = Shade.FontColour
            .Font.Bold = True
        End With
        With xl.Cells(c.Row, 2)
            .Interior.ColorIndex = Shade.ColourCode
            .Interior.Pattern = Shade.PatternCode
            .Font.ColorIndex = Shade.FontColour
            .Font.Bold = True
        End With
    End If
Jump:
    If Not ThemeTest And MyTest > GrayGreater Then
        With xl.Range(c.Address)
            .Interior.ColorIndex = 15
            .Interior.Pattern = xlSolid
            .Font.Bold = True
        End With
        With xl.Cells(c.Row, 2)
            .Interior.ColorIndex = 15
            .Interior.Pattern = xlSolid
            .Font.Bold = True
        End With
    End If

Next c

MyExit:
On Error GoTo 0
Debug.Print clsStopWatch.EndTimer / 1000

End Sub

Private Function LoadSppInfo(CurrSpp As String, rs As Recordset) As Boolean
Dim Criteria As String
Criteria = "ScientificName = '" & CurrSpp & "'"
rs.FindFirst Criteria

If rs.NoMatch Then
    LoadSppInfo = False
Else
    Shade.ColourCode = rs!ColourCode
    Shade.FontColour = rs!FontColour
    Shade.PatternCode = rs!PatternCode
    LoadSppInfo = True
End If

End Function
