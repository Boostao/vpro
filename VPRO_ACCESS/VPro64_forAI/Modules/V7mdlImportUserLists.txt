Option Compare Database
Option Explicit
Function ImportUserSiteUnits()

'Dec 4, 2008
'Imports users' site unit list
On Error Resume Next
    DoCmd.DeleteObject acTable, "ImportedUnitList"
On Error GoTo errImportUserSiteUnits
Dim MyDB As Database, RemoteDB As Database, i As Integer
Dim MyFile As String, MyTitle As String, strTestTable As String
Dim ListInDB As Boolean, SqlText As String
ListInDB = False

    Set MyDB = CurrentDb()
    MyFile = ""
    MyTitle = ProgramName() & " - pick a database that contains a user site unit list"
    MyFile = GetFileName(MyTitle, True)
    If MyFile = "" Then GoTo MyExit
    Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(MyFile)
        
        DoCmd.Hourglass True
        For i = 0 To RemoteDB.TableDefs.count - 1
            strTestTable = RemoteDB.TableDefs(i).Name
            If strTestTable = "UserSiteUnitList" Then
                ListInDB = True
            End If
        Next i
        DoCmd.Hourglass False

        If ListInDB Then
            DoCmd.TransferDatabase acLink, "Microsoft Access", RemoteDB.Name, acTable, "UserSiteUnitList", "ImportedUnitList"
        Else
            MsgBox "This database didn't contain a user site unit list.", vbInformation, clsVProReg.ProgramName
            GoTo MyExit
        End If

MergeLists:
        SqlText = "INSERT INTO USysUserSiteUnitList"
        SqlText = SqlText & " SELECT ImportedUnitList.*"
        SqlText = SqlText & " FROM ImportedUnitList;"
        DoCmd.RunSQL SqlText

MyExit:
    On Error Resume Next
    DoCmd.DeleteObject acTable, "ImportedUnitList"
    Set MyDB = Nothing
    Set RemoteDB = Nothing
    Exit Function
errImportUserSiteUnits:
    If Err.Number = 2501 Then
        'do nothing
    Else
        MsgBox Err.Description
    End If
    Resume MyExit
End Function
