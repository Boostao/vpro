Public Sub ProfilePlots(SubVarSppAsSpp As Boolean)

Dim MyDB As Database, MyRS As Recordset
Dim SqlText As String
Set MyDB = CurrentDb

    'clear previous run
    DoCmd.SetWarnings False
    SqlText = "DELETE USysPlotProfilePlotsTemp.* FROM USysPlotProfilePlotsTemp;"
    DoCmd.RunSQL SqlText
    SqlText = "DELETE * FROM USysPlotProfilePlots;"
    DoCmd.RunSQL SqlText
    DoCmd.SetWarnings True
    
    'create profile table recordset
    SqlText = "SELECT Order, Table, Field, Operator, Layer, Species, Criteria, Operation, PlotCount"
    SqlText = SqlText & " From " & clsVProReg.CurrVegProfile & "_Profile"
    SqlText = SqlText & " ORDER BY " & clsVProReg.CurrVegProfile & "_Profile.Order;"
    Set MyRS = MyDB.OpenRecordset(SqlText) ' change this later to the current profile
    'MyDB.CreateQueryDef "herman", SqlText
    'clean up
    CleanUp

    'clear plot counts in prfile table
    ClearPlotCounts clsVProReg.CurrVegProfile & "_Profile"

    'set current lump table
    SetLump SubVarSppAsSpp

    'update form labels
    UpDateFormLabels "lblFilteredPlots", "Filtered plots: 0"
    UpDateFormLabels "lblTotalPlots", "Project plots: " & nTotalPlots(MyDB)

    'loop through profile records
    Do Until MyRS.EOF
        Select Case MyRS!Table
            Case "Env"
                ProfileEnv MyDB, MyRS!Order
            Case "Veg"
                ProfileVeg MyDB, MyRS!Order
            Case "Lump"
                ProfileVeg MyDB, MyRS!Order
            Case Else
                MsgBox "Error finding " & MyRS!Table, vbInformation, "VPro"
        End Select
        UpdateFilterCount MyDB
        Forms("USysPlotProfiling").Recalc
        Forms("USysPlotProfiling").Repaint
        MyRS.MoveNext
    Loop

    'update forms
    UpdateForms

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    CleanUp
    Exit Sub

End Sub

Private Function CurrCount(MyDB As Database) As Long

Dim MyRS As Recordset
    Set MyRS = MyDB.OpenRecordset("USysPlotProfilePlots")
    CurrCount = MyRS.RecordCount
    MyRS.Close

End Function

Private Sub WriteVegProfile2Filter(FormName As String) 'rs As Recordset, nProfileRecs As Integer)
On Error GoTo errWriteVegProfile2Filter
Dim i As Long, FilterText As String, MyForm As Form
Dim MyDB As Database, SuRecords As Recordset, MyRS As Recordset
Dim Msg As String, dgdef As Integer, Response As Integer, Title As String, X

Set MyDB = CurrentDb()
Set MyForm = Forms(FormName)
Set MyRS = MyDB.OpenRecordset("USysPlotProfilePlots")
MyForm.FilterOn = False

If MyRS.RecordCount > 200 Then GoTo TooComplex
    
    FilterText = "(("
    Do Until MyRS.EOF
        If Len(FilterText) = 2 Then
            FilterText = FilterText & "PlotNumber='" & MyRS!PlotNumber & "'"
        Else
            FilterText = FilterText & " OR PlotNumber='" & MyRS!PlotNumber & "'"
        End If
        MyRS.MoveNext
    Loop
    FilterText = FilterText & "))"

    If FilterText <> "(())" Then
        MyForm.Filter = FilterText
        MyForm.FilterOn = True
    Else
        'MsgBox "Zero plots matched.  Filter not applied.", vbInformation, "VPro"
        MsgBox "Profile resulted in zero plots.  Filter not applied to data form.", vbInformation, "VPro"
    End If

GoTo MyExit

TooComplex:
Set MyDB = CurrentDb()
    Title = ProgramName()
    Msg = "This filter exceeded Access's abilities.  Do you wish VPro to save the selected plots as a site unit table?"
    dgdef = MB_YESNO + MB_ICONSTOP + MB_DEFBUTTON2   ' Describe dialog box.
    Response = MsgBox(Msg, dgdef, Title)    ' Get user response.
    If Response = 6 Then
        sysSaveFilter2 = 2
        sysBreakSU = "bla"
        X = CreateSuTable()
        If sysBreakSU = "bla2" Then GoTo MyExit
        MyRS.MoveFirst
        Set SuRecords = MyDB.OpenRecordset(sysBreakSU)
        For i = 1 To MyRS.RecordCount
            SuRecords.AddNew
            SuRecords!PlotNumber = MyRS!PlotNumber
            SuRecords.Update
            MyRS.MoveNext
        Next i
        MsgBox "Plots matching your vegetation profile have been inserted into " & Left(SuRecords.Name, Len(SuRecords.Name) - 3)
        SuRecords.Close
    End If
    
MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Sub
errWriteVegProfile2Filter:
    If Err.Number = 2176 Then
        GoTo TooComplex
    Else
        MsgBox Err.Description
        GoTo MyExit
    End If
End Sub

            SqlText = SqlText & " GROUP BY UsysEnv.PlotNumber;"
            DoCmd.RunSQL SqlText
            
            'delete plots without species
            SqlText = "DELETE DISTINCTROW USysPlotProfilePlots.*, USysPlotProfilePlotsTemp.PlotNumber"
            SqlText = SqlText & " FROM USysPlotProfilePlots LEFT JOIN USysPlotProfilePlotsTemp ON USysPlotProfilePlots.PlotNumber = USysPlotProfilePlotsTemp.PlotNumber"
            SqlText = SqlText & " WHERE (((USysPlotProfilePlotsTemp.PlotNumber) Is Null));"
            DoCmd.RunSQL SqlText
        End If
    
        MyRS.MoveNext
    Loop
    DoCmd.SetWarnings True

    MyRS.Close
    Set MyDB = Nothing
