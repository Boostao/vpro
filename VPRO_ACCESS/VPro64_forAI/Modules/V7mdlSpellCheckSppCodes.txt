Option Compare Database
Option Explicit
Dim MyDB As Database
Global sysSpeciesTestNext As Integer
Global sysSpeciesToChange
Global sysActiveForm As Form
Dim X

Function CheckSpeciesCodes(Optional PlotNumber) As Boolean
'Oct. 4, 2002
'checks species codes
'On Error GoTo errCheckSpeciesCodes
sysSpeciesTestNext = 0 '0 = continue, 1 = wait for dialog, 3 = change species, 4 = full stop
Dim Plot As String, OldSpp As String, NewSpp As String
Dim test As Integer, VegRS As Recordset, SqlText As String, MyForm As Form
Dim MyProject As String, MySU As String
MyProject = clsVProReg.CurrProject
MySU = clsVProReg.CurrPlotlist
Set MyDB = CurrentDb()
If Not IsMissing(PlotNumber) Then
    SqlText = "SELECT * FROM [" & CurrProject() & "_Veg] WHERE (((Species) Is Not Null) AND ((PlotNumber)='" & PlotNumber & "'));"
Else
    If clsVProReg.CurrPlotlist = "None" Then
        SqlText = "SELECT * FROM [" & CurrProject() & "_Veg] WHERE (((Species) Is Not Null));"
    Else
        SqlText = "SELECT [" & MyProject & "_Veg].* FROM [" & MyProject & "_Veg] INNER JOIN [" & MySU & "_SU] ON [" & MyProject & "_Veg].PlotNumber = [" & MySU & "_SU].PlotNumber;"
    End If
End If
Set VegRS = MyDB.OpenRecordset(SqlText)
VegRS.MoveFirst
DoCmd.Hourglass True
Set sysActiveForm = Screen.ActiveForm

Do Until VegRS.EOF
    test = CheckCode(VegRS!Species, "USysAllSpecs", "Code")
        If test < 2 Then
            test = CheckCode(VegRS!Species, "USysUserSpp", "Code")
        End If
        If test < 2 Then
            Do Until Check4Synonym(VegRS!Species, PlotNumber)
                DoEvents
            Loop
        End If
        If test < 2 Then
            If Not IsLoaded("USysCodeCheck") Then
                DoCmd.OpenForm "USysCodeCheck"
                    If Not IsMissing(PlotNumber) Then
                        Forms![USysCodeCheck]![btnChangeAll].Enabled = False
                      End If
            End If
            Forms![USysCodeCheck].Repaint
            DoCmd.Hourglass False
            Forms!USysCodeCheck!txtNotInList = VegRS!Species
            X = PointToPlot(VegRS!PlotNumber, VegRS!Species)
            sysSpeciesTestNext = 1
                Do Until sysSpeciesTestNext = 0
                    If sysSpeciesTestNext = 3 Then
                        Plot = VegRS!PlotNumber
                        OldSpp = VegRS!Species
                        NewSpp = sysSpeciesToChange
                        AddSppToAudit Plot, OldSpp, NewSpp, VegRS!ID
                        VegRS.Edit
'======================================================================================================================================
                        VegRS!Species = sysSpeciesToChange
                        VegRS.Update
                        sysActiveForm.Recalc
                        sysSpeciesTestNext = 0
                    End If
                    If sysSpeciesTestNext = 4 Then
                        GoTo MyExit
                        sysSpeciesTestNext = 0
                    End If
                    DoEvents
                Loop
        End If
    VegRS.MoveNext
    DoCmd.Hourglass True
Loop

DoCmd.Close acForm, "USysCodeCheck"
    MsgBox "Code check complete"
    DoCmd.SetWarnings False
    SqlText = "DELETE FROM USysUserSpp WHERE (((LifeForm)=999));"
    DoCmd.RunSQL SqlText
    DoCmd.SetWarnings True

MyExit:
    On Error Resume Next
    DoCmd.Hourglass False
    VegRS.Close
    Set MyDB = Nothing
    Exit Function
errCheckSpeciesCodes:
    DoCmd.Hourglass False
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function CheckCode(Code2Check As String, CodeTable As String, CodeField As String) As Integer
'Oct. 4, 2002
'Checks codes
CheckCode = 0
On Error GoTo errCheckCode
Dim CheckRS As Recordset, CodeRS As Recordset, Criteria As String
Set MyDB = CurrentDb()
Set CodeRS = MyDB.OpenRecordset(CodeTable)

    Criteria = CodeField & " = '" & Code2Check & "'"
    CodeRS.MoveFirst
    CodeRS.FindFirst Criteria
    If CodeRS.NoMatch Then
        CheckCode = 1
    Else
        CheckCode = 2
    End If

MyExit:
    On Error Resume Next
    CodeRS.Close
    Set MyDB = Nothing
    Exit Function
errCheckCode:
    MsgBox Err.Description
    Resume MyExit
End Function

Function ChangeAllCodes(ChangeFrom As String, ChangeTo As String, DataTable As String, DataField As String) As Boolean
'Nov. 15, 1998
'Changes the code in the datatable
ChangeAllCodes = False
'On Error GoTo errChangeAllCodes
Dim MyRS As Recordset, Criteria As String, MyField As Field
Set MyDB = CurrentDb()
Set MyRS = MyDB.OpenRecordset(DataTable, dbOpenDynaset)
Set MyField = MyRS.Fields(DataField)
Criteria = DataField & " = '" & ChangeFrom & "'"
MyRS.MoveFirst
MyRS.FindFirst Criteria

Do Until MyRS.NoMatch
    AddSppToAudit MyRS!PlotNumber, ChangeFrom, ChangeTo, MyRS!ID
    MyRS.Edit
    MyField = ChangeTo
    MyRS.Update
    MyRS.FindNext Criteria
Loop
sysActiveForm.Recalc
ChangeAllCodes = True
        
MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
errChangeAllCodes:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function PointToPlot(PlotNumber As String, SearchSpecies As String) As Boolean
'Oct. 4, 2002
    On Error Resume Next
    Dim FoundIt As Boolean
    sysActiveForm.SetFocus
    DoCmd.GoToControl "VegPlotNumber"
    DoCmd.FindRecord PlotNumber
    'highlight species
    
    DoCmd.GoToControl "SubVegA"
    DoCmd.FindRecord SearchSpecies
    If Screen.ActiveControl = SearchSpecies Then GoTo MyExit
    DoCmd.GoToControl "SubVegC"
    DoCmd.FindRecord SearchSpecies
    If Screen.ActiveControl = SearchSpecies Then GoTo MyExit
    DoCmd.GoToControl "SubVegD"
    DoCmd.FindRecord SearchSpecies
    'DoCmd.OpenForm "USysCodeCheck"
MyExit:
    On Error GoTo 0
    
End Function
Private Function Check4Synonym(Code As String, Optional PlotNumber) As Boolean

'Oct. 6, 2008
On Error GoTo errCheck4Synonym

    If Not IsLoaded("USysCodeCheck") Then
        DoCmd.OpenForm "USysCodeCheck"
        If Not IsMissing(PlotNumber) Then
            Forms![USysCodeCheck]![btnChangeAll].Enabled = False
        End If
    End If

Check4Synonym = False
Dim MyRS As Recordset, Criteria As String
Set MyDB = CurrentDb()
Set MyRS = MyDB.OpenRecordset("USysAllSpecs")
Criteria = "OldCode = '" & Code & "'"

MyRS.FindFirst Criteria

If Not MyRS.NoMatch Then
    Forms![USysCodeCheck]![txtChangeTo] = MyRS!Code
    Forms![USysCodeCheck]![txtNotInList] = Code
    If Forms![USysCodeCheck]![btnChangeAll].Enabled Then
        MsgBox Code & " is an old code and has been replaced with " & MyRS!Code & ".  Click Change to replace this one instance or Change All to replace all instances of this code.", vbInformation, "VPro"
    Else
        MsgBox Code & " is an old code and has been replaced with " & MyRS!Code & ".", vbInformation, "VPro"
    End If
Else

End If

Check4Synonym = True

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Function
errCheck4Synonym:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Sub AddSppToAudit(Plot As String, OldSpp As String, NewSpp As Variant, ID As Long)
'May 31, 2002
'GoTo MyExit
'On Error GoTo 0
Dim db As Database, MyRS As Recordset
Set db = CurrentDb
Set MyRS = db.OpenRecordset(clsVProReg.CurrProject & "_Audit")

MyRS.AddNew
MyRS!Project = clsVProReg.CurrProject
MyRS!User = clsVProReg.UserName
MyRS!PlotNumber = Plot
MyRS!Table = "_Veg"
MyRS!EditField = "Species"
MyRS!EditWhen = Now()
MyRS!BeforeEdit = OldSpp
MyRS!AfterEdit = NewSpp
MyRS!ID = ID
MyRS.Update

MyExit:
    On Error Resume Next
    MyRS.Close
    Set db = Nothing
    Exit Sub
End Sub
