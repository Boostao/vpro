Public Sub UnattachProfileTables(Optional ProfileName As String)
Dim MyDB As Database, MyRS As Recordset
Dim i As Integer, TableToDelete As String, SqlText As String
Set MyDB = CurrentDb

If ProfileName = "" Then
    SqlText = "SELECT ProjectName FROM tblUnattach WHERE (((Unattach)=True));"
Else
    SqlText = "SELECT ProjectName FROM tblUnattach WHERE (((ProjectName)='" & ProfileName & "'));"
End If

Set MyRS = MyDB.OpenRecordset(SqlText)
If MyRS.RecordCount = 0 Then GoTo MyExit
MyRS.MoveLast
MyRS.MoveFirst

For i = 1 To MyRS.RecordCount
    TableToDelete = MyRS!ProjectName
    DoCmd.DeleteObject A_TABLE, TableToDelete & "_Profile"
    MyRS.MoveNext
Next i

UnattachDupTables

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Exit Sub
End Sub

Public Sub ListUnattachProfileTables()
Dim MyDB As Database
Dim TestConnect As String, i As Integer, TblName As String, SqlText As String
Dim MyRS As Recordset
Set MyDB = CurrentDb
DoCmd.SetWarnings False
SqlText = "DELETE * FROM tblUnattach;"
DoCmd.RunSQL SqlText
DoCmd.SetWarnings True

Set MyRS = MyDB.OpenRecordset("tblUnattach", dbOpenDynaset)

For i = 0 To MyDB.TableDefs.count - 1
    TblName = MyDB.TableDefs(i).Name
    If Right(TblName, 8) = "_Profile" And MyDB.TableDefs(i).Connect > "" And TblName <> clsVProReg.CurrVegProfile & "_Profile" Then
        MyRS.AddNew
        MyRS!ProjectName = Left(TblName, Len(TblName) - 8)
        MyRS.Update
    End If
Next i

MyRS.Close
Set MyDB = Nothing

End Sub
