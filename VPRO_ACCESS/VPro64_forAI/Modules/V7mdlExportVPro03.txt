Public Sub ExportVPro03()

'Feb 18, 2009
Dim RemoteDB As Database, MyDB As Database
Dim ProjectName As String, MyFile As String, Msg As String, Title As String
Set MyDB = CurrentDb

Msg = "Enter a name for your exported project.  The name must start with a letter, contain no punctuation characters, preferably no spaces, and be less than 40 characters in length"
Title = "Export your project"
ProjectName = InputBox(Msg, Title)
If ProjectName = "" Then GoTo MyExit
If CheckForExistingProject(MyDB, ProjectName & "_Env") Then 'project exists in the local db
    MsgBox "A project of that name is currently attached to VPro.  Please detach or try a different name.", vbInformation, "VPro"
    GoTo MyExit
End If

MyFile = GetDbName
If MyFile = "" Then GoTo MyExit
Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(MyFile)

If CheckForExistingProject(RemoteDB, ProjectName & "_Env") Then 'project exists in the remote db
    MsgBox "A project of that name already exists in that db.  Please try a different name or db.", vbInformation, "VPro"
    GoTo MyExit
End If

DeleteXXXRK 'clean up any failed attempt

AttachVPro03Tables RemoteDB

AppendVPro03Data clsVProReg.CurrProject 'transfer the data

RenameProject ProjectName, RemoteDB 'rename the project

RemoveLinks ProjectName 'unlink the tables

DeleteXXXRK 'cleanup

MyExit:
    On Error Resume Next
    DoCmd.SetWarnings True
    Set RemoteDB = Nothing
    Set MyDB = Nothing

End Sub

    RemoteDB.TableDefs("xxxrk_CWD").Name = ProjectName & "_CWD"
    RemoteDB.TableDefs("xxxrk_CWDPiling").Name = ProjectName & "_CWDPiling"

    RemoteDB.TableDefs("xxxrk_Mineral").Name = ProjectName & "_Mineral"

    RemoteDB.TableDefs("xxxrk_Trees").Name = ProjectName & "_Trees"

End Sub

Private Function CheckForExistingProject(db As Database, ProjectName As String) As Boolean

Dim MyName As String

    On Error Resume Next
    
    Exit Function
errCheckForExistingProject:
    If Err.Number = 3265 Then

Private Function Delete03Relationship(RemoteDB As String, Project As String)

'Feb 17, 2009
On Error GoTo errDeleteRelationship
Dim MyDB As Database, MyRel As Relation, MyFld As Field, RelationName As String
Dim VegTable As String, HumusTable As String, MineralTable As String
Dim TreePlotTable As String, TreesTable As String, CwdTable As String, CwdPileTable As String
Dim AuditTable As String, OtherTable As String, i As Integer
Dim MyRelName As String, X As Integer
Set MyDB = CurrentDb()
Dim EnvTable As String, db As Database ', MyWS As Workspace
EnvTable = Project
    
Set db = DBEngine.Workspaces(0).OpenDatabase(RemoteDB)

RelationName = EnvTable
EnvTable = RelationName & "_Env"
VegTable = RelationName & "_Veg"
HumusTable = RelationName & "_Humus"
MineralTable = RelationName & "_Mineral"
OtherTable = RelationName & "_Other"
AuditTable = RelationName & "_Audit"
TreePlotTable = RelationName & "_TreePlot"
TreesTable = RelationName & "_Trees"
CwdTable = RelationName & "_CWD"
CwdPileTable = RelationName & "_CWDPiling"
If Find03Relation(RelationName) = True Then GoTo MyExit

'delete old relationships
If db.Relations.count > 0 Then
    On Error Resume Next
    For i = 1 To 9
        db.Relations.Delete (Project & i)
    Next i
End If

MyExit:
    On Error Resume Next
    Set db = Nothing
    Set MyDB = Nothing
    Exit Function
errDeleteRelationship:
    MsgBox Err.Description
    Resume MyExit
End Function

Private Function Find03Relation(RelationName As String) As Boolean

'Feb 17, 2009
Dim MyDB As Database, MyRel As Relation
Set MyDB = CurrentDb()
Find03Relation = False

For Each MyRel In MyDB.Relations
    If MyRel.Table = RelationName Then
        Find03Relation = True
    End If
Next

Set MyDB = Nothing

End Function
