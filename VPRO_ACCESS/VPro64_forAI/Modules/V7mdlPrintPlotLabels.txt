Option Compare Database
Option Explicit
'This module contains code for printing plot identification labels to the default printer

Sub SetLabelRecord()
'Jan. 3, 2008
'prints a lable containg plot info - suitable for paper files
On Error GoTo errSetLabelRecord
Dim MyDB As Database, MyRS As Recordset
Dim MyFrm As Form
Dim SqlText As String
Set MyDB = CurrentDb

If IsLoaded("FS882-6x4XL") Then
    Set MyFrm = Forms("FS882-6x4XL")
End If
If IsLoaded("FS882-8x6XL") Then
    Set MyFrm = Forms("FS882-8x6XL")
End If
DoCmd.SetWarnings False
SqlText = "Delete * From BecLabels;"
DoCmd.RunSQL SqlText
DoCmd.SetWarnings True
Set MyRS = MyDB.OpenRecordset("BecLabels")

'MyRS.Edit
MyRS.AddNew
MyRS!PlotNumber1 = MyFrm.PlotNumber
MyRS!Zone1 = NullToNothing(MyFrm.Zone) & Space(4 - Len(NullToNothing(MyFrm.Zone))) & MyFrm.SubZone & "/" & NullToNothing(MyFrm.SiteSeries)
MyRS!Representing1 = NullToNothing(MyFrm.PlotRepresenting)
MyRS!ProjectID1 = NullToNothing(MyFrm.ProjectID)
MyRS.Update

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    DoCmd.SetWarnings True
    Exit Sub
errSetLabelRecord:
    MsgBox Err.Description
    Resume MyExit
End Sub
