Public Function GetDefPrinter() As String

'Sep 18, 2008
'used in report options
Dim sLen As Long, hResult As Long, GetDP As String

GetDP = Space$(255)
sLen = 255
hResult = GetDefaultPrinter(ByVal GetDP, sLen)

If hResult <> 0 Then
    GetDefPrinter = Left(GetDP, sLen - 1)
Else
    GetDefPrinter = ""
End If
End Function
