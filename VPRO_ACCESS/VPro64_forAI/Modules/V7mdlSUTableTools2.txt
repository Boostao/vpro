Sub UpdateOneSuTable(TblName As String, dbName As String)
Dim SqlText As String, qPos As Integer, OriginalTableName
DoCmd.TransferDatabase acLink, "Microsoft Access", dbName, acTable, TblName, TblName
OriginalTableName = TblName
DoCmd.SetWarnings False
SqlText = "DELETE * FROM [" & TblName & "];"
DoCmd.RunSQL SqlText

qPos = InStr(1, TblName, "?")
SqlText = "INSERT INTO [" & TblName & "] ( PlotNumber, SiteUnit )"
SqlText = SqlText & " SELECT PlotNumber, AssignedSiteUnit"
SqlText = SqlText & " FROM [" & CurrProject & "_ENV]"
If qPos > 0 Then
    TblName = Left(TblName, qPos - 1) & "[?]" & Mid(TblName, qPos + 1)
    SqlText = SqlText & " WHERE ((([zone] & [SubZone]) Like '" & Left(TblName, Len(TblName) - 6) & "'));"
Else
    SqlText = SqlText & " WHERE ((([zone] & [SubZone])='" & Left(TblName, Len(TblName) - 6) & "'));"
End If
DoCmd.RunSQL SqlText
DoCmd.DeleteObject acTable, OriginalTableName
DoCmd.SetWarnings True

End Sub
Function ControlGenerateSuTables() As String

'Sep 22, 2008
Dim NewDBName As String, Prompt As String, KillFile As Variant
Dim NewDB As Database
ControlGenerateSuTables = ""
On Error GoTo errtestit
    Prompt = "Create New Database"
    NewDBName = GetFileName(Prompt, False)
    If NewDBName = "" Then GoTo MyExit
    If NewDBName = CurrentDb.Name Then GoTo errtestit
Create:
    Set NewDB = DBEngine.Workspaces(0).CreateDatabase(Trim(NewDBName), DB_LANG_GENERAL)
    ControlGenerateSuTables = NewDB.Name
    GenerateTables CurrProject & "_ENV", NewDB.Name
    
MyExit:
    On Error Resume Next
    Set NewDB = Nothing
    Exit Function
errtestit:
    If Err.Number = 3204 Then
        KillFile = MsgBox("A database by that name already exists.  Do you want to replace it?", vbYesNo)
        If KillFile = 6 Then
        Kill NewDBName
        Resume Create
        End If
    Else
        MsgBox Err.Description
    End If
    Resume MyExit
End Function

Private Sub GenerateTables(ProjectName As String, dbName As String)
Dim RemoteDB As Database, MyDB As Database, MyRS As Recordset, MyQD As QueryDef
Dim i As Integer
Dim SqlText As String, TblName As String, MyTime As New clsStopWatch
Dim MyStr As String, qryStr As String, qPos As Integer
Set MyDB = CurrentDb
Set RemoteDB = DBEngine.Workspaces(0).OpenDatabase(dbName)
MyTime.StartTimer

'first get ris of zero-length strings!
DoCmd.SetWarnings False
SqlText = "UPDATE [" & ProjectName & "] SET Zone = Null WHERE (((Zone)=''));"
DoCmd.RunSQL SqlText
SqlText = "UPDATE [" & ProjectName & "] SET SubZone = Null WHERE (((SubZone)=''));"
DoCmd.RunSQL SqlText

'generate a unique list of zone/subzones and create a recordset
SqlText = "SELECT DISTINCT Zone, SubZone"
SqlText = SqlText & " FROM [" & ProjectName & "]"
SqlText = SqlText & " WHERE (((Zone) Is Not Null));"
Set MyRS = MyDB.OpenRecordset(SqlText, dbOpenDynaset)
MyRS.MoveLast
MyRS.MoveFirst

'loop through records and generate the tables
For i = 1 To MyRS.RecordCount
    If Not IsNull(MyRS!SubZone) Then
        qPos = InStr(1, MyRS!SubZone, "?")
    End If
    TblName = MyRS!Zone & MyRS!SubZone & "BEC_SU"
    'Forms![frmSuTableTools]![lblStatus].Caption = tblName
    'Forms![frmSuTableTools]![lblnDone].Caption = "Tables created: " & i
    'Forms![frmSuTableTools]![lblTime].Caption = "Elapsed time: " & MyTime.EndTimer / 1000 & " seconds"
    'Forms("frmSuTableTools").Repaint
    DoCmd.CopyObject , TblName, acTable, "USysSuTable"
    SqlText = "INSERT INTO [" & TblName & "] ( PlotNumber, SiteUnit )"
    SqlText = SqlText & " SELECT PlotNumber, AssignedSiteUnit"
    SqlText = SqlText & " FROM [" & ProjectName & "]"
    If IsNull(MyRS!SubZone) Then
        SqlText = SqlText & " WHERE (((Zone)='" & MyRS!Zone & "') AND ((SubZone) Is Null));"
    Else
        If qPos = 0 Then
            SqlText = SqlText & " WHERE (((Zone)='" & MyRS!Zone & "') AND ((SubZone)='" & MyRS!SubZone & "'));"
        Else
            MyStr = MyRS!SubZone
            qryStr = "Like '" & Left(MyStr, qPos - 1) & "[?]" & Mid(MyStr, qPos + 1) & "'"
            SqlText = SqlText & " WHERE (((Zone)='" & MyRS!Zone & "') AND ((SubZone) " & qryStr & "));"
        End If
    End If
    DoCmd.RunSQL SqlText
    DoCmd.TransferDatabase acExport, "Microsoft Access", RemoteDB.Name, acTable, TblName, TblName
    DoCmd.DeleteObject acTable, TblName
    MyRS.MoveNext
Next i
'Forms![frmSuTableTools]![lblStatus].Caption = " "
DoCmd.SetWarnings True

MyExit:
    On Error Resume Next
    MyRS.Close
    Set MyDB = Nothing
    Set RemoteDB = Nothing
    Exit Sub

End Sub
